**Plan:** ## Plan

**What makes this task unique:** Pawpularity is a *human-judged* aesthetic score, and the 12 binary metadata columns are themselves **derived from the image** (Blur, Eyes, Face, Group, Human, Near, Subject Focus, etc.). That means the tabular features are redundant with the visual signal — so blindly concatenating them adds noise. The key insight: the image carries most of the information; the metadata's value lies in **gating/modulating** what the vision encoder sees (e.g., a "Group" photo has different aesthetic criteria than a solo portrait).

**Why my approach fits:** Instead of the prior attempts' frozen-ResNet50 + MLP concat, I use a **trainable ConvNeXt-Tiny backbone** (stronger spatial inductive bias for fine-grained pet aesthetics) fused with tabular features via a **cross-modal gating network**: the tabular vector produces per-channel attention gates that scale the pooled image embedding before regression. This directly models "metadata tells the model *how to read* the image." MSELoss on [0,100] matches the RMSE metric exactly.

**How I'll validate:** Reproducible stratified-by-decile train/val split; if val RMSE beats a plain image-only baseline substantially, the gating hypothesis holds; if not, the gate collapses to ~uniform and we fall back to image-dominant behavior gracefully.

Stage 1 - Model Design for PetFinder.my Pawpularity Score (regression, RMSE).

Design summary
--------------
Modality : image (JPEG) + 12 binary tabular metadata columns
Target   : continuous Pawpularity in [0, 100], RMSE metric
Family   : CNN (ConvNeXt-Tiny backbone) fused with tabular via CROSS-MODAL GATING
Loss     : MSELoss on raw [0,100] targets
Output   : single float per sample in [0, 100]

Key idea
--------
The 12 metadata columns (Blur, Eyes, Face, Group, Human, Near, Subject Focus, ...)
are themselves DERIVED FROM THE IMAGE, so they are highly correlated with the visual
signal. Rather than naively concatenating them (which double-counts the same
information), we let the tabular vector produce PER-CHANNEL ATTENTION GATES that
modulate the pooled image embedding before the regression head. This encodes the
hypothesis that metadata tells the model HOW to read the image (a 'Group' photo is
judged differently than a solo portrait), instead of merely adding redundant features.

This module is import-safe: it only defines constants, transforms, the dataset, the

# --------------------------------------------------------------------------- #
# Tabular feature engineering (fit ONLY on training data to avoid leakage)
# --------------------------------------------------------------------------- #
class TabularScaler:
    """Standard scaler fitted on training rows; applied identically to val/test.

    Even though the columns are already binary {0,1}, we keep a fitted mean/std
    interface so the pipeline generalizes if the schema changes, and so the
    adapter can reuse the exact same object used during real training.
    """

    def __init__(self):
        self.mean_ = None
        self.std_ = None

    def fit(self, X: np.ndarray) -> "TabularScaler":
        X = np.asarray(X, dtype=np.float32)
        self.mean_ = X.mean(axis=0)
        std = X.std(axis=0)
        # Guard against zero-variance columns.
        std[std < 1e-8] = 1.0
        self.std_ = std
        return self

    def transform(self, X: np.ndarray) -> np.ndarray:
        X = np.asarray(X, dtype=np.float32)
        return (X - self.mean_) / self.std_

    def fit_transform(self, X: np.ndarray) -> np.ndarray:
        return self.fit(X).transform(X)


# --------------------------------------------------------------------------- #
# Dataset
# --------------------------------------------------------------------------- #
class PawpularityDataset(Dataset):
    """Loads JPEG images + scaled tabular features + target for one split."""

    def __init__(
        self,
        df: pd.DataFrame,
        img_dir: str,
        scaler: TabularScaler,
        transform: transforms.Compose,
        target_col: str = "Pawpularity",
    ):
        self.df = df.reset_index(drop=True)
        self.img_dir = img_dir
        self.scaler = scaler
        self.transform = transform
        self.target_col = target_col
        self.has_target = target_col in self.df.columns

    def __len__(self):
        return len(self.df)

    def _load_image(self, path: str) -> Image.Image:
        try:
            img = Image.open(path).convert("RGB")
        except Exception:
            # Corrupt / missing file fallback: neutral gray tile.
            img = Image.new("RGB", (IMG_SIZE, IMG_SIZE), (128, 128, 128))
        return img

    def __getitem__(self, idx):
        row = self.df.iloc[idx]
        img_path = os.path.join(self.img_dir, f"{row['Id']}.jpg")
        img = self._load_image(img_path)
        image = self.transform(img)

        tab_raw = row[TABULAR_COLS].to_numpy(dtype=np.float32)
        tabular = self.scaler.transform(tab_raw.reshape(1, -1))[0]

        item = {"image": image, "tabular": torch.from_numpy(tabular)}
        if self.has_target:
            item["target"] = torch.tensor(
                float(row[self.target_col]), dtype=torch.float32
            )
        return item


def collate_fn(batch):
    """Stack into tensors; keeps target optional (absent for test set)."""
    images = torch.stack([b["image"] for b in batch])
    tabular = torch.stack([b["tabular"] for b in batch])
    out = {"image": images, "tabular": tabular}
    if "target" in batch[0]:
        out["target"] = torch.stack([b["target"] for b in batch])
    return out


# --------------------------------------------------------------------------- #
# Splitting (reproducible, decile-stratified on target to cover full score range)
# --------------------------------------------------------------------------- #
def make_stratified_split(
    df: pd.DataFrame, val_fraction: float = VAL_FRACTION, seed: int = SEED
):
    """Stratify by Pawpularity decile so validation mirrors the full distribution."""
    rng = np.random.default_rng(seed)
    df = df.copy().reset_index(drop=True)
    df["_decile"] = pd.qcut(df["Pawpularity"], q=10, labels=False, duplicates="drop")

    train_idx, val_idx = [], []
    for d, grp in df.groupby("_decile"):
        n_val = max(1, int(round(len(grp) * val_fraction)))
        chosen = rng.choice(grp.index.to_numpy(), size=n_val, replace=False)
        val_set = set(chosen.tolist())
        val_idx.extend([i for i in grp.index if i in val_set])
        train_idx.extend([i for i in grp.index if i not in val_set])

    train_df = df.loc[train_idx].drop(columns=["_decile"]).reset_index(drop=True)
    val_df = df.loc[val_idx].drop(columns=["_decile"]).reset_index(drop=True)
    return train_df, val_df


# --------------------------------------------------------------------------- #
# Model architecture: ConvNeXt-Tiny backbone + cross-modal tabular gating
# --------------------------------------------------------------------------- #
def _load_convnext_tiny_pretrained():
    """Load pretrained ConvNeXt-Tiny weights WITHOUT any remote download.

    Uses torchvision's local weight cache when present; otherwise returns None
    so the caller falls back to random initialization (offline-safe).
    """
    try:
        from torchvision.models import convnext_tiny, ConvNeXt_Tiny_Weights

        w = ConvNeXt_Tiny_Weights.IMAGENET1K_V1
        # get_state_dict triggers a download only if not cached; wrap to stay safe.
        sd = w.get_state_dict()
        return sd
    except Exception:
        return None


class CrossModalGatedFusion(nn.Module):
    """Fuse an image embedding with tabular features via learned channel gates.

    The tabular vector produces per-channel attention gates (sigmoid) that scale
    the image embedding element-wise, then a small MLP regresses to [0, 100].
    This lets metadata MODULATE the visual representation rather than merely
    concatenate onto it.
    """

    def __init__(
        self,
        img_dim: int,
        tab_dim: int,
        hidden: int = FUSION_HIDDEN,
        dropout: float = DROPOUT_P,
    ):
        super().__init__()
        # Gate producer: tabular -> per-image-channel attention (same length as img_dim).
        self.tab_bn = nn.BatchNorm1d(tab_dim)
        self.gate_net = nn.Sequential(
            nn.Linear(tab_dim, hidden),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(hidden, img_dim),
            nn.Sigmoid(),
        )
        # Regression head over gated image embedding + tabular residual.
        self.head = nn.Sequential(
            nn.LayerNorm(img_dim + tab_dim),
            nn.Linear(img_dim + tab_dim, hidden),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(hidden, hidden // 2),
            nn.GELU(),
            nn.Dropout(dropout * 0.5),
            nn.Linear(hidden // 2, 1),
        )

    def forward(self, img_emb: torch.Tensor, tab: torch.Tensor) -> torch.Tensor:
        tab_n = self.tab_bn(tab)
        gate = self.gate_net(tab_n)  # [B, img_dim] in (0,1)
        gated_img = img_emb * gate  # per-channel modulation
        fused = torch.cat([gated_img, tab_n], dim=1)
        out = self.head(fused).squeeze(-1)  # [B]
        return torch.clamp(out, 0.0, 100.0)  # output interface: [0,100]


class PawpularityModel(nn.Module):
    """ConvNeXt-Tiny image encoder + cross-modal tabular-gated regression head."""

    def __init__(
        self,
        tab_dim: int = TABBAR_DIM,
        hidden: int = FUSION_HIDDEN,
        dropout: float = DROPOUT_P,
        pretrained: bool = True,
    ):
        super().__init__()
        from torchvision.models import convnext_tiny

        self.backbone = convnext_tiny(pretrained=False)
        img_dim = self.backbone.classifier[-1].in_features  # 768 for tiny
        # Replace classification head with global average pooling (we fuse manually).
        self.backbone.classifier = nn.Identity()

        if pretrained:
            sd = _load_convnext_tiny_pretrained()
            if sd is not None:
                # Load only the parts that match our modified backbone.
                own = self.backbone.state_dict()
                matched = {
                    k: v for k, v in sd.items() if k in own and own[k].shape == v.shape
                }
                missing, unexpected = self.backbone.load_state_dict(
                    matched, strict=False
                )
                # Random-init anything unmatched (e.g., dropped classifier layers).
                for name, param in self.backbone.named_parameters():
                    if name not in matched:
                        nn.init.kaiming_normal_(
                            param, mode="fan_out", nonlinearity="relu"
                        )
        else:
            for m in self.backbone.modules():
                if isinstance(m, (nn.Conv2d, nn.Linear)):
                    nn.init.kaiming_normal_(
                        m.weight, mode="fan_out", nonlinearity="relu"
                    )
                    if m.bias is not None:
                        nn.init.zeros_(m.bias)

        self.fusion = CrossModalGatedFusion(
            img_dim=img_dim, tab_dim=tab_dim, hidden=hidden, dropout=dropout
        )

    def forward(self, image: torch.Tensor, tabular: torch.Tensor) -> torch.Tensor:
        img_emb = self.backbone(image)  # [B, img_dim]
        return self.fusion(img_emb, tabular)  # [B] in [0,100]

# Data-loading helpers (used by both the adapter and the main training pipeline)

def load_datasets(
    input_dir: str = INPUT_DIR, val_fraction: float = VAL_FRACTION, seed: int = SEED
):
    """Build train/val datasets, test dataset, and the fitted scaler.

    Scaler is fit ONLY on the TRAIN split (leakage-safe). Returns a dict of
    datasets plus the scaler and the raw test dataframe (for submission order).
    """
    train_full = pd.read_csv(os.path.join(input_dir, "train.csv"))
    test_df = pd.read_csv(os.path.join(input_dir, "test.csv"))

    train_df, val_df = make_stratified_split(
        train_full, val_fraction=val_fraction, seed=seed
    )

    # Fit scaler strictly on the training portion.
    scaler = TabularScaler().fit(train_df[TABULAR_COLS].to_numpy(dtype=np.float32))

    train_ds = PawpularityDataset(
        train_df, os.path.join(input_dir, "train"), scaler, build_train_transform()
    )
    val_ds = PawpularityDataset(
        val_df, os.path.join(input_dir, "train"), scaler, build_eval_transform()
    )
    test_ds = PawpularityDataset(
        test_df,
        os.path.join(input_dir, "test"),
        scaler,
        build_eval_transform(),
        target_col=None,
    )

    return {
        "train_ds": train_ds,
        "val_ds": val_ds,
        "test_ds": test_ds,
        "scaler": scaler,
        "test_df": test_df,
        "train_df": train_df,
        "val_df": val_df,
    }


def build_dataloader(
    ds: Dataset,
    batch_size: int,
    shuffle: bool,
    num_workers: int = 2,
    pin_memory: bool = False,
) -> DataLoader:
    return DataLoader(
        ds,
        batch_size=batch_size,
        shuffle=shuffle,
        num_workers=num_workers,
        pin_memory=pin_memory,
        collate_fn=collate_fn,
    )


# --------------------------------------------------------------------------- #
# CPU preflight adapter (mandatory contract)
# --------------------------------------------------------------------------- #
class CandidateAdapter:
    """No-argument adapter exercising the REAL model, loss, transforms and optimizer.

    Every method treats the caller-supplied `context` as a PARTIAL mapping: it is
    merged over adapter defaults, and caller-provided values always win. `training_step`
    and `validation_step` resolve the real script-level criterion when the partial
    context omits it or supplies None.
    """

    DEFAULTS = {
        "img_size": IMG_SIZE,
        "tab_dim": TABBAR_DIM,
        "hidden": FUSION_HIDDEN,
        "dropout": DROPOUT_P,
        "pretrained": False,  # CPU preflight must not require weights/network
        "lr": 1e-3,
        "weight_decay": 1e-4,
        "input_dir": INPUT_DIR,
        "num_workers": 0,  # CPU preflight: no worker processes
        "pin_memory": False,
        "seed": SEED,
    }

    def __init__(self):
        self._datasets_cache = {}
        self._scaler = None

    # -- context merging ---------------------------------------------------- #
    def _ctx(self, context):
        merged = dict(self.DEFAULTS)
        if context:
            merged.update({k: v for k, v in context.items() if v is not None})
        return merged

    def _criterion(self, context):
        crit = None
        if context:
            crit = context.get("criterion", None)
        if crit is None:
            crit = build_criterion()
        return crit

    # -- lazy dataset construction (CPU-safe, no downloads) ----------------- #
    def _get_datasets(self, ctx):
        key = (ctx["input_dir"], ctx["img_size"])
        if key not in self._datasets_cache:
            ds = load_datasets(
                input_dir=ctx["input_dir"], val_fraction=VAL_FRACTION, seed=ctx["seed"]
            )
            self._datasets_cache[key] = ds
            self._scaler = ds["scaler"]
        return self._datasets_cache[key]

    # -- required methods --------------------------------------------------- #
    def build_model(self, context):
        ctx = self._ctx(context)
        model = PawpularityModel(
            tab_dim=ctx["tab_dim"],
            hidden=ctx["hidden"],
            dropout=ctx["dropout"],
            pretrained=ctx["pretrained"],
        )
        return model

    def build_optimizer(self, model, context):
        ctx = self._ctx(context)
        # AdamW from torch.optim (NOT transformers). Real optimizer used in training.
        return torch.optim.AdamW(
            model.parameters(), lr=ctx["lr"], weight_decay=ctx["weight_decay"]
        )

    def build_train_batch(self, scenario, device):
        ctx = self._ctx(scenario)
        bs = int(scenario.get("batch_size", 4))
        ds = self._get_datasets(ctx)["train_ds"]
        dl = build_dataloader(
            ds,
            batch_size=bs,
            shuffle=True,
            num_workers=ctx["num_workers"],
            pin_memory=ctx["pin_memory"],
        )
        batch = next(iter(dl))
        return self._to_device(batch, device)

    def build_validation_batch(self, scenario, device):
        ctx = self._ctx(scenario)
        bs = int(scenario.get("batch_size", 4))
        ds = self._get_datasets(ctx)["val_ds"]
        dl = build_dataloader(
            ds,
            batch_size=bs,
            shuffle=False,
            num_workers=ctx["num_workers"],
            pin_memory=ctx["pin_memory"],
        )
        batch = next(iter(dl))
        return self._to_device(batch, device)

    @staticmethod
    def _to_device(batch, device):
        dev = torch.device(device) if isinstance(device, str) else device
        out = {}
        for k, v in batch.items():
            if torch.is_tensor(v):
                out[k] = v.to(dev)
            else:
                out[k] = v
        return out

class TabularScaler:
    """Standard scaler fitted on training rows; applied identically to val/test.

    Even though the columns are already binary {0,1}, we keep a fitted mean/std
    interface so the pipeline generalizes if the schema changes, and so the
    adapter can reuse the exact same object used during real training.
    """

    def __init__(self):
        self.mean_ = None
        self.std_ = None

    def fit(self, X: np.ndarray) -> "TabularScaler":
        X = np.asarray(X, dtype=np.float32)
        self.mean_ = X.mean(axis=0)
        std = X.std(axis=0)
        # Guard against zero-variance columns.
        std[std < 1e-8] = 1.0
        self.std_ = std
        return self

    def transform(self, X: np.ndarray) -> np.ndarray:
        X = np.asarray(X, dtype=np.float32)
        return (X - self.mean_) / self.std_

    def fit_transform(self, X: np.ndarray) -> np.ndarray:
        return self.fit(X).transform(X)

def make_stratified_split(
    df: pd.DataFrame, val_fraction: float = VAL_FRACTION, seed: int = SEED
):
    """Stratify by Pawpularity decile so validation mirrors the full distribution."""
    rng = np.random.default_rng(seed)
    df = df.copy().reset_index(drop=True)
    df["_decile"] = pd.qcut(df["Pawpularity"], q=10, labels=False, duplicates="drop")

    train_idx, val_idx = [], []
    for d, grp in df.groupby("_decile"):
        n_val = max(1, int(round(len(grp) * val_fraction)))
        chosen = rng.choice(grp.index.to_numpy(), size=n_val, replace=False)
        val_set = set(chosen.tolist())
        val_idx.extend([i for i in grp.index if i in val_set])
        train_idx.extend([i for i in grp.index if i not in val_set])

    train_df = df.loc[train_idx].drop(columns=["_decile"]).reset_index(drop=True)
    val_df = df.loc[val_idx].drop(columns=["_decile"]).reset_index(drop=True)

def _load_convnext_tiny_pretrained():
    """Load pretrained ConvNeXt-Tiny weights WITHOUT any remote download.

    Uses torchvision's local weight cache when present; otherwise returns None
    so the caller falls back to random initialization (offline-safe).
    """
    try:
        from torchvision.models import convnext_tiny, ConvNeXt_Tiny_Weights

        w = ConvNeXt_Tiny_Weights.IMAGENET1K_V1
        # get_state_dict triggers a download only if not cached; wrap to stay safe.
        sd = w.get_state_dict()
        return sd
    except Exception:
        return None


class CrossModalGatedFusion(nn.Module):
    """Fuse an image embedding with tabular features via learned channel gates.

    The tabular vector produces per-channel attention gates (sigmoid) that scale
    the image embedding element-wise, then a small MLP regresses to [0, 100].
    This lets metadata MODULATE the visual representation rather than merely
    concatenate onto it.
    """

    def __init__(
        self,
        img_dim: int,
        tab_dim: int,
        hidden: int = FUSION_HIDDEN,
        dropout: float = DROPOUT_P,
    ):
        super().__init__()
        # Gate producer: tabular -> per-image-channel attention (same length as img_dim).
        self.tab_bn = nn.BatchNorm1d(tab_dim)
        self.gate_net = nn.Sequential(
            nn.Linear(tab_dim, hidden),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(hidden, img_dim),
            nn.Sigmoid(),
        )
        # Regression head over gated image embedding + tabular residual.
        self.head = nn.Sequential(
            nn.LayerNorm(img_dim + tab_dim),
            nn.Linear(img_dim + tab_dim, hidden),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(hidden, hidden // 2),
            nn.GELU(),
            nn.Dropout(dropout * 0.5),
            nn.Linear(hidden // 2, 1),
        )

    def forward(self, img_emb: torch.Tensor, tab: torch.Tensor) -> torch.Tensor:
        tab_n = self.tab_bn(tab)
        gate = self.gate_net(tab_n)  # [B, img_dim] in (0,1)
        gated_img = img_emb * gate  # per-channel modulation
        fused = torch.cat([gated_img, tab_n], dim=1)
        out = self.head(fused).squeeze(-1)  # [B]
        return torch.clamp(out, 0.0, 100.0)  # output interface: [0,100]


class PawpularityModel(nn.Module):
    """ConvNeXt-Tiny image encoder + cross-modal tabular-gated regression head."""

    def __init__(
        self,
        tab_dim: int = TABBAR_DIM,
        hidden: int = FUSION_HIDDEN,
        dropout: float = DROPOUT_P,
        pretrained: bool = True,
    ):
        super().__init__()
        from torchvision.models import convnext_tiny

        self.backbone = convnext_tiny(pretrained=False)
        img_dim = self.backbone.classifier[-1].in_features  # 768 for tiny
        # Replace classification head with global average pooling (we fuse manually).
        self.backbone.classifier = nn.Identity()

        if pretrained:
            sd = _load_convnext_tiny_pretrained()
            if sd is not None:
                # Load only the parts that match our modified backbone.
                own = self.backbone.state_dict()
                matched = {
                    k: v for k, v in sd.items() if k in own and own[k].shape == v.shape
                }
                missing, unexpected = self.backbone.load_state_dict(
                    matched, strict=False
                )
                # Random-init anything unmatched (e.g., dropped classifier layers).
                for name, param in self.backbone.named_parameters():
                    if name not in matched:
                        nn.init.kaiming_normal_(
                            param, mode="fan_out", nonlinearity="relu"
                        )
        else:
            for m in self.backbone.modules():
                if isinstance(m, (nn.Conv2d, nn.Linear)):
                    nn.init.kaiming_normal_(
                        m.weight, mode="fan_out", nonlinearity="relu"
                    )
                    if m.bias is not None:
                        nn.init.zeros_(m.bias)

        self.fusion = CrossModalGatedFusion(
            img_dim=img_dim, tab_dim=tab_dim, hidden=hidden, dropout=dropout
        )

    def forward(self, image: torch.Tensor, tabular: torch.Tensor) -> torch.Tensor:
        img_emb = self.backbone(image)  # [B, img_dim]
        return self.fusion(img_emb, tabular)  # [B] in [0,100]

        return merged

    def _criterion(self, context):

    def build_model(self, context):

        model = PawpularityModel(
            tab_dim=ctx["tab_dim"],
            hidden=ctx["hidden"],
            dropout=ctx["dropout"],
            pretrained=ctx["pretrained"],
        )
        return model

    def build_optimizer(self, model, context):
        ctx = self._ctx(context)
        # AdamW from torch.optim (NOT transformers). Real optimizer used in training.
        return torch.optim.AdamW(
            model.parameters(), lr=ctx["lr"], weight_decay=ctx["weight_decay"]
        )

    def build_train_batch(self, scenario, device):

        bs = int(scenario.get("batch_size", 4))
        ds = self._get_datasets(ctx)["train_ds"]
        dl = build_dataloader(
            ds,
            batch_size=bs,
            shuffle=True,
            num_workers=ctx["num_workers"],
            pin_memory=ctx["pin_memory"],
        )
        batch = next(iter(dl))
        return self._to_device(batch, device)

    def build_validation_batch(self, scenario, device):
        ctx = self._ctx(scenario)
        bs = int(scenario.get("batch_size", 4))
        ds = self._get_datasets(ctx)["val_ds"]
        dl = build_dataloader(
            ds,
            batch_size=bs,
            shuffle=False,
            num_workers=ctx["num_workers"],
            pin_memory=ctx["pin_memory"],
        )
        batch = next(iter(dl))
        return self._to_device(batch, device)

    @staticmethod
    def _to_device(batch, device):
        dev = torch.device(device) if isinstance(device, str) else device
        out = {}
        for k, v in batch.items():
            if torch.is_tensor(v):
                out[k] = v.to(dev)
            else:
                out[k] = v
        return out

    def training_step(self, model, batch, context):
        crit = self._criterion(context)
        image = batch["image"]
        tabular = batch["tabular"]
        target = batch["target"]
        pred = model(image, tabular)
        loss = crit(pred, target)
        return loss

    def validation_step(self, model, batch, context):

        image = batch["image"]
        tabular = batch["tabular"]
        target = batch["target"]
        with torch.no_grad():
            pred = model(image, tabular)
            loss = crit(pred, target)
        return loss

    class CrossModalGatedFusion(nn.Module):
        def __init__(self, img_dim, tab_dim, hidden=HIDDEN, dropout=DROPOUT_P):
            super().__init__()
            self.tab_bn = nn.BatchNorm1d(tab_dim)
            self.gate_net = nn.Sequential(
                nn.Linear(tab_dim, hidden),
                nn.GELU(),
                nn.Dropout(dropout),
                nn.Linear(hidden, img_dim),
                nn.Sigmoid(),
            )
            self.head = nn.Sequential(
                nn.LayerNorm(img_dim + tab_dim),
                nn.Linear(img_dim + tab_dim, hidden),
                nn.GELU(),
                nn.Dropout(dropout),
                nn.Linear(hidden, hidden // 2),
                nn.GELU(),
                nn.Dropout(dropout * 0.5),
                nn.Linear(hidden // 2, 1),
            )

        def forward(self, img_emb, tab):
            tab_n = self.tab_bn(tab)
            gate = self.gate_net(tab_n)
            gated_img = img_emb * gate
            fused = torch.cat([gated_img, tab_n], dim=1)
            out = self.head(fused).squeeze(-1)
            return torch.clamp(out, 0.0, 100.0)

# Differential-learning-rate parameter groups (backbone vs fusion head)

def build_param_groups(model, lr_backbone, lr_head):
    """Backbone params get a low LR; fusion-head/gate params get a high LR."""
    backbone_ids = set(id(p) for p in model.backbone.parameters())
    backbone_params, head_params = [], []
    for name, p in model.named_parameters():
        if not p.requires_grad:
            continue
        (backbone_params if id(p) in backbone_ids else head_params).append(p)
    return [
        {"params": backbone_params, "lr": lr_backbone},
        {"params": head_params, "lr": lr_head},
    ]


def build_optimizer(
    model, lr_backbone=LR_BACKBONE, lr_head=LR_HEAD, weight_decay=WEIGHT_DECAY
):
    groups = build_param_groups(model, lr_backbone, lr_head)
    return torch.optim.AdamW(groups, lr=lr_head, weight_decay=weight_decay)


class WarmupCosineScheduler:
    """Linear warmup for `warmup_epochs`, then cosine decay to ~0 over remaining."""

    def __init__(self, optimizer, total_epochs, warmup_epochs=WARMUP_EPOCHS):
        self.optimizer = optimizer
        self.total = max(int(total_epochs), 1)
        self.warmup = max(int(warmup_epochs), 0)
        self.base_lrs = [g["lr"] for g in optimizer.param_groups]
        self.epoch = 0

    def step(self):
        self.epoch += 1
        e = self.epoch
        if e <= self.warmup:
            frac = e / max(self.warmup, 1)
        else:
            span = max(self.total - self.warmup, 1)
            prog = (e - self.warmup) / span
            frac = 0.5 * (1.0 + math.cos(math.pi * min(prog, 1.0)))
        for group, base in zip(self.optimizer.param_groups, self.base_lrs):
            group["lr"] = base * max(frac, 1e-8)
        return self.optimizer.param_groups[0]["lr"]
