def _load_pretrained_backbone(model_name="convnext_tiny.in12k_ft_in1k", num_classes=0):
    """Offline-safe backbone loader.

    Tries a locally cached timm weight first; falls back to random init so the
    script always runs without network access.
    """
    try:
        import timm

        try:
            model = timm.create_model(
                model_name, pretrained=True, num_classes=num_classes
            )
        except Exception:
            # No cached weights available -> random init (still a real CNN).
            model = timm.create_model(
                model_name, pretrained=False, num_classes=num_classes
            )
        return model
    except Exception:
        # Last-resort fallback: torchvision ResNet50 (random init, offline).
        import torchvision.models as tvm

        m = tvm.resnet50(weights=None)
        m.fc = nn.Identity()
        return m


class GatedTabularFusion(nn.Module):
    """Cross-attention fusion of a tabular embedding into vision features.

    The tabular vector becomes a single query token that attends over the
    spatial vision tokens; a learned gate blends the residual stream with the
    attended context so the model can down-weight noisy modalities.
    """

    def __init__(self, vision_dim, tab_dim, num_heads=4, dropout=0.1):
        super().__init__()
        self.tab_proj = nn.Sequential(
            nn.Linear(tab_dim, vision_dim),
            nn.LayerNorm(vision_dim),
            nn.GELU(),
        )
        self.attn = nn.MultiheadAttention(
            embed_dim=vision_dim, num_heads=num_heads, dropout=dropout, batch_first=True
        )
        self.norm = nn.LayerNorm(vision_dim)
        self.gate = nn.Sequential(
            nn.Linear(vision_dim * 2, vision_dim),
            nn.Sigmoid(),
        )
        self.drop = nn.Dropout(dropout)

    def forward(self, vision_feat, tabular):
        # vision_feat: [B, D] ; tabular: [B, tab_dim]
        q = self.tab_proj(tabular).unsqueeze(1)  # [B, 1, D]
        kv = vision_feat.unsqueeze(1)  # [B, 1, D]
        ctx, _ = self.attn(q, kv, kv)  # [B, 1, D]
        ctx = ctx.squeeze(1)  # [B, D]
        g = self.gate(torch.cat([vision_feat, ctx], dim=-1))
        fused = self.norm(vision_feat + g * self.drop(ctx))
        return fused


class PawpularityModel(nn.Module):
    """convnext backbone + gated cross-attention tabular fusion -> scalar in [0,100]."""

    def __init__(self, tab_dim=NUM_TABULAR, dropout=0.3):
        super().__init__()
        self.backbone = _load_pretrained_backbone(num_classes=0)
        # Probe the backbone's feature dimension robustly.
        with torch.no_grad():
            probe = torch.zeros(1, 3, IMAGE_SIZE, IMAGE_SIZE)
            feat = self.backbone(probe)
            if isinstance(feat, (list, tuple)):
                feat = feat[0]
            self.vision_dim = int(feat.shape[-1])
        self.fusion = GatedTabularFusion(
            self.vision_dim, tab_dim, num_heads=4, dropout=0.1
        )
        self.head = nn.Sequential(
            nn.Linear(self.vision_dim, 256),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(256, 64),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(64, 1),
        )

    def forward(self, image, tabular):
        feat = self.backbone(image)
        if isinstance(feat, (list, tuple)):
            feat = feat[0]
        if feat.dim() == 4:  # spatial map -> global pool
            feat = feat.flatten(1)
        elif feat.dim() == 3:  # sequence tokens -> mean pool
            feat = feat.mean(dim=1)
        fused = self.fusion(feat, tabular)
        out = self.head(fused).squeeze(-1)
        return out

def mixup_features(images, tabular, targets, alpha=0.2):
    """Feature-level Mixup: jointly interpolate images, tabular vectors, targets.

    Regularizes the subjective regression target by creating smooth synthetic
    examples consistent across both modalities.
    """
    lam = float(np.random.beta(alpha, alpha))
    n = images.size(0)
    idx = torch.randperm(n, device=images.device)
    mixed_img = lam * images + (1.0 - lam) * images[idx]
    mixed_tab = lam * tabular + (1.0 - lam) * tabular[idx]
    mixed_tgt = lam * targets + (1.0 - lam) * targets[idx]
    return mixed_img, mixed_tab, mixed_tgt

class CandidateAdapter:
    """Exercises the REAL model, loss, inputs and optimizer on tiny CPU batches."""

    DEFAULT_CONTEXT = {
        "criterion": None,
        "model": None,
        "optimizer": None,
        "lr": 1e-4,
        "batch_size": 2,
        "num_workers": 0,
    }

    def _ctx(self, context):
        merged = dict(self.DEFAULT_CONTEXT)
        if context:
            merged.update(context)
        return merged

    def build_model(self, context):
        ctx = self._ctx(context)
        if ctx.get("model") is None:
            ctx["model"] = build_model()
        return ctx["model"]

    def build_optimizer(self, model, context):
        ctx = self._ctx(context)
        lr = float(ctx.get("lr", 1e-4))
        opt = torch.optim.AdamW(model.parameters(), lr=lr, weight_decay=1e-4)
        ctx["optimizer"] = opt
        return opt

    def _tiny_batch(self, scenario, device):
        bs = int(scenario.get("batch_size", 2))
        image = torch.randn(bs, 3, IMAGE_SIZE, IMAGE_SIZE, device=device)
        tabular = torch.randint(0, 2, (bs, NUM_TABULAR)).to(
            torch.float32, device=device
        )
        target = torch.rand(bs, device=device) * 100.0
        return {"image": image, "tabular": tabular, "target": target}

    def build_train_batch(self, scenario, device):
        return self._tiny_batch(scenario, device)

    def build_validation_batch(self, scenario, device):
        return self._tiny_batch(scenario, device)

    def training_step(self, model, batch, context):
        ctx = self._ctx(context)
        criterion = ctx.get("criterion")
        if criterion is None:
            criterion = get_criterion()
        optimizer = ctx.get("optimizer")
        if optimizer is None:
            optimizer = self.build_optimizer(model, ctx)

        optimizer.zero_grad(set_to_none=True)
        out = model(batch["image"], batch["tabular"])
        loss = criterion(out, batch["target"])
        loss.backward()
        optimizer.step()
        return loss.detach()

    def validation_step(self, model, batch, context):
        ctx = self._ctx(context)
        criterion = ctx.get("criterion")
        if criterion is None:
            criterion = get_criterion()
        model.eval()
        with torch.no_grad():
            out = model(batch["image"], batch["tabular"])
            loss = criterion(out, batch["target"])
        return loss.detach()

def _load_pretrained_backbone(model_name="convnext_tiny.in12k_ft_in1k", num_classes=0):
    """Offline-safe backbone loader.

    Tries a locally cached timm weight first; falls back to random init so the
    script always runs without network access.
    """
    try:
        import timm

        try:
            model = timm.create_model(
                model_name, pretrained=True, num_classes=num_classes
            )
        except Exception:
            # No cached weights available -> random init (still a real CNN).
            model = timm.create_model(
                model_name, pretrained=False, num_classes=num_classes
            )
        return model
    except Exception:
        # Last-resort fallback: torchvision ResNet50 (random init, offline).
        import torchvision.models as tvm

        m = tvm.resnet50(weights=None)
        m.fc = nn.Identity()
        return m


class GatedTabularFusion(nn.Module):
    """Cross-attention fusion of a tabular embedding into vision features.

    The tabular vector becomes a single query token that attends over the
    spatial vision tokens; a learned gate blends the residual stream with the
    attended context so the model can down-weight noisy modalities.
    """

    def __init__(self, vision_dim, tab_dim, num_heads=4, dropout=0.1):
        super().__init__()
        self.tab_proj = nn.Sequential(
            nn.Linear(tab_dim, vision_dim),
            nn.LayerNorm(vision_dim),
            nn.GELU(),
        )
        self.attn = nn.MultiheadAttention(
            embed_dim=vision_dim, num_heads=num_heads, dropout=dropout, batch_first=True
        )
        self.norm = nn.LayerNorm(vision_dim)
        self.gate = nn.Sequential(
            nn.Linear(vision_dim * 2, vision_dim),
            nn.Sigmoid(),
        )
        self.drop = nn.Dropout(dropout)

    def forward(self, vision_feat, tabular):
        # vision_feat: [B, D] ; tabular: [B, tab_dim]
        q = self.tab_proj(tabular).unsqueeze(1)  # [B, 1, D]
        kv = vision_feat.unsqueeze(1)  # [B, 1, D]
        ctx, _ = self.attn(q, kv, kv)  # [B, 1, D]
        ctx = ctx.squeeze(1)  # [B, D]
        g = self.gate(torch.cat([vision_feat, ctx], dim=-1))
        fused = self.norm(vision_feat + g * self.drop(ctx))
        return fused


class PawpularityModel(nn.Module):
    """convnext backbone + gated cross-attention tabular fusion -> scalar in [0,100]."""

    def __init__(self, tab_dim=NUM_TABULAR, dropout=0.3):
        super().__init__()
        self.backbone = _load_pretrained_backbone(num_classes=0)
        # Probe the backbone's feature dimension robustly.
        with torch.no_grad():
            probe = torch.zeros(1, 3, IMAGE_SIZE, IMAGE_SIZE)
            feat = self.backbone(probe)
            if isinstance(feat, (list, tuple)):
                feat = feat[0]
            self.vision_dim = int(feat.shape[-1])
        self.fusion = GatedTabularFusion(
            self.vision_dim, tab_dim, num_heads=4, dropout=0.1
        )
        self.head = nn.Sequential(
            nn.Linear(self.vision_dim, 256),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(256, 64),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(64, 1),
        )

    def forward(self, image, tabular):
        feat = self.backbone(image)
        if isinstance(feat, (list, tuple)):
            feat = feat[0]
        if feat.dim() == 4:  # spatial map -> global pool
            feat = feat.flatten(1)
        elif feat.dim() == 3:  # sequence tokens -> mean pool
            feat = feat.mean(dim=1)
        fused = self.fusion(feat, tabular)
        out = self.head(fused).squeeze(-1)
        return out

def mixup_features(images, tabular, targets, alpha=0.2):
    """Feature-level Mixup: jointly interpolate images, tabular vectors, targets.

    Regularizes the subjective regression target by creating smooth synthetic
    examples consistent across both modalities.
    """
    lam = float(np.random.beta(alpha, alpha))
    n = images.size(0)
    idx = torch.randperm(n, device=images.device)
    mixed_img = lam * images + (1.0 - lam) * images[idx]
    mixed_tab = lam * tabular + (1.0 - lam) * tabular[idx]
    mixed_tgt = lam * targets + (1.0 - lam) * targets[idx]
    return mixed_img, mixed_tab, mixed_tgt

class CandidateAdapter:
    """Exercises the REAL model, loss, inputs and optimizer on tiny CPU batches."""

    DEFAULT_CONTEXT = {
        "criterion": None,
        "model": None,
        "optimizer": None,
        "lr": 1e-4,
        "batch_size": 2,
        "num_workers": 0,
    }

    def _ctx(self, context):
        merged = dict(self.DEFAULT_CONTEXT)
        if context:
            merged.update(context)
        return merged

    def build_model(self, context):
        ctx = self._ctx(context)
        if ctx.get("model") is None:
            ctx["model"] = build_model()
        return ctx["model"]

    def build_optimizer(self, model, context):
        ctx = self._ctx(context)
        lr = float(ctx.get("lr", 1e-4))
        opt = torch.optim.AdamW(model.parameters(), lr=lr, weight_decay=1e-4)
        ctx["optimizer"] = opt
        return opt

    def _tiny_batch(self, scenario, device):
        bs = int(scenario.get("batch_size", 2))
        image = torch.randn(bs, 3, IMAGE_SIZE, IMAGE_SIZE, device=device)
        tabular = torch.randint(0, 2, (bs, NUM_TABULAR)).to(
            torch.float32, device=device
        )
        target = torch.rand(bs, device=device) * 100.0
        return {"image": image, "tabular": tabular, "target": target}

    def build_train_batch(self, scenario, device):
        return self._tiny_batch(scenario, device)

    def build_validation_batch(self, scenario, device):
        return self._tiny_batch(scenario, device)

    def training_step(self, model, batch, context):
        ctx = self._ctx(context)
        criterion = ctx.get("criterion")
        if criterion is None:
            criterion = get_criterion()
        optimizer = ctx.get("optimizer")
        if optimizer is None:
            optimizer = self.build_optimizer(model, ctx)

        optimizer.zero_grad(set_to_none=True)
        out = model(batch["image"], batch["tabular"])
        loss = criterion(out, batch["target"])
        loss.backward()
        optimizer.step()
        return loss.detach()

    def validation_step(self, model, batch, context):
        ctx = self._ctx(context)
        criterion = ctx.get("criterion")
        if criterion is None:
            criterion = get_criterion()
        model.eval()
        with torch.no_grad():
            out = model(batch["image"], batch["tabular"])
            loss = criterion(out, batch["target"])
        return loss.detach()

def _param_groups(model):
    """Differential learning rates: backbone vs fusion+head."""
    backbone_params = []
    head_params = []
    for name, p in model.named_parameters():
        if not p.requires_grad:
            continue
        if name.startswith("backbone"):
            backbone_params.append(p)
        else:
            head_params.append(p)
    groups = [
        {"params": head_params, "lr": HEAD_LR},
        {"params": backbone_params, "lr": BACKBONE_LR},
    ]
    return groups


def build_optimizer(model):
    groups = _param_groups(model)
    return torch.optim.AdamW(groups, weight_decay=WEIGHT_DECAY)

# Scheduler: linear warmup -> CosineAnnealingWarmRestarts
# --------------------------------------------------------------------------- #
class WarmupCosineScheduler:
    """Linear warmup for `warmup_steps`, then CosineAnnealingWarmRestarts."""

    def __init__(
        self, optimizer, warmup_steps=WARMUP_STEPS, t0=COSINE_T0, t_mult=COSINE_T_MULT
    ):
        self.optimizer = optimizer
        self.warmup_steps = max(1, int(warmup_steps))
        self.t0 = t0
        self.t_mult = t_mult
        self.step_count = 0
        self.base_lrs = [g["lr"] for g in optimizer.param_groups]
        self.cosine = torch.optim.lr_scheduler.CosineAnnealingWarmRestarts(
            optimizer, T_0=t0, T_mult=t_mult
        )

    def step(self):
        self.step_count += 1
        if self.step_count <= self.warmup_steps:
            frac = self.step_count / float(self.warmup_steps)
            for pg, base_lr in zip(self.optimizer.param_groups, self.base_lrs):
                pg["lr"] = base_lr * frac
        else:
            self.cosine.step()

    def get_last_lr(self):
        return [pg["lr"] for pg in self.optimizer.param_groups]

# Metric: RMSE (the leaderboard metric)

def compute_rmse(preds, targets):
    preds = np.asarray(preds, dtype=np.float64).ravel()
    targets = np.asarray(targets, dtype=np.float64).ravel()
    if len(preds) == 0:
        return float("nan")
    return float(np.sqrt(np.mean((preds - targets) ** 2)))

@torch.no_grad()
def evaluate_rmse(model, loader, device):
    """Run model forward on every sample; return RMSE (lower is better)."""
    model.eval()
    all_preds, all_targets = [], []
    for batch in loader:
        images = batch["image"].to(device, non_blocking=True)
        tabular = batch["tabular"].to(device, non_blocking=True)
        out = model(images, tabular)
        out = torch.clamp(out, 0.0, 100.0)
        all_preds.append(out.cpu().numpy())
        all_targets.append(batch["target"].cpu().numpy())
    preds = np.concatenate(all_preds)
    targets = np.concatenate(all_targets)
    return compute_rmse(preds, targets), preds, targets

def train_one_epoch(
    model, loader, criterion, optimizer, scheduler, device, use_mixup=True
):
    model.train()
    total_loss, n_batches = 0.0, 0
    for batch in loader:
        images = batch["image"].to(device, non_blocking=True)
        tabular = batch["tabular"].to(device, non_blocking=True)
        targets = batch["target"].to(device, non_blocking=True)

        # Feature-level Mixup (jointly interpolates both modalities + target).
        if use_mixup and images.size(0) > 1 and np.random.rand() < MIXUP_PROB:
            from solution import mixup_features

            images, tabular, targets = mixup_features(
                images, tabular, targets, alpha=MIXUP_ALPHA
            )

        optimizer.zero_grad(set_to_none=True)
        out = model(images, tabular)
        loss = criterion(out, targets)
        loss.backward()
        if GRAD_CLIP is not None:
            torch.nn.utils.clip_grad_norm_(model.parameters(), GRAD_CLIP)
        optimizer.step()
        scheduler.step()

        total_loss += float(loss.item())
        n_batches += 1
    return total_loss / max(1, n_batches)


def fit(model, train_loader, val_loader, device):
    criterion = get_criterion()
    optimizer = build_optimizer(model)
    scheduler = WarmupCosineScheduler(optimizer)

    best_rmse = float("inf")
    best_state = None
    patience_counter = 0

    for epoch in range(1, MAX_EPOCHS + 1):
        t0 = time.time()
        train_loss = train_one_epoch(
            model,
            train_loader,
            criterion,
            optimizer,
            scheduler,
            device,
            use_mixup=True,
        )
        val_rmse, _, _ = evaluate_rmse(model, val_loader, device)
        elapsed = time.time() - t0
        cur_lr = scheduler.get_last_lr()[0]
        print(
            f"[epoch {epoch:02d}] train_mse={train_loss:.4f} "
            f"val_rmse={val_rmse:.4f} lr={cur_lr:.2e} time={elapsed:.1f}s"
        )

        if val_rmse < best_rmse - 1e-6:
            best_rmse = val_rmse
            best_state = copy.deepcopy(model.state_dict())
            patience_counter = 0
            try:
                os.makedirs(WORKING_DIR, exist_ok=True)
                torch.save(best_state, BEST_CKPT)
            except Exception:
                pass
        else:
            patience_counter += 1
            if patience_counter >= PATIENCE:
                print(f"[early-stop] no improvement for {PATIENCE} epochs.")
                break

    # Restore best weights for final evaluation/inference.
    if best_state is not None:
        model.load_state_dict(best_state)
    return best_rmse

# CPU preflight adapter (no-argument, import-safe, no downloads, no CUDA)

class CandidateAdapter:
    """Exercises the REAL model, loss, inputs and optimizer on tiny CPU batches."""

    DEFAULT_CONTEXT = {
        "criterion": None,
        "model": None,
        "optimizer": None,
        "lr": HEAD_LR,
        "batch_size": 2,
        "num_workers": 0,
    }

    def _ctx(self, context):
        merged = dict(self.DEFAULT_CONTEXT)
        if context:
            merged.update(context)
        return merged

    def build_model(self, context):
        ctx = self._ctx(context)
        if ctx.get("model") is None:
            ctx["model"] = build_model(device=None)
        return ctx["model"]

    def build_optimizer(self, model, context):
        ctx = self._ctx(context)
        opt = build_optimizer(model)
        ctx["optimizer"] = opt
        return opt

    def _tiny_batch(self, scenario, device):
        bs = int(scenario.get("batch_size", 2))
        image = torch.randn(bs, 3, IMAGE_SIZE, IMAGE_SIZE, device=device)
        tabular = torch.randint(0, 2, (bs, NUM_TABULAR)).to(
            torch.float32, device=device
        )
        target = torch.rand(bs, device=device) * 100.0
        return {"image": image, "tabular": tabular, "target": target}

    def build_train_batch(self, scenario, device):
        return self._tiny_batch(scenario, device)

    def build_validation_batch(self, scenario, device):
        return self._tiny_batch(scenario, device)

    def training_step(self, model, batch, context):
        ctx = self._ctx(context)
        criterion = ctx.get("criterion")
        if criterion is None:
            criterion = get_criterion()
        optimizer = ctx.get("optimizer")
        if optimizer is None:
            optimizer = self.build_optimizer(model, ctx)

        optimizer.zero_grad(set_to_none=True)
        out = model(batch["image"], batch["tabular"])
        loss = criterion(out, batch["target"])
        loss.backward()
        optimizer.step()
        return loss.detach()

    def validation_step(self, model, batch, context):
        ctx = self._ctx(context)
        criterion = ctx.get("criterion")
        if criterion is None:
            criterion = get_criterion()
        model.eval()
        with torch.no_grad():
            out = model(batch["image"], batch["tabular"])
            loss = criterion(out, batch["target"])
        return loss.detach()
