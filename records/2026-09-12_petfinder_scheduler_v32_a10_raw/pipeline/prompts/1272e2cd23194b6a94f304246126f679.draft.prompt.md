class AttrCalibrator:
    """Learns the additive contribution of each binary attribute to score.

    For each attribute we fit mean(score | attr==1) - mean(score | attr==0).
    A global offset anchors the sum so predictions stay in scale.
    """

    def __init__(self):
        self.offset = float(TARGET_MEAN)
        self.coefs = np.zeros(12, dtype=np.float64)

    def fit(self, X, y):
        X = np.asarray(X, dtype=np.float64)
        y = np.asarray(y, dtype=np.float64)
        coefs = np.zeros(X.shape[1], dtype=np.float64)
        for j in range(X.shape[1]):
            mask = X[:, j] > 0.5
            if mask.sum() == 0 or (~mask).sum() == 0:
                coefs[j] = 0.0
            else:
                coefs[j] = y[mask].mean() - y[~mask].mean()
        self.coefs = coefs
        self.offset = float(TARGET_MEAN)
        return self

    def predict(self, X):
        X = np.asarray(X, dtype=np.float64)
        return self.offset + X @ self.coefs

class PetModel(nn.Module):
    """Hybrid: frozen CNN visual features + learned attribute->residual head.

    Final score = attr_score + sigmoid(residual_head) * 100
    The head predicts how far the true score deviates from the calibrated
    attribute baseline, letting the network focus on visual residuals.
    """

    def __init__(self, tab_dim=12, feat_dim=512, hidden=128, dropout=0.3):
        super().__init__()
        self.backbone = _load_resnet18_backbone()
        # Freeze the backbone (small optimizer state, fast, robust)
        for p in self.backbone.parameters():
            p.requires_grad_(False)

        self.feat_proj = nn.Sequential(
            nn.Linear(feat_dim, hidden),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
        )
        self.tab_proj = nn.Sequential(
            nn.Linear(tab_dim, hidden),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
        )
        self.head = nn.Sequential(
            nn.Linear(hidden * 2, hidden),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
            nn.Linear(hidden, 1),
        )

    def forward(self, image, tabular, attr_score):
        feats = self.backbone(image)  # [B, 512]
        v = self.feat_proj(feats)  # [B, H]
        t = self.tab_proj(tabular)  # [B, H]
        x = torch.cat([v, t], dim=1)  # [B, 2H]
        residual = torch.sigmoid(self.head(x)).squeeze(-1)  # [B] in (0,1)
        score = attr_score + residual * 100.0  # [B]
        return score

class NormalizedMSELoss(nn.Module):
    def __init__(self, mean=TARGET_MEAN, std=TARGET_STD):
        super().__init__()
        self.mean = float(mean)
        self.std = float(std)

    def forward(self, pred, target):
        # pred is in raw [0,100]; target here is the normalized training target.
        # Convert pred to the same normalized space before comparing.
        pred_norm = (pred - self.mean) / self.std
        return torch.nn.functional.mse_loss(pred_norm, target)

class AttrCalibrator:
    """Learns the additive contribution of each binary attribute to score.

    For each attribute we fit mean(score | attr==1) - mean(score | attr==0).
    A global offset anchors the sum so predictions stay in scale.
    """

    def __init__(self):
        self.offset = float(TARGET_MEAN)
        self.coefs = np.zeros(12, dtype=np.float64)

    def fit(self, X, y):
        X = np.asarray(X, dtype=np.float64)
        y = np.asarray(y, dtype=np.float64)
        coefs = np.zeros(X.shape[1], dtype=np.float64)
        for j in range(X.shape[1]):
            mask = X[:, j] > 0.5
            if mask.sum() == 0 or (~mask).sum() == 0:
                coefs[j] = 0.0
            else:
                coefs[j] = y[mask].mean() - y[~mask].mean()
        self.coefs = coefs
        self.offset = float(TARGET_MEAN)
        return self

    def predict(self, X):
        X = np.asarray(X, dtype=np.float64)
        return self.offset + X @ self.coefs

class PetModel(nn.Module):
    """Hybrid: frozen CNN visual features + learned attribute->residual head.

    Final score = attr_score + sigmoid(residual_head) * 100
    The head predicts how far the true score deviates from the calibrated
    attribute baseline, letting the network focus on visual residuals.
    """

    def __init__(self, tab_dim=12, feat_dim=512, hidden=128, dropout=0.3):
        super().__init__()
        self.backbone = _load_resnet18_backbone()
        # Freeze the backbone (small optimizer state, fast, robust)
        for p in self.backbone.parameters():
            p.requires_grad_(False)

        self.feat_proj = nn.Sequential(
            nn.Linear(feat_dim, hidden),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
        )
        self.tab_proj = nn.Sequential(
            nn.Linear(tab_dim, hidden),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
        )
        self.head = nn.Sequential(
            nn.Linear(hidden * 2, hidden),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
            nn.Linear(hidden, 1),
        )

    def forward(self, image, tabular, attr_score):
        feats = self.backbone(image)  # [B, 512]
        v = self.feat_proj(feats)  # [B, H]
        t = self.tab_proj(tabular)  # [B, H]
        x = torch.cat([v, t], dim=1)  # [B, 2H]
        residual = torch.sigmoid(self.head(x)).squeeze(-1)  # [B] in (0,1)
        score = attr_score + residual * 100.0  # [B]
        return score

class NormalizedMSELoss(nn.Module):
    def __init__(self, mean=TARGET_MEAN, std=TARGET_STD):
        super().__init__()
        self.mean = float(mean)
        self.std = float(std)

    def forward(self, pred, target):
        # pred is in raw [0,100]; target here is the normalized training target.
        # Convert pred to the same normalized space before comparing.
        pred_norm = (pred - self.mean) / self.std
        return torch.nn.functional.mse_loss(pred_norm, target)

class AttrCalibrator:
    def __init__(self):
        self.offset = float(TARGET_MEAN)
        self.coefs = np.zeros(12, dtype=np.float64)

    def fit(self, X, y):
        X = np.asarray(X, dtype=np.float64)
        y = np.asarray(y, dtype=np.float64)
        coefs = np.zeros(X.shape[1], dtype=np.float64)
        for j in range(X.shape[1]):
            mask = X[:, j] > 0.5
            if mask.sum() == 0 or (~mask).sum() == 0:
                coefs[j] = 0.0
            else:
                coefs[j] = y[mask].mean() - y[~mask].mean()
        self.coefs = coefs
        self.offset = float(TARGET_MEAN)
        return self

    def predict(self, X):
        X = np.asarray(X, dtype=np.float64)
        return self.offset + X @ self.coefs

class PetModel(nn.Module):
    def __init__(self, tab_dim=12, feat_dim=512, hidden=128, dropout=0.3):
        super().__init__()
        self.backbone = _load_resnet18_backbone()
        for p in self.backbone.parameters():
            p.requires_grad_(False)

        self.feat_proj = nn.Sequential(
            nn.Linear(feat_dim, hidden),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
        )
        self.tab_proj = nn.Sequential(
            nn.Linear(tab_dim, hidden),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
        )
        self.head = nn.Sequential(
            nn.Linear(hidden * 2, hidden),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
            nn.Linear(hidden, 1),
        )

    def forward(self, image, tabular, attr_score):
        feats = self.backbone(image)
        v = self.feat_proj(feats)
        t = self.tab_proj(tabular)
        x = torch.cat([v, t], dim=1)
        residual = torch.sigmoid(self.head(x)).squeeze(-1)
        score = attr_score + residual * 100.0
        return score

class NormalizedMSELoss(nn.Module):
    def __init__(self, mean=TARGET_MEAN, std=TARGET_STD):
        super().__init__()

def build_optimizer(model, lr_head=1e-3, lr_tab=5e-4, wd=1e-4):
    """Differential LR: higher LR on the small trainable head, lower on tabular."""
    head_params = list(model.feat_proj.parameters()) + list(model.head.parameters())
    tab_params = list(model.tab_proj.parameters())
    head_ids = {id(p) for p in head_params}
    groups = [
        {"params": head_params, "lr": lr_head},
        {"params": tab_params, "lr": lr_tab},
    ]
    return torch.optim.AdamW(groups, lr=lr_head, weight_decay=wd)


def make_scheduler(optimizer, warmup_epochs, total_epochs, T_mult=4, eta_min=1e-5):
    """Linear warmup then CosineAnnealingWarmRestarts."""

    def lr_lambda(epoch):
        if epoch < warmup_epochs:
            return (epoch + 1) / max(1, warmup_epochs)
        return 1.0  # cosine restarts handled by the scheduler below

    warmup = torch.optim.lr_scheduler.LambdaLR(optimizer, lr_lambda)
    cosine = torch.optim.lr_scheduler.CosineAnnealingWarmRestarts(
        optimizer, T_0=max(1, total_epochs // 4), T_mult=T_mult, eta_min=eta_min
    )
    return warmup, cosine


def train_one_epoch(model, loader, criterion, optimizer, device, grad_clip=5.0):
    model.train()
    running = 0.0
    n = 0
    for batch in loader:
        image = batch["image"].to(device, non_blocking=True)
        tab = batch["tabular"].to(device, non_blocking=True)
        attr = batch["attr_score"].to(device, non_blocking=True)
        target = batch["target"].to(device, non_blocking=True)

        optimizer.zero_grad(set_to_none=True)
        out = model(image, tab, attr)
        loss = criterion(out, target)

        if not torch.isfinite(loss):
            # NaN guard: skip this batch, keep going
            optimizer.zero_grad(set_to_none=True)
            continue

        loss.backward()
        if grad_clip:
            torch.nn.utils.clip_grad_norm_(
                [
                    p
                    for g in optimizer.param_groups
                    for p in g["params"]
                    if p.requires_grad
                ],
                grad_clip,
            )
        optimizer.step()
        running += loss.item() * len(image)
        n += len(image)
    return running / max(1, n)

    best_state = None
    patience_counter = 0
