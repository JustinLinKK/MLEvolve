def compute_rank_smoothed_targets(y, sigma=0.15):
    """Denoise noisy integer labels via rank smoothing (fit on TRAIN only).

    Maps each label to its percentile rank then applies a Gaussian kernel over
    the sorted ranks. This shrinks idiosyncratic outliers toward the local
    neighbourhood while preserving the global ordering, which lowers RMSE.
    Returns smoothed values on the original [0,100] scale.
    """
    y = np.asarray(y, dtype=np.float32)
    order = np.argsort(y, kind="mergesort")
    ranks = np.empty_like(order, dtype=np.float32)
    ranks[order] = np.arange(len(y), dtype=np.float32)
    n = len(y)
    pos = ranks / max(n - 1, 1)  # [0,1]
    k = int(max(1, round(sigma * n)))  # half-width of kernel
    idx = np.arange(-k, k + 1, dtype=np.float32)
    kernel = np.exp(-(idx**2) / (2.0 * (sigma * n) ** 2))
    kernel /= kernel.sum()
    smooth_pos = np.zeros_like(pos)
    for i in range(n):
        lo = max(0, i - k)
        hi = min(n, i + k + 1)
        seg = pos[lo:hi]
        w = kernel[k - (i - lo) : k - (i - lo) + (hi - lo)]
        smooth_pos[i] = float(np.dot(seg, w))
    ymin, ymax = float(y.min()), float(y.max())
    return (smooth_pos * (ymax - ymin) + ymin).astype(np.float32)

class GeometricAttentionReadout(nn.Module):
    """Soft spatial attention over the CNN feature map with a learned
    center-bias + size-gain prior. Encodes 'prominent, well-centred subjects
    score higher' directly in the geometry of the pooling."""

    def __init__(self, in_ch, feat_h, feat_w, hidden=128):
        super().__init__()
        self.feat_h, self.feat_w = feat_h, feat_w
        self.proj = nn.Conv2d(in_ch, hidden, kernel_size=1)
        self.score = nn.Conv2d(hidden, 1, kernel_size=1)
        # Learned priors over the HxW grid: bias favours centre, gain favours
        # larger (more central) regions -> captures subject prominence.
        self.center_bias = nn.Parameter(torch.zeros(feat_h, feat_w))
        self.size_gain = nn.Parameter(torch.zeros(feat_h, feat_w))
        self.pool = nn.AdaptiveAvgPool2d((feat_h, feat_w))

    def forward(self, x):
        h = self.pool(x)  # [B,C,H,W]
        s = self.score(self.proj(h)).squeeze(1)  # [B,H,W]
        s = s + self.center_bias[None]  # add learned geometry
        attn = F.softmax(s.view(s.size(0), -1), dim=1)  # [B,H*W]
        feat = h.flatten(2).transpose(1, 2)  # [B,H*W,C]
        pooled = (attn.unsqueeze(-1) * feat).sum(dim=1)  # [B,C]
        return pooled, attn


class PetFinderModel(nn.Module):
    """ResNet18 backbone (early blocks frozen) + geometric attention readout
    fused with tabular metadata through a gated MLP head."""

    def __init__(self, num_tabular=NUM_TABULAR, freeze_blocks=2, dropout=0.3):
        super().__init__()
        backbone = resnet18(weights=None)
        self.stem = nn.Sequential(*list(backbone.children())[:freeze_blocks])
        feat_ch = list(self.stem.parameters()) and self._probe_channels()
        self.attn_readout = GeometricAttentionReadout(
            feat_ch, feat_h=IMG_SIZE // 32, feat_w=IMG_SIZE // 32
        )

        vision_dim = feat_ch
        tab_dim = num_tabular
        fused_dim = vision_dim + tab_dim
        self.gate = nn.Sequential(
            nn.Linear(fused_dim, 128),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
            nn.Sigmoid(),
        )
        self.head = nn.Sequential(
            nn.Linear(fused_dim, 128),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
            nn.Linear(128, 64),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout * 0.5),
            nn.Linear(64, 1),
        )

    def _probe_channels(self):
        dummy = torch.randn(1, 3, IMG_SIZE, IMG_SIZE)
        with torch.no_grad():
            out = self.stem(dummy)
        return out.shape[1]

    def forward(self, images, tabular):
        feat_map = self.stem(images)  # [B,C,h,w]
        vision, _attn = self.attn_readout(feat_map)  # [B,C]
        fused = torch.cat([vision, tabular], dim=1)
        g = self.gate(fused)
        out = self.head(fused * g)
        return out.squeeze(-1)  # [B] normalized logits

def load_pretrained_backbone(model):
    """Load local ResNet18 weights into the stem if available; else keep init."""
    if not os.path.exists(PRETRAINED_WEIGHTS_PATH):
        return False
    try:
        state = torch.load(PRETRAINED_WEIGHTS_PATH, map_location="cpu")
        if isinstance(state, dict) and "state_dict" in state:
            state = state["state_dict"]
        # Map full-resnet18 keys onto our stem (first `len(stem)` children).
        stem_state = {}
        prefix_map = {f"{i}": i for i in range(len(list(model.stem.children())))}
        for k, v in state.items():
            top = k.split(".")[0]
            if top in prefix_map:
                stem_state[k] = v
        missing, unexpected = model.stem.load_state_dict(stem_state, strict=False)
        return len(missing) == 0
    except Exception:
        return False

def prepare_datasets():

def mixup_batch(batch, alpha=0.2):
    """Apply Mixup to images + tabular + targets (all in normalized space)."""
    lam = float(np.random.beta(alpha, alpha))
    b = batch["images"].size(0)
    idx = torch.randperm(b, device=batch["images"].device)
    mixed_img = lam * batch["images"] + (1 - lam) * batch["images"][idx]
    mixed_tab = lam * batch["tabular"] + (1 - lam) * batch["tabular"][idx]
    mixed_tgt = lam * batch["targets"] + (1 - lam) * batch["targets"][idx]
    return {"images": mixed_img, "tabular": mixed_tab, "targets": mixed_tgt}

    for epoch in range(EPOCHS):
        set_collate_mode("train")

        running = 0.0

        for batch in train_loader:
            batch = {

    final = np.clip(np.array(final, dtype=np.float64), 0.0, 100.0)

    sub.to_csv(os.path.join(SUBMISSION_DIR, "submission.csv"), index=False)

    print(f"Final Validation Score: {score}")

def compute_rank_smoothed_targets(y, sigma=0.15):
    """Denoise noisy integer labels via rank smoothing (fit on TRAIN only).

    Maps each label to its percentile rank then applies a Gaussian kernel over
    the sorted ranks. This shrinks idiosyncratic outliers toward the local
    neighbourhood while preserving the global ordering, which lowers RMSE.
    Returns smoothed values on the original [0,100] scale.
    """
    y = np.asarray(y, dtype=np.float32)
    order = np.argsort(y, kind="mergesort")
    ranks = np.empty_like(order, dtype=np.float32)
    ranks[order] = np.arange(len(y), dtype=np.float32)
    n = len(y)
    pos = ranks / max(n - 1, 1)  # [0,1]
    k = int(max(1, round(sigma * n)))  # half-width of kernel
    idx = np.arange(-k, k + 1, dtype=np.float32)
    kernel = np.exp(-(idx**2) / (2.0 * (sigma * n) ** 2))
    kernel /= kernel.sum()
    smooth_pos = np.zeros_like(pos)
    for i in range(n):
        lo = max(0, i - k)
        hi = min(n, i + k + 1)
        seg = pos[lo:hi]
        w = kernel[k - (i - lo) : k - (i - lo) + (hi - lo)]
        smooth_pos[i] = float(np.dot(seg, w))
    ymin, ymax = float(y.min()), float(y.max())
    return (smooth_pos * (ymax - ymin) + ymin).astype(np.float32)

class GeometricAttentionReadout(nn.Module):
    """Soft spatial attention over the CNN feature map with a learned
    center-bias + size-gain prior. Encodes 'prominent, well-centred subjects
    score higher' directly in the geometry of the pooling."""

    def __init__(self, in_ch, feat_h, feat_w, hidden=128):
        super().__init__()
        self.feat_h, self.feat_w = feat_h, feat_w
        self.proj = nn.Conv2d(in_ch, hidden, kernel_size=1)
        self.score = nn.Conv2d(hidden, 1, kernel_size=1)
        # Learned priors over the HxW grid: bias favours centre, gain favours
        # larger (more central) regions -> captures subject prominence.
        self.center_bias = nn.Parameter(torch.zeros(feat_h, feat_w))
        self.size_gain = nn.Parameter(torch.zeros(feat_h, feat_w))
        self.pool = nn.AdaptiveAvgPool2d((feat_h, feat_w))

    def forward(self, x):
        h = self.pool(x)  # [B,C,H,W]
        s = self.score(self.proj(h)).squeeze(1)  # [B,H,W]
        s = s + self.center_bias[None]  # add learned geometry
        attn = F.softmax(s.view(s.size(0), -1), dim=1)  # [B,H*W]
        feat = h.flatten(2).transpose(1, 2)  # [B,H*W,C]
        pooled = (attn.unsqueeze(-1) * feat).sum(dim=1)  # [B,C]
        return pooled, attn


class PetFinderModel(nn.Module):
    """ResNet18 backbone (early blocks frozen) + geometric attention readout
    fused with tabular metadata through a gated MLP head."""

    def __init__(self, num_tabular=NUM_TABULAR, freeze_blocks=2, dropout=0.3):
        super().__init__()
        backbone = resnet18(weights=None)
        self.stem = nn.Sequential(*list(backbone.children())[:freeze_blocks])
        feat_ch = list(self.stem.parameters()) and self._probe_channels()
        self.attn_readout = GeometricAttentionReadout(
            feat_ch, feat_h=IMG_SIZE // 32, feat_w=IMG_SIZE // 32
        )

        vision_dim = feat_ch
        tab_dim = num_tabular
        fused_dim = vision_dim + tab_dim
        self.gate = nn.Sequential(
            nn.Linear(fused_dim, 128),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
            nn.Sigmoid(),
        )
        self.head = nn.Sequential(
            nn.Linear(fused_dim, 128),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
            nn.Linear(128, 64),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout * 0.5),
            nn.Linear(64, 1),
        )

    def _probe_channels(self):
        dummy = torch.randn(1, 3, IMG_SIZE, IMG_SIZE)
        with torch.no_grad():
            out = self.stem(dummy)
        return out.shape[1]

    def forward(self, images, tabular):
        feat_map = self.stem(images)  # [B,C,h,w]
        vision, _attn = self.attn_readout(feat_map)  # [B,C]
        fused = torch.cat([vision, tabular], dim=1)
        g = self.gate(fused)
        out = self.head(fused * g)
        return out.squeeze(-1)  # [B] normalized logits

def load_pretrained_backbone(model):
    """Load local ResNet18 weights into the stem if available; else keep init."""
    if not os.path.exists(PRETRAINED_WEIGHTS_PATH):
        return False
    try:
        state = torch.load(PRETRAINED_WEIGHTS_PATH, map_location="cpu")
        if isinstance(state, dict) and "state_dict" in state:
            state = state["state_dict"]
        # Map full-resnet18 keys onto our stem (first `len(stem)` children).
        stem_state = {}
        prefix_map = {f"{i}": i for i in range(len(list(model.stem.children())))}
        for k, v in state.items():
            top = k.split(".")[0]
            if top in prefix_map:
                stem_state[k] = v
        missing, unexpected = model.stem.load_state_dict(stem_state, strict=False)
        return len(missing) == 0
    except Exception:
        return False

def compute_rank_smoothed_targets(y, sigma=0.15):
    y = np.asarray(y, dtype=np.float32)
    order = np.argsort(y, kind="mergesort")
    ranks = np.empty_like(order, dtype=np.float32)
    ranks[order] = np.arange(len(y), dtype=np.float32)
    n = len(y)
    pos = ranks / max(n - 1, 1)
    k = int(max(1, round(sigma * n)))
    idx = np.arange(-k, k + 1, dtype=np.float32)
    kernel = np.exp(-(idx**2) / (2.0 * (sigma * n) ** 2))
    kernel /= kernel.sum()
    smooth_pos = np.zeros_like(pos)
    for i in range(n):
        lo = max(0, i - k)
        hi = min(n, i + k + 1)
        seg = pos[lo:hi]
        w = kernel[k - (i - lo) : k - (i - lo) + (hi - lo)]
        smooth_pos[i] = float(np.dot(seg, w))
    ymin, ymax = float(y.min()), float(y.max())
    return (smooth_pos * (ymax - ymin) + ymin).astype(np.float32)

class GeometricAttentionReadout(nn.Module):
    def __init__(self, in_ch, feat_h, feat_w, hidden=128):
        super().__init__()
        self.feat_h, self.feat_w = feat_h, feat_w
        self.proj = nn.Conv2d(in_ch, hidden, kernel_size=1)
        self.score = nn.Conv2d(hidden, 1, kernel_size=1)
        self.center_bias = nn.Parameter(torch.zeros(feat_h, feat_w))
        self.size_gain = nn.Parameter(torch.zeros(feat_h, feat_w))
        self.pool = nn.AdaptiveAvgPool2d((feat_h, feat_w))

    def forward(self, x):
        h = self.pool(x)
        s = self.score(self.proj(h)).squeeze(1)
        s = s + self.center_bias[None]
        attn = F.softmax(s.view(s.size(0), -1), dim=1)
        feat = h.flatten(2).transpose(1, 2)
        pooled = (attn.unsqueeze(-1) * feat).sum(dim=1)
        return pooled, attn


class PetFinderModel(nn.Module):
    def __init__(self, num_tabular=NUM_TABULAR, freeze_blocks=2, dropout=0.3):
        super().__init__()
        backbone = resnet18(weights=None)
        self.stem = nn.Sequential(*list(backbone.children())[:freeze_blocks])
        feat_ch = self._probe_channels()
        self.attn_readout = GeometricAttentionReadout(
            feat_ch, feat_h=IMG_SIZE // 32, feat_w=IMG_SIZE // 32
        )
        vision_dim = feat_ch
        fused_dim = vision_dim + num_tabular
        self.gate = nn.Sequential(
            nn.Linear(fused_dim, 128),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
            nn.Sigmoid(),
        )
        self.head = nn.Sequential(
            nn.Linear(fused_dim, 128),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
            nn.Linear(128, 64),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout * 0.5),
            nn.Linear(64, 1),
        )

    def _probe_channels(self):
        dummy = torch.randn(1, 3, IMG_SIZE, IMG_SIZE)
        with torch.no_grad():
            out = self.stem(dummy)
        return out.shape[1]

    def forward(self, images, tabular):
        feat_map = self.stem(images)
        vision, _attn = self.attn_readout(feat_map)
        fused = torch.cat([vision, tabular], dim=1)
        g = self.gate(fused)
        out = self.head(fused * g)
        return out.squeeze(-1)

def build_param_groups(model, lr_head=1e-3, lr_backbone=1e-4, wd=1e-4):
    """Differential LR: freshly-initialised attention/gate/head get a higher LR
    than the (possibly pretrained) frozen-then-thawed backbone."""
    backbone_ids = set(id(p) for p in model.stem.parameters())
    head_params, back_params = [], []
    for name, p in model.named_parameters():
        if not p.requires_grad:
            continue
        (back_params if id(p) in backbone_ids else head_params).append(p)
    groups = [{"params": head_params, "lr": lr_head}]
    if back_params:
        groups.append({"params": back_params, "lr": lr_backbone})
    return groups


def build_scheduler(optimizer, total_steps, warmup_frac=0.1, eta_min_ratio=0.01):
    """Linear warmup followed by cosine-annealing-with-restarts over steps."""
    warmup_steps = max(1, int(total_steps * warmup_frac))

    def lr_lambda(step):
        if step < warmup_steps:
            return (step + 1) / warmup_steps
        progress = (step - warmup_steps) / max(1, total_steps - warmup_steps)
        cos_term = 0.5 * (1.0 + math.cos(math.pi * progress))
        return eta_min_ratio + (1.0 - eta_min_ratio) * cos_term

    return torch.optim.lr_scheduler.LambdaLR(optimizer, lr_lambda=lr_lambda)
