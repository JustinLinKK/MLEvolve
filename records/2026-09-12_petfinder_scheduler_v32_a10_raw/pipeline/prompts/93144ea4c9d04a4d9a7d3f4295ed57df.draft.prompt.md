# Model: EfficientNet-B0 + attention-gated tabular fusion

class TabularAttentionFusion(nn.Module):
    """Fuse a visual embedding with tabular features via cross-attention gating.

    The tabular vector produces per-channel gates that emphasize/suppress
    parts of the visual embedding, letting flags like Face/Eyes/Near steer
    which visual cues dominate the prediction.
    """

    def __init__(self, n_tab, vis_dim, hidden=128, dropout=0.2):
        super().__init__()
        self.vis_proj = nn.Sequential(
            nn.Linear(vis_dim, hidden), nn.ReLU(inplace=True), nn.Dropout(dropout)
        )
        self.tab_proj = nn.Sequential(
            nn.Linear(n_tab, hidden), nn.ReLU(inplace=True), nn.Dropout(dropout)
        )
        # Gating: tabular -> per-channel gate in [0,1]
        self.gate = nn.Sequential(nn.Linear(hidden, hidden), nn.Sigmoid())
        self.head = nn.Sequential(
            nn.Linear(hidden * 2, hidden),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
            nn.Linear(hidden, 1),
        )

    def forward(self, vis_feat, tab_feat):
        v = self.vis_proj(vis_feat)  # [B, H]
        t = self.tab_proj(tab_feat)  # [B, H]
        g = self.gate(t)  # [B, H]
        fused = v * g  # gated visual
        x = torch.cat([fused, t], dim=1)
        return self.head(x).squeeze(-1)  # [B]


class PawpularityModel(nn.Module):
    def __init__(self, n_tab, backbone_name="efficientnet_b0", pretrained=False):
        super().__init__()
        from torchvision.models import efficientnet_b0, EfficientNet_B0_Weights

        if pretrained:
            weights = EfficientNet_B0_Weights.DEFAULT
        else:
            weights = None
        self.backbone = efficientnet_b0(weights=weights)
        vis_dim = self.backbone.classifier[1].in_features
        self.backbone.classifier = nn.Identity()
        self.fusion = TabularAttentionFusion(n_tab=n_tab, vis_dim=vis_dim)

    def forward(self, image, tabular):
        vis_feat = self.backbone(image)
        return self.fusion(vis_feat, tabular)

        ctx = dict(self.DEFAULT_CTX)
        if context:
            ctx.update({k: v for k, v in context.items() if v is not None})
        return ctx

    def build_model(self, context=None):
        ctx = self._ctx(context)
        model = build_model(pretrained=bool(ctx["pretrained"]), n_tab=int(ctx["n_tab"]))
        self._model = model
        self._criterion = make_criterion()
        return model

    def build_optimizer(self, model, context=None):
        ctx = self._ctx(context)
        return torch.optim.AdamW(
            model.parameters(), lr=float(ctx["lr"]), weight_decay=1e-4
        )

    def _make_batch(self, scenario, device, with_target=True):
        bs = int(scenario.get("batch_size", 2))
        n_tab = int(self._ctx(None)["n_tab"])
        image = torch.randn(bs, 3, IMG_SIZE, IMG_SIZE, device=device)
        tabular = torch.randint(0, 2, (bs, n_tab), device=device).float()
        batch = {"image": image, "tabular": tabular}
        if with_target:
            batch["target"] = torch.rand(bs, device=device)
        return batch

    def build_train_batch(self, scenario, device):
        return self._make_batch(scenario, device, with_target=True)

    def build_validation_batch(self, scenario, device):
        return self._make_batch(scenario, device, with_target=True)

    def training_step(self, model, batch, context=None):
        crit = self._criterion if self._criterion is not None else make_criterion()
        if context and context.get("criterion") is not None:
            crit = context["criterion"]
        out = model(batch["image"], batch["tabular"])
        loss = crit(out, batch["target"])
        return loss

        crit = self._criterion if self._criterion is not None else make_criterion()
        if context and context.get("criterion") is not None:
            crit = context["criterion"]
        out = model(batch["image"], batch["tabular"])

class TabularAttentionFusion(nn.Module):
    """Fuse a visual embedding with tabular features via cross-attention gating.

    The tabular vector produces per-channel gates that emphasize/suppress
    parts of the visual embedding, letting flags like Face/Eyes/Near steer
    which visual cues dominate the prediction.
    """

    def __init__(self, n_tab, vis_dim, hidden=128, dropout=0.2):
        super().__init__()
        self.vis_proj = nn.Sequential(
            nn.Linear(vis_dim, hidden), nn.ReLU(inplace=True), nn.Dropout(dropout)
        )
        self.tab_proj = nn.Sequential(
            nn.Linear(n_tab, hidden), nn.ReLU(inplace=True), nn.Dropout(dropout)
        )
        # Gating: tabular -> per-channel gate in [0,1]
        self.gate = nn.Sequential(nn.Linear(hidden, hidden), nn.Sigmoid())
        self.head = nn.Sequential(
            nn.Linear(hidden * 2, hidden),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
            nn.Linear(hidden, 1),
        )

    def forward(self, vis_feat, tab_feat):
        v = self.vis_proj(vis_feat)  # [B, H]
        t = self.tab_proj(tab_feat)  # [B, H]
        g = self.gate(t)  # [B, H]
        fused = v * g  # gated visual
        x = torch.cat([fused, t], dim=1)
        return self.head(x).squeeze(-1)  # [B]


class PawpularityModel(nn.Module):
    def __init__(self, n_tab, backbone_name="efficientnet_b0", pretrained=False):
        super().__init__()
        from torchvision.models import efficientnet_b0, EfficientNet_B0_Weights

        if pretrained:
            weights = EfficientNet_B0_Weights.DEFAULT
        else:
            weights = None
        self.backbone = efficientnet_b0(weights=weights)
        vis_dim = self.backbone.classifier[1].in_features
        self.backbone.classifier = nn.Identity()
        self.fusion = TabularAttentionFusion(n_tab=n_tab, vis_dim=vis_dim)

    def forward(self, image, tabular):
        vis_feat = self.backbone(image)
        return self.fusion(vis_feat, tabular)

        "lr": 1e-3,
    }

        self._model = None
        self._criterion = None

    def _ctx(self, context):
        ctx = dict(self.DEFAULT_CTX)
        if context:
            ctx.update({k: v for k, v in context.items() if v is not None})
        return ctx

    def build_model(self, context=None):
        ctx = self._ctx(context)
        model = build_model(pretrained=bool(ctx["pretrained"]), n_tab=int(ctx["n_tab"]))
        self._model = model
        self._criterion = make_criterion()
        return model

    def build_optimizer(self, model, context=None):
        ctx = self._ctx(context)
        return torch.optim.AdamW(
            model.parameters(), lr=float(ctx["lr"]), weight_decay=1e-4
        )

    def _make_batch(self, scenario, device, with_target=True):
        bs = int(scenario.get("batch_size", 2))
        n_tab = int(self._ctx(None)["n_tab"])
        image = torch.randn(bs, 3, IMG_SIZE, IMG_SIZE, device=device)
        tabular = torch.randint(0, 2, (bs, n_tab), device=device).float()
        batch = {"image": image, "tabular": tabular}
        if with_target:
            batch["target"] = torch.rand(bs, device=device)
        return batch

    def build_train_batch(self, scenario, device):
        return self._make_batch(scenario, device, with_target=True)

    def build_validation_batch(self, scenario, device):
        return self._make_batch(scenario, device, with_target=True)

    def training_step(self, model, batch, context=None):
        crit = self._criterion if self._criterion is not None else make_criterion()
        if context and context.get("criterion") is not None:
            crit = context["criterion"]
        out = model(batch["image"], batch["tabular"])
        loss = crit(out, batch["target"])
        return loss

    def validation_step(self, model, batch, context=None):
        crit = self._criterion if self._criterion is not None else make_criterion()
        if context and context.get("criterion") is not None:
            crit = context["criterion"]
        out = model(batch["image"], batch["tabular"])
        loss = crit(out, batch["target"])
        return loss

# Optimizer builder with differential learning rates
# --------------------------------------------------------------------------- #
def build_optimizer(
    model, backbone_lr=BACKBONE_LR, fusion_lr=FUSION_LR, weight_decay=WEIGHT_DECAY
):
    backbone_params = list(model.backbone.parameters())
    fusion_params = list(model.fusion.parameters())
    param_groups = [
        {"params": backbone_params, "lr": backbone_lr},
        {"params": fusion_params, "lr": fusion_lr},
    ]
    return torch.optim.AdamW(param_groups, lr=fusion_lr, weight_decay=weight_decay)


# --------------------------------------------------------------------------- #
# Warmup + CosineAnnealingWarmRestarts scheduler
# --------------------------------------------------------------------------- #
class WarmupCosineScheduler:
    """Linear warmup followed by CosineAnnealingWarmRestarts."""

    def __init__(self, optimizer, warmup_epochs, t_max, t_mult=1, last_epoch=-1):
        self.optimizer = optimizer
        self.warmup_epochs = warmup_epochs
        self.t_max = t_max
        self.t_mult = t_mult
        self.last_epoch = last_epoch
        self.base_lrs = [pg["lr"] for pg in optimizer.param_groups]
        self.step_count = 0

    def step(self):
        self.step_count += 1
        epoch = self.step_count
        if epoch <= self.warmup_epochs:
            frac = epoch / max(1, self.warmup_epochs)
            for pg, base_lr in zip(self.optimizer.param_groups, self.base_lrs):
                pg["lr"] = base_lr * frac
        else:
            e = epoch - self.warmup_epochs
            # Cosine annealing with warm restarts
            T_cur = e % self.t_max
            T_0 = self.t_max
            cosine_factor = 0.5 * (1 + math.cos(math.pi * T_cur / T_0))
            for pg, base_lr in zip(self.optimizer.param_groups, self.base_lrs):
                pg["lr"] = base_lr * cosine_factor

    def get_last_lr(self):
        return [pg["lr"] for pg in self.optimizer.param_groups]

# Training loop
# --------------------------------------------------------------------------- #
def train_one_epoch(model, loader, criterion, optimizer, device, grad_accum_steps=1):
    model.train()
    total_loss = 0.0
    n_batches = 0
    optimizer.zero_grad()

    for batch_idx, batch in enumerate(loader):
        images = batch["image"].to(device, non_blocking=True)
        tabular = batch["tabular"].to(device, non_blocking=True)
        targets = batch["target"].to(device, non_blocking=True)

        outputs = model(images, tabular)
        loss = criterion(outputs, targets) / grad_accum_steps
        loss.backward()

        if (batch_idx + 1) % grad_accum_steps == 0:
            torch.nn.utils.clip_grad_norm_(model.parameters(), max_norm=1.0)
            optimizer.step()
            optimizer.zero_grad()

        total_loss += loss.item() * grad_accum_steps
        n_batches += 1

    # Handle leftover gradients if dataset size not divisible by accum steps
    if n_batches % grad_accum_steps != 0:
        torch.nn.utils.clip_grad_norm_(model.parameters(), max_norm=1.0)
        optimizer.step()
        optimizer.zero_grad()

    return total_loss / max(1, n_batches)


@torch.no_grad()
def evaluate(model, loader, criterion, device):
    model.eval()
    total_loss = 0.0
    n_batches = 0
    all_preds = []
    all_targets = []

    for batch in loader:
        images = batch["image"].to(device, non_blocking=True)
        tabular = batch["tabular"].to(device, non_blocking=True)
        targets = batch["target"].to(device, non_blocking=True)

        outputs = model(images, tabular)
        loss = criterion(outputs, targets)

        total_loss += loss.item()
        n_batches += 1
        all_preds.append(outputs.cpu().numpy())
        all_targets.append(targets.cpu().numpy())

    avg_loss = total_loss / max(1, n_batches)
    preds = np.concatenate(all_preds, axis=0)
    tgts = np.concatenate(all_targets, axis=0)
    return avg_loss, preds, tgts

        if context:
            ctx.update({k: v for k, v in context.items() if v is not None})
        return ctx

        model = build_model(pretrained=bool(ctx["pretrained"]), n_tab=int(ctx["n_tab"]))

        self._criterion = make_criterion()
        return model

        ctx = self._ctx(context)
        return torch.optim.AdamW(
            model.parameters(), lr=float(ctx["lr"]), weight_decay=1e-4
        )

    def _make_batch(self, scenario, device, with_target=True):
        bs = int(scenario.get("batch_size", 2))
        n_tab = int(self._ctx(None)["n_tab"])
        image = torch.randn(bs, 3, IMG_SIZE, IMG_SIZE, device=device)
        tabular = torch.randint(0, 2, (bs, n_tab), device=device).float()
        batch = {"image": image, "tabular": tabular}
        if with_target:
            batch["target"] = torch.rand(bs, device=device)
        return batch

        return self._make_batch(scenario, device, with_target=True)

    def build_validation_batch(self, scenario, device):
        return self._make_batch(scenario, device, with_target=True)

    def training_step(self, model, batch, context=None):
        crit = self._criterion if self._criterion is not None else make_criterion()
        if context and context.get("criterion") is not None:
            crit = context["criterion"]
        out = model(batch["image"], batch["tabular"])
        loss = crit(out, batch["target"])
        return loss

    def validation_step(self, model, batch, context=None):
        crit = self._criterion if self._criterion is not None else make_criterion()
        if context and context.get("criterion") is not None:
            crit = context["criterion"]
        out = model(batch["image"], batch["tabular"])
        loss = crit(out, batch["target"])
