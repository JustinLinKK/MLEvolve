def compute_rank_targets(y):
    """Rank-based smoothed regression target (monotonic, robust to outliers).

    Maps each sample to its percentile rank in [0, 100], then blends with the
    raw value to keep the scale aligned with the evaluation metric.
    """
    y = np.asarray(y, dtype=np.float32)
    order = np.argsort(np.argsort(y))  # ranks 0..n-1
    n = max(len(y) - 1, 1)
    ranks = (order + 1.0) / (n + 1.0) * 100.0
    # Blend: keep most of the raw signal, add rank smoothing.
    return 0.5 * y + 0.5 * ranks

class CrossAttentionFusion(nn.Module):
    """Cross-attention fusion of tabular queries over image patch tokens."""

    def __init__(self, d_model, num_heads=4, dropout=0.1):
        super().__init__()
        self.attn = nn.MultiheadAttention(
            d_model, num_heads, dropout=dropout, batch_first=True
        )
        self.norm = nn.LayerNorm(d_model)
        self.dropout = nn.Dropout(dropout)

    def forward(self, query, kv):
        # query: [B, Lq, D]; kv: [B, Lk, D]
        out, _ = self.attn(query, kv, kv, need_weights=False)
        out = self.norm(out + query)
        return self.dropout(out)


class PawpularityModel(nn.Module):
    """Frozen EfficientNet-B0 image encoder + cross-attention tabular fusion.

    Forward inputs:
      images   : [B, 3, H, W]
      tabular  : [B, T]
    Output:
      pred     : [B]          (raw Pawpularity estimate, clamped later)
      log_var  : [B]          (learned per-sample log-variance for NLL loss)
    """

    def __init__(
        self, num_tabular=12, d_model=512, num_heads=4, dropout=0.2, pretrained=True
    ):
        super().__init__()
        try:
            import timm

            weights = "IMAGENET1K_V1" if pretrained else None
            self.backbone = timm.create_model(
                "efficientnet_b0",
                pretrained=(pretrained and weights is not None),
                num_classes=0,
                global_pool="avg",
            )
            feat_dim = self.backbone.num_features
        except Exception:
            # Offline fallback: torchvision ResNet18 (no download needed).
            from torchvision.models import resnet18

            base = resnet18(weights=None)
            self.backbone = nn.Sequential(*list(base.children())[:-1])
            feat_dim = 512

        # Freeze the backbone for speed + stability.
        for p in self.backbone.parameters():
            p.requires_grad_(False)

        self.tab_embed = nn.Sequential(
            nn.Linear(num_tabular, d_model // 2),
            nn.GELU(),
            nn.Linear(d_model // 2, d_model),
        )
        self.image_proj = nn.Linear(feat_dim, d_model)
        self.fusion = CrossAttentionFusion(
            d_model, num_heads=num_heads, dropout=dropout
        )

        self.head = nn.Sequential(
            nn.LayerNorm(d_model),
            nn.Linear(d_model, d_model // 2),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(d_model // 2, 2),  # [mean, log_var]
        )

    def forward(self, images, tabular):
        with torch.no_grad():
            img_feat = self.backbone(images)  # [B, feat_dim]
        img_tok = self.image_proj(img_feat).unsqueeze(1)  # [B, 1, D]
        tab_tok = self.tab_embed(tabular).unsqueeze(1)  # [B, 1, D]

        # Tabular attends to image tokens (query=tab, kv=image).
        fused = self.fusion(tab_tok, img_tok)  # [B, 1, D]
        fused = fused.squeeze(1)  # [B, D]

        out = self.head(fused)  # [B, 2]
        mean = out[:, 0]
        log_var = out[:, 1].clamp(-4.0, 4.0)
        return mean, log_var


def build_model(pretrained=True, num_tabular=12, device="cpu"):
    model = PawpularityModel(num_tabular=num_tabular, pretrained=pretrained)
    return model.to(device)

# Loss: hybrid Gaussian NLL + MSE
# ----------------------------------------------------------------------------
class HybridGaussianLoss(nn.Module):
    """Gaussian negative log-likelihood blended with plain MSE.

    The NLL term learns a per-sample variance that down-weights high-noise
    (mid-range) samples, while the MSE term anchors the scale to the RMSE
    evaluation metric.
    """

    def __init__(self, nll_weight=0.5, min_log_var=-4.0, max_log_var=4.0):
        super().__init__()
        self.nll_weight = nll_weight
        self.min_log_var = min_log_var
        self.max_log_var = max_log_var

    def forward(self, mean, log_var, target):
        log_var = log_var.clamp(self.min_log_var, self.max_log_var)
        var = torch.exp(log_var)
        # Gaussian NLL (reduction mean).
        nll = 0.5 * (log_var + ((target - mean) ** 2) / var)
        nll = nll.mean()
        mse = F.mse_loss(mean, target)
        return self.nll_weight * nll + (1.0 - self.nll_weight) * mse


def build_criterion(nll_weight=0.5):
    return HybridGaussianLoss(nll_weight=nll_weight)

    """Root mean squared error (the competition metric)."""
    pred = np.asarray(pred, dtype=np.float64)
    target = np.asarray(target, dtype=np.float64)
    return float(np.sqrt(np.mean((pred - target) ** 2)))

def compute_rank_targets(y):
    """Rank-based smoothed regression target (monotonic, robust to outliers).

    Maps each sample to its percentile rank in [0, 100], then blends with the
    raw value to keep the scale aligned with the evaluation metric.
    """
    y = np.asarray(y, dtype=np.float32)
    order = np.argsort(np.argsort(y))  # ranks 0..n-1
    n = max(len(y) - 1, 1)
    ranks = (order + 1.0) / (n + 1.0) * 100.0
    # Blend: keep most of the raw signal, add rank smoothing.
    return 0.5 * y + 0.5 * ranks

class CrossAttentionFusion(nn.Module):
    """Cross-attention fusion of tabular queries over image patch tokens."""

    def __init__(self, d_model, num_heads=4, dropout=0.1):
        super().__init__()
        self.attn = nn.MultiheadAttention(
            d_model, num_heads, dropout=dropout, batch_first=True
        )
        self.norm = nn.LayerNorm(d_model)
        self.dropout = nn.Dropout(dropout)

    def forward(self, query, kv):
        # query: [B, Lq, D]; kv: [B, Lk, D]
        out, _ = self.attn(query, kv, kv, need_weights=False)
        out = self.norm(out + query)
        return self.dropout(out)


class PawpularityModel(nn.Module):
    """Frozen EfficientNet-B0 image encoder + cross-attention tabular fusion.

    Forward inputs:
      images   : [B, 3, H, W]
      tabular  : [B, T]
    Output:
      pred     : [B]          (raw Pawpularity estimate, clamped later)
      log_var  : [B]          (learned per-sample log-variance for NLL loss)
    """

    def __init__(
        self, num_tabular=12, d_model=512, num_heads=4, dropout=0.2, pretrained=True
    ):
        super().__init__()
        try:
            import timm

            weights = "IMAGENET1K_V1" if pretrained else None
            self.backbone = timm.create_model(
                "efficientnet_b0",
                pretrained=(pretrained and weights is not None),
                num_classes=0,
                global_pool="avg",
            )
            feat_dim = self.backbone.num_features
        except Exception:
            # Offline fallback: torchvision ResNet18 (no download needed).
            from torchvision.models import resnet18

            base = resnet18(weights=None)
            self.backbone = nn.Sequential(*list(base.children())[:-1])
            feat_dim = 512

        # Freeze the backbone for speed + stability.
        for p in self.backbone.parameters():
            p.requires_grad_(False)

        self.tab_embed = nn.Sequential(
            nn.Linear(num_tabular, d_model // 2),
            nn.GELU(),
            nn.Linear(d_model // 2, d_model),
        )
        self.image_proj = nn.Linear(feat_dim, d_model)
        self.fusion = CrossAttentionFusion(
            d_model, num_heads=num_heads, dropout=dropout
        )

        self.head = nn.Sequential(
            nn.LayerNorm(d_model),
            nn.Linear(d_model, d_model // 2),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(d_model // 2, 2),  # [mean, log_var]
        )

    def forward(self, images, tabular):
        with torch.no_grad():
            img_feat = self.backbone(images)  # [B, feat_dim]
        img_tok = self.image_proj(img_feat).unsqueeze(1)  # [B, 1, D]
        tab_tok = self.tab_embed(tabular).unsqueeze(1)  # [B, 1, D]

        # Tabular attends to image tokens (query=tab, kv=image).
        fused = self.fusion(tab_tok, img_tok)  # [B, 1, D]
        fused = fused.squeeze(1)  # [B, D]

        out = self.head(fused)  # [B, 2]
        mean = out[:, 0]
        log_var = out[:, 1].clamp(-4.0, 4.0)
        return mean, log_var


def build_model(pretrained=True, num_tabular=12, device="cpu"):
    model = PawpularityModel(num_tabular=num_tabular, pretrained=pretrained)
    return model.to(device)

# Loss: hybrid Gaussian NLL + MSE
# ----------------------------------------------------------------------------
class HybridGaussianLoss(nn.Module):
    """Gaussian negative log-likelihood blended with plain MSE.

    The NLL term learns a per-sample variance that down-weights high-noise
    (mid-range) samples, while the MSE term anchors the scale to the RMSE
    evaluation metric.
    """

    def __init__(self, nll_weight=0.5, min_log_var=-4.0, max_log_var=4.0):
        super().__init__()
        self.nll_weight = nll_weight
        self.min_log_var = min_log_var
        self.max_log_var = max_log_var

    def forward(self, mean, log_var, target):
        log_var = log_var.clamp(self.min_log_var, self.max_log_var)
        var = torch.exp(log_var)
        # Gaussian NLL (reduction mean).
        nll = 0.5 * (log_var + ((target - mean) ** 2) / var)
        nll = nll.mean()
        mse = F.mse_loss(mean, target)
        return self.nll_weight * nll + (1.0 - self.nll_weight) * mse


def build_criterion(nll_weight=0.5):
    return HybridGaussianLoss(nll_weight=nll_weight)

    """Rank-based smoothed regression target (monotonic, robust to outliers)."""
    y = np.asarray(y, dtype=np.float32)

    return 0.5 * y + 0.5 * ranks

class CrossAttentionFusion(nn.Module):
    def __init__(self, d_model, num_heads=4, dropout=0.1):
        super().__init__()
        self.attn = nn.MultiheadAttention(
            d_model, num_heads, dropout=dropout, batch_first=True
        )
        self.norm = nn.LayerNorm(d_model)
        self.dropout = nn.Dropout(dropout)

    def forward(self, query, kv):
        out, _ = self.attn(query, kv, kv, need_weights=False)
        out = self.norm(out + query)
        return self.dropout(out)


class PawpularityModel(nn.Module):
    """Frozen EfficientNet-B0 image encoder + cross-attention tabular fusion."""

    def __init__(
        self, num_tabular=12, d_model=512, num_heads=4, dropout=0.2, pretrained=True
    ):
        super().__init__()
        try:
            import timm

            weights = "IMAGENET1K_V1" if pretrained else None
            self.backbone = timm.create_model(
                "efficientnet_b0",
                pretrained=(pretrained and weights is not None),
                num_classes=0,
                global_pool="avg",
            )
            feat_dim = self.backbone.num_features
        except Exception:
            from torchvision.models import resnet18

            base = resnet18(weights=None)
            self.backbone = nn.Sequential(*list(base.children())[:-1])
            feat_dim = 512

        for p in self.backbone.parameters():
            p.requires_grad_(False)

        self.tab_embed = nn.Sequential(
            nn.Linear(num_tabular, d_model // 2),
            nn.GELU(),
            nn.Linear(d_model // 2, d_model),
        )
        self.image_proj = nn.Linear(feat_dim, d_model)
        self.fusion = CrossAttentionFusion(
            d_model, num_heads=num_heads, dropout=dropout
        )
        self.head = nn.Sequential(
            nn.LayerNorm(d_model),
            nn.Linear(d_model, d_model // 2),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(d_model // 2, 2),
        )

    def forward(self, images, tabular):
        with torch.no_grad():
            img_feat = self.backbone(images)
        img_tok = self.image_proj(img_feat).unsqueeze(1)
        tab_tok = self.tab_embed(tabular).unsqueeze(1)
        fused = self.fusion(tab_tok, img_tok)
        fused = fused.squeeze(1)
        out = self.head(fused)
        mean = out[:, 0]
        log_var = out[:, 1].clamp(-4.0, 4.0)
        return mean, log_var

        if name.startswith("head.") or name.startswith("fusion."):

            proj_params.append(p)
    groups = [{"params": head_params, "lr": head_lr}]
    if proj_params:
        groups.append({"params": proj_params, "lr": proj_lr})
    return torch.optim.AdamW(groups, lr=head_lr, weight_decay=weight_decay)


def build_scheduler(optimizer, total_epochs, warmup_epochs=WARMUP_EPOCHS):
    """Linear warmup followed by cosine annealing with warm restarts."""
    warmup_steps = max(warmup_epochs, 1)
    base_eta_min = HEAD_LR * COSINE_ETA_MIN_RATIO

    def lr_lambda(epoch):
        if epoch < warmup_steps:
            return (epoch + 1) / float(warmup_steps)
        # Cosine restarts measured from the end of warmup.
        e = epoch - warmup_steps
        progress = (e % COSINE_T_MAX) / float(COSINE_T_MAX)
        cos_term = 0.5 * (1.0 + math.cos(math.pi * progress))
        eta_ratio = COSINE_ETA_MIN_RATIO + (1.0 - COSINE_ETA_MIN_RATIO) * cos_term
        return eta_ratio

    return torch.optim.lr_scheduler.LambdaLR(optimizer, lr_lambda=lr_lambda)

def _is_oom(err):
    msg = str(err).lower()
    return "out of memory" in msg or "cuda oom" in msg


def run_training(
    model,
    criterion,
    train_ds,
    val_ds,
    test_ds,
    device,
    val_raw_targets,
    test_ids,
    num_tabular,
):
    start_time = time.time()
    best_val_rmse = float("inf")
    best_state = None
    patience_counter = 0

    batch_size = INITIAL_BATCH_SIZE
    num_workers = 4 if device.type == "cuda" else 2

    def make_loader(ds, bs, shuffle):
        return torch.utils.data.DataLoader(
            ds,
            batch_size=bs,
            shuffle=shuffle,
            num_workers=num_workers,
            pin_memory=(device.type == "cuda"),
            drop_last=False,
            collate_fn=collate_fn,
        )

    # Rebuild loaders when batch size changes (elastic).
    train_loader = make_loader(train_ds, batch_size, shuffle=True)
    val_loader = make_loader(val_ds, batch_size, shuffle=False)

    for epoch in range(1, MAX_EPOCHS + 1):
        # ---------------- Train ----------------
        model.train()
        running_loss = 0.0
        n_batches = 0
        oom_this_epoch = False
        for batch in train_loader:
            images = batch["images"].to(device, non_blocking=True)
            tabular = batch["tabular"].to(device, non_blocking=True)
            target = batch["target"].to(device, non_blocking=True)
            optimizer.zero_grad(set_to_none=True)
            try:
                mean, log_var = model(images, tabular)
                loss = criterion(mean, log_var, target)
                loss.backward()
                torch.nn.utils.clip_grad_norm_(
                    [p for p in model.parameters() if p.requires_grad], max_norm=1.0
                )
                optimizer.step()
            except RuntimeError as err:
                if _is_oom(err):
                    oom_this_epoch = True
                    if device.type == "cuda":
                        torch.cuda.empty_cache()
                    break
                raise
            running_loss += float(loss.item())
            n_batches += 1
        avg_loss = running_loss / max(n_batches, 1)

        # ---------------- Validate (RAW targets -> true RMSE) ----------------
        val_preds = predict_dataset(
            model,
            val_ds,
            device,
            batch_size=max(batch_size, 16),
            num_workers=num_workers,
        )
        val_rmse = compute_rmse(val_preds, val_raw_targets)
        elapsed = time.time() - start_time

        print(
            f"Epoch {epoch:02d}/{MAX_EPOCHS} | train_loss={avg_loss:.4f} | "
            f"val_rmse={val_rmse:.4f} | bs={batch_size} | elapsed={elapsed:.0f}s"
        )

        # ---------------- Best checkpoint ----------------
        if val_rmse < best_val_rmse:
            best_val_rmse = val_rmse
            best_state = {
                k: v.detach().cpu().clone() for k, v in model.state_dict().items()
            }
            patience_counter = 0
        else:
            patience_counter += 1

        # ---------------- Early stopping ----------------
        if patience_counter >= PATIENCE:
            print(f"Early stopping at epoch {epoch} (no improvement for {PATIENCE}).")
            break

        # ---------------- Elastic batch: halve on OOM ----------------
        if oom_this_epoch:
            new_bs = max(MIN_BATCH_SIZE, batch_size // 2)
            if new_bs != batch_size:
                print(f"OOM detected -> reducing batch size {batch_size} -> {new_bs}")
                batch_size = new_bs
                train_loader = make_loader(train_ds, batch_size, shuffle=True)
                val_loader = make_loader(val_ds, batch_size, shuffle=False)
            continue

        # ---------------- Scheduler step (per epoch) ----------------
        scheduler.step()

    # Restore best weights.
    if best_state is not None:
        model.load_state_dict(best_state)

    final_val_rmse = best_val_rmse

    # ---------------- Test inference (same path as validation) ----------------
    test_preds = predict_dataset(
        model, test_ds, device, batch_size=max(batch_size, 16), num_workers=num_workers
    )
    test_preds = np.clip(test_preds, 0.0, 100.0)

    # ---------------- Write submission ----------------
    os.makedirs(SUBMISSION_DIR, exist_ok=True)
    sub_df = pd.DataFrame(
        {
            "Id": test_ids,
            "Pawpularity": test_preds.astype(float),
        }
    )
    sub_path = os.path.join(SUBMISSION_DIR, "submission.csv")
    sub_df.to_csv(sub_path, index=False)
    print(f"Wrote {len(sub_df)} predictions to {sub_path}")

    return final_val_rmse
