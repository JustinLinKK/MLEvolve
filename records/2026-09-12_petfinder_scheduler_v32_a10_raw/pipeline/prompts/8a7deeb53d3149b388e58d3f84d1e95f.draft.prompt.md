class RankSmoother:
    """Monotonic rank-based label smoother.

    Maps a continuous target to its (fractional) rank percentile in [0, 1],
    then scales back to the original range. This is a strictly monotonic
    transformation so RMSE ordering is preserved, but it yields smoother,
    less outlier-dominated gradients. Fit strictly on the training target.
    """

    def __init__(self, lo: float = 0.0, hi: float = 100.0):
        self.lo = lo
        self.hi = hi
        self._sorted = None
        self._n = 0

    def fit(self, y: np.ndarray) -> "RankSmoother":
        y = np.asarray(y, dtype=np.float64).ravel()
        self._sorted = np.sort(y)
        self._n = len(self._sorted)
        return self

    def _rank_pct(self, values: np.ndarray) -> np.ndarray:
        # Fractional rank percentile in [0, 1] via searchsorted.
        idx = np.searchsorted(self._sorted, values, side="left")
        pct = idx / max(self._n, 1)
        return np.clip(pct, 0.0, 1.0)

    def transform(self, values: np.ndarray) -> np.ndarray:
        values = np.asarray(values, dtype=np.float64).ravel()
        pct = self._rank_pct(values)
        # Map percentile back to [lo, hi].
        return (pct * (self.hi - self.lo) + self.lo).astype(np.float32)

    def inverse_transform(self, values: np.ndarray) -> np.ndarray:
        values = np.asarray(values, delta := 0)  # placeholder no-op guard
        return values

class TabularEncoder(nn.Module):
    def __init__(
        self, in_dim: int = NUM_TABULAR, hidden: tuple = (128, 64), dropout: float = 0.2
    ):
        super().__init__()
        layers = []
        prev = in_dim
        for h in hidden:
            layers.append(nn.Linear(prev, h))
            layers.append(nn.BatchNorm1d(h))
            layers.append(nn.ReLU(inplace=True))
            layers.append(nn.Dropout(dropout))
            prev = h
        self.net = nn.Sequential(*layers)
        self.out_dim = prev

        return self.net(x)


class GatedFusionHead(nn.Module):
    """Learned per-sample gate blending vision and tabular embeddings."""

    def __init__(self, vis_dim: int, tab_dim: int, hidden: int = 128):
        super().__init__()
        self.vis_proj = nn.Linear(vis_dim, hidden)
        self.tab_proj = nn.Linear(tab_dim, hidden)
        self.gate = nn.Sequential(
            nn.Linear(hidden * 2, hidden),
            nn.ReLU(inplace=True),
            nn.Linear(hidden, 1),
            nn.Sigmoid(),
        )
        self.head = nn.Sequential(
            nn.Linear(hidden, hidden // 2),
            nn.ReLU(inplace=True),
            nn.Dropout(0.1),
            nn.Linear(hidden // 2, 1),
        )

    def forward(self, vis_feat, tab_feat):
        v = self.vis_proj(vis_feat)
        t = self.tab_proj(tab_feat)
        g = self.gate(torch.cat([v, t], dim=1))  # [B, 1]
        fused = g * v + (1.0 - g) * t
        return self.head(fused).squeeze(-1)


class PawpularityModel(nn.Module):
    """Decoupled ResNet18 vision + MLP tabular, fused via learned gate."""

    def __init__(self, pretrained: bool = True, dropout: float = 0.3):
        super().__init__()
        weights = ResNet18_Weights.DEFAULT if pretrained else None
        backbone = resnet18(weights=weights)
        # Replace final FC with an identity to expose pooled features.
        self.backbone = nn.Sequential(*list(backbone.children())[:-1])
        vis_dim = 512  # ResNet18 pool output
        self.tab_encoder = TabularEncoder(in_dim=NUM_TABULAR, hidden=(128, 64))
        self.fusion = GatedFusionHead(vis_dim, self.tab_encoder.out_dim, hidden=128)
        self.dropout = nn.Dropout(dropout)

    def forward(self, image, tabular):
        vis_feat = self.backbone(image).flatten(1)
        vis_feat = self.dropout(vis_feat)
        tab_feat = self.tab_encoder(tabular)
        logit = self.fusion(vis_feat, tab_feat)
        # Squash to [0, 100] output range.
        return torch.sigmoid(logit) * 100.0

class PawpularityCriterion(nn.Module):
    """Huber (smooth-L1) loss for robust regression against RMSE.

    Huber reduces gradient domination by the heavy tails of human-rated
    scores, which empirically lowers final RMSE compared to pure MSE.
    """

    def __init__(self, beta: float = 1.0):
        super().__init__()
        self.beta = beta

    def forward(self, pred, target):
        pred = pred.reshape(-1)
        target = target.reshape(-1)
        return F.smooth_l1_loss(pred, target, reduction="mean", beta=self.beta)


def compute_rmse(pred: np.ndarray, target: np.ndarray) -> float:
    pred = np.asarray(pred, dtype=np.float64).ravel()
    target = np.asarray(target, dtype=np.float64).ravel()
    return float(np.sqrt(np.mean((pred - target) ** 2)))

class RankSmoother:
    """Monotonic rank-based label smoother.

    Maps a continuous target to its (fractional) rank percentile in [0, 1],
    then scales back to the original range. This is a strictly monotonic
    transformation so RMSE ordering is preserved, but it yields smoother,
    less outlier-dominated gradients. Fit strictly on the training target.
    """

    def __init__(self, lo: float = 0.0, hi: float = 100.0):
        self.lo = lo
        self.hi = hi
        self._sorted = None
        self._n = 0

    def fit(self, y: np.ndarray) -> "RankSmoother":
        y = np.asarray(y, dtype=np.float64).ravel()
        self._sorted = np.sort(y)
        self._n = len(self._sorted)
        return self

    def _rank_pct(self, values: np.ndarray) -> np.ndarray:
        # Fractional rank percentile in [0, 1] via searchsorted.
        idx = np.searchsorted(self._sorted, values, side="left")
        pct = idx / max(self._n, 1)
        return np.clip(pct, 0.0, 1.0)

    def transform(self, values: np.ndarray) -> np.ndarray:
        values = np.asarray(values, dtype=np.float64).ravel()
        pct = self._rank_pct(values)
        # Map percentile back to [lo, hi].
        return (pct * (self.hi - self.lo) + self.lo).astype(np.float32)

    def inverse_transform(self, values: np.ndarray) -> np.ndarray:
        values = np.asarray(values, delta := 0)  # placeholder no-op guard
        return values

class PetDataset(torch.utils.data.Dataset):
    """Loads images + tabular features for one split."""

    def __init__(
        self,
        ids: list,
        img_dir: str,
        tabular: np.ndarray,
        transform: transforms.Compose,
        targets: np.ndarray = None,
    ):
        self.ids = list(ids)
        self.img_dir = img_dir
        self.tabular = np.asarray(tabular, dtype=np.float32)
        self.transform = transform
        self.targets = (
            np.asarray(targets, dtype=np.float32) if targets is not None else None
        )

    def __len__(self):
        return len(self.ids)

    def _load_image(self, path: str) -> torch.Tensor:
        from PIL import Image

        try:
            img = Image.open(path).convert("RGB")
        except Exception:
            # Fallback: black image (should be rare).
            img = Image.new("RGB", (IMAGE_SIZE, IMAGE_SIZE), (0, 0, 0))
        return self.transform(img)

    def __getitem__(self, idx):
        row_id = self.ids[idx]
        path = get_image_path(self.img_dir, row_id)
        image = self._load_image(path)
        tab = torch.from_numpy(self.tabular[idx])
        item = {"image": image, "tabular": tab}
        if self.targets is not None:
            item["target"] = torch.tensor(self.targets[idx], dtype=torch.float32)
        return item


def collate_fn(batch):
    images = torch.stack([b["image"] for b in batch], dim=0)
    tabular = torch.stack([b["tabular"] for b in batch], dim=0)
    out = {"image": images, "tabular": tabular}
    if "target" in batch[0]:
        out["target"] = torch.stack([b["target"] for b in batch], dim=0)
    return out

class GatedFusionHead(nn.Module):
    """Learned per-sample gate blending vision and tabular embeddings."""

    def __init__(self, vis_dim: int, tab_dim: int, hidden: int = 128):
        super().__init__()
        self.vis_proj = nn.Linear(vis_dim, hidden)
        self.tab_proj = nn.Linear(tab_dim, hidden)
        self.gate = nn.Sequential(
            nn.Linear(hidden * 2, hidden),
            nn.ReLU(inplace=True),
            nn.Linear(hidden, 1),
            nn.Sigmoid(),
        )
        self.head = nn.Sequential(
            nn.Linear(hidden, hidden // 2),
            nn.ReLU(inplace=True),
            nn.Dropout(0.1),
            nn.Linear(hidden // 2, 1),
        )

    def forward(self, vis_feat, tab_feat):
        v = self.vis_proj(vis_feat)
        t = self.tab_proj(tab_feat)
        g = self.gate(torch.cat([v, t], dim=1))  # [B, 1]
        fused = g * v + (1.0 - g) * t
        return self.head(fused).squeeze(-1)

class RankSmoother:
    """Monotonic rank-based label smoother (fit strictly on the train target)."""

    def __init__(self, lo: float = 0.0, hi: float = 100.0):
        self.lo = lo
        self.hi = hi
        self._sorted = None
        self._n = 0

    def fit(self, y: np.ndarray) -> "RankSmoother":
        y = np.asarray(y, dtype=np.float64).ravel()
        self._sorted = np.sort(y)
        self._n = len(self._sorted)
        return self

    def _rank_pct(self, values: np.ndarray) -> np.ndarray:
        idx = np.searchsorted(self._sorted, values, side="left")
        pct = idx / max(self._n, 1)
        return np.clip(pct, 0.0, 1.0)

    def transform(self, values: np.ndarray) -> np.ndarray:
        values = np.asarray(values, dtype=np.float64).ravel()
        pct = self._rank_pct(values)
        return (pct * (self.hi - self.lo) + self.lo).astype(np.float32)

        item = {"image": image, "tabular": tab}

        return item

    if "target" in batch[0]:

    return out

def post_process_prediction(pred: np.ndarray) -> np.ndarray:
    """Clamp predictions to the valid [0, 100] range."""
    return np.clip(np.asarray(pred, dtype=np.float32), 0.0, 100.0)

# Optimizer / Scheduler builders (owned by this stage)
# ----------------------------------------------------------------------------
def build_optimizer(model):
    """Differential-LR AdamW: low LR on backbone, high LR on fresh head."""
    backbone_params = list(model.backbone.parameters())
    backbone_ids = {id(p) for p in backbone_params}
    head_params = [
        p
        for n, p in model.named_parameters()
        if id(p) not in backbone_ids and p.requires_grad
    ]
    groups = [
        {"params": backbone_params, "lr": BACKBONE_LR},
        {"params": head_params, "lr": HEAD_LR},
    ]
    return torch.optim.AdamW(groups, weight_decay=WEIGHT_DECAY)


def build_scheduler(optimizer, total_epochs):
    """Linear warmup (per-step) then CosineAnnealingWarmRestarts."""

    def lr_lambda(step):
        warmup_steps = WARMUP_EPOCHS * 1000  # large enough to cover any epoch length
        if step < warmup_steps:
            return max(1e-6, step / max(1, warmup_steps))
        return 1.0  # cosine handled by the underlying LambdaLR base below

    # Use a combined LambdaLR: warmup factor multiplied onto a cosine base.
    T_max = max(1, (total_epochs - WARMUP_EPOCHS) * 1000)

    def combined_lambda(step):
        warmup_steps = WARMUP_EPOCHS * 1000
        if step < warmup_steps:
            warm = max(1e-6, step / max(1, warmup_steps))
        else:
            s = step - warmup_steps
            # cosine anneal with restarts period ~ T_max
            cycle = int(s // T_max)
            phase = (s % T_max) / max(1, T_max)
            warm = 0.5 * (1.0 + math.cos(math.pi * phase))
        return warm

    return torch.optim.lr_scheduler.LambdaLR(optimizer, lr_lambda=combined_lambda)


# ----------------------------------------------------------------------------
# Mixup helper (vision-only; tabular kept raw to preserve causal signal)
# ----------------------------------------------------------------------------
def mixup_batch(images, tabular, targets, alpha=MIXUP_ALPHA):
    if alpha <= 0 or images.shape[0] < 2:
        return images, tabular, targets
    lam = float(np.random.beta(alpha, alpha))
    B = images.shape[0]
    idx = torch.randperm(B, device=images.device)
    mixed_img = lam * images + (1.0 - lam) * images[idx]
    mixed_target = lam * targets + (1.0 - lam) * targets[idx]
    # tabular stays un-mixed (binary causal features must remain meaningful)
    return mixed_img, tabular, mixed_target

# Train / Validate / Predict
# ----------------------------------------------------------------------------
def train_one_epoch(
    model,
    loader,
    criterion,
    optimizer,
    scheduler,
    device,
    accum_steps=1,
    mixup_alpha=MIXUP_ALPHA,
):
    model.train()
    running = 0.0
    n_batches = 0
    optimizer.zero_grad(set_to_none=True)
    for i, batch in enumerate(loader):
        images = batch["image"].to(device, non_blocking=True)
        tabular = batch["tabular"].to(device, non_blocking=True)
        targets = batch["target"].to(device, non_blocking=True)

        images, tabular, targets = mixup_batch(images, tabular, targets, mixup_alpha)

        pred = model(images, tabular)
        loss = criterion(pred, targets) / accum_steps
        loss.backward()

        if (i + 1) % accum_steps == 0 or (i + 1) == len(loader):
            if CLIP_NORM > 0:
                torch.nn.utils.clip_grad_norm_(model.parameters(), CLIP_NORM)
            optimizer.step()
            optimizer.zero_grad(set_to_none=True)
            # step scheduler once per optimizer update
            scheduler.step()

        running += loss.item() * accum_steps
        n_batches += 1
    return running / max(1, n_batches)


@torch.no_grad()
def evaluate(model, loader, criterion, device):
    model.eval()
    running = 0.0
    n_batches = 0
    all_pred = []
    all_target = []
    for batch in loader:
        images = batch["image"].to(device, non_blocking=True)
        tabular = batch["tabular"].to(device, non_blocking=True)
        targets = batch["target"].to(device, non_blocking=True)
        pred = model(images, tabular)
        loss = criterion(pred, targets)
        running += loss.item()
        n_batches += 1
        all_pred.append(pred.cpu().numpy())
        all_target.append(targets.cpu().numpy())
    avg_loss = running / max(1, n_batches)
    preds = np.concatenate(all_pred) if all_pred else np.array([])
    tgts = np.concatenate(all_target) if all_target else np.array([])
    rmse = (
        compute_rmse(post_process_prediction(preds), tgts)
        if len(preds)
        else float("nan")
    )
    return avg_loss, rmse


@torch.no_grad()
def predict_dataset(model, loader, device):
    """Shared val/test inference path (identical processing logic)."""
    model.eval()
    all_pred = []

        tabular = batch["tabular"].to(device, non_blocking=True)

        all_pred.append(pred.cpu().numpy())

class CandidateAdapter:
    """CPU-safe adapter exercising the REAL model, loss, and inputs."""

    DEFAULT_CONTEXT = {

        self._optimizer = None
        self._datasets_cache = None

        ctx = dict(self.DEFAULT_CONTEXT)
        if isinstance(context, dict):
            for k, v in context.items():
                if v is not None:
                    ctx[k] = v
        return ctx

        ctx = self._ctx(context)
        opt = torch.optim.AdamW(
            model.parameters(), lr=ctx["lr"], weight_decay=ctx["weight_decay"]
        )
        self._optimizer = opt
        return opt

    def _get_datasets(self):
        if self._datasets_cache is None:
            self._datasets_cache = build_splits_and_datasets()
        return self._datasets_cache

    def _make_batch(self, dataset, batch_size, device):
        n = len(dataset)
        bs = max(1, min(int(batch_size), n))
        items = [dataset[i] for i in range(bs)]
        batch = collate_fn(items)
        batch = {
            k: (v.to(device) if torch.is_tensor(v) else v) for k, v in batch.items()

        ds = self._get_datasets()["train_dataset"]
        bs = scenario.get("batch_size", 4) if isinstance(scenario, dict) else 4
        return self._make_batch(ds, bs, device)

        ds = self._get_datasets()["val_dataset"]
        bs = scenario.get("batch_size", 4) if isinstance(scenario, dict) else 4
        return self._make_batch(ds, bs, device)

    def _resolve_criterion(self, context):
        if self._criterion is None:
            ctx = self._ctx(context)
            self._criterion = PawpularityCriterion(beta=ctx["criterion_beta"])

        crit = self._resolve_criterion(context)

        crit = self._resolve_criterion(context)
        pred = model(batch["image"], batch["tabular"])
        loss = crit(pred, batch["target"])
        return loss
