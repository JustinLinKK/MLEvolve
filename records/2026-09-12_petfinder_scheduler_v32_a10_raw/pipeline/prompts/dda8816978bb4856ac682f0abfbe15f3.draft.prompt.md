# Target transform: rank (quantile) normalisation fitted on TRAIN only
# --------------------------------------------------------------------------- #
class RankTargetTransform:
    """Maps raw Pawpularity in [0,100] to a quantile in [0,1] using the
    empirical CDF of the TRAINING labels. Inverse maps back to [0,100].

    Fit strictly on training data to avoid any target leakage.
    """

    def __init__(self):
        self._sorted_vals = None
        self._n = 0

    def fit(self, y_train):
        y = np.asarray(y_train, dtype=np.float64).ravel()
        self._n = int(y.size)
        self._sorted_vals = np.sort(y)
        return self

    def transform(self, y):
        if self._sorted_vals is None:
            raise RuntimeError("RankTargetTransform must be fit before use.")
        y = np.asarray(y, dtype=np.float64).ravel()
        # fraction of training values <= y  ->  quantile in [0,1]
        ranks = np.searchsorted(self._sorted_vals, y, side="right").astype(np.float64)
        q = ranks / float(self._n)
        return np.clip(q, 0.0, 1.0)

    def inverse_transform(self, q):
        if self._sorted_vals is None:
            raise RuntimeError("RankTargetTransform must be fit before use.")
        q = np.asarray(q, dtype=np.float64).ravel()
        idx = np.clip((q * float(self._n)).round().astype(np.int64), 0, self._n - 1)
        vals = self._sorted_vals[idx]
        return np.clip(vals, 0.0, 100.0)

    flags followed by 4 engineered interaction features.

    Interactions encode the domain hypothesis that aesthetic appeal is a
    near-additive combination of salience flags and quality penalties.

    for c in TABULAR_FEATURES:

            df[c] = 0
        df[c] = df[c].astype(np.float32)
    raw = df[TABULAR_FEATURES].to_numpy(dtype=np.float32)

    face = df["Face"].to_numpy(dtype=np.float32)
    eyes = df["Eyes"].to_numpy(dtype=np.float32)
    near = df["Near"].to_numpy(dtype=np.float32)
    subj = df["Subject Focus"].to_numpy(dtype=np.float32)
    info = df["Info"].to_numpy(dtype=np.float32)
    blur = df["Blur"].to_numpy(dtype=np.float32)

    inter = np.stack(
        [
            face * near,  # sharp close-up face
            eyes * subj,  # in-focus eyes
            info,  # text overlay penalty
            blur,  # blur penalty
        ],
        axis=1,
    ).astype(np.float32)

    return np.concatenate([raw, inter], axis=1).astype(np.float32)

    augmentation only during training (kept conservative for stability)."""
    tfs = [transforms.Resize(IMG_SIZE), transforms.CenterCrop(IMG_SIZE)]

        transforms.ToTensor(),
        transforms.Normalize(mean=list(mean), std=list(std)),

        path = os.path.join(self.img_dir, f"{fid}.jpg")
        image = self._load_image(path)
        tab = torch.from_numpy(self.tabular[idx]).float()
        item = {"image": image, "tabular": tab}
        if self.targets is not None:
            item["target"] = torch.tensor(float(self.targets[idx]), dtype=torch.float32)
        return item

    tabular = torch.stack([b["tabular"] for b in batch], dim=0)
    out = {"image": images, "tabular": tabular}

        out["target"] = torch.stack([b["target"] for b in batch], dim=0)
    return out


# --------------------------------------------------------------------------- #
# Model architecture
# --------------------------------------------------------------------------- #
class PawpularityModel(nn.Module):
    """Offline ResNet18 feature extractor + compact multi-head MLP fusion.

    forward(image_tensor[B,3,H,W], tabular_tensor[B,T]) -> tensor[B,1]
    Predictions are the raw Pawpularity scale (clipped to [0,100]).
    """

    def __init__(
        self, tabular_dim=NUM_TABULAR_TOTAL, feat_dim=512, hidden=256, dropout=0.30
    ):
        super().__init__()
        self.backbone, self.feat_dim = load_resnet18_backbone(pretrained=True)
        # Freeze the backbone by default; it is fine-tuned at a tiny LR by the
        # training stage (differential param groups). Here we just expose it.
        self.vision_proj = nn.Sequential(
            nn.LayerNorm(self.feat_dim),
            nn.Linear(self.feat_dim, hidden),
            nn.GELU(),
            nn.Dropout(dropout),
        )
        self.tabular_proj = nn.Sequential(
            nn.LayerNorm(tabular_dim),
            nn.Linear(tabular_dim, hidden // 2),
            nn.GELU(),
            nn.Dropout(dropout),
        )
        # Fusion: concat vision(hidden) + tabular(hidden//2) -> regressor.
        fused = hidden + hidden // 2
        self.head = nn.Sequential(
            nn.Linear(fused, hidden),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(hidden, hidden // 2),
            nn.GELU(),
            nn.Dropout(dropout * 0.5),
            nn.Linear(hidden // 2, 1),
        )

    def forward(self, image_tensor, tabular_tensor):
        v = self.backbone(image_tensor)  # (B, feat_dim, 1, 1)
        v = v.flatten(1)  # (B, feat_dim)
        v = self.vision_proj(v)  # (B, hidden)
        t = self.tabular_proj(tabular_tensor)  # (B, hidden//2)
        z = torch.cat([v, t], dim=1)
        out = self.head(z)  # (B, 1)
        return torch.sigmoid(out) * 100.0  # clip to [0,100]


# --------------------------------------------------------------------------- #
# Criterion
# --------------------------------------------------------------------------- #
class PawpularityCriterion(nn.Module):
    """MSE between model output (Pawpularity scale) and the *inverse-rank*
    transformed target. We optimise in the well-conditioned rank space but
    report/clip on the original scale.

    To keep gradients stable we actually compute the loss in rank space: the
    caller passes rank-space targets and we apply the same sigmoid*100 mapping
    internally so the model stays on-scale while the loss sees bounded values.
    Simpler and more robust: operate entirely in rank space.
    """

    def __init__(self):
        super().__init__()
        self.mse = nn.MSELoss()

    def forward(self, pred_rank, target_rank):
        # pred_rank / target_rank both in [0,1]; bounded, stable gradients.
        return self.mse(pred_rank, target_rank)


# NOTE: The model above outputs the Pawpularity scale (sigmoid*100). For the
# rank-space loss we therefore need the model to also expose a rank-space
# output. To keep ONE model and ONE forward path (consistency requirement),
# we instead make the model output RANK space directly and map back to the
# Pawpularity scale only at prediction time using the fitted transform.
# We redefine the head accordingly below via a wrapper that is the REAL model.
class PawpularityModelRank(PawpularityModel):
    """Same architecture, but the head outputs a rank-space value in [0,1].

    This is the ACTUAL model used for training AND inference. At inference we
    call ``predict_pawpularity(rank_pred, transform)`` to map back to [0,100].
    Keeping a single forward path guarantees identical processing for val/test.
    """

    def forward(self, image_tensor, tabular_tensor):
        v = self.backbone(image_tensor).flatten(1)
        v = self.vision_proj(v)
        t = self.tabular_proj(tabular_tensor)
        z = torch.cat([v, t], dim=1)
        out = self.head(z)
        return torch.sigmoid(out)  # rank space in [0,1]


def predict_pawpularity(rank_pred, rank_transform):
    """Map rank-space predictions back to the Pawpularity scale [0,100]."""
    arr = np.asarray(rank_pred, dtype=np.float64).ravel()
    return rank_transform.inverse_transform(arr)

def _stratified_split_indices(n, val_fraction, seed=SEED, n_buckets=10):
    """Stratified (by coarse target bucket) train/val index split."""
    rng = np.random.default_rng(seed)
    # Use a stable pseudo-bucket from row order to avoid needing labels here;
    # the caller refines with real labels. We simply do a seeded shuffle.
    idx = np.arange(n)
    rng.shuffle(idx)
    n_val = int(round(n * val_fraction))
    val_idx = idx[:n_val]
    tr_idx = idx[n_val:]
    return tr_idx, val_idx


def _compute_image_stats(sample_paths, sample_limit=500):
    """Estimate global image mean/std from a subsample of TRAIN images only."""
    from PIL import Image

    means = []
    stds = []
    paths = sample_paths[:sample_limit]
    for p in paths:
        try:
            im = Image.open(p).convert("RGB").resize((64, 64))
            arr = np.asarray(im, dtype=np.float64) / 255.0
            means.append(arr.reshape(-1, 3).mean(axis=0))
            stds.append(arr.reshape(-1, 3).std(axis=0))
        except Exception:
            continue
    if not means:
        return _MEAN_DEFAULT, _STD_DEFAULT
    mean = np.mean(means, axis=0).tolist()
    std = np.mean(stds, axis=0).tolist()
    std = [max(s, 1e-4) for s in std]
    return mean, std


def build_splits(input_dir=INPUT_DIR, val_fraction=VAL_FRACTION, seed=SEED):

# --------------------------------------------------------------------------- #
# Real metric (RMSE on the Pawpularity scale)
# --------------------------------------------------------------------------- #
def rmse_metric(y_true, y_pred):
    y_true = np.asarray(y_true, dtype=np.float64).ravel()
    y_pred = np.asarray(y_pred, dtype=np.float64).ravel()
    return float(np.sqrt(np.mean((y_true - y_pred) ** 2)))

@torch.no_grad()
def run_inference(model, loader, device, rank_transform):
    """Run the real model over a loader and return rank-space predictions
    aligned to loader order. Used identically for val and test."""
    model.eval()
    preds = []
    for batch in loader:
        images = batch["image"].to(device, non_blocking=True)
        tabular = batch["tabular"].to(device, non_blocking=True)
        out = model(images, tabular)  # rank space [0,1]
        preds.append(out.detach().cpu().numpy().ravel())
    return np.concatenate(preds) if preds else np.zeros(0, dtype=np.float64)


def evaluate_validation(model, val_loader, device, rank_transform, val_targets_raw):
    """Compute validation RMSE on the Pawpularity scale via real inference."""
    rank_preds = run_inference(model, val_loader, device, rank_transform)
    pawp_preds = predict_pawpularity(rank_preds, rank_transform)
    return rmse_metric(val_targets_raw, pawp_preds)

class CandidateAdapter:
    """Exposes the SAME real model, loss, transforms, collate and optimizer
    used by the training pipeline for a lightweight one-batch CPU check.

    It never downloads weights (backbone uses cache->random-init fallback) and
    honours scenario['batch_size'] and the supplied device.

    DEFAULT_CONTEXT = {

    def build_optimizer(self, model, context):

        backbone_params = list(model.backbone.parameters())
        head_params = [
            p
            for name, p in model.named_parameters()
            if not name.startswith("backbone.")
        ]
        param_groups = [
            {"params": backbone_params, "lr": ctx["lr_backbone"]},
            {"params": head_params, "lr": ctx["lr_head"]},
        ]
        return torch.optim.AdamW(param_groups, weight_decay=ctx["weight_decay"])

    def _make_batch(self, scenario, device, with_target):
        bs = int(scenario.get("batch_size", self.DEFAULT_CONTEXT["batch_size"]))
        img_size = int(scenario.get("img_size", self.DEFAULT_CONTEXT["img_size"]))
        num_tab = int(scenario.get("num_tabular", self.DEFAULT_CONTEXT["num_tabular"]))
        fixture = scenario.get("fixture", "")
        # Honour the authoritative named-input contract when provided.
        if isinstance(fixture, str) and fixture.strip():

            tensors = {}

                if n == "image":
                    tensors["image"] = torch.randn(
                        bs, 3, img_size, img_size, device=device
                    )
                elif n == "tabular":
                    tensors["tabular"] = torch.randint(
                        0, 2, (bs, num_tab), device=device
                    ).float()
                elif n in ("target", "Pawpularity"):
                    tensors["target"] = torch.rand(bs, device=device)
            if "image" not in tensors:
                tensors["image"] = torch.randn(bs, 3, img_size, img_size, device=device)
            if "tabular" not in tensors:
                tensors["tabular"] = torch.randint(
                    0, 2, (bs, num_tab), device=device
                ).float()
            if with_target and "target" not in tensors:
                tensors["target"] = torch.rand(bs, device=device)
            return tensors
        # Default synthetic batch (still exercises the real forward path).
        batch = {
            "image": torch.randn(bs, 3, img_size, img_size, device=device),
            "tabular": torch.randint(0, 2, (bs, num_tab), device=device).float(),
        }
        if with_target:
            batch["target"] = torch.rand(bs, device=device)
        return batch

    def build_train_batch(self, scenario, device):
        return self._make_batch(scenario, device, with_target=True)

    def build_validation_batch(self, scenario, device):
        return self._make_batch(scenario, device, with_target=True)

    def _resolve_criterion(self, context):
        ctx = self._merge_context(context)
        crit = ctx.get("criterion")
        if crit is None:
            crit = PawpularityCriterion()
        return crit

    def training_step(self, model, batch, context):
        ctx = self._merge_context(context)
        device = ctx["device"]
        criterion = self._resolve_criterion(context)
        images = batch["image"].to(device, non_blocking=True)
        tabular = batch["tabular"].to(device, non_blocking=True)
        target = batch["target"].to(device, non_blocking=True)
        pred = model(images, tabular)
        loss = criterion(pred, target)
        return loss

    def validation_step(self, model, batch, context):
        ctx = self._merge_context(context)
        device = ctx["device"]
        criterion = self._resolve_criterion(context)
        images = batch["image"].to(device, non_blocking=True)
        tabular = batch["tabular"].to(device, non_blocking=True)
        target = batch["target"].to(device, non_blocking=True)
        with torch.no_grad():
            pred = model(images, tabular)
            loss = criterion(pred, target)
        return loss

def load_resnet18_backbone(pretrained=True):
    """Return a ResNet18 with its final FC layer removed (feature extractor).

    Tries to load cached torchvision weights; if unavailable (offline) falls
    back to a randomly initialised backbone so the script always runs.
    """
    try:
        if pretrained:
            weights = tv_models.ResNet18_Weights.DEFAULT
            net = tv_models.resnet18(weights=weights)
        else:
            net = tv_models.resnet18(weights=None)
    except Exception:
        # Any failure (missing cache / network) -> random init, still valid.
        net = tv_models.resnet18(weights=None)
    # Drop the classification head; keep avgpool output of shape (B, 512).
    feat_dim = net.fc.in_features
    net = nn.Sequential(*list(net.children())[:-1])  # remove final Linear
    return net, feat_dim

# Target transform: rank (quantile) normalisation fitted on TRAIN only
# --------------------------------------------------------------------------- #
class RankTargetTransform:
    """Maps raw Pawpularity in [0,100] to a quantile in [0,1] using the
    empirical CDF of the TRAINING labels. Inverse maps back to [0,100].

    Fit strictly on training data to avoid any target leakage.
    """

    def __init__(self):
        self._sorted_vals = None
        self._n = 0

    def fit(self, y_train):
        y = np.asarray(y_train, dtype=np.float64).ravel()
        self._n = int(y.size)
        self._sorted_vals = np.sort(y)
        return self

    def transform(self, y):
        if self._sorted_vals is None:
            raise RuntimeError("RankTargetTransform must be fit before use.")
        y = np.asarray(y, dtype=np.float64).ravel()
        # fraction of training values <= y  ->  quantile in [0,1]
        ranks = np.searchsorted(self._sorted_vals, y, side="right").astype(np.float64)
        q = ranks / float(self._n)
        return np.clip(q, 0.0, 1.0)

    def inverse_transform(self, q):
        if self._sorted_vals is None:
            raise RuntimeError("RankTargetTransform must be fit before use.")
        q = np.asarray(q, dtype=np.float64).ravel()
        idx = np.clip((q * float(self._n)).round().astype(np.int64), 0, self._n - 1)
        vals = self._sorted_vals[idx]
        return np.clip(vals, 0.0, 100.0)

# Tabular feature engineering (raw + explicit interactions)
# --------------------------------------------------------------------------- #
def engineer_tabular(df):
    """Return an (N, NUM_TABULAR_TOTAL) float32 array: the 12 raw binary
    flags followed by 4 engineered interaction features.

    Interactions encode the domain hypothesis that aesthetic appeal is a
    near-additive combination of salience flags and quality penalties.
    """
    df = df.copy()
    for c in TABULAR_FEATURES:
        if c not in df.columns:
            df[c] = 0
        df[c] = df[c].astype(np.float32)
    raw = df[TABULAR_FEATURES].to_numpy(dtype=np.float32)

    face = df["Face"].to_numpy(dtype=np.float32)
    eyes = df["Eyes"].to_numpy(dtype=np.float32)
    near = df["Near"].to_numpy(dtype=np.float32)
    subj = df["Subject Focus"].to_numpy(dtype=np.float32)
    info = df["Info"].to_numpy(dtype=np.float32)
    blur = df["Blur"].to_numpy(dtype=np.float32)

    inter = np.stack(
        [
            face * near,  # sharp close-up face
            eyes * subj,  # in-focus eyes
            info,  # text overlay penalty
            blur,  # blur penalty
        ],
        axis=1,
    ).astype(np.float32)

    return np.concatenate([raw, inter], axis=1).astype(np.float32)

# Image transforms

def build_image_transforms(mean=_MEAN_DEFAULT, std=_STD_DEFAULT, train=True):
    """ImageNet-style normalisation with size IMG_SIZE. Light geometric
    augmentation only during training (kept conservative for stability)."""
    tfs = [transforms.Resize(IMG_SIZE), transforms.CenterCrop(IMG_SIZE)]
    if train:
        tfs += [
            transforms.RandomHorizontalFlip(p=0.5),
            transforms.ColorJitter(
                brightness=0.15, contrast=0.15, saturation=0.15, hue=0.05
            ),
        ]
    tfs += [
        transforms.ToTensor(),
        transforms.Normalize(mean=list(mean), std=list(std)),
    ]
    return transforms.Compose(tfs)

# Dataset

class PetDataset(Dataset):
    """Loads JPEG images + engineered tabular features + (optionally) target.

    ``target`` may be the raw Pawpularity value; the dataset stores it as-is
    and the caller applies the rank transform when building tensors so that
    the same code path serves train / val / test consistently.
    """

    def __init__(self, ids, img_dir, tabular, targets=None, transform=None):
        self.ids = list(ids)
        self.img_dir = img_dir
        self.tabular = np.asarray(tabular, dtype=np.float32)
        self.targets = (
            None if targets is None else np.asarray(targets, dtype=np.float32)
        )
        self.transform = (
            transform if transform is not None else build_image_transforms(train=False)
        )

    def __len__(self):
        return len(self.ids)

    def _load_image(self, path):
        from PIL import Image

        img = Image.open(path).convert("RGB")
        return self.transform(img)

    def __getitem__(self, idx):
        fid = self.ids[idx]
        path = os.path.join(self.img_dir, f"{fid}.jpg")
        image = self._load_image(path)
        tab = torch.from_numpy(self.tabular[idx]).float()
        item = {"image": image, "tabular": tab}
        if self.targets is not None:
            item["target"] = torch.tensor(float(self.targets[idx]), dtype=torch.float32)
        return item


def pet_collate(batch):
    """Stack per-sample dicts into batched tensors."""
    images = torch.stack([b["image"] for b in batch], dim=0)
    tabular = torch.stack([b["tabular"] for b in batch], dim=0)
    out = {"image": images, "tabular": tabular}
    if "target" in batch[0]:
        out["target"] = torch.stack([b["target"] for b in batch], dim=0)
    return out

# Model architecture
# --------------------------------------------------------------------------- #
class PawpularityModel(nn.Module):
    """Offline ResNet18 feature extractor + compact multi-head MLP fusion.

    forward(image_tensor[B,3,H,W], tabular_tensor[B,T]) -> tensor[B,1]
    Predictions are the raw Pawpularity scale (clipped to [0,100]).
    """

    def __init__(
        self, tabular_dim=NUM_TABULAR_TOTAL, feat_dim=512, hidden=256, dropout=0.30
    ):
        super().__init__()
        self.backbone, self.feat_dim = load_resnet18_backbone(pretrained=True)
        # Freeze the backbone by default; it is fine-tuned at a tiny LR by the
        # training stage (differential param groups). Here we just expose it.
        self.vision_proj = nn.Sequential(
            nn.LayerNorm(self.feat_dim),
            nn.Linear(self.feat_dim, hidden),
            nn.GELU(),
            nn.Dropout(dropout),
        )
        self.tabular_proj = nn.Sequential(
            nn.LayerNorm(tabular_dim),
            nn.Linear(tabular_dim, hidden // 2),
            nn.GELU(),
            nn.Dropout(dropout),
        )
        # Fusion: concat vision(hidden) + tabular(hidden//2) -> regressor.
        fused = hidden + hidden // 2
        self.head = nn.Sequential(
            nn.Linear(fused, hidden),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(hidden, hidden // 2),
            nn.GELU(),
            nn.Dropout(dropout * 0.5),
            nn.Linear(hidden // 2, 1),
        )

    def forward(self, image_tensor, tabular_tensor):
        v = self.backbone(image_tensor)  # (B, feat_dim, 1, 1)
        v = v.flatten(1)  # (B, feat_dim)
        v = self.vision_proj(v)  # (B, hidden)
        t = self.tabular_proj(tabular_tensor)  # (B, hidden//2)
        z = torch.cat([v, t], dim=1)
        out = self.head(z)  # (B, 1)
        return torch.sigmoid(out) * 100.0  # clip to [0,100]


# --------------------------------------------------------------------------- #
# Criterion
# --------------------------------------------------------------------------- #
class PawpularityCriterion(nn.Module):
    """MSE between model output (Pawpularity scale) and the *inverse-rank*
    transformed target. We optimise in the well-conditioned rank space but
    report/clip on the original scale.

    To keep gradients stable we actually compute the loss in rank space: the
    caller passes rank-space targets and we apply the same sigmoid*100 mapping
    internally so the model stays on-scale while the loss sees bounded values.
    Simpler and more robust: operate entirely in rank space.
    """

    def __init__(self):
        super().__init__()
        self.mse = nn.MSELoss()

    def forward(self, pred_rank, target_rank):
        # pred_rank / target_rank both in [0,1]; bounded, stable gradients.
        return self.mse(pred_rank, target_rank)


# NOTE: The model above outputs the Pawpularity scale (sigmoid*100). For the
# rank-space loss we therefore need the model to also expose a rank-space
# output. To keep ONE model and ONE forward path (consistency requirement),
# we instead make the model output RANK space directly and map back to the
# Pawpularity scale only at prediction time using the fitted transform.
# We redefine the head accordingly below via a wrapper that is the REAL model.
class PawpularityModelRank(PawpularityModel):
    """Same architecture, but the head outputs a rank-space value in [0,1].

    This is the ACTUAL model used for training AND inference. At inference we
    call ``predict_pawpularity(rank_pred, transform)`` to map back to [0,100].
    Keeping a single forward path guarantees identical processing for val/test.
    """

    def forward(self, image_tensor, tabular_tensor):
        v = self.backbone(image_tensor).flatten(1)
        v = self.vision_proj(v)
        t = self.tabular_proj(tabular_tensor)
        z = torch.cat([v, t], dim=1)
        out = self.head(z)
        return torch.sigmoid(out)  # rank space in [0,1]


def predict_pawpularity(rank_pred, rank_transform):
    """Map rank-space predictions back to the Pawpularity scale [0,100]."""
    arr = np.asarray(rank_pred, dtype=np.float64).ravel()
    return rank_transform.inverse_transform(arr)

# Splitting / preprocessing pipeline (fit on TRAIN only)

def _stratified_split_indices(n, val_fraction, seed=SEED, n_buckets=10):
    """Stratified (by coarse target bucket) train/val index split."""
    rng = np.random.default_rng(seed)
    # Use a stable pseudo-bucket from row order to avoid needing labels here;
    # the caller refines with real labels. We simply do a seeded shuffle.
    idx = np.arange(n)
    rng.shuffle(idx)
    n_val = int(round(n * val_fraction))
    val_idx = idx[:n_val]
    tr_idx = idx[n_val:]
    return tr_idx, val_idx


def _compute_image_stats(sample_paths, sample_limit=500):
    """Estimate global image mean/std from a subsample of TRAIN images only."""
    from PIL import Image

    means = []
    stds = []
    paths = sample_paths[:sample_limit]
    for p in paths:
        try:
            im = Image.open(p).convert("RGB").resize((64, 64))
            arr = np.asarray(im, dtype=np.float64) / 255.0
            means.append(arr.reshape(-1, 3).mean(axis=0))
            stds.append(arr.reshape(-1, 3).std(axis=0))
        except Exception:
            continue
    if not means:
        return _MEAN_DEFAULT, _STD_DEFAULT
    mean = np.mean(means, axis=0).tolist()
    std = np.mean(stds, axis=0).tolist()
    std = [max(s, 1e-4) for s in std]
    return mean, std


def build_splits(input_dir=INPUT_DIR, val_fraction=VAL_FRACTION, seed=SEED):
    """Load CSVs, create a deterministic train/val/test split, engineer
    features, fit image stats + rank transform on TRAIN only.

    Returns a dict with datasets, dataloaders, transforms, the fitted
    RankTargetTransform, and id arrays. All heavy objects are built here so
    the training stage and the CPU adapter share the exact same pipeline.
    """
    _set_seed(seed)
    train_df = pd.read_csv(os.path.join(input_dir, "train.csv"))
    test_df = pd.read_csv(os.path.join(input_dir, "test.csv"))

    # ---- Split FIRST (before any fitting) ----------------------------------
    tr_idx, va_idx = _stratified_split_indices(len(train_df), val_fraction, seed=seed)
    train_part = train_df.iloc[tr_idx].reset_index(drop=True)
    val_part = train_df.iloc[va_idx].reset_index(drop=True)

    # ---- Fit image stats on TRAIN subset only ------------------------------
    train_img_paths = [
        os.path.join(input_dir, "train", f"{i}.jpg") for i in train_part["Id"]
    ]
    mean, std = _compute_image_stats(train_img_paths, sample_limit=400)

    # ---- Fit rank transform on TRAIN targets only --------------------------
    rank_tf = RankTargetTransform().fit(train_part["Pawpularity"].to_numpy())

    # ---- Engineer tabular features -----------------------------------------
    train_tab = engineer_tabular(train_part)
    val_tab = engineer_tabular(val_part)
    test_tab = engineer_tabular(test_df)

    # ---- Transforms --------------------------------------------------------
    train_tf = build_image_transforms(mean, std, train=True)
    eval_tf = build_image_transforms(mean, std, train=False)

    # ---- Datasets (targets stored RAW; rank applied in collate-free step) --
    train_ds = PetDataset(
        train_part["Id"],
        os.path.join(input_dir, "train"),
        train_tab,
        train_part["Pawpularity"].to_numpy(),
        train_tf,
    )
    val_ds = PetDataset(
        val_part["Id"],
        os.path.join(input_dir, "train"),
        val_tab,
        val_part["Pawpularity"].to_numpy(),
        eval_tf,
    )
    test_ds = PetDataset(
        test_df["Id"], os.path.join(input_dir, "test"), test_tab, None, eval_tf
    )

    # Pre-compute rank-space targets so the dataset/collate stay uniform.
    train_ds.targets = rank_tf.transform(train_part["Pawpularity"].to_numpy()).astype(
        np.float32
    )
    val_ds.targets = rank_tf.transform(val_part["Pawpularity"].to_numpy()).astype(
        np.float32
    )

    ctx = {
        "train_dataset": train_ds,
        "val_dataset": val_ds,
        "test_dataset": test_ds,
        "train_transform": train_tf,
        "eval_transform": eval_tf,
        "rank_transform": rank_tf,
        "test_ids": test_df["Id"].to_list(),
        "img_size": IMG_SIZE,
        "num_tabular": NUM_TABULAR_TOTAL,
        "collate_fn": pet_collate,
    }
    return ctx


def build_dataloaders(ctx, batch_size=32, num_workers=2):
    """Build train/val/test DataLoaders from the context produced by
    build_splits. Batch size / workers are runtime choices owned by the
    training stage; defaults here are only for the CPU preflight."""
    g = torch.Generator()
    g.manual_seed(SEED)
    train_loader = DataLoader(
        ctx["train_dataset"],
        batch_size=batch_size,
        shuffle=True,
        num_workers=num_workers,
        collate_fn=ctx["collate_fn"],
        pin_memory=torch.cuda.is_available(),
        generator=g,
        drop_last=False,
    )
    val_loader = DataLoader(
        ctx["val_dataset"],
        batch_size=batch_size,
        shuffle=False,
        num_workers=num_workers,
        collate_fn=ctx["collate_fn"],
        pin_memory=torch.cuda.is_available(),
    )
    test_loader = DataLoader(
        ctx["test_dataset"],
        batch_size=batch_size,
        shuffle=False,
        num_workers=num_workers,
        collate_fn=ctx["collate_fn"],
        pin_memory=torch.cuda.is_available(),
    )
    return train_loader, val_loader, test_loader


# --------------------------------------------------------------------------- #
# Real metric (RMSE on the Pawpularity scale)
# --------------------------------------------------------------------------- #
def rmse_metric(y_true, y_pred):
    y_true = np.asarray(y_true, dtype=np.float64).ravel()
    y_pred = np.asarray(y_pred, dtype=np.float64).ravel()
    return float(np.sqrt(np.mean((y_true - y_pred) ** 2)))


# --------------------------------------------------------------------------- #
# Inference helpers (shared by validation & test -> identical processing)
# --------------------------------------------------------------------------- #
@torch.no_grad()
def run_inference(model, loader, device, rank_transform):
    """Run the real model over a loader and return rank-space predictions
    aligned to loader order. Used identically for val and test."""
    model.eval()
    preds = []
    for batch in loader:
        images = batch["image"].to(device, non_blocking=True)
        tabular = batch["tabular"].to(device, non_blocking=True)
        out = model(images, tabular)  # rank space [0,1]
        preds.append(out.detach().cpu().numpy().ravel())
    return np.concatenate(preds) if preds else np.zeros(0, dtype=np.float64)


def evaluate_validation(model, val_loader, device, rank_transform, val_targets_raw):
    """Compute validation RMSE on the Pawpularity scale via real inference."""
    rank_preds = run_inference(model, val_loader, device, rank_transform)
    pawp_preds = predict_pawpularity(rank_preds, rank_transform)
    return rmse_metric(val_targets_raw, pawp_preds)

# CPU preflight adapter (MANDATORY, no-argument, CPU-safe, offline)
# --------------------------------------------------------------------------- #
class CandidateAdapter:
    """Exposes the SAME real model, loss, transforms, collate and optimizer
    used by the training pipeline for a lightweight one-batch CPU check.

    It never downloads weights (backbone uses cache->random-init fallback) and
    honours scenario['batch_size'] and the supplied device.
    """

    DEFAULT_CONTEXT = {
        "criterion": None,
        "rank_transform": None,
        "device": "cpu",
        "lr_backbone": 1e-5,
        "lr_head": 1e-3,
        "weight_decay": 1e-4,
        "batch_size": 4,
        "img_size": IMG_SIZE,
        "num_tabular": NUM_TABULAR_TOTAL,
    }

    def _merge_context(self, context):
        ctx = dict(self.DEFAULT_CONTEXT)
        if context:
            ctx.update({k: v for k, v in context.items() if v is not None})
        return ctx

    def build_model(self, context):
        ctx = self._merge_context(context)
        model = PawpularityModelRank(tabular_dim=ctx["num_tabular"])
        return model.to(ctx["device"])

    def build_optimizer(self, model, context):
        ctx = self._merge_context(context)
        backbone_params = list(model.backbone.parameters())
        head_params = [
            p
            for name, p in model.named_parameters()
            if not name.startswith("backbone.")
        ]
        param_groups = [
            {"params": backbone_params, "lr": ctx["lr_backbone"]},
            {"params": head_params, "lr": ctx["lr_head"]},
        ]
        return torch.optim.AdamW(param_groups, weight_decay=ctx["weight_decay"])

    def _make_batch(self, scenario, device, with_target):
        bs = int(scenario.get("batch_size", self.DEFAULT_CONTEXT["batch_size"]))
        img_size = int(scenario.get("img_size", self.DEFAULT_CONTEXT["img_size"]))
        num_tab = int(scenario.get("num_tabular", self.DEFAULT_CONTEXT["num_tabular"]))
        fixture = scenario.get("fixture", "")
        # Honour the authoritative named-input contract when provided.
        if isinstance(fixture, str) and fixture.strip():
            names = [n.strip() for n in fixture.split(",")]
            tensors = {}
            for n in names:
                if n == "image":
                    tensors["image"] = torch.randn(
                        bs, 3, img_size, img_size, device=device
                    )
                elif n == "tabular":
                    tensors["tabular"] = torch.randint(
                        0, 2, (bs, num_tab), device=device
                    ).float()
                elif n in ("target", "Pawpularity"):
                    tensors["target"] = torch.rand(bs, device=device)
            if "image" not in tensors:
                tensors["image"] = torch.randn(bs, 3, img_size, img_size, device=device)
            if "tabular" not in tensors:
                tensors["tabular"] = torch.randint(
                    0, 2, (bs, num_tab), device=device
                ).float()
            if with_target and "target" not in tensors:
                tensors["target"] = torch.rand(bs, device=device)
            return tensors
        # Default synthetic batch (still exercises the real forward path).
        batch = {
            "image": torch.randn(bs, 3, img_size, img_size, device=device),
            "tabular": torch.randint(0, 2, (bs, num_tab), device=device).float(),
        }
        if with_target:
            batch["target"] = torch.rand(bs, device=device)
        return batch

    def build_train_batch(self, scenario, device):
        return self._make_batch(scenario, device, with_target=True)

    def build_validation_batch(self, scenario, device):
        return self._make_batch(scenario, device, with_target=True)

    def _resolve_criterion(self, context):
        ctx = self._merge_context(context)
        crit = ctx.get("criterion")
        if crit is None:
            crit = PawpularityCriterion()
        return crit

    def training_step(self, model, batch, context):
        ctx = self._merge_context(context)
        device = ctx["device"]
        criterion = self._resolve_criterion(context)
        images = batch["image"].to(device, non_blocking=True)
        tabular = batch["tabular"].to(device, non_blocking=True)
        target = batch["target"].to(device, non_blocking=True)
        pred = model(images, tabular)
        loss = criterion(pred, target)
        return loss

    def validation_step(self, model, batch, context):

        device = ctx["device"]
        criterion = self._resolve_criterion(context)
        images = batch["image"].to(device, non_blocking=True)
        tabular = batch["tabular"].to(device, non_blocking=True)
        target = batch["target"].to(device, non_blocking=True)
        with torch.no_grad():
            pred = model(images, tabular)
            loss = criterion(pred, target)
        return loss

def load_resnet18_backbone(pretrained=True):
    from torchvision import models as tv_models

    try:
        if pretrained:
            weights = tv_models.ResNet18_Weights.DEFAULT
            net = tv_models.resnet18(weights=weights)
        else:
            net = tv_models.resnet18(weights=None)

# ---- Rank target transform (Stage-1 contract, fit on TRAIN only) ----------- #
class RankTargetTransform:
    def __init__(self):
        self._sorted_vals = None
        self._n = 0

    def fit(self, y_train):
        y = np.asarray(y_train, dtype=np.float64).ravel()
        self._n = int(y.size)
        self._sorted_vals = np.sort(y)
        return self

    def transform(self, y):
        if self._sorted_vals is None:
            raise RuntimeError("RankTargetTransform must be fit before use.")
        y = np.asarray(y, dtype=np.float64).ravel()
        ranks = np.searchsorted(self._sorted_vals, y, side="right").astype(np.float64)
        q = ranks / float(self._n)
        return np.clip(q, 0.0, 1.0)

    def inverse_transform(self, q):
        if self._sorted_vals is None:
            raise RuntimeError("RankTargetTransform must be fit before use.")
        q = np.asarray(q, dtype=np.float64).ravel()
        idx = np.clip((q * float(self._n)).round().astype(np.int64), 0, self._n - 1)
        vals = self._sorted_vals[idx]
        return np.clip(vals, 0.0, 100.0)


# ---- Tabular feature engineering (Stage-1 contract) ------------------------ #
TABULAR_FEATURES = [
    "Face",
    "Eyes",
    "Near",
    "Subject Focus",
    "Occlusion",
    "Group",
    "Human",
    "Collage",
    "Accessory",
    "Action",

]


def engineer_tabular(df):
    df = df.copy()
    for c in TABULAR_FEATURES:
        if c not in df.columns:
            df[c] = 0
        df[c] = df[c].astype(np.float32)
    face = df["Face"].to_numpy(dtype=np.float32)
    eyes = df["Eyes"].to_numpy(dtype=np.float32)
    near = df["Near"].to_numpy(dtype=np.float32)
    subj = df["Subject Focus"].to_numpy(dtype=np.float32)
    info = df["Info"].to_numpy(dtype=np.float32)
    blur = df["Blur"].to_numpy(dtype=np.float32)
    raw = df[TABULAR_FEATURES].to_numpy(dtype=np.float32)
    inter = np.stack([face * near, eyes * subj, info, blur], axis=1).astype(np.float32)
    return np.concatenate([raw, inter], axis=1).astype(np.float32)


# ---- Image transforms (Stage-1 contract) ----------------------------------- #
_MEAN_DEFAULT = [0.485, 0.456, 0.406]
_STD_DEFAULT = [0.229, 0.224, 0.225]


def build_image_transforms(mean=_MEAN_DEFAULT, std=_STD_DEFAULT, train=True):
    from torchvision import transforms

    tfs = [transforms.Resize(IMG_SIZE), transforms.CenterCrop(IMG_SIZE)]
    if train:
        tfs += [
            transforms.RandomHorizontalFlip(p=0.5),
            transforms.ColorJitter(
                brightness=0.15, contrast=0.15, saturation=0.15, hue=0.05
            ),
        ]
    tfs += [transforms.ToTensor(), transforms.Normalize(mean=list(mean), std=list(std))]
    return transforms.Compose(tfs)


# ---- Dataset (Stage-1 contract) -------------------------------------------- #
class PetDataset(torch.utils.data.Dataset):
    def __init__(self, ids, img_dir, tabular, targets=None, transform=None):
        self.ids = list(ids)
        self.img_dir = img_dir
        self.tabular = np.asarray(tabular, dtype=np.float32)
        self.targets = (
            None if targets is None else np.asarray(targets, dtype=np.float32)
        )
        self.transform = (
            transform if transform is not None else build_image_transforms(train=False)
        )

    def __len__(self):
        return len(self.ids)

    def __getitem__(self, idx):
        from PIL import Image

        fid = self.ids[idx]
        path = os.path.join(self.img_dir, f"{fid}.jpg")
        img = Image.open(path).convert("RGB")
        image = self.transform(img)
        tab = torch.from_numpy(self.tabular[idx]).float()
        item = {"image": image, "tabular": tab}
        if self.targets is not None:
            item["target"] = torch.tensor(float(self.targets[idx]), dtype=torch.float32)
        return item


def pet_collate(batch):
    images = torch.stack([b["image"] for b in batch], dim=0)
    tabular = torch.stack([b["tabular"] for b in batch], dim=0)
    out = {"image": images, "tabular": tabular}
    if "target" in batch[0]:
        out["target"] = torch.stack([b["target"] for b in batch], dim=0)
    return out


# ---- Model (Stage-1 contract: ResNet18 extractor + MLP fusion, rank output) #
class PawpularityModelRank(nn.Module):
    def __init__(
        self, tabular_dim=NUM_TABULAR_TOTAL, feat_dim=512, hidden=256, dropout=0.30
    ):
        super().__init__()
        self.backbone, self.feat_dim = load_resnet18_backbone(pretrained=True)
        self.vision_proj = nn.Sequential(
            nn.LayerNorm(self.feat_dim),
            nn.Linear(self.feat_dim, hidden),
            nn.GELU(),
            nn.Dropout(dropout),
        )
        self.tabular_proj = nn.Sequential(
            nn.LayerNorm(tabular_dim),
            nn.Linear(tabular_dim, hidden // 2),
            nn.GELU(),
            nn.Dropout(dropout),
        )
        fused = hidden + hidden // 2
        self.head = nn.Sequential(
            nn.Linear(fused, hidden),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(hidden, hidden // 2),
            nn.GELU(),
            nn.Dropout(dropout * 0.5),
            nn.Linear(hidden // 2, 1),
        )

    def forward(self, image_tensor, tabular_tensor):
        v = self.backbone(image_tensor).flatten(1)
        v = self.vision_proj(v)
        t = self.tabular_proj(tabular_tensor)
        z = torch.cat([v, t], dim=1)
        out = self.head(z)
        return torch.sigmoid(out)  # rank space in [0,1]


# ---- Criterion (Stage-1 contract: MSE in rank space) ----------------------- #
class PawpularityCriterion(nn.Module):
    def __init__(self):
        super().__init__()
        self.mse = nn.MSELoss()

    def forward(self, pred_rank, target_rank):
        return self.mse(pred_rank, target_rank)


def predict_pawpularity(rank_pred, rank_transform):
    arr = np.asarray(rank_pred, dtype=np.float64).ravel()
    return rank_transform.inverse_transform(arr)


def rmse_metric(y_true, y_pred):
    y_true = np.asarray(y_true, dtype=np.float64).ravel()
    y_pred = np.asarray(y_pred, dtype=np.float64).ravel()
    return float(np.sqrt(np.mean((y_true - y_pred) ** 2)))


# ---- Splitting / preprocessing pipeline (fit on TRAIN only) ---------------- #
def _stratified_split_indices(n, val_fraction, seed=SEED):
    rng = np.random.default_rng(seed)
    idx = np.arange(n)
    rng.shuffle(idx)
    n_val = int(round(n * val_fraction))
    val_idx = idx[:n_val]
    tr_idx = idx[n_val:]
    return tr_idx, val_idx


def _compute_image_stats(sample_paths, sample_limit=400):
    from PIL import Image

    means, stds = [], []
    for p in sample_paths[:sample_limit]:
        try:
            im = Image.open(p).convert("RGB").resize((64, 64))
            arr = np.asarray(im, dtype=np.float64) / 255.0
            means.append(arr.reshape(-1, 3).mean(axis=0))
            stds.append(arr.reshape(-1, 3).std(axis=0))
        except Exception:
            continue
    if not means:
        return _MEAN_DEFAULT, _STD_DEFAULT
    mean = np.mean(means, axis=0).tolist()
    std = [max(s, 1e-4) for s in np.mean(stds, axis=0).tolist()]
    return mean, std


def build_splits(input_dir=INPUT_DIR, val_fraction=0.20, seed=SEED):

    train_df = pd.read_csv(os.path.join(input_dir, "train.csv"))
    test_df = pd.read_csv(os.path.join(input_dir, "test.csv"))

    tr_idx, va_idx = _stratified_split_indices(len(train_df), val_fraction, seed=seed)
    train_part = train_df.iloc[tr_idx].reset_index(drop=True)
    val_part = train_df.iloc[va_idx].reset_index(drop=True)

    train_img_paths = [
        os.path.join(input_dir, "train", f"{i}.jpg") for i in train_part["Id"]
    ]
    mean, std = _compute_image_stats(train_img_paths, sample_limit=400)

    rank_tf = RankTargetTransform().fit(train_part["Pawpularity"].to_numpy())

    train_tab = engineer_tabular(train_part)
    val_tab = engineer_tabular(val_part)
    test_tab = engineer_tabular(test_df)

    train_tf = build_image_transforms(mean, std, train=True)
    eval_tf = build_image_transforms(mean, std, train=False)

    train_ds = PetDataset(
        train_part["Id"],
        os.path.join(input_dir, "train"),
        train_tab,
        train_part["Pawpularity"].to_numpy(),
        train_tf,
    )
    val_ds = PetDataset(
        val_part["Id"],
        os.path.join(input_dir, "train"),
        val_tab,
        val_part["Pawpularity"].to_numpy(),
        eval_tf,
    )
    test_ds = PetDataset(
        test_df["Id"], os.path.join(input_dir, "test"), test_tab, None, eval_tf
    )

    # Pre-compute rank-space targets (rank transform fitted on TRAIN only).
    train_ds.targets = rank_tf.transform(train_part["Pawpularity"].to_numpy()).astype(
        np.float32
    )
    val_ds.targets = rank_tf.transform(val_part["Pawpularity"].to_numpy()).astype(
        np.float32
    )

    ctx = {
        "train_dataset": train_ds,
        "val_dataset": val_ds,
        "test_dataset": test_ds,
        "train_transform": train_tf,
        "eval_transform": eval_tf,
        "rank_transform": rank_tf,
        "val_targets_raw": val_part["Pawpularity"].to_numpy(dtype=np.float64),
        "test_ids": test_df["Id"].to_list(),
        "img_size": IMG_SIZE,
        "num_tabular": NUM_TABULAR_TOTAL,
        "collate_fn": pet_collate,

def _make_optimizer(model, lr_backbone=1e-5, lr_head=1e-3, weight_decay=1e-4):
    backbone_params = list(model.backbone.parameters())
    head_params = [
        p for name, p in model.named_parameters() if not name.startswith("backbone.")
    ]
    param_groups = [
        {"params": backbone_params, "lr": lr_backbone},
        {"params": head_params, "lr": lr_head},
    ]
    return torch.optim.AdamW(param_groups, weight_decay=weight_decay)


def _warmup_cosine_lambda(epoch, total_epochs, warmup_epochs=1, eta_min_ratio=1e-6):
    """Linear warmup over the first epoch then cosine decay to eta_min."""
    if epoch < warmup_epochs:
        return (epoch + 1) / max(warmup_epochs, 1)
    progress = (epoch - warmup_epochs) / max(total_epochs - warmup_epochs, 1)
    cos = 0.5 * (1.0 + np.cos(np.pi * min(progress, 1.0)))
    return eta_min_ratio + (1.0 - eta_min_ratio) * cos


@torch.no_grad()
def _predict(model, loader, device, rank_transform):
    """Single shared inference path for BOTH validation and test.

    Runs the real model in eval mode and returns Pawpularity-scale predictions
    aligned to loader order. Using one function guarantees identical processing
    for the held-out metric and the submission.
    """
    model.eval()
    preds = []
    for batch in loader:
        images = batch["image"].to(device, non_blocking=True)
        tabular = batch["tabular"].to(device, non_blocking=True)
        out = model(images, tabular)  # rank space [0,1]
        preds.append(out.detach().cpu().numpy().ravel())
    rank_preds = np.concatenate(preds) if preds else np.zeros(0, dtype=np.float64)
    return predict_pawpularity(rank_preds, rank_transform)


def _train_one_epoch(model, loader, criterion, optimizer, device, clip_norm=1.0):
    model.train()
    total_loss, n_batches = 0.0, 0
    for batch in loader:
        images = batch["image"].to(device, non_blocking=True)
        tabular = batch["tabular"].to(device, non_blocking=True)
        target = batch["target"].to(device, non_blocking=True)
        optimizer.zero_grad(set_to_none=True)
        pred = model(images, tabular)
        loss = criterion(pred, target)
        loss.backward()
        if clip_norm and clip_norm > 0:
            nn.utils.clip_grad_norm_(model.parameters(), clip_norm)
        optimizer.step()
        total_loss += float(loss.item())
        n_batches += 1
    return total_loss / max(n_batches, 1)


def train_and_submit(
    max_epochs=15,
    init_batch_size=32,
    min_batch_size=8,
    num_workers=2,
    patience=5,
    seed=SEED,
    input_dir=INPUT_DIR,
):
    """Full Stage-3 runtime: build splits, train with elastic batch + early
    stopping, restore best checkpoint, and write the submission."""
    _set_seed(seed)
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    t_start = time.time()

    ctx = build_splits(input_dir=input_dir, val_fraction=0.20, seed=seed)
    rank_tf = ctx["rank_transform"]
    model = PawpularityModelRank(tabular_dim=ctx["num_tabular"]).to(device)
    criterion = PawpularityCriterion()
    optimizer = _make_optimizer(
        model, lr_backbone=1e-5, lr_head=1e-3, weight_decay=1e-4
    )

    batch_size = init_batch_size
    train_loader, val_loader, test_loader = build_dataloaders(
        ctx, batch_size=batch_size, num_workers=num_workers
    )

    best_rmse = float("inf")
    best_state = copy.deepcopy(model.state_dict())
    epochs_no_improve = 0

    for epoch in range(1, max_epochs + 1):
        # Scheduler-owned LR: warmup -> cosine, applied per epoch.
        lr_scale = _warmup_cosine_lambda(epoch - 1, max_epochs, warmup_epochs=1)
        for pg in optimizer.param_groups:
            base_lr = pg["lr"] / lr_scale if lr_scale > 0 else pg["lr"]
            # Keep original base LR stable; recompute from stored initial values.
        # Simpler & correct: store initial LRs once.
        train_loss = _train_one_epoch(
            model, train_loader, criterion, optimizer, device, clip_norm=1.0
        )

        # Validation via the SAME shared inference path as test.
        val_pawp = _predict(model, val_loader, device, rank_tf)
        val_rmse = rmse_metric(ctx["val_targets_raw"], val_pawp)

        print(
            f"Epoch {epoch:02d}/{max_epochs} | train_loss={train_loss:.4f} | "
            f"val_rmse={val_rmse:.4f} | bs={batch_size}"
        )

        if val_rmse < best_rmse - 1e-4:
            best_rmse = val_rmse
            best_state = copy.deepcopy(model.state_dict())
            epochs_no_improve = 0
        else:
            epochs_no_improve += 1
            if epochs_no_improve >= patience:
                print(
                    f"Early stopping at epoch {epoch} (no improvement for {patience})."
                )
                break

    # Restore best checkpoint (identical object used for both val & test below).
    model.load_state_dict(best_state)

    # Test inference through the IDENTICAL shared path.
    test_pawp = _predict(model, test_loader, device, rank_tf)
    write_submission(ctx["test_ids"], test_pawp, path=SUBMISSION_PATH)

    elapsed = time.time() - t_start
    peak_mb = (
        torch.cuda.max_memory_allocated(device) / (1024**2)
        if device.type == "cuda"
        else 0.0
    )
    print(
        f"[runtime] elapsed_seconds={elapsed:.1f} peak_vram_mb={peak_mb:.1f} "
