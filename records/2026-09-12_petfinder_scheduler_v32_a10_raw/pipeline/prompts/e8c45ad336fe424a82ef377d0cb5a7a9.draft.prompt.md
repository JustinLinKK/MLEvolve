class TabularEncoder:
    """Encodes the 11 binary flags into a 12-dim engineered vector.

    Adds a 'positive appeal sum' interaction feature.  No target information is
    used, so there is no leakage.  Fit is trivially identity-safe but we still
    keep an explicit fit/transform contract for pipeline hygiene.
    """

    def __init__(self):
        self.fitted = False

    def _encode(self, df):
        X = df[FLAG_COLS].astype(np.float32).to_numpy()
        pos_sum = X[:, [FLAG_COLS.index(c) for c in POSITIVE_FLAGS]].sum(
            axis=1, keepdims=True
        )
        return np.concatenate([X, pos_sum], axis=1).astype(np.float32)

    def fit(self, df):
        self.fitted = True
        return self

    def transform(self, df):
        if not self.fitted:
            self.fit(df)
        return self._encode(df)

def compute_rank_targets(labels):
    """Map raw Pawpularity to its percentile rank in [0, 100].

    This is a monotone transform fitted on the training labels; it damps label
    noise and keeps the scale aligned with the RMSE metric.  Validation/test
    predictions are compared against their RAW labels (RMSE), so this is a pure
    training-target regulariser, not a leak.
    """
    labels = np.asarray(labels, dtype=np.float64)
    order = np.argsort(np.argsort(labels))
    n = len(labels)
    ranks = (order + 1) / (n + 1) * 100.0
    return ranks.astype(np.float32)

class GatedCrossAttentionFusion(nn.Module):
    """Tabular query attends over spatial image features via gated cross-attention."""

    def __init__(self, tab_dim, feat_dim, hidden, num_heads, dropout):
        super().__init__()
        self.tab_proj = nn.Sequential(
            nn.Linear(tab_dim, hidden),
            nn.GELU(),
            nn.Dropout(dropout),
        )
        self.kv_proj = nn.Linear(feat_dim, hidden)
        self.attn = nn.MultiheadAttention(
            hidden, num_heads, dropout=dropout, batch_first=True
        )
        self.gate = nn.Sequential(nn.Linear(hidden * 2, hidden), nn.Sigmoid())
        self.out_norm = nn.LayerNorm(hidden)

    def forward(self, tab, feat_seq):
        # tab: [B, tab_dim]; feat_seq: [B, N, feat_dim]
        q = self.tab_proj(tab).unsqueeze(1)  # [B, 1, H]
        kv = self.kv_proj(feat_seq)  # [B, N, H]
        attn_out, _ = self.attn(q, kv, kv)  # [B, 1, H]
        attn_out = attn_out.squeeze(1)  # [B, H]
        g = self.gate(torch.cat([q.squeeze(1), attn_out], dim=1))
        fused = self.out_norm(g * attn_out)  # [B, H]
        return fused

    """ConvNeXt-B backbone + gated cross-attention tabular fusion -> scalar."""

    def __init__(
        self,
        backbone,
        tab_dim=TAB_DIM,
        feat_dim=BACKBONE_FEATURES,
        hidden=256,
        num_heads=NUM_HEADS,
        dropout=DROPOUT,
    ):
        super().__init__()
        self.backbone = backbone
        self.fusion = GatedCrossAttentionFusion(
            tab_dim, feat_dim, hidden, num_heads, dropout

            nn.GELU(),
            nn.Dropout(dropout),

        fused = self.fusion(tabular, seq)  # [B, hidden]
        return self.head(fused).squeeze(-1)  # [B]

class TabularEncoder:
    """Encodes the 11 binary flags into a 12-dim engineered vector.

    Adds a 'positive appeal sum' interaction feature.  No target information is
    used, so there is no leakage.  Fit is trivially identity-safe but we still
    keep an explicit fit/transform contract for pipeline hygiene.
    """

    def __init__(self):
        self.fitted = False

    def _encode(self, df):
        X = df[FLAG_COLS].astype(np.float32).to_numpy()
        pos_sum = X[:, [FLAG_COLS.index(c) for c in POSITIVE_FLAGS]].sum(
            axis=1, keepdims=True
        )
        return np.concatenate([X, pos_sum], axis=1).astype(np.float32)

    def fit(self, df):
        self.fitted = True
        return self

    def transform(self, df):
        if not self.fitted:
            self.fit(df)
        return self._encode(df)

def compute_rank_targets(labels):
    """Map raw Pawpularity to its percentile rank in [0, 100].

    This is a monotone transform fitted on the training labels; it damps label
    noise and keeps the scale aligned with the RMSE metric.  Validation/test
    predictions are compared against their RAW labels (RMSE), so this is a pure
    training-target regulariser, not a leak.
    """
    labels = np.asarray(labels, dtype=np.float64)
    order = np.argsort(np.argsort(labels))
    n = len(labels)
    ranks = (order + 1) / (n + 1) * 100.0
    return ranks.astype(np.float32)

class GatedCrossAttentionFusion(nn.Module):
    """Tabular query attends over spatial image features via gated cross-attention."""

    def __init__(self, tab_dim, feat_dim, hidden, num_heads, dropout):
        super().__init__()
        self.tab_proj = nn.Sequential(
            nn.Linear(tab_dim, hidden),
            nn.GELU(),
            nn.Dropout(dropout),
        )
        self.kv_proj = nn.Linear(feat_dim, hidden)
        self.attn = nn.MultiheadAttention(
            hidden, num_heads, dropout=dropout, batch_first=True
        )
        self.gate = nn.Sequential(nn.Linear(hidden * 2, hidden), nn.Sigmoid())
        self.out_norm = nn.LayerNorm(hidden)

    def forward(self, tab, feat_seq):
        # tab: [B, tab_dim]; feat_seq: [B, N, feat_dim]
        q = self.tab_proj(tab).unsqueeze(1)  # [B, 1, H]
        kv = self.kv_proj(feat_seq)  # [B, N, H]
        attn_out, _ = self.attn(q, kv, kv)  # [B, 1, H]
        attn_out = attn_out.squeeze(1)  # [B, H]
        g = self.gate(torch.cat([q.squeeze(1), attn_out], dim=1))
        fused = self.out_norm(g * attn_out)  # [B, H]
        return fused


class PawpularityModel(nn.Module):
    """ConvNeXt-B backbone + gated cross-attention tabular fusion -> scalar."""

    def __init__(
        self,
        backbone,
        tab_dim=TAB_DIM,
        feat_dim=BACKBONE_FEATURES,
        hidden=256,
        num_heads=NUM_HEADS,
        dropout=DROPOUT,
    ):
        super().__init__()
        self.backbone = backbone
        self.fusion = GatedCrossAttentionFusion(
            tab_dim, feat_dim, hidden, num_heads, dropout
        )
        self.head = nn.Sequential(
            nn.Linear(hidden, hidden // 2),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(hidden // 2, 1),
        )

    def forward(self, image, tabular):
        feats = self.backbone(image)  # [B, C, H, W]
        B, C, H, W = feats.shape
        seq = feats.flatten(2).transpose(1, 2)  # [B, HW, C]
        fused = self.fusion(tabular, seq)  # [B, hidden]
        return self.head(fused).squeeze(-1)  # [B]

class GatedCrossAttentionFusion(nn.Module):
    def __init__(self, tab_dim, feat_dim, hidden, num_heads, dropout):
        super().__init__()
        self.tab_proj = nn.Sequential(
            nn.Linear(tab_dim, hidden), nn.GELU(), nn.Dropout(dropout)
        )
        self.kv_proj = nn.Linear(feat_dim, hidden)
        self.attn = nn.MultiheadAttention(
            hidden, num_heads, dropout=dropout, batch_first=True
        )
        self.gate = nn.Sequential(nn.Linear(hidden * 2, hidden), nn.Sigmoid())
        self.out_norm = nn.LayerNorm(hidden)

    def forward(self, tab, feat_seq):
        q = self.tab_proj(tab).unsqueeze(1)
        kv = self.kv_proj(feat_seq)
        attn_out, _ = self.attn(q, kv, kv)
        attn_out = attn_out.squeeze(1)
        g = self.gate(torch.cat([q.squeeze(1), attn_out], dim=1))
        return self.out_norm(g * attn_out)


class PawpularityModel(nn.Module):
    def __init__(
        self,
        backbone,
        tab_dim=TAB_DIM,
        feat_dim=BACKBONE_FEATURES,
        hidden=256,
        num_heads=NUM_HEADS,
        dropout=DROPOUT,
    ):
        super().__init__()
        self.backbone = backbone
        self.fusion = GatedCrossAttentionFusion(
            tab_dim, feat_dim, hidden, num_heads, dropout
        )
        self.head = nn.Sequential(
            nn.Linear(hidden, hidden // 2),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(hidden // 2, 1),
        )

    def forward(self, image, tabular):
        feats = self.backbone(image)
        B, C, H, W = feats.shape
        seq = feats.flatten(2).transpose(1, 2)
        fused = self.fusion(tabular, seq)
        return self.head(fused).squeeze(-1)

def mixup_batch(images, tabs, targets, alpha=0.2):
    """Standard Mixup. Returns mixed images/tabs and interpolated targets."""
    B = images.size(0)
    lam = float(np.random.beta(alpha, alpha)) if B > 1 else 1.0
    idx = torch.randperm(B, device=images.device)
    im = lam * images + (1.0 - lam) * images[idx]
    tb = lam * tabs + (1.0 - lam) * tabs[idx]
    tg = lam * targets + (1.0 - lam) * targets[idx]
    return im, tb, tg


class WarmupCosineScheduler:
    """Linear warmup then cosine decay to eta_min. step() once per epoch."""

    def __init__(self, optimizer, warmup_epochs, total_epochs, eta_min=1e-7):
        self.optimizer = optimizer
        self.warmup = max(1, int(warmup_epochs))
        self.total = max(self.warmup + 1, int(total_epochs))
        self.eta_min = float(eta_min)
        self.base_lrs = [g["lr"] for g in optimizer.param_groups]
        self.epoch = 0

    def _factor(self):
        e = self.epoch
        if e < self.warmup:
            return (e + 1) / self.warmup
        p = (e - self.warmup) / max(1, (self.total - self.warmup))
        cos = 0.5 * (1.0 + np.cos(np.pi * min(1.0, p)))
        return (
            self.eta_min / max(self.base_lrs[0], 1e-12)
            + (1.0 - self.eta_min / max(self.base_lrs[0], 1e-12)) * cos
        )

    def step(self):
        f = self._factor()
        for g, base in zip(self.optimizer.param_groups, self.base_lrs):
            g["lr"] = base * f
        self.epoch += 1

def _train_one_epoch(
    model, loader, opt, crit, device, use_mixup=True, mix_alpha=0.2, clip=1.0
):
    model.train()
    run_loss, n = 0.0, 0
    for batch in loader:
        img = batch["image"].to(device, non_blocking=True)
        tab = batch["tabular"].to(device, non_blocking=True)
        tgt = batch["target"].to(device, non_blocking=True)
        if use_mixup:
            img, tab, tgt = mixup_batch(img, tab, tgt, mix_alpha)
        opt.zero_grad(set_to_none=True)
        pred = model(img, tab)
        loss = crit(pred, tgt)
        loss.backward()
        if clip:
            torch.nn.utils.clip_grad_norm_(model.parameters(), clip)
        opt.step()
        run_loss += loss.item() * img.size(0)
        n += img.size(0)
    return run_loss / max(1, n)
