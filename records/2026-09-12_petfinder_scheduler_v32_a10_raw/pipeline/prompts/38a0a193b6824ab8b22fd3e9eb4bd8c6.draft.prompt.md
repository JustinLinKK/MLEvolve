class TabularExpert(nn.Module):
    def __init__(self, in_dim=N_TABULAR, hidden=(256, 128, 64)):
        super().__init__()
        layers, d = [], in_dim
        for h in hidden:
            layers += [nn.Linear(d, h), nn.LayerNorm(h), nn.GELU(), nn.Dropout(0.2)]
            d = h
        layers.append(nn.Linear(d, 1))
        self.net = nn.Sequential(*layers)

    def forward(self, x):
        return self.net(x).squeeze(-1)


class VisionExpert(nn.Module):
    def __init__(self, embed_dim=768):
        super().__init__()
        try:
            self.vit = torch.hub.load(

            )
        except Exception:
            self.vit = None
        self.proj = nn.Sequential(nn.AdaptiveAvgPool2d(1), nn.Flatten())
        self.head = nn.Sequential(
            nn.Linear(embed_dim, 128), nn.GELU(), nn.Dropout(0.2), nn.Linear(128, 1)
        )

    def forward(self, x):
        if self.vit is not None:
            f = self.vit.forward_features(x)
            if isinstance(f, dict):
                f = f.get("x_norm_patchtokens", f.get("cls_token"))
            f = self.proj(f)
        else:
            f = self.proj(x)
        return self.head(f).squeeze(-1)


class PawpularityModel(nn.Module):
    """Decoupled experts + learned residual fusion head."""

    def __init__(self, img_size=IMG_SIZE, n_tabular=N_TABULAR):
        super().__init__()
        self.tab_expert = TabularExpert(n_tabular)
        self.vis_expert = VisionExpert()
        self.fusion = nn.Sequential(
            nn.Linear(2, 128), nn.GELU(), nn.Dropout(0.2), nn.Linear(128, 1)
        )

    def forward(self, image, tabular):
        t = self.tab_expert(tabular)  # [B]
        v = self.vis_expert(image)  # [B]
        z = torch.stack([t, v], dim=-1)  # [B,2]
        resid = self.fusion(z).squeeze(-1)  # [B]
        out = t + resid  # tabular anchor + learned correction
        return torch.clamp(out, 0.0, 100.0)

        ### Step 1: model_design
        **Plan:** **Plan.** The task is a *subjective* aesthetic-score regression where the 12 binary profile flags are known to be strong, near-deterministic drivers (Face/Eyes/Near/Group/Human etc. shift Pawpularity by tens of points), while the image adds residual detail. Prior attempts all fused tabular into a CNN via attention/gates and hit ~20.75–21.2 RMSE; several later variants crashed. My novel hypothesis: instead of fusing tabular *into* the vision path, run two **decoupled experts** — a frozen pretrained ViT-B/16 for spatial/aesthetic cues and a deep MLP on engineered tabular features — then combine them with a small learned **residual head** that lets the model correct the tabular expert's systematic bias using image evidence. This matches the task structure (tabular carries most signal, image refines) and avoids fragile cross-attention fusion. I validate with a reproducible stratified split and target RMSE directly.

Memory strategy: fp32 throughout, no AMP (per trace); single fixed batch (no elasticity in this stage). Operator/normalization notes: LayerNorm in the MLP + GELU activations, BatchNorm-free ViT, Dropout(0.2) regularization. Configurable dims: `IMG_SIZE=256`, MLP hidden widths `[256,128,64]`, fusion width `128`. Fallback: if pretrained ViT weights aren't cached locally, fall back to random-init ViT (still learns from data). Scheduler-owned controls (optimizer/scheduler/batch/epochs/lr/AMP/checkpointing/validation loop/submission) are NOT implemented here.

class _PatchEmbed(nn.Module):
    def __init__(self, patch=16, in_ch=3, dim=768):
        super().__init__()
        self.proj = nn.Conv2d(in_ch, dim, kernel_size=patch, stride=patch)

    def forward(self, x):
        x = self.proj(x)  # [B, dim, H, W]
        x = x.flatten(2).transpose(1, 2)  # [B, N, dim]
        return x


class _Block(nn.Module):
    def __init__(self, dim=768, heads=12, mlp_ratio=4.0, drop=0.0):
        super().__init__()
        self.norm1 = nn.LayerNorm(dim)
        self.attn = nn.MultiheadAttention(dim, heads, dropout=drop, batch_first=True)
        self.drop1 = nn.Dropout(drop)
        self.norm2 = nn.LayerNorm(dim)
        self.mlp = nn.Sequential(
            nn.Linear(dim, int(dim * mlp_ratio)),
            nn.GELU(),
            nn.Dropout(drop),
            nn.Linear(int(dim * mlp_ratio), dim),
            nn.Dropout(drop),
        )

    def forward(self, x):
        h = self.norm1(x)
        a, _ = self.attn(h, h, h, need_weights=False)
        x = x + self.drop1(a)
        x = x + self.mlp(self.norm2(x))
        return x


class _LocalViT(nn.Module):
    """Self-contained ViT-B/16 built locally (no remote weight download)."""

    def __init__(self, img_size=256, patch=16, dim=768, depth=12, heads=12):
        super().__init__()
        self.patch_embed = _PatchEmbed(patch, 3, dim)
        num_patches = (img_size // patch) ** 2
        self.cls_token = nn.Parameter(torch.zeros(1, 1, dim))
        self.pos_embed = nn.Parameter(torch.zeros(1, num_patches + 1, dim))
        self.blocks = nn.ModuleList([_Block(dim, heads) for _ in range(depth)])
        self.norm = nn.LayerNorm(dim)
        nn.init.trunc_normal_(self.pos_embed, std=0.02)
        nn.init.normal_(self.cls_token, std=0.02)
        self.apply(self._init_weights)

    @staticmethod
    def _init_weights(m):
        if isinstance(m, nn.Linear):
            nn.init.trunc_normal_(m.weight, std=0.02)
            if m.bias is not None:
                nn.init.zeros_(m.bias)
        elif isinstance(m, nn.LayerNorm):
            nn.init.ones_(m.weight)
            nn.init.zeros_(m.bias)
        elif isinstance(m, nn.Conv2d):
            nn.init.trunc_normal_(m.weight, std=0.02)
            if m.bias is not None:
                nn.init.zeros_(m.bias)

    def forward_features(self, x):
        B = x.shape[0]
        x = self.patch_embed(x)  # [B, N, dim]
        cls = self.cls_token.expand(B, -1, -1)
        x = torch.cat([cls, x], dim=1)
        x = x + self.pos_embed
        for blk in self.blocks:
            x = blk(x)
        x = self.norm(x)
        return x[:, 0]  # CLS token [B, dim]

def _param_groups(model):
    backbone_params, other_params = [], []
    for name, p in model.named_parameters():
        if not p.requires_grad:
            continue
        if name.startswith("vis_expert.vit"):

            other_params.append(p)
    groups = [
        {"params": other_params, "lr": LR_HEAD},

    if backbone_params:
        groups.append({"params": backbone_params, "lr": LR_BACKBONE})
    return groups


# ------------------------------- Scheduler --------------------------------- #
def build_scheduler(optimizer, warmup_epochs, total_epochs):
    def lr_lambda(epoch):
        if epoch < warmup_epochs:
            return max(1e-6, (epoch + 1) / max(1, warmup_epochs))
        progress = (epoch - warmup_epochs) / max(1, (total_epochs - warmup_epochs))
        return 0.5 * (1.0 + math.cos(math.pi * min(1.0, progress)))

    return torch.optim.lr_scheduler.LambdaLR(optimizer, lr_lambda)


# --------------------------- Train / Eval helpers -------------------------- #
def _run_epoch(model, loader, criterion, device, train=True):
    model.train() if train else model.eval()
    total, count = 0.0, 0
    ctx = torch.enable_grad() if train else torch.no_grad()
    with ctx:
        for batch in loader:
            img = batch["image"].to(device, non_blocking=True)
            tab = batch["tabular"].to(device, non_blocking=True)
            tgt = batch["target"].to(device, non_blocking=True)
            pred = model(img, tab)
            loss = criterion(pred, tgt)
            total += loss.item() * tgt.size(0)
            count += tgt.size(0)
    return total / max(1, count)
