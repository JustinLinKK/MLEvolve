def build_tab_features(raw):
    """
    Expand raw binary metadata (B, 12) into (B, 16) by appending engineered
    interaction terms that encode composition quality cues:
        t12 = Near * Eyes            (close-up AND eyes open -> high appeal)
        t13 = Near * Face             (close-up AND face visible)
        t14 = Face * Eyes             (face AND eyes)
        t15 = 1 - Blur                (sharpness bonus)
    Returns a float32 ndarray of shape (B, NUM_TAB_FEATURES).
    """
    raw = np.asarray(raw, dtype=np.float32)
    if raw.ndim == 1:
        raw = raw.reshape(1, -1)
    # Column indices (robust to ordering via TAB_COLUMNS definition).
    idx = {c: i for i, c in enumerate(TAB_COLUMNS)}
    face = raw[:, idx["Face"]]
    eyes = raw[:, idx["Eyes"]]
    near = raw[:, idx["Near"]]
    blur = raw[:, idx["Blur"]]

    inter = np.stack(
        [
            near * eyes,
            near * face,
            face * eyes,
            1.0 - blur,
        ],
        axis=1,
    ).astype(np.float32)

    return np.concatenate([raw, inter], axis=1).astype(np.float32)

# --------------------------------------------------------------------------- #
class RankSmoother:
    """
    Maps raw integer labels (1..100) to smooth continuous targets in (0,100)
    using empirical ranks, and provides the inverse mapping back to [0,100].
    Fit strictly on the training target to avoid leakage.
    """

    def __init__(self, alpha=RANK_SMOOTH_ALPHA):
        self.alpha = alpha
        self._rank_to_target = None
        self._target_to_rank = None
        self._n = 0

    def fit(self, y):
        y = np.asarray(y, dtype=float)
        n = len(y)
        order = np.argsort(y, kind="mergesort")
        ranks = np.empty(n, dtype=np.float64)
        ranks[order] = np.arange(1, n + 1, dtype=np.float64)
        # Smoothed percentile in (0, 1).
        smoothed = (ranks * self.alpha + (n + 1) * (1.0 - self.alpha)) / (n + 1)
        # Map back to the 1..100 scale.
        targets = smoothed * 100.0
        self._rank_to_target = targets
        self._target_to_rank = ranks.astype(np.int64)
        self._n = n
        # Build inverse lookup: given a predicted smoothed value, recover a
        # label by nearest-rank interpolation.
        self._inv_sorted_targets = targets
        return self

    def transform(self, y):
        y = np.asarray(y, dtype=float)
        # Nearest-rank mapping to the smoothed target grid.
        idx = np.searchsorted(self._inv_sorted_targets, y, side="left")
        idx = np.clip(idx, 0, self._n - 1)
        return self._rank_to_target[idx].astype(np.float32)

    def inverse_transform(self, t):
        t = np.asarray(t, dtype=float)
        idx = np.searchsorted(self._inv_sorted_targets, t, side="left")
        idx = np.clip(idx, 0, self._n - 1)
        return self._rank_to_target[idx].astype(np.float32)

      tab_score    = MLP_tab(engineered_tabular)
      gate         = sigmoid(Wg * [image_score, tab_score, tab] + bg)
      output       = clamp(gate * image_score + (1 - gate) * tab_score, 0, 100)

    The gate lets the model lean on the strong tabular prior while retaining
    image detail as a residual, which is well matched to this task where the
    metadata flags are a near-deterministic composition prior.
    """

    def __init__(
        self, num_tab_features=NUM_TAB_FEATURES, dropout=0.25, freeze_image=True
    ):
        super().__init__()
        self.image_expert = _ResNet50ImageExpert(freeze=freeze_image)
        img_dim = 2048

        self.image_head = nn.Sequential(
            nn.Linear(img_dim, 512),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
            nn.Linear(512, 128),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
            nn.Linear(128, 1),
        )

        self.tab_head = nn.Sequential(
            nn.Linear(num_tab_features, 128),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
            nn.Linear(128, 64),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
            nn.Linear(64, 1),
        )

        # Gate input: image score (1) + tab score (1) + tab features.
        self.gate = nn.Sequential(
            nn.Linear(2 + num_tab_features, 64),
            nn.ReLU(inplace=True),
            nn.Linear(64, 1),
        )

    def forward(self, image, tabular):
        img_emb = self.image_expert(image)
        img_score = self.image_head(img_emb)  # (B, 1)
        tab_score = self.tab_head(tabular)  # (B, 1)
        gate_in = torch.cat([img_score, tab_score, tabular], dim=1)
        gate = torch.sigmoid(self.gate(gate_in))  # (B, 1)
        out = gate * img_score + (1.0 - gate) * tab_score
        out = torch.clamp(out, 0.0, 100.0)
        return out.squeeze(-1)

def make_split_indices(train_df, val_frac=0.10, seed=42):
    """

    return np.asarray(tr), np.asarray(va)

def build_tab_features(raw):
    """
    Expand raw binary metadata (B, 12) into (B, 16) by appending engineered
    interaction terms that encode composition quality cues:
        t12 = Near * Eyes            (close-up AND eyes open -> high appeal)
        t13 = Near * Face             (close-up AND face visible)
        t14 = Face * Eyes             (face AND eyes)
        t15 = 1 - Blur                (sharpness bonus)
    Returns a float32 ndarray of shape (B, NUM_TAB_FEATURES).
    """
    raw = np.asarray(raw, dtype=np.float32)
    if raw.ndim == 1:
        raw = raw.reshape(1, -1)
    # Column indices (robust to ordering via TAB_COLUMNS definition).
    idx = {c: i for i, c in enumerate(TAB_COLUMNS)}
    face = raw[:, idx["Face"]]
    eyes = raw[:, idx["Eyes"]]
    near = raw[:, idx["Near"]]
    blur = raw[:, idx["Blur"]]

    inter = np.stack(
        [
            near * eyes,
            near * face,
            face * eyes,
            1.0 - blur,
        ],
        axis=1,
    ).astype(np.float32)

    return np.concatenate([raw, inter], axis=1).astype(np.float32)


# --------------------------------------------------------------------------- #
# Rank smoothing (label denoising) -- fit on TRAIN only
# --------------------------------------------------------------------------- #
class RankSmoother:
    """
    Maps raw integer labels (1..100) to smooth continuous targets in (0,100)
    using empirical ranks, and provides the inverse mapping back to [0,100].
    Fit strictly on the training target to avoid leakage.
    """

    def __init__(self, alpha=RANK_SMOOTH_ALPHA):
        self.alpha = alpha
        self._rank_to_target = None
        self._target_to_rank = None
        self._n = 0

    def fit(self, y):
        y = np.asarray(y, dtype=float)
        n = len(y)
        order = np.argsort(y, kind="mergesort")
        ranks = np.empty(n, dtype=np.float64)
        ranks[order] = np.arange(1, n + 1, dtype=np.float64)
        # Smoothed percentile in (0, 1).
        smoothed = (ranks * self.alpha + (n + 1) * (1.0 - self.alpha)) / (n + 1)
        # Map back to the 1..100 scale.
        targets = smoothed * 100.0
        self._rank_to_target = targets
        self._target_to_rank = ranks.astype(np.int64)
        self._n = n
        # Build inverse lookup: given a predicted smoothed value, recover a
        # label by nearest-rank interpolation.
        self._inv_sorted_targets = targets
        return self

    def transform(self, y):
        y = np.asarray(y, dtype=float)
        # Nearest-rank mapping to the smoothed target grid.
        idx = np.searchsorted(self._inv_sorted_targets, y, side="left")
        idx = np.clip(idx, 0, self._n - 1)
        return self._rank_to_target[idx].astype(np.float32)

    def inverse_transform(self, t):
        t = np.asarray(t, dtype=float)
        idx = np.searchsorted(self._inv_sorted_targets, t, side="left")
        idx = np.clip(idx, 0, self._n - 1)
        return self._rank_to_target[idx].astype(np.float32)

class _ResNet50ImageExpert(nn.Module):
    """Frozen-by-default pretrained ResNet50 producing a 2048-dim embedding."""

    def __init__(self, freeze=True):
        super().__init__()
        self.backbone = self._build_backbone()
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.freeze = freeze
        if freeze:
            for p in self.backbone.parameters():
                p.requires_grad_(False)

    @staticmethod
    def _build_backbone():
        try:
            # Offline-safe: load_state_dict from local cache if available,
            # otherwise fall back to random init (no network access).
            m = tv_models.resnet50(weights=None)
            try:
                sd = tv_models.ResNet50_Weights.DEFAULT.get_state_dict(progress=False)
                m.load_state_dict(sd)
            except Exception:
                pass  # keep random init
            return m
        except Exception:
            return tv_models.resnet50(pretrained=False)

    def forward(self, x):
        x = self.backbone(x)
        x = self.pool(x)
        x = torch.flatten(x, 1)
        return x


class PetPawpularityModel(nn.Module):
    """
    Decoupled dual-expert architecture with a learned soft-convex fusion gate.

      image_score  = MLP_img(frozen_resnet50(image))
      tab_score    = MLP_tab(engineered_tabular)
      gate         = sigmoid(Wg * [image_score, tab_score, tab] + bg)
      output       = clamp(gate * image_score + (1 - gate) * tab_score, 0, 100)

    The gate lets the model lean on the strong tabular prior while retaining
    image detail as a residual, which is well matched to this task where the
    metadata flags are a near-deterministic composition prior.
    """

    def __init__(
        self, num_tab_features=NUM_TAB_FEATURES, dropout=0.25, freeze_image=True
    ):
        super().__init__()
        self.image_expert = _ResNet50ImageExpert(freeze=freeze_image)
        img_dim = 2048

        self.image_head = nn.Sequential(
            nn.Linear(img_dim, 512),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
            nn.Linear(512, 128),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
            nn.Linear(128, 1),
        )

        self.tab_head = nn.Sequential(
            nn.Linear(num_tab_features, 128),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
            nn.Linear(128, 64),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
            nn.Linear(64, 1),
        )

        # Gate input: image score (1) + tab score (1) + tab features.
        self.gate = nn.Sequential(
            nn.Linear(2 + num_tab_features, 64),
            nn.ReLU(inplace=True),
            nn.Linear(64, 1),
        )

    def forward(self, image, tabular):
        img_emb = self.image_expert(image)
        img_score = self.image_head(img_emb)  # (B, 1)
        tab_score = self.tab_head(tabular)  # (B, 1)
        gate_in = torch.cat([img_score, tab_score, tabular], dim=1)
        gate = torch.sigmoid(self.gate(gate_in))  # (B, 1)
        out = gate * img_score + (1.0 - gate) * tab_score
        out = torch.clamp(out, 0.0, 100.0)
        return out.squeeze(-1)

# Metric (real RMSE on the native [0,100] scale)

def rmse_score(y_true, y_pred):
    """True root-mean-squared error over the full array."""
    y_true = np.asarray(y_true, dtype=np.float64)
    y_pred = np.asarray(y_pred, dtype=np.float64)
    return float(np.sqrt(np.mean((y_true - y_pred) ** 2)))

# Optimizer: differential learning rates
#   - high LR for the small trainable heads (image_head/tab_head/gate)
#   - low LR for any (optionally) unfrozen image backbone parameters
# --------------------------------------------------------------------------- #
def build_optimizer(model, head_lr=1e-3, backbone_lr=1e-4, weight_decay=1e-4):
    head_params, backbone_params = [], []
    for name, p in model.named_parameters():
        if not p.requires_grad:
            continue
        if "image_expert" in name:
            backbone_params.append(p)
        else:
            head_params.append(p)
    groups = [{"params": head_params, "lr": head_lr}]
    if backbone_params:
        groups.append({"params": backbone_params, "lr": backbone_lr})
    return torch.optim.AdamW(groups, lr=head_lr, weight_decay=weight_decay)


# --------------------------------------------------------------------------- #
# Scheduler: linear warmup then cosine decay to a floor
# --------------------------------------------------------------------------- #
class WarmupCosineScheduler:
    def __init__(self, optimizer, warmup_epochs, total_epochs, min_ratio=0.1):
        self.optimizer = optimizer
        self.warmup = max(1, int(warmup_epochs))
        self.total = max(self.warmup + 1, int(total_epochs))
        self.min_ratio = min_ratio
        self.base_lrs = [g["lr"] for g in optimizer.param_groups]
        self._step = 0

    def step(self):
        self._step += 1
        e = self._step
        if e <= self.warmup:
            factor = e / self.warmup
        else:
            progress = (e - self.warmup) / max(1, (self.total - self.warmup))
            cos = 0.5 * (1.0 + math.cos(math.pi * min(1.0, progress)))
            factor = self.min_ratio + (1.0 - self.min_ratio) * cos
        for g, base in zip(self.optimizer.param_groups, self.base_lrs):
            g["lr"] = base * factor

@torch.no_grad()
def _predict(model, loader, device):
    model.eval()
    preds = []
    for batch in loader:
        img = batch["image"].to(device, non_blocking=True)
        tab = batch["tabular"].to(device, non_blocking=True)
        out = model(img, tab)
        preds.append(out.detach().cpu())
    return torch.cat(preds, dim=0).numpy().astype(np.float32)


# --------------------------------------------------------------------------- #
# One training epoch (with optional Mixup + gradient clipping + OOM halving)
# --------------------------------------------------------------------------- #
def train_one_epoch(
    model, loader, optimizer, criterion, device, mixup_alpha=0.2, grad_clip=1.0
):
    model.train()
    total_loss, n_batches = 0.0, 0
    bs = loader.batch_size
    for batch in loader:
        img = batch["image"].to(device, non_blocking=True)
        tab = batch["tabular"].to(device, non_blocking=True)
        target = batch["target"].to(device, non_blocking=True)

        try:
            # Optional Mixup over both modalities (targets are continuous).
            if mixup_alpha > 0 and img.shape[0] > 1:
                lam = float(np.random.beta(mixup_alpha, mixup_alpha))
                idx = torch.randperm(img.shape[0], device=device)
                img = lam * img + (1.0 - lam) * img[idx]
                tab = lam * tab + (1.0 - lam) * tab[idx]
                tgt = lam * target + (1.0 - lam) * target[idx]
            else:
                tgt = target

            optimizer.zero_grad(set_to_none=True)
            pred = model(img, tab)
            loss = criterion(pred, tgt)
            loss.backward()
            if grad_clip > 0:
                torch.nn.utils.clip_grad_norm_(
                    [p for p in model.parameters() if p.requires_grad], grad_clip
                )
            optimizer.step()
        except RuntimeError as exc:
            if "out of memory" in str(exc).lower():
                torch.cuda.empty_cache()
                # Halve the effective batch size for the remainder of the epoch.
                bs = max(4, bs // 2)
                new_loader = DataLoader(
                    loader.dataset,
                    batch_size=bs,
                    shuffle=False,
                    num_workers=loader.num_workers,
                    collate_fn=default_collate,
                )
                print(f"[OOM] reducing batch size to {bs}")
                # Restart this epoch with the smaller loader (best-effort).
                return train_one_epoch(
                    model,
                    new_loader,
                    optimizer,
                    criterion,
                    device,
                    mixup_alpha,
                    grad_clip,
                )
            raise

        total_loss += float(loss.item())
        n_batches += 1
    return total_loss / max(1, n_batches)
