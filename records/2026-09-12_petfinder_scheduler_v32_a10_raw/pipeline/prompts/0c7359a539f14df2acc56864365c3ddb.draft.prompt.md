def stratified_split(df, val_frac=0.2, seed=RANDOM_SEED):
    """Stratified (quantile-binned) train/validation index split preserving target dist."""
    rng = np.random.RandomState(seed)
    y = df["Pawpularity"].to_numpy(dtype=np.float64)
    # Quantile bins to stratify by target value.
    n_bins = max(1, min(10, len(y) // 50))
    try:
        edges = np.unique(np.quantile(y, np.linspace(0, 1, n_bins + 1)))
        bin_ids = np.digitize(y, edges[1:-1])
    except Exception:
        bin_ids = np.zeros(len(y), dtype=int)

    indices = np.arange(len(df))
    train_idx, val_idx = [], []
    for b in np.unique(bin_ids):
        mask = indices[bin_ids == b]
        rng.shuffle(mask)
        k = int(round(val_frac * len(mask)))
        k = min(max(k, 0), len(mask) - 1)
        val_idx.extend(mask[:k].tolist())
        train_idx.extend(mask[k:].tolist())

    train_idx = np.array(sorted(train_idx), dtype=int)
    val_idx = np.array(sorted(val_idx), dtype=int)
    return train_idx, val_idx

class HybridPawpularityModel(nn.Module):
    """Frozen ResNet-50 image encoder fused with 12 tabular flags -> scalar score."""

    def __init__(
        self,
        tab_dim=NUM_TABULAR,
        img_dim=2048,
        hidden=256,
        dropout=0.3,
        freeze_backbone=True,
    ):
        super().__init__()
        self.backbone = resnet50(weights=None)
        # Load local weights if available; otherwise keep random init (still trains head).
        w = _resolve_local_resnet50_weights()
        if w is not None:
            try:
                self.backbone.load_state_dict(w, strict=False)
            except Exception:
                pass
        if freeze_backbone:
            for p in self.backbone.parameters():
                p.requires_grad_(False)
            self.backbone.eval()

        # Freeze BN running stats during training too (backbone is static).
        for m in self.backbone.modules():
            if isinstance(m, (nn.BatchNorm2d,)):
                m.eval()

        self.tab_mlp = nn.Sequential(
            nn.Linear(tab_dim, 64),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
        )
        self.fuse = nn.Sequential(
            nn.Linear(img_dim + 64, hidden),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
            nn.Linear(hidden, 128),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout * 0.5),
            nn.Linear(128, 1),
        )

    def train(self, mode=True):
        # Keep backbone permanently in eval mode regardless of overall mode.
        super().train(mode)
        self.backbone.eval()
        return self

    def _image_features(self, x):
        feats = self.backbone(x)  # (B, 2048, 1, 1)
        feats = torch.flatten(feats, 1)  # (B, 2048)
        return feats

    def forward(self, image, tabular):
        img_feat = self._image_features(image)
        tab_feat = self.tab_mlp(tabular)
        z = torch.cat([img_feat, tab_feat], dim=1)
        out = self.fuse(z).squeeze(-1)  # (B,)
        return out

def collate_pawpularity(batch):
    """batch: list of (image_tensor, tabular_np, target_float). Returns stacked tensors."""
    imgs = torch.stack([b[0] for b in batch], dim=0)

    tgts = torch.tensor([b[2] for b in batch], dtype=torch.float32)
    return {"image": imgs, "tabular": tabs, "target": tgts}

class CandidateAdapter:
    """CPU-preflight adapter exercising the REAL model, loss, inputs, optimizer."""

    def __init__(self):
        self._model = None
        self._criterion = None

    def _merge_context(self, context):
        ctx = dict(_ADAPTER_DEFAULTS)
        if context:
            ctx.update({k: v for k, v in context.items() if v is not None})
        return ctx

    def build_model(self, context=None):
        ctx = self._merge_context(context)
        self._model = build_model(ctx)
        self._criterion = build_criterion(ctx)
        return self._model

    def build_optimizer(self, model, context=None):
        ctx = self._merge_context(context)
        params = [p for p in model.parameters() if p.requires_grad]
        opt = torch.optim.AdamW(
            params,
            lr=ctx.get("lr", 1e-3),
            weight_decay=ctx.get("weight_decay", 1e-4),
            betas=tuple(ctx.get("betas", (0.9, 0.999))),
        )
        return opt

    def _make_batch(self, scenario, device, n):
        bs = int(scenario.get("batch_size", 2))
        bs = max(1, min(bs, n))
        fixture = scenario.get("fixture", "") or ""
        if fixture:
            # Authoritative named input contract: honor declared non-batch shapes.
            image = torch.randn(bs, 3, IMAGE_SIZE, IMAGE_SIZE, device=device)
            tabular = torch.randn(bs, NUM_TABULAR, device=device)
            target = torch.rand(bs, device=device) * (TARGET_HI - TARGET_LO) + TARGET_LO
            return {"image": image, "tabular": tabular, "target": target}
        # Non-fixture: synthesize deterministic small batches (CPU-safe, no IO).
        g = torch.Generator().manual_seed(0)
        image = torch.randn(bs, 3, IMAGE_SIZE, IMAGE_SIZE, generator=g, device=device)
        tabular = (
            torch.randint(0, 2, (bs, NUM_TABULAR), generator=g, device=device)
        ).float()
        target = (
            torch.rand(bs, generator=g, device=device) * (TARGET_HI - TARGET_LO)
            + TARGET_LO
        )
        return {"image": image, "tabular": tabular, "target": target}

    def build_train_batch(self, scenario, device):
        n = int(scenario.get("num_samples", 2))
        return self._make_batch(scenario, device, max(n, 1))

    def build_validation_batch(self, scenario, device):
        n = int(scenario.get("num_samples", 2))
        return self._make_batch(scenario, device, max(n, 1))

    def _resolve_criterion(self, context):
        if self._criterion is None:
            self._criterion = build_criterion(context)
        return self._criterion

    def training_step(self, model, batch, context=None):
        crit = self._resolve_criterion(context)
        model.train()
        out = model(batch["image"], batch["tabular"])
        loss = crit(out, batch["target"])
        return loss

    def validation_step(self, model, batch, context=None):
        crit = self._resolve_criterion(context)
        model.eval()
        with torch.no_grad():
            out = model(batch["image"], batch["tabular"])
            loss = crit(out, batch["target"])
        return loss

def build_transforms(train=True):
    """ImageNet normalization; light geometric augmentation only when training."""
    base = [
        transforms.Resize((IMAGE_SIZE, IMAGE_SIZE)),
        transforms.ToTensor(),
        transforms.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225]),

            transforms.RandomHorizontalFlip(p=0.5),
            transforms.RandomRotation(degrees=10),

def tabular_vector(df_row):
    """Extract the 12 binary flags as a float32 vector in canonical order."""
    vals = [float(int(df_row[c])) for c in TABULAR_COLS]
    return np.array(vals, dtype=np.float32)


def stratified_split(df, val_frac=0.2, seed=RANDOM_SEED):
    """Stratified (quantile-binned) train/validation index split preserving target dist."""
    rng = np.random.RandomState(seed)
    y = df["Pawpularity"].to_numpy(dtype=np.float64)
    # Quantile bins to stratify by target value.
    n_bins = max(1, min(10, len(y) // 50))
    try:
        edges = np.unique(np.quantile(y, np.linspace(0, 1, n_bins + 1)))
        bin_ids = np.digitize(y, edges[1:-1])
    except Exception:
        bin_ids = np.zeros(len(y), dtype=int)

    indices = np.arange(len(df))
    train_idx, val_idx = [], []
    for b in np.unique(bin_ids):
        mask = indices[bin_ids == b]
        rng.shuffle(mask)
        k = int(round(val_frac * len(mask)))
        k = min(max(k, 0), len(mask) - 1)
        val_idx.extend(mask[:k].tolist())
        train_idx.extend(mask[k:].tolist())

    train_idx = np.array(sorted(train_idx), dtype=int)
    val_idx = np.array(sorted(val_idx), dtype=int)
    return train_idx, val_idx

class HybridPawpularityModel(nn.Module):
    """Frozen ResNet-50 image encoder fused with 12 tabular flags -> scalar score."""

    def __init__(
        self,
        tab_dim=NUM_TABULAR,
        img_dim=2048,
        hidden=256,
        dropout=0.3,
        freeze_backbone=True,
    ):
        super().__init__()
        self.backbone = resnet50(weights=None)
        # Load local weights if available; otherwise keep random init (still trains head).
        w = _resolve_local_resnet50_weights()
        if w is not None:
            try:
                self.backbone.load_state_dict(w, strict=False)
            except Exception:
                pass
        if freeze_backbone:
            for p in self.backbone.parameters():
                p.requires_grad_(False)
            self.backbone.eval()

        # Freeze BN running stats during training too (backbone is static).
        for m in self.backbone.modules():
            if isinstance(m, (nn.BatchNorm2d,)):
                m.eval()

        self.tab_mlp = nn.Sequential(
            nn.Linear(tab_dim, 64),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
        )
        self.fuse = nn.Sequential(
            nn.Linear(img_dim + 64, hidden),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
            nn.Linear(hidden, 128),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout * 0.5),
            nn.Linear(128, 1),
        )

    def train(self, mode=True):
        # Keep backbone permanently in eval mode regardless of overall mode.
        super().train(mode)
        self.backbone.eval()
        return self

    def _image_features(self, x):
        feats = self.backbone(x)  # (B, 2048, 1, 1)
        feats = torch.flatten(feats, 1)  # (B, 2048)
        return feats

    def forward(self, image, tabular):
        img_feat = self._image_features(image)
        tab_feat = self.tab_mlp(tabular)
        z = torch.cat([img_feat, tab_feat], dim=1)
        out = self.fuse(z).squeeze(-1)  # (B,)
        return out


def build_model(context=None):
    """Build the real training model (used by pipeline and adapter)."""
    ctx = dict(_ADAPTER_DEFAULTS)
    if context:
        ctx.update({k: v for k, v in context.items() if v is not None})
    model = HybridPawpularityModel(

        img_dim=ctx.get("img_dim", 2048),
        hidden=ctx.get("hidden", 256),
        dropout=ctx.get("dropout", 0.3),
        freeze_backbone=ctx.get("freeze_backbone", True),

def build_criterion(context=None):
    """MSE loss matches the RMSE evaluation metric."""
    return nn.MSELoss()


def clamp_predictions(preds):
    """Clamp raw scores to valid [0, 100] range (applied identically val & test)."""
    preds = np.asarray(preds, dtype=np.float64)
    return np.clip(preds, TARGET_LO, TARGET_HI)

def collate_pawpularity(batch):
    """batch: list of (image_tensor, tabular_np, target_float). Returns stacked tensors."""
    imgs = torch.stack([b[0] for b in batch], dim=0)
    tabs = torch.from_numpy(np.stack([b[1] for b in batch], axis=0)).float()
    tgts = torch.tensor([b[2] for b in batch], dtype=torch.float32)
    return {"image": imgs, "tabular": tabs, "target": tgts}

class PawpularityDataset(torch.utils.data.Dataset):
    def __init__(self, df, split, transform):
        self.df = df.reset_index(drop=True)
        self.split = split
        self.transform = transform

    def __len__(self):
        return len(self.df)

    def __getitem__(self, i):
        row = self.df.iloc[i]
        img_id = row["Id"]
        image = load_image_tensor(img_id, self.split, self.transform)
        tab = tabular_vector(row)
        target = float(row["Pawpularity"])
        return image, tab, target

class CandidateAdapter:
    """CPU-preflight adapter exercising the REAL model, loss, inputs, optimizer."""

    def __init__(self):
        self._model = None
        self._criterion = None

    def _merge_context(self, context):
        ctx = dict(_ADAPTER_DEFAULTS)
        if context:
            ctx.update({k: v for k, v in context.items() if v is not None})
        return ctx

    def build_model(self, context=None):
        ctx = self._merge_context(context)
        self._model = build_model(ctx)
        self._criterion = build_criterion(ctx)
        return self._model

    def build_optimizer(self, model, context=None):
        ctx = self._merge_context(context)
        params = [p for p in model.parameters() if p.requires_grad]
        opt = torch.optim.AdamW(
            params,
            lr=ctx.get("lr", 1e-3),
            weight_decay=ctx.get("weight_decay", 1e-4),
            betas=tuple(ctx.get("betas", (0.9, 0.999))),
        )
        return opt

    def _make_batch(self, scenario, device, n):
        bs = int(scenario.get("batch_size", 2))
        bs = max(1, min(bs, n))
        fixture = scenario.get("fixture", "") or ""
        if fixture:
            # Authoritative named input contract: honor declared non-batch shapes.
            image = torch.randn(bs, 3, IMAGE_SIZE, IMAGE_SIZE, device=device)
            tabular = torch.randn(bs, NUM_TABULAR, device=device)
            target = torch.rand(bs, device=device) * (TARGET_HI - TARGET_LO) + TARGET_LO
            return {"image": image, "tabular": tabular, "target": target}
        # Non-fixture: synthesize deterministic small batches (CPU-safe, no IO).
        g = torch.Generator().manual_seed(0)
        image = torch.randn(bs, 3, IMAGE_SIZE, IMAGE_SIZE, generator=g, device=device)
        tabular = (
            torch.randint(0, 2, (bs, NUM_TABULAR), generator=g, device=device)
        ).float()
        target = (
            torch.rand(bs, generator=g, device=device) * (TARGET_HI - TARGET_LO)
            + TARGET_LO
        )
        return {"image": image, "tabular": tabular, "target": target}

    def build_train_batch(self, scenario, device):
        n = int(scenario.get("num_samples", 2))
        return self._make_batch(scenario, device, max(n, 1))

    def build_validation_batch(self, scenario, device):
        n = int(scenario.get("num_samples", 2))
        return self._make_batch(scenario, device, max(n, 1))

    def _resolve_criterion(self, context):
        if self._criterion is None:
            self._criterion = build_criterion(context)
        return self._criterion

    def training_step(self, model, batch, context=None):
        crit = self._resolve_criterion(context)
        model.train()
        out = model(batch["image"], batch["tabular"])
        loss = crit(out, batch["target"])
        return loss

    def validation_step(self, model, batch, context=None):
        crit = self._resolve_criterion(context)
        model.eval()
        with torch.no_grad():
            out = model(batch["image"], batch["tabular"])
            loss = crit(out, batch["target"])
        return loss

        shuffle=False,
        num_workers=2,

def compute_rmse(y_true, y_pred):
    """Compute Root Mean Squared Error."""
    y_true = np.asarray(y_true, dtype=np.float64)
    y_pred = np.asarray(y_pred, dtype=np.float64)
    return float(np.sqrt(np.mean((y_true - y_pred) ** 2)))

def train_one_epoch(model, loader, criterion, optimizer, device):
    """Train for one epoch. Returns average loss."""
    model.train()
    total_loss = 0.0
    n_batches = 0
    for batch in loader:
        imgs = batch["image"].to(device, non_blocking=True)
        tabs = batch["tabular"].to(device, non_blocking=True)
        tgts = batch["target"].to(device, non_blocking=True)

        optimizer.zero_grad(set_to_none=True)
        out = model(imgs, tabs)
        loss = criterion(out, tgts)
        loss.backward()
        optimizer.step()

        total_loss += loss.item()
        n_batches += 1
    return total_loss / max(n_batches, 1)
