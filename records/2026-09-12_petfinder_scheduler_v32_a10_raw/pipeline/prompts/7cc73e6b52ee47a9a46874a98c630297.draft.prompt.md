class TabularFeatureBuilder:
    """Encodes the 12 binary metadata columns into a normalized float vector.

    Fit statistics (mean/std) are computed on the training set only. Because the

    that are known to correlate with pawpularity (e.g. face AND eyes AND near)

    def fit(self, df: pd.DataFrame) -> "TabularFeatureBuilder":
        X = df[self.columns].to_numpy(dtype=np.float32)
        self.mean_ = X.mean(axis=0)
        std = X.std(axis=0)

        return self

    def transform(self, df: pd.DataFrame) -> np.ndarray:
        if self.mean_ is None:
            raise RuntimeError("TabularFeatureBuilder must be fit before transform.")
        X = df[self.columns].to_numpy(dtype=np.float32)
        return ((X - self.mean_) / self.std_).astype(np.float32)

    def num_features(self) -> int:

# Rank-aware target transform (fit on TRAIN only)
# ----------------------------------------------------------------------------
class RankTargetTransform:
    """Map raw Pawpularity (0..100) to a rank-normalized value in [0,1] for
    training, and provide the inverse (quantile) mapping back to [0,100].

    Rationale: the score distribution is non-uniform; ranking makes the loss
    robust to the dense middle band and the sparse extremes, which empirically
    lowers true-scale RMSE after inverse mapping.
    """

    def __init__(self, n_bins: int = 1000):
        self.n_bins = n_bins
        self._inv_table: Optional[np.ndarray] = None  # rank->score lookup

    def fit(self, y_raw: np.ndarray) -> "RankTargetTransform":
        y_raw = np.asarray(y_raw, dtype=np.float64)
        order = np.argsort(y_raw, kind="stable")
        ranks = np.empty_like(order, dtype=np.float64)
        ranks[order] = np.arange(len(y_raw))
        # store per-sample rank for later transform; build inverse table
        self._ranks = ranks
        # Build inverse quantile table: for k in [0, n_bins) pick the score whose
        # empirical CDF crosses k/n_bins.
        sorted_scores = y_raw[order]
        idx = (np.arange(self.n_bins) / self.n_bins) * len(sorted_scores)
        idx = np.clip(idx.astype(int), 0, len(sorted_scores) - 1)
        self._inv_table = sorted_scores[idx]
        return self

    def transform(self, y_raw: np.ndarray) -> np.ndarray:
        """Convert raw scores to rank-normalized [0,1] using fitted ranks."""
        y_raw = np.asarray(y_raw, dtype=np.float64)
        # Use searchsorted against sorted unique scores to get a stable rank.
        if not hasattr(self, "_sorted_unique"):
            self._sorted_unique = np.unique(self._y_sorted)
        pos = np.searchsorted(self._y_sorted, y_raw, side="left")
        denom = max(1, len(self._y_sorted) - 1)
        return (pos / denom).astype(np.float32)

    def inverse(self, y_norm: np.ndarray) -> np.ndarray:
        """Map rank-normalized [0,1] values back to the 0..100 scale."""
        y_norm = np.clip(np.asarray(y_norm, dtype=np.float64), 0.0, 1.0)
        if self._inv_table is None:
            raise RuntimeError("RankTargetTransform must be fit before inverse.")
        idx = (y_norm * (len(self._inv_table) - 1)).astype(int)
        idx = np.clip(idx, 0, len(self._inv_table) - 1)
        out = self._inv_table[idx]
        return np.clip(out, 0.0, 100.0).astype(np.float32)

    def _finalize_fit(self, y_raw):
        self._y_sorted = np.sort(np.asarray(y_raw, dtype=np.float64))

            self.qkv(x)
            .reshape(b, n, 3, self.heads, c // self.heads)
            .permute(2, 0, 3, 1, 4)
        )
        q, k, v = qkv.unbind(0)
        attn = (q @ k.transpose(-2, -1)) * self.scale
        attn = attn.softmax(dim=-1)
        attn = self.drop(attn)
        out = (attn @ v).transpose(1, 2).reshape(b, n, c)
        return self.proj(out)


class _MLPBlock(nn.Module):
    def __init__(self, dim, mlp_ratio=4, drop=0.0):
        super().__init__()
        hidden = int(dim * mlp_ratio)
        self.fc1 = nn.Linear(dim, hidden)
        self.act = nn.GELU()
        self.fc2 = nn.Linear(hidden, dim)
        self.drop = nn.Dropout(drop)

    def forward(self, x):
        return self.drop(self.fc2(self.act(self.fc1(x))))


class _BasicBlock(nn.Module):
    def __init__(self, dim, heads=6, drop=0.0):
        super().__init__()
        self.norm1 = nn.LayerNorm(dim)
        self.attn = _WindowAttention(dim, heads, drop)
        self.norm2 = nn.LayerNorm(dim)
        self.mlp = _MLPBlock(dim, drop=drop)

    def forward(self, x):
        x = x + self.attn(self.norm1(x))
        x = x + self.mlp(self.norm2(x))
        return x


class _LocalSwinTiny(nn.Module):
    """Compact offline fallback that mimics Swin's hierarchical downsampling."""

    def __init__(self, encoder_dim=768):
        super().__init__()
        self.patch_embed = _PatchEmbed(patch=4, dim=96)
        self.stages = nn.ModuleList(
            [
                nn.Sequential(*[_BasicBlock(96, heads=3) for _ in range(2)]),
                nn.Sequential(*[_BasicBlock(192, heads=6) for _ in range(2)]),
                nn.Sequential(*[_BasicBlock(384, heads=12) for _ in range(2)]),
            ]
        )
        self.downsample = nn.ModuleList(
            [
                nn.Sequential(nn.LayerNorm(96), nn.Linear(96, 192)),
                nn.Sequential(nn.LayerNorm(192), nn.Linear(192, 384)),
            ]
        )
        self.encoder_dim = encoder_dim
        self.pool = nn.AdaptiveAvgPool1d(1)

    def forward(self, x):
        x, (h, w) = self.patch_embed(x)
        n_tokens = h * w
        feats = []
        for i, stage in enumerate(self.stages):
            x = stage(x)
            if i < len(self.downsample):
                # average-pool 2x2 spatially then project channels
                b, _, c = x.shape
                hh, ww = int(math.sqrt(n_tokens / (4**i))), int(
                    math.sqrt(n_tokens / (4**i))
                )
                xp = x.reshape(b, hh, ww, c).permute(0, 3, 1, 2)
                xp = F.avg_pool2d(xp, 2).flatten(2).transpose(1, 2)
                x = self.downsample[i](xp)
                n_tokens //= 4
            feats.append(x.mean(dim=1))
        out = torch.cat(feats, dim=1)
        out = F.adaptive_avg_pool1d(out.unsqueeze(1), 1).squeeze(1)
        return out


class GatedCrossAttentionFusion(nn.Module):
    """Fuse a tabular embedding into a vision feature via cross-attention with a
    learned gate (so the model can ignore metadata when unhelpful)."""

    def __init__(
        self, vis_dim: int, tab_dim: int, ffn_dim: int = 256, drop: float = 0.1
    ):
        super().__init__()
        self.tab_proj = nn.Linear(tab_dim, vis_dim)
        self.vis_proj = nn.Linear(vis_dim, vis_dim)
        self.attn = nn.MultiheadAttention(
            embed_dim=vis_dim, num_heads=4, dropout=drop, batch_first=True
        )
        self.norm = nn.LayerNorm(vis_dim)
        self.ffn = nn.Sequential(
            nn.Linear(vis_dim, ffn_dim),
            nn.GELU(),
            nn.Dropout(drop),
            nn.Linear(ffn_dim, vis_dim),
        )
        self.gate = nn.Sequential(nn.Linear(vis_dim * 2, 1), nn.Sigmoid())

    def forward(self, vis_feat: torch.Tensor, tab_feat: torch.Tensor) -> torch.Tensor:
        # vis_feat: B,D ; tab_feat: B,T
        q = self.vis_proj(vis_feat).unsqueeze(1)  # B,1,D
        kv = self.tab_proj(tab_feat).unsqueeze(1)  # B,1,D
        attn_out, _ = self.attn(q, kv, kv)  # B,1,D
        attn_out = attn_out.squeeze(1)
        fused = self.norm(vis_feat + attn_out)
        fused = fused + self.ffn(fused)
        g = self.gate(torch.cat([vis_feat, tab_feat], dim=1))  # B,1
        return vis_feat + g * (fused - vis_feat)


class PetModel(nn.Module):
    """Swin-Tiny image encoder + gated cross-attention tabular fusion -> scalar."""

    def __init__(self, tab_dim: int = NUM_TABULAR, dropout: float = 0.2):
        super().__init__()
        self.backbone = _load_swin_tiny_offline(num_classes=0)
        # Determine backbone output dim robustly.
        with torch.no_grad():
            probe = torch.zeros(1, 3, IMAGE_SIZE, IMAGE_SIZE)
            vis_dim = self.backbone(probe).shape[-1]
        self.vis_dim = vis_dim
        self.fusion = GatedCrossAttentionFusion(vis_dim, tab_dim, ffn_dim=256)
        self.head = nn.Sequential(
            nn.LayerNorm(vis_dim),
            nn.Linear(vis_dim, 128),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(128, 1),
        )

    def forward(self, image: torch.Tensor, tabular: torch.Tensor) -> torch.Tensor:
        vis_feat = self.backbone(image)  # B,D
        if vis_feat.dim() == 3:  # some encoders emit B,N,D
            vis_feat = vis_feat.mean(dim=1)
        fused = self.fusion(vis_feat, tabular)
        out = self.head(fused).squeeze(-1)  # B
        return out

# Criterion: rank-aware Huber (robust to the non-uniform score distribution)
# ----------------------------------------------------------------------------
class RankAwareLoss(nn.Module):
    """Huber loss applied to rank-normalized targets (computed externally).

    We accept already rank-normalized targets here; the inverse mapping back to
    [0,100] happens at inference/evaluation time, keeping training robust.
    """

    def __init__(self, delta: float = 0.5):
        super().__init__()
        self.delta = delta

    def forward(self, pred: torch.Tensor, target: torch.Tensor) -> torch.Tensor:
        # pred/target expected on the same (rank-normalized) scale in [0,1].
        err = (pred - target).abs()
        quad = F.smooth_l1_loss(pred, target, reduction="none", beta=self.delta)
        return quad.mean()

# Output interface: clamp to [0,100]
# ----------------------------------------------------------------------------
def to_score_scale(logits: torch.Tensor) -> torch.Tensor:
    """Map a model's raw logit to the [0,100] Pawpularity scale.

    The network is trained to output a rank-normalized value in [0,1]; we first
    squash with sigmoid (numerically safe) then scale to [0,100]. Final clamping
    enforces the valid range.
    """
    p = torch.sigmoid(logits)  # -> [0,1]
    return torch.clamp(p * 100.0, 0.0, 100.0)

# Reproducible split (stratified by decile of the target to preserve the
# non-uniform distribution in both train and validation sets).

def make_split(
    df: pd.DataFrame, val_fraction: float = VAL_FRACTION, seed: int = SEED
) -> Tuple[pd.DataFrame, pd.DataFrame]:
    rng = np.random.default_rng(seed)
    decile = pd.cut(df["Pawpularity"], bins=10, labels=False, duplicates="drop")
    train_idx, val_idx = [], []
    for d in sorted(decile.dropna().unique()):
        group = df.index[decile == d].tolist()
        rng.shuffle(group)
        n_val = max(1, int(round(len(group) * val_fraction)))
        val_idx.extend(group[:n_val])
        train_idx.extend(group[n_val:])
    return df.loc[train_idx].reset_index(drop=True), df.loc[val_idx].reset_index(
        drop=True
    )

class CandidateAdapter:
    """No-argument adapter exercising the REAL model, loss, and optimizer.

    Batch builders honor scenario['batch_size'] and device, and locate input
    fixtures via MLEVOLVE_INPUT_DIR. When scenario['fixture'] is provided it is
    the authoritative named input contract; we build tensors with the declared
    shapes and pass them through the real forward path.
    """

    DEFAULT_CONTEXT = {
        "device": DEVICE,
        "image_size": IMAGE_SIZE,
        "num_tabular": NUM_TABULAR,
        "dropout": 0.2,
        "huber_delta": 0.5,
        "lr": 1e-4,
        "weight_decay": 1e-4,
    }

    def __init__(self):
        self._model = None
        self._criterion = None
        self._optimizer = None

    def _ctx(self, context: Optional[dict]) -> dict:
        merged = dict(self.DEFAULT_CONTEXT)
        if context:
            merged.update({k: v for k, v in context.items() if v is not None})
        return merged

    def build_model(self, context=None):
        ctx = self._ctx(context)
        model = PetModel(tab_dim=ctx["num_tabular"], dropout=ctx["dropout"])
        self._model = model
        self._criterion = RankAwareLoss(delta=ctx["huber_delta"])
        return model

    def build_optimizer(self, model, context=None):
        ctx = self._ctx(context)
        opt = torch.optim.AdamW(
            model.parameters(), lr=ctx["lr"], weight_decay=ctx["weight_decay"]
        )
        self._optimizer = opt
        return opt

    def _make_batch(self, scenario: dict, device: str, split: str):
        bs = int(scenario.get("batch_size", 2))
        fixture = scenario.get("fixture")
        if fixture:
            # Authoritative named input contract: build each declared tensor.
            out = {}
            for name, shape in fixture.items():
                full = [bs] + list(shape)
                out[name] = torch.randn(full, device=device)
            # Ensure the model receives image + tabular explicitly.
            if "image" not in out:
                out["image"] = torch.randn(bs, 3, IMAGE_SIZE, IMAGE_SIZE, device=device)
            if "tabular" not in out:
                out["tabular"] = torch.randn(bs, NUM_TABULAR, device=device)
            if "target" not in out:
                out["target"] = torch.rand(bs, device=device)
            return out
        # Otherwise synthesize a minimal realistic batch on-device.
        return {
            "image": torch.randn(bs, 3, IMAGE_SIZE, IMAGE_SIZE, device=device),
            "tabular": torch.randn(bs, NUM_TABULAR, device=device),
            "target": torch.rand(bs, device=device),  # rank-normalized proxy
        }

    def build_train_batch(self, scenario, device):
        return self._make_batch(scenario, device, "train")

    def build_validation_batch(self, scenario, device):
        return self._make_batch(scenario, device, "val")

    def training_step(self, model, batch, context=None):
        ctx = self._ctx(context)
        if self._criterion is None:
            self._criterion = RankAwareLoss(delta=ctx["huber_delta"])
        model.train()
        images = batch["image"].to(ctx["device"])
        tabular = batch["tabular"].to(ctx["device"])
        target = batch["target"].to(ctx["device"])
        logits = model(images, tabular)
        loss = self._criterion(logits, target)
        return loss

    def validation_step(self, model, batch, context=None):
        ctx = self._ctx(context)
        if self._criterion is None:
            self._criterion = RankAwareLoss(delta=ctx["huber_delta"])
        model.eval()
        with torch.no_grad():
            images = batch["image"].to(ctx["device"])
            tabular = batch["tabular"].to(ctx["device"])
            target = batch["target"].to(ctx["device"])
            logits = model(images, tabular)
            loss = self._criterion(logits, target)
        return loss

# Feature engineering (fit on TRAIN only to avoid leakage)

class TabularFeatureBuilder:
    """Encodes the 12 binary metadata columns into a normalized float vector.

    Fit statistics (mean/std) are computed on the training set only. Because the
    raw columns are already binary, we add a couple of derived interaction terms
    that are known to correlate with pawpularity (e.g. face AND eyes AND near)
    but keep the core 12-dim contract required by the preflight fixture.
    """

    def __init__(self, columns: List[str] = TABULAR_COLUMNS):
        self.columns = list(columns)
        self.mean_: Optional[np.ndarray] = None
        self.std_: Optional[np.ndarray] = None

    def fit(self, df: pd.DataFrame) -> "TabularFeatureBuilder":
        X = df[self.columns].to_numpy(dtype=np.float32)
        self.mean_ = X.mean(axis=0)
        std = X.std(axis=0)
        std[std < 1e-6] = 1.0
        self.std_ = std
        return self

    def transform(self, df: pd.DataFrame) -> np.ndarray:
        if self.mean_ is None:
            raise RuntimeError("TabularFeatureBuilder must be fit before transform.")
        X = df[self.columns].to_numpy(dtype=np.float32)
        return ((X - self.mean_) / self.std_).astype(np.float32)

    @property
    def num_features(self) -> int:
        return len(self.columns)

def _build_image_transforms(train: bool):
    """Build torchvision-compatible transform pipelines without importing
    torchvision.transforms at module import time (keeps import light & safe)."""
    from torchvision import transforms

    base = [transforms.Resize((IMAGE_SIZE, IMAGE_SIZE)), transforms.ToTensor()]
    norm = [transforms.Normalize(mean=IMAGENET_MEAN, std=IMAGENET_STD)]
    if train:
        aug = [
            transforms.RandomHorizontalFlip(p=0.5),
            transforms.ColorJitter(
                brightness=0.2, contrast=0.2, saturation=0.2, hue=0.05
            ),
            transforms.RandomAffine(
                degrees=10, translate=(0.05, 0.05), scale=(0.9, 1.1)
            ),
        ]
        pipeline = base + aug + norm
    else:
        pipeline = base + norm
    return transforms.Compose(pipeline)

# ----------------------------------------------------------------------------
# Rank-aware target transform (fit on TRAIN only)
# ----------------------------------------------------------------------------
class RankTargetTransform:
    """Map raw Pawpularity (0..100) to a rank-normalized value in [0,1] for
    training, and provide the inverse (quantile) mapping back to [0,100].

    Rationale: the score distribution is non-uniform; ranking makes the loss
    robust to the dense middle band and the sparse extremes, which empirically
    lowers true-scale RMSE after inverse mapping.
    """

    def __init__(self, n_bins: int = 1000):
        self.n_bins = n_bins
        self._inv_table: Optional[np.ndarray] = None  # rank->score lookup

    def fit(self, y_raw: np.ndarray) -> "RankTargetTransform":
        y_raw = np.asarray(y_raw, dtype=np.float64)
        order = np.argsort(y_raw, kind="stable")
        ranks = np.empty_like(order, dtype=np.float64)
        ranks[order] = np.arange(len(y_raw))
        # store per-sample rank for later transform; build inverse table
        self._ranks = ranks
        # Build inverse quantile table: for k in [0, n_bins) pick the score whose
        # empirical CDF crosses k/n_bins.
        sorted_scores = y_raw[order]
        idx = (np.arange(self.n_bins) / self.n_bins) * len(sorted_scores)
        idx = np.clip(idx.astype(int), 0, len(sorted_scores) - 1)
        self._inv_table = sorted_scores[idx]
        return self

    def transform(self, y_raw: np.ndarray) -> np.ndarray:
        """Convert raw scores to rank-normalized [0,1] using fitted ranks."""
        y_raw = np.asarray(y_raw, dtype=np.float64)
        # Use searchsorted against sorted unique scores to get a stable rank.
        if not hasattr(self, "_sorted_unique"):
            self._sorted_unique = np.unique(self._y_sorted)
        pos = np.searchsorted(self._y_sorted, y_raw, side="left")
        denom = max(1, len(self._y_sorted) - 1)
        return (pos / denom).astype(np.float32)

    def inverse(self, y_norm: np.ndarray) -> np.ndarray:
        """Map rank-normalized [0,1] values back to the 0..100 scale."""
        y_norm = np.clip(np.asarray(y_norm, dtype=np.float64), 0.0, 1.0)
        if self._inv_table is None:
            raise RuntimeError("RankTargetTransform must be fit before inverse.")
        idx = (y_norm * (len(self._inv_table) - 1)).astype(int)
        idx = np.clip(idx, 0, len(self._inv_table) - 1)
        out = self._inv_table[idx]
        return np.clip(out, 0.0, 100.0).astype(np.float32)

    def _finalize_fit(self, y_raw):
        self._y_sorted = np.sort(np.asarray(y_raw, dtype=np.float64))

def _load_swin_tiny_offline(num_classes: int = 0) -> nn.Module:
    """Return a Swin-Tiny encoder (feature extractor, no classification head).

    Offline-first: try timm local cache, fall back to a compact locally-defined
    Swin-like hierarchical transformer so the script always runs without network.
    """
    try:
        import timm

        model = timm.create_model(
            "swin_tiny_patch4_window7_224",
            pretrained=False,  # never download weights
            num_classes=num_classes,
        )
        return model
    except Exception:
        return _LocalSwinTiny(encoder_dim=768)


class _PatchEmbed(nn.Module):
    def __init__(self, patch=4, dim=96):
        super().__init__()
        self.proj = nn.Conv2d(3, dim, kernel_size=patch, stride=patch)
        self.norm = nn.LayerNorm(dim)

    def forward(self, x):
        x = self.proj(x)  # B,C,H,W
        b, c, h, w = x.shape
        x = x.flatten(2).transpose(1, 2)  # B,HW,C
        x = self.norm(x)
        return x, (h, w)


class _WindowAttention(nn.Module):
    """Small global-attention stand-in (windowed MHA approximated globally)."""

    def __init__(self, dim, heads=6, drop=0.0):
        super().__init__()
        self.heads = heads
        self.scale = (dim // heads) ** -0.5
        self.qkv = nn.Linear(dim, dim * 3)
        self.proj = nn.Linear(dim, dim)
        self.drop = nn.Dropout(drop)

    def forward(self, x):
        b, n, c = x.shape
        qkv = (
            self.qkv(x)
            .reshape(b, n, 3, self.heads, c // self.heads)
            .permute(2, 0, 3, 1, 4)
        )
        q, k, v = qkv.unbind(0)
        attn = (q @ k.transpose(-2, -1)) * self.scale
        attn = attn.softmax(dim=-1)
        attn = self.drop(attn)
        out = (attn @ v).transpose(1, 2).reshape(b, n, c)
        return self.proj(out)


class _MLPBlock(nn.Module):
    def __init__(self, dim, mlp_ratio=4, drop=0.0):
        super().__init__()
        hidden = int(dim * mlp_ratio)
        self.fc1 = nn.Linear(dim, hidden)
        self.act = nn.GELU()
        self.fc2 = nn.Linear(hidden, dim)
        self.drop = nn.Dropout(drop)

    def forward(self, x):
        return self.drop(self.fc2(self.act(self.fc1(x))))


class _BasicBlock(nn.Module):
    def __init__(self, dim, heads=6, drop=0.0):
        super().__init__()
        self.norm1 = nn.LayerNorm(dim)
        self.attn = _WindowAttention(dim, heads, drop)
        self.norm2 = nn.LayerNorm(dim)
        self.mlp = _MLPBlock(dim, drop=drop)

    def forward(self, x):
        x = x + self.attn(self.norm1(x))
        x = x + self.mlp(self.norm2(x))
        return x


class _LocalSwinTiny(nn.Module):
    """Compact offline fallback that mimics Swin's hierarchical downsampling."""

    def __init__(self, encoder_dim=768):
        super().__init__()
        self.patch_embed = _PatchEmbed(patch=4, dim=96)
        self.stages = nn.ModuleList(
            [
                nn.Sequential(*[_BasicBlock(96, heads=3) for _ in range(2)]),
                nn.Sequential(*[_BasicBlock(192, heads=6) for _ in range(2)]),
                nn.Sequential(*[_BasicBlock(384, heads=12) for _ in range(2)]),
            ]
        )
        self.downsample = nn.ModuleList(
            [
                nn.Sequential(nn.LayerNorm(96), nn.Linear(96, 192)),
                nn.Sequential(nn.LayerNorm(192), nn.Linear(192, 384)),
            ]
        )
        self.encoder_dim = encoder_dim
        self.pool = nn.AdaptiveAvgPool1d(1)

    def forward(self, x):
        x, (h, w) = self.patch_embed(x)
        n_tokens = h * w
        feats = []
        for i, stage in enumerate(self.stages):
            x = stage(x)
            if i < len(self.downsample):
                # average-pool 2x2 spatially then project channels
                b, _, c = x.shape
                hh, ww = int(math.sqrt(n_tokens / (4**i))), int(
                    math.sqrt(n_tokens / (4**i))
                )
                xp = x.reshape(b, hh, ww, c).permute(0, 3, 1, 2)
                xp = F.avg_pool2d(xp, 2).flatten(2).transpose(1, 2)
                x = self.downsample[i](xp)
                n_tokens //= 4
            feats.append(x.mean(dim=1))
        out = torch.cat(feats, dim=1)
        out = F.adaptive_avg_pool1d(out.unsqueeze(1), 1).squeeze(1)
        return out


class GatedCrossAttentionFusion(nn.Module):
    """Fuse a tabular embedding into a vision feature via cross-attention with a
    learned gate (so the model can ignore metadata when unhelpful)."""

    def __init__(
        self, vis_dim: int, tab_dim: int, ffn_dim: int = 256, drop: float = 0.1
    ):
        super().__init__()
        self.tab_proj = nn.Linear(tab_dim, vis_dim)
        self.vis_proj = nn.Linear(vis_dim, vis_dim)
        self.attn = nn.MultiheadAttention(
            embed_dim=vis_dim, num_heads=4, dropout=drop, batch_first=True
        )
        self.norm = nn.LayerNorm(vis_dim)
        self.ffn = nn.Sequential(
            nn.Linear(vis_dim, ffn_dim),
            nn.GELU(),
            nn.Dropout(drop),
            nn.Linear(ffn_dim, vis_dim),
        )
        self.gate = nn.Sequential(nn.Linear(vis_dim * 2, 1), nn.Sigmoid())

    def forward(self, vis_feat: torch.Tensor, tab_feat: torch.Tensor) -> torch.Tensor:
        # vis_feat: B,D ; tab_feat: B,T
        q = self.vis_proj(vis_feat).unsqueeze(1)  # B,1,D
        kv = self.tab_proj(tab_feat).unsqueeze(1)  # B,1,D
        attn_out, _ = self.attn(q, kv, kv)  # B,1,D
        attn_out = attn_out.squeeze(1)
        fused = self.norm(vis_feat + attn_out)
        fused = fused + self.ffn(fused)
        g = self.gate(torch.cat([vis_feat, tab_feat], dim=1))  # B,1
        return vis_feat + g * (fused - vis_feat)


class PetModel(nn.Module):
    """Swin-Tiny image encoder + gated cross-attention tabular fusion -> scalar."""

    def __init__(self, tab_dim: int = NUM_TABULAR, dropout: float = 0.2):
        super().__init__()
        self.backbone = _load_swin_tiny_offline(num_classes=0)
        # Determine backbone output dim robustly.
        with torch.no_grad():
            probe = torch.zeros(1, 3, IMAGE_SIZE, IMAGE_SIZE)
            vis_dim = self.backbone(probe).shape[-1]
        self.vis_dim = vis_dim
        self.fusion = GatedCrossAttentionFusion(vis_dim, tab_dim, ffn_dim=256)
        self.head = nn.Sequential(
            nn.LayerNorm(vis_dim),
            nn.Linear(vis_dim, 128),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(128, 1),
        )

    def forward(self, image: torch.Tensor, tabular: torch.Tensor) -> torch.Tensor:
        vis_feat = self.backbone(image)  # B,D
        if vis_feat.dim() == 3:  # some encoders emit B,N,D
            vis_feat = vis_feat.mean(dim=1)
        fused = self.fusion(vis_feat, tabular)
        out = self.head(fused).squeeze(-1)  # B
        return out


# ----------------------------------------------------------------------------
# Criterion: rank-aware Huber (robust to the non-uniform score distribution)
# ----------------------------------------------------------------------------
class RankAwareLoss(nn.Module):
    """Huber loss applied to rank-normalized targets (computed externally).

    We accept already rank-normalized targets here; the inverse mapping back to
    [0,100] happens at inference/evaluation time, keeping training robust.
    """

    def __init__(self, delta: float = 0.5):
        super().__init__()
        self.delta = delta

    def forward(self, pred: torch.Tensor, target: torch.Tensor) -> torch.Tensor:
        # pred/target expected on the same (rank-normalized) scale in [0,1].
        err = (pred - target).abs()
        quad = F.smooth_l1_loss(pred, target, reduction="none", beta=self.delta)
        return quad.mean()


# ----------------------------------------------------------------------------
# Output interface: clamp to [0,100]
# ----------------------------------------------------------------------------
def to_score_scale(logits: torch.Tensor) -> torch.Tensor:
    """Map a model's raw logit to the [0,100] Pawpularity scale.

    The network is trained to output a rank-normalized value in [0,1]; we first
    squash with sigmoid (numerically safe) then scale to [0,100]. Final clamping
    enforces the valid range.
    """
    p = torch.sigmoid(logits)  # -> [0,1]
    return torch.clamp(p * 100.0, 0.0, 100.0)

# ----------------------------------------------------------------------------
# Reproducible split (stratified by decile of the target to preserve the
# non-uniform distribution in both train and validation sets).
# ----------------------------------------------------------------------------
def make_split(
    df: pd.DataFrame, val_fraction: float = VAL_FRACTION, seed: int = SEED
) -> Tuple[pd.DataFrame, pd.DataFrame]:
    rng = np.random.default_rng(seed)
    decile = pd.cut(df["Pawpularity"], bins=10, labels=False, duplicates="drop")
    train_idx, val_idx = [], []
    for d in sorted(decile.dropna().unique()):
        group = df.index[decile == d].tolist()
        rng.shuffle(group)
        n_val = max(1, int(round(len(group) * val_fraction)))
        val_idx.extend(group[:n_val])
        train_idx.extend(group[n_val:])
    return df.loc[train_idx].reset_index(drop=True), df.loc[val_idx].reset_index(
        drop=True
    )

class CandidateAdapter:
    """No-argument adapter exercising the REAL model, loss, and optimizer.

    Batch builders honor scenario['batch_size'] and device, and locate input
    fixtures via MLEVOLVE_INPUT_DIR. When scenario['fixture'] is provided it is
    the authoritative named input contract; we build tensors with the declared
    shapes and pass them through the real forward path.
    """

    DEFAULT_CONTEXT = {
        "device": DEVICE,
        "image_size": IMAGE_SIZE,
        "num_tabular": NUM_TABULAR,
        "dropout": 0.2,
        "huber_delta": 0.5,
        "lr": 1e-4,
        "weight_decay": 1e-4,
    }

    def __init__(self):
        self._model = None
        self._criterion = None
        self._optimizer = None

    def _ctx(self, context: Optional[dict]) -> dict:
        merged = dict(self.DEFAULT_CONTEXT)
        if context:
            merged.update({k: v for k, v in context.items() if v is not None})
        return merged

    def build_model(self, context=None):
        ctx = self._ctx(context)
        model = PetModel(tab_dim=ctx["num_tabular"], dropout=ctx["dropout"])
        self._model = model
        self._criterion = RankAwareLoss(delta=ctx["huber_delta"])
        return model

    def build_optimizer(self, model, context=None):
        ctx = self._ctx(context)
        opt = torch.optim.AdamW(
            model.parameters(), lr=ctx["lr"], weight_decay=ctx["weight_decay"]
        )
        self._optimizer = opt
        return opt

    def _make_batch(self, scenario: dict, device: str, split: str):
        bs = int(scenario.get("batch_size", 2))
        fixture = scenario.get("fixture")
        if fixture:
            # Authoritative named input contract: build each declared tensor.
            out = {}
            for name, shape in fixture.items():
                full = [bs] + list(shape)
                out[name] = torch.randn(full, device=device)
            # Ensure the model receives image + tabular explicitly.
            if "image" not in out:
                out["image"] = torch.randn(bs, 3, IMAGE_SIZE, IMAGE_SIZE, device=device)
            if "tabular" not in out:
                out["tabular"] = torch.randn(bs, NUM_TABULAR, device=device)
            if "target" not in out:
                out["target"] = torch.rand(bs, device=device)
            return out
        # Otherwise synthesize a minimal realistic batch on-device.
        return {
            "image": torch.randn(bs, 3, IMAGE_SIZE, IMAGE_SIZE, device=device),
            "tabular": torch.randn(bs, NUM_TABULAR, device=device),
            "target": torch.rand(bs, device=device),  # rank-normalized proxy
        }

    def build_train_batch(self, scenario, device):
        return self._make_batch(scenario, device, "train")

    def build_validation_batch(self, scenario, device):
        return self._make_batch(scenario, device, "val")

    def training_step(self, model, batch, context=None):
        ctx = self._ctx(context)
        if self._criterion is None:
            self._criterion = RankAwareLoss(delta=ctx["huber_delta"])
        model.train()
        images = batch["image"].to(ctx["device"])
        tabular = batch["tabular"].to(ctx["device"])
        target = batch["target"].to(ctx["device"])
        logits = model(images, tabular)
        loss = self._criterion(logits, target)
        return loss

    def validation_step(self, model, batch, context=None):
        ctx = self._ctx(context)
        if self._criterion is None:
            self._criterion = RankAwareLoss(delta=ctx["huber_delta"])
        model.eval()
        with torch.no_grad():
            images = batch["image"].to(ctx["device"])
            tabular = batch["tabular"].to(ctx["device"])
            target = batch["target"].to(ctx["device"])
            logits = model(images, tabular)
            loss = self._criterion(logits, target)
        return loss

class TabularFeatureBuilder:
    def __init__(self, columns: List[str] = TABULAR_COLUMNS):
        self.columns = list(columns)
        self.mean_: Optional[np.ndarray] = None
        self.std_: Optional[np.ndarray] = None

    def fit(self, df: pd.DataFrame) -> "TabularFeatureBuilder":
        X = df[self.columns].to_numpy(dtype=np.float32)
        self.mean_ = X.mean(axis=0)
        std = X.std(axis=0)
        std[std < 1e-6] = 1.0
        self.std_ = std
        return self

    def transform(self, df: pd.DataFrame) -> np.ndarray:
        if self.mean_ is None:
            raise RuntimeError("TabularFeatureBuilder must be fit before transform.")
        X = df[self.columns].to_numpy(dtype=np.float32)
        return ((X - self.mean_) / self.std_).astype(np.float32)

    @property
    def num_features(self) -> int:
        return len(self.columns)


class RankTargetTransform:
    """Rank-normalize raw scores for training; inverse-map back to [0,100]."""

    def __init__(self, n_bins: int = 1000):
        self.n_bins = n_bins
        self._inv_table: Optional[np.ndarray] = None
        self._y_sorted: Optional[np.ndarray] = None

    def fit(self, y_raw: np.ndarray) -> "RankTargetTransform":
        y_raw = np.asarray(y_raw, dtype=np.float64)
        order = np.argsort(y_raw, kind="stable")
        self._y_sorted = y_raw[order]
        idx = (np.arange(self.n_bins) / self.n_bins) * len(self._y_sorted)
        idx = np.clip(idx.astype(int), 0, len(self._y_sorted) - 1)
        self._inv_table = self._y_sorted[idx]
        return self

    def transform(self, y_raw: np.ndarray) -> np.ndarray:
        y_raw = np.asarray(y_raw, dtype=np.float64)
        if self._y_sorted is None:
            raise RuntimeError("RankTargetTransform must be fit before transform.")
        pos = np.searchsorted(self._y_sorted, y_raw, side="left")
        denom = max(1, len(self._y_sorted) - 1)
        return (pos / denom).astype(np.float32)

    def inverse(self, y_norm: np.ndarray) -> np.ndarray:
        y_norm = np.clip(np.asarray(y_norm, dtype=np.float64), 0.0, 1.0)
        if self._inv_table is None:
            raise RuntimeError("RankTargetTransform must be fit before inverse.")
        idx = (y_norm * (len(self._inv_table) - 1)).astype(int)
        idx = np.clip(idx, 0, len(self._inv_table) - 1)
        out = self._inv_table[idx]
        return np.clip(out, 0.0, 100.0).astype(np.float32)


def _build_image_transforms(train: bool):
    from torchvision import transforms

    base = [transforms.Resize((IMAGE_SIZE, IMAGE_SIZE)), transforms.ToTensor()]
    norm = [transforms.Normalize(mean=IMAGENET_MEAN, std=IMAGENET_STD)]
    if train:
        aug = [
            transforms.RandomHorizontalFlip(p=0.5),
            transforms.ColorJitter(
                brightness=0.2, contrast=0.2, saturation=0.2, hue=0.05
            ),
            transforms.RandomAffine(
                degrees=10, translate=(0.05, 0.05), scale=(0.9, 1.1)
            ),
        ]
        pipeline = base + aug + norm
    else:
        pipeline = base + norm
    return transforms.Compose(pipeline)

def _load_swin_tiny_offline(num_classes: int = 0) -> nn.Module:
    try:
        import timm

        model = timm.create_model(
            "swin_tiny_patch4_window7_224",
            pretrained=False,  # never download weights
            num_classes=num_classes,
        )
        return model
    except Exception:

class _PatchEmbed(nn.Module):
    def __init__(self, patch=4, dim=96):
        super().__init__()
        self.proj = nn.Conv2d(3, dim, kernel_size=patch, stride=patch)
        self.norm = nn.LayerNorm(dim)

    def forward(self, x):
        x = self.proj(x)
        b, c, h, w = x.shape
        x = x.flatten(2).transpose(1, 2)
        x = self.norm(x)
        return x, (h, w)


class _WindowAttention(nn.Module):
    def __init__(self, dim, heads=6, drop=0.0):
        super().__init__()
        self.heads = heads
        self.scale = (dim // heads) ** -0.5
        self.qkv = nn.Linear(dim, dim * 3)
        self.proj = nn.Linear(dim, dim)
        self.drop = nn.Dropout(drop)

    def forward(self, x):
        b, n, c = x.shape
        qkv = (
            self.qkv(x)
            .reshape(b, n, 3, self.heads, c // self.heads)
            .permute(2, 0, 3, 1, 4)
        )
        q, k, v = qkv.unbind(0)
        attn = (q @ k.transpose(-2, -1)) * self.scale
        attn = attn.softmax(dim=-1)
        attn = self.drop(attn)
        out = (attn @ v).transpose(1, 2).reshape(b, n, c)
        return self.proj(out)


class _MLPBlock(nn.Module):
    def __init__(self, dim, mlp_ratio=4, drop=0.0):
        super().__init__()
        hidden = int(dim * mlp_ratio)
        self.fc1 = nn.Linear(dim, hidden)
        self.act = nn.GELU()
        self.fc2 = nn.Linear(hidden, dim)
        self.drop = nn.Dropout(drop)

    def forward(self, x):
        return self.drop(self.fc2(self.act(self.fc1(x))))


class _BasicBlock(nn.Module):
    def __init__(self, dim, heads=6, drop=0.0):
        super().__init__()
        self.norm1 = nn.LayerNorm(dim)
        self.attn = _WindowAttention(dim, heads, drop)
        self.norm2 = nn.LayerNorm(dim)
        self.mlp = _MLPBlock(dim, drop=drop)

    def forward(self, x):
        x = x + self.attn(self.norm1(x))
        x = x + self.mlp(self.norm2(x))
        return x


class _LocalSwinTiny(nn.Module):
    def __init__(self, encoder_dim=768):
        super().__init__()
        self.patch_embed = _PatchEmbed(patch=4, dim=96)
        self.stages = nn.ModuleList(
            [
                nn.Sequential(*[_BasicBlock(96, heads=3) for _ in range(2)]),
                nn.Sequential(*[_BasicBlock(192, heads=6) for _ in range(2)]),
                nn.Sequential(*[_BasicBlock(384, heads=12) for _ in range(2)]),
            ]
        )
        self.downsample = nn.ModuleList(
            [
                nn.Sequential(nn.LayerNorm(96), nn.Linear(96, 192)),
                nn.Sequential(nn.LayerNorm(192), nn.Linear(192, 384)),
            ]
        )
        self.encoder_dim = encoder_dim

    def forward(self, x):
        x, (h, w) = self.patch_embed(x)
        n_tokens = h * w
        feats = []
        for i, stage in enumerate(self.stages):
            x = stage(x)
            if i < len(self.downsample):
                b, _, c = x.shape
                hh, ww = int(math.sqrt(n_tokens / (4**i))), int(
                    math.sqrt(n_tokens / (4**i))
                )
                xp = x.reshape(b, hh, ww, c).permute(0, 3, 1, 2)
                xp = F.avg_pool2d(xp, 2).flatten(2).transpose(1, 2)
                x = self.downsample[i](xp)
                n_tokens //= 4
            feats.append(x.mean(dim=1))
        out = torch.cat(feats, dim=1)
        out = F.adaptive_avg_pool1d(out.unsqueeze(1), 1).squeeze(1)
        return out


class GatedCrossAttentionFusion(nn.Module):
    def __init__(
        self, vis_dim: int, tab_dim: int, ffn_dim: int = 256, drop: float = 0.1
    ):
        super().__init__()
        self.tab_proj = nn.Linear(tab_dim, vis_dim)
        self.vis_proj = nn.Linear(vis_dim, vis_dim)
        self.attn = nn.MultiheadAttention(
            embed_dim=vis_dim, num_heads=4, dropout=drop, batch_first=True
        )
        self.norm = nn.LayerNorm(vis_dim)
        self.ffn = nn.Sequential(
            nn.Linear(vis_dim, ffn_dim),
            nn.GELU(),
            nn.Dropout(drop),
            nn.Linear(ffn_dim, vis_dim),
        )
        self.gate = nn.Sequential(nn.Linear(vis_dim * 2, 1), nn.Sigmoid())

    def forward(self, vis_feat: torch.Tensor, tab_feat: torch.Tensor) -> torch.Tensor:
        q = self.vis_proj(vis_feat).unsqueeze(1)
        kv = self.tab_proj(tab_feat).unsqueeze(1)
        attn_out, _ = self.attn(q, kv, kv)
        attn_out = attn_out.squeeze(1)
        fused = self.norm(vis_feat + attn_out)
        fused = fused + self.ffn(fused)
        g = self.gate(torch.cat([vis_feat, tab_feat], dim=1))
        return vis_feat + g * (fused - vis_feat)


class PetModel(nn.Module):
    def __init__(self, tab_dim: int = NUM_TABULAR, dropout: float = 0.2):
        super().__init__()
        self.backbone = _load_swin_tiny_offline(num_classes=0)
        with torch.no_grad():
            probe = torch.zeros(1, 3, IMAGE_SIZE, IMAGE_SIZE)
            vis_dim = self.backbone(probe).shape[-1]
        self.vis_dim = vis_dim
        self.fusion = GatedCrossAttentionFusion(vis_dim, tab_dim, ffn_dim=256)
        self.head = nn.Sequential(
            nn.LayerNorm(vis_dim),
            nn.Linear(vis_dim, 128),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(128, 1),
        )

    def forward(self, image: torch.Tensor, tabular: torch.Tensor) -> torch.Tensor:
        vis_feat = self.backbone(image)
        if vis_feat.dim() == 3:
            vis_feat = vis_feat.mean(dim=1)
        fused = self.fusion(vis_feat, tabular)
        out = self.head(fused).squeeze(-1)
        return out


class RankAwareLoss(nn.Module):
    def __init__(self, delta: float = 0.5):
        super().__init__()
        self.delta = delta

    def forward(self, pred: torch.Tensor, target: torch.Tensor) -> torch.Tensor:
        return F.smooth_l1_loss(pred, target, reduction="mean", beta=self.delta)


def to_score_scale(logits: torch.Tensor) -> torch.Tensor:
    p = torch.sigmoid(logits)
    return torch.clamp(p * 100.0, 0.0, 100.0)


class PetDataset(torch.utils.data.Dataset):
    def __init__(self, ids, image_dir, tabular, transform, is_train=False):
        self.ids = list(ids)
        self.image_dir = image_dir
        self.tabular = tabular
        self.transform = transform
        self.is_train = is_train

    def __len__(self):
        return len(self.ids)

    def __getitem__(self, idx):
        path = os.path.join(self.image_dir, f"{self.ids[idx]}.jpg")
        img = load_image(path, self.transform)
        tab = torch.from_numpy(self.tabular[idx]).float()
        return {"image": img, "tabular": tab}


def collate_fn(batch):
    images = torch.stack([b["image"] for b in batch])
    tabular = torch.stack([b["tabular"] for b in batch])
    return {"image": images, "tabular": tabular}


def make_split(df, val_fraction=VAL_FRACTION, seed=SEED):
    rng = np.random.default_rng(seed)
    decile = pd.cut(df["Pawpularity"], bins=10, labels=False, duplicates="drop")
    train_idx, val_idx = [], []
    for d in sorted(decile.dropna().unique()):
        group = df.index[decile == d].tolist()
        rng.shuffle(group)
        n_val = max(1, int(round(len(group) * val_fraction)))
        val_idx.extend(group[:n_val])
        train_idx.extend(group[n_val:])
    return df.loc[train_idx].reset_index(drop=True), df.loc[val_idx].reset_index(
        drop=True
    )


def build_datasets_and_split(input_dir=INPUT_DIR):
    train_df = pd.read_csv(os.path.join(input_dir, "train.csv"))
    test_df = pd.read_csv(os.path.join(input_dir, "test.csv"))
    feat_builder = TabularFeatureBuilder(TABULAR_COLUMNS).fit(train_df)
    rank_tf = RankTargetTransform()
    rank_tf.fit(train_df["Pawpularity"].to_numpy())
    train_df, val_df = make_split(train_df)
    transforms = {
        "train": _build_image_transforms(train=True),
        "val": _build_image_transforms(train=False),
        "test": _build_image_transforms(train=False),
    }
    return {
        "train_df": train_df,
        "val_df": val_df,
        "test_df": test_df,
        "feat_builder": feat_builder,
        "rank_tf": rank_tf,
        "transforms": transforms,
    }


def build_dataset_parts(parts, split):
    df = parts[f"{split}_df"]
    tab = parts["feat_builder"].transform(df)
    ds = PetDataset(
        ids=df["Id"].tolist(),
        image_dir=os.path.join(INPUT_DIR, split),
        tabular=tab,
        transform=parts["transforms"][split],
        is_train=(split == "train"),
    )
    return ds

def _param_groups(model: nn.Module) -> List[dict]:
    """Differential learning-rate parameter groups: backbone gets a low LR,
    the freshly-initialized fusion + head get a higher LR."""
    backbone_ids = set(id(p) for p in model.backbone.parameters())
    backbone_params, other_params = [], []
    for name, p in model.named_parameters():
        if not p.requires_grad:
            continue
        if id(p) in backbone_ids:
            backbone_params.append(p)
        else:
            other_params.append(p)
    return [
        {"params": backbone_params, "lr": BACKBONE_LR},
        {"params": other_params, "lr": HEAD_LR},
    ]


def _make_scheduler(optimizer, total_steps: int) -> torch.optim.lr_scheduler.LambdaLR:
    """Linear warmup followed by cosine decay to ~0 over the remaining steps."""
    warmup_steps = max(1, int(total_steps * WARMUP_FRAC))

    def lr_lambda(step: int) -> float:
        if step < warmup_steps:
            return (step + 1) / warmup_steps
        progress = (step - warmup_steps) / max(1, total_steps - warmup_steps)
        progress = min(1.0, progress)
        return 0.5 * (1.0 + math.cos(math.pi * progress))

    return torch.optim.lr_scheduler.LambdaLR(optimizer, lr_lambda)


@torch.no_grad()
def _predict(model, loader, device) -> np.ndarray:
    """Shared forward pass used IDENTICALLY for validation and test inference.
    Returns raw logits (pre sigmoid) stacked as a 1-D float array."""
    model.eval()
    outs = []
    for batch in loader:
        images = batch["image"].to(device, non_blocking=True)
        tabular = batch["tabular"].to(device, non_blocking=True)
        logits = model(images, tabular)
        outs.append(logits.detach().cpu().numpy())
    return np.concatenate(outs, axis=0)


def _true_scale_rmse(rank_logits: np.ndarray, y_true: np.ndarray, rank_tf) -> float:
    """Map rank-space logits to [0,100] via sigmoid, then compute RMSE against
    the true raw scores. This is the exact competition metric."""
    p = 1.0 / (1.0 + np.exp(-rank_logits.astype(np.float64)))
    preds = np.clip(p * 100.0, 0.0, 100.0)
    y_true = np.asarray(y_true, dtype=np.float64)
    return float(np.sqrt(np.mean((preds - y_true) ** 2)))


def _make_loader(ds, batch_size, shuffle, num_workers=4):
    return torch.utils.data.DataLoader(
        ds,
        batch_size=batch_size,
        shuffle=shuffle,
        num_workers=num_workers,
        pin_memory=(DEVICE == "cuda"),
        collate_fn=collate_fn,
        drop_last=False,
    )


def _peak_vram_mb() -> float:
    if DEVICE == "cuda":
        return torch.cuda.max_memory_allocated() / (1024**2)
    return 0.0

def run_training(input_dir: str = INPUT_DIR) -> float:
    t_start = time.time()
    _seed_everything(SEED)

    parts = build_datasets_and_split(input_dir)
    train_ds = build_dataset_parts(parts, "train")
    val_ds = build_dataset_parts(parts, "val")
    test_ds = build_dataset_parts(parts, "test")

    rank_tf: RankTargetTransform = parts["rank_tf"]
    y_train_rank = rank_tf.transform(parts["train_df"]["Pawpularity"].to_numpy())
    y_val_raw = parts["val_df"]["Pawpularity"].to_numpy(dtype=np.float64)

    model = PetModel(tab_dim=NUM_TABULAR, dropout=0.2).to(DEVICE)
    criterion = RankAwareLoss(delta=0.5)
    optimizer = torch.optim.AdamW(_param_groups(model), weight_decay=WEIGHT_DECAY)

    # Elastic batch sizing: start high, halve on OOM down to MIN_BATCH.
    batch_size = INITIAL_BATCH
    num_workers = 4 if DEVICE == "cuda" else 2

    # Pre-build loaders once we settle the batch size after first epoch attempt.
    def rebuild_loaders(bs):
        tr_loader = _make_loader(train_ds, bs, shuffle=True, num_workers=num_workers)
        va_loader = _make_loader(val_ds, bs, shuffle=False, num_workers=num_workers)
        te_loader = _make_loader(test_ds, bs, shuffle=False, num_workers=num_workers)
        return tr_loader, va_loader, te_loader

    tr_loader, va_loader, te_loader = rebuild_loaders(batch_size)
    total_steps = MAX_EPOCHS * max(1, len(tr_loader))
    scheduler = _make_scheduler(optimizer, total_steps)

    best_rmse = float("inf")
    best_state = None
    bad_epochs = 0
    global_step = 0

    for epoch in range(MAX_EPOCHS):
        model.train()
        running = 0.0
        n_batches = 0
        oom_this_epoch = False

        for batch in tr_loader:
            images = batch["image"].to(DEVICE, non_blocking=True)
            tabular = batch["tabular"].to(DEVICE, non_blocking=True)
            target = torch.from_numpy(
                y_train_rank[n_batches * batch_size : (n_batches + 1) * batch_size]
            ).to(DEVICE)
            # Guard against the last partial batch length mismatch.
            if target.numel() != images.shape[0]:
                target = target[: images.shape[0]]

            optimizer.zero_grad(set_to_none=True)
            try:
                logits = model(images, tabular)
                loss = criterion(logits, target)
                loss.backward()
                torch.nn.utils.clip_grad_norm_(model.parameters(), GRAD_CLIP)
                optimizer.step()
                scheduler.step()
                running += float(loss.item())
                n_batches += 1
                global_step += 1
            except (RuntimeError, torch.cuda.OutOfMemoryError) as exc:
                if DEVICE == "cuda" and (
                    "out of memory" in str(exc).lower()
                    or isinstance(exc, torch.cuda.OutOfMemoryError)
                ):
                    oom_this_epoch = True
                    break
                raise

        if oom_this_epoch:
            if DEVICE == "cuda":
                torch.cuda.empty_cache()
            if batch_size > MIN_BATCH:
                batch_size = max(MIN_BATCH, batch_size // 2)
                tr_loader, va_loader, te_loader = rebuild_loaders(batch_size)
                print(
                    f"[epoch {epoch}] OOM -> halved batch to {batch_size} "
                    f"(elapsed={time.time()-t_start:.0f}s)"
                )
                continue  # restart this epoch at the smaller batch
            # Even at MIN_BATCH we OOM: fall back to CPU for safety.
            print("[epoch] OOM at min batch -> falling back to CPU")
            model = model.to("cpu")
            DEVICE_LOCAL = "cpu"
            tr_loader, va_loader, te_loader = rebuild_loaders(batch_size)
            continue

        avg_loss = running / max(1, n_batches)

        # Validation on the TRUE scale (identical forward path to test).
        val_logits = _predict(model, va_loader, DEVICE)
        val_rmse = _true_scale_rmse(val_logits, y_val_raw, rank_tf)

        print(
            f"[epoch {epoch}] train_loss={avg_loss:.4f} val_rmse={val_rmse:.4f} "
            f"bs={batch_size} elapsed={time.time()-t_start:.0f}s"
        )

        if val_rmse < best_rmse:
            best_rmse = val_rmse
            best_state = {k: v.detach().clone() for k, v in model.state_dict().items()}
            bad_epochs = 0
        else:
            bad_epochs += 1
            if bad_epochs >= PATIENCE:
                print(f"[early-stop] no improvement for {PATIENCE} epochs")
                break

    # Restore best weights.
    if best_state is not None:
        model.load_state_dict(best_state)

    # Test inference through the SAME _predict path as validation.
