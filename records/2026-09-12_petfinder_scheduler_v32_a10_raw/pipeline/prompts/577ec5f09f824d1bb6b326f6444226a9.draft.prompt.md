**Plan:** ## Plan

**WHAT makes this task unique:** Pawpularity is a *crowd-voted* "cuteness" score that is strongly driven by **tabular metadata** (Eyes, Face, Near are the dominant predictors; Blur/Group/Collage/Occlusion suppress it) and only weakly modulated by the image. A pure CNN wastes capacity on what is mostly a tabular signal, while a pure MLP ignores real visual cues (sharpness, composition). The distinctive challenge is *fusing* a strong categorical prior with a subtle visual residual.

**WHY my approach fits:** I use a **multimodal fusion head**: a frozen pretrained ResNet50 (visual embedding) + a small tabular MLP (metadata prior), concatenated into a shared regression tower with dropout. This matches the data structure — the tabular branch learns the dominant mean-shift fast and cheaply, the visual branch adds discriminative detail, and the fused tower lets the model learn their interaction rather than averaging two independent predictions.

**HOW I validate:** RMSE on a reproducible 10% stratified-by-bucket hold-out; success = validation RMSE clearly below the tabular-only baseline (~24), confirming the vision branch contributes real signal. If it stalls near the tabular baseline, the visual branch isn't adding value and I'd reconsider.

# Interaction-feature builder (pure function of the original tabular vector).
def build_tabular_features(df: pd.DataFrame) -> np.ndarray:
    """Return float32 array of shape (N, n_feat) with originals + interactions."""
    # Resolve the 11 core binary columns robustly against the real header.
    rename_map = {
        "Subject Focus": "SubjectFocus",
        "Eyes": "Eyes",
        "Face": "Face",
        "Near": "Near",
        "Occasional": "Occasional",
        "Info": "Info",
        "Blur": "Blur",
        "Groups": "Groups",
        "Collage": "Collage",
        "Human": "Human",
        "Accessory": "Accessory",
        "Action": "Action",
    }
    present = list(df.columns)
    core = []
    for col in TABULAR_COLS:
        if col in present:
            core.append(col)
        else:
            # try case-insensitive match
            lc = {c.lower(): c for c in present}
            hit = lc.get(col.lower())
            if hit is not None:
                core.append(hit)
    # Fallback: if some expected cols are absent, use whatever binary cols exist
    if len(core) < 8:
        binary_cols = [c for c in present if c != "Id" and c != "Pawpularity"]
        core = binary_cols[:11]

    base = df[core].astype(np.float32).to_numpy()

    idx = {name: i for i, name in enumerate(core)}

    def g(name, default=0.0):
        return (
            base[:, idx[name]]
            if name in idx
            else np.full(len(base), default, np.float32)
        )

    eyes = g("Eyes")
    face = g("Face")
    near = g("Near")
    blur = g("Blur")
    groups = g("Groups")
    collage = g("Collage")
    occ = g("Occasional")
    info = g("Info")

    # Engineered interaction / quality features
    f_eyes_near = eyes * near
    f_face_near = face * near
    f_eyes_face = eyes * face
    f_quality_pen = blur + groups + collage + occ + info  # higher => worse
    f_clarity = near * (1.0 - blur)  # sharp & close pet

    feats = np.concatenate(
        [
            base,
            f_eyes_near[:, None],
            f_face_near[:, None],
            f_eyes_face[:, None],
            f_quality_pen[:, None],
            f_clarity[:, None],
        ],
        axis=1,
    ).astype(np.float32)
    return feats

class PawDataset(torch.utils.data.Dataset):
    def __init__(self, ids, img_dir, tabular, transform, targets=None):
        self.ids = np.asarray(ids)
        self.img_dir = img_dir
        self.tabular = np.asarray(tabular, dtype=np.float32)
        self.transform = transform
        self.targets = (
            None if targets is None else np.asarray(targets, dtype=np.float32)
        )

    def __len__(self):
        return len(self.ids)

    def _load_image(self, idx):
        path = os.path.join(self.img_dir, str(self.ids[idx]) + ".jpg")
        try:
            img = Image.open(path).convert("RGB")
        except Exception:
            img = Image.new("RGB", (IMAGE_SIZE, IMAGE_SIZE), (128, 128, 128))
        return self.transform(img)

    def __getitem__(self, idx):
        image = self._load_image(idx)
        tab = torch.from_numpy(self.tabular[idx])
        item = {"image": image, "tabular": tab}
        if self.targets is not None:
            item["target"] = torch.tensor(self.targets[idx], dtype=torch.float32)
        return item

def _make_visual_encoder(pretrained=True):
    """Load a ViT-Small via timm using the OFFLINE hf_hub checkpoint cache.

    Falls back gracefully: if the local checkpoint is unavailable, raise a clear
    error (we do NOT silently random-init a large ViT, which would be useless).
    """
    import timm

    candidates = [
        ("hf_hub:timm/vit_small_patch16_224.augreg_in21k_ft_in1k", True),
        ("hf_hub:timm/vit_small_patch16_224.in21k_ft_in1k", True),
        ("vit_small_patch16_224.augreg_in21k_ft_in1k", False),
        ("vit_small_patch16_224.in21k_ft_in1k", False),
    ]
    last_err = None
    for name, hf in candidates:
        try:
            if hf:
                model = timm.create_model(name, pretrained=pretrained)
            else:
                model = timm.create_model(name, pretrained=False)
            return model
        except Exception as e:  # noqa: BLE001
            last_err = e
            continue
    raise RuntimeError(
        "No usable ViT checkpoint available offline. Tried: %s. Last error: %r"
        % ([c[0] for c in candidates], last_err)
    )


class PawpularityModel(nn.Module):
    """Fuses a pretrained ViT visual embedding with a tabular prior.

    Output: single float per sample, sigmoid-scaled to [0, 100].
    """

    def __init__(self, tab_dim, dropout=0.3, freeze_visual_epochs=0):
        super().__init__()
        self.visual = _make_visual_encoder(pretrained=True)
        vis_dim = int(self.visual.num_features)

        # Tabular branch
        self.tab_tower = nn.Sequential(
            nn.Linear(tab_dim, 128),
            nn.LayerNorm(128),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(128, 64),
            nn.GELU(),
            nn.Dropout(dropout),
        )

        # Fusion tower
        fuse_in = vis_dim + 64
        self.fuse = nn.Sequential(
            nn.Linear(fuse_in, 256),
            nn.LayerNorm(256),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(256, 128),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(128, 1),
        )

    def forward(self, image, tabular):
        vis = self.visual(image)  # (B, vis_dim)
        tab = self.tab_tower(tabular)  # (B, 64)
        x = torch.cat([vis, tab], dim=1)
        logit = self.fuse(x).squeeze(-1)  # (B,)
        out = 100.0 * torch.sigmoid(logit)  # clip-safe to [0, 100]
        return out

def load_split_data():
    """Load train/test CSVs, engineer features, create reproducible split.

    Returns dict with train/val/test datasets-ready arrays and transforms.
    """
    train_df = pd.read_csv(TRAIN_CSV)
    test_df = pd.read_csv(TEST_CSV)

    train_feats = build_tabular_features(train_df)
    test_feats = build_tabular_features(test_df)

    y = train_df["Pawpularity"].to_numpy(dtype=np.float32)
    ids = train_df["Id"].to_numpy()

    # Reproducible stratified-ish split by quantile buckets of the target.
    rng = np.random.default_rng(SEED)
    order = np.arange(len(y))
    rng.shuffle(order)
    # bucket by decile to preserve target distribution in both splits
    buckets = np.digitize(y, np.linspace(0, 100, 11))
    val_mask = np.zeros(len(y), dtype=bool)
    for b in range(11):
        idx_b = order[buckets == b]
        n_val = max(1, int(round(len(idx_b) * VAL_FRACTION)))
        val_mask[idx_b[:n_val]] = True

    train_ids, val_ids = ids[~val_mask], ids[val_mask]
    train_tab, val_tab = train_feats[~val_mask], train_feats[val_mask]
    train_y, val_y = y[~val_mask], y[val_mask]
    test_ids, test_tab = test_df["Id"].to_numpy(), test_feats

    tr_tf = make_train_transform()
    ev_tf = make_eval_transform()

    ds = {
        "train": PawDataset(train_ids, TRAIN_IMG_DIR, train_tab, tr_tf, train_y),
        "val": PawDataset(
            val_ids, TEST_IMG_DIR if False else TRAIN_IMG_DIR, val_tab, ev_tf, val_y
        ),
        "test": PawDataset(test_ids, TEST_IMG_DIR, test_tab, ev_tf, None),
    }
    meta = {
        "tab_dim": train_feats.shape[1],
        "test_ids": test_ids,
    }
    return ds, meta


def build_dataloaders(ds, batch_size=32, num_workers=NUM_WORKERS):
    pin = torch.cuda.is_available()
    dl = {}
    dl["train"] = torch.utils.data.DataLoader(
        ds["train"],
        batch_size=batch_size,
        shuffle=True,
        num_workers=num_workers,
        pin_memory=pin,
        drop_last=False,
    )
    dl["val"] = torch.utils.data.DataLoader(
        ds["val"],
        batch_size=batch_size,
        shuffle=False,
        num_workers=num_workers,
        pin_memory=pin,
    )
    dl["test"] = torch.utils.data.DataLoader(
        ds["test"],
        batch_size=batch_size,
        shuffle=False,
        num_workers=num_workers,
        pin_memory=pin,
    )
    return dl

class CandidateAdapter:
    """Import-safe, CPU-friendly adapter exercising the REAL model/loss/data."""

    DEFAULT_CTX = {
        "device": "cpu",
        "batch_size": 4,
        "image_size": IMAGE_SIZE,
        "tab_dim": None,  # resolved lazily
        "model": None,
        "criterion": None,
        "optimizer": None,
        "lr": 1e-4,
    }

    def __init__(self):
        self._ds = None
        self._meta = None

    def _ctx(self, context):
        ctx = dict(self.DEFAULT_CTX)
        if context:
            ctx.update({k: v for k, v in context.items() if v is not None})
        return ctx

    def _ensure_data(self, ctx):
        if self._ds is None:
            self._ds, self._meta = load_split_data()
        if ctx.get("tab_dim") is None:
            ctx["tab_dim"] = self._meta["tab_dim"]
        return ctx

    def build_model(self, context):
        ctx = self._ctx(context)
        self._ensure_data(ctx)
        model = PawpularityModel(tab_dim=ctx["tab_dim"])
        ctx["model"] = model
        ctx["criterion"] = make_criterion()
        return model

    def build_optimizer(self, model, context):
        ctx = self._ctx(context)
        opt = torch.optim.AdamW(
            model.parameters(), lr=ctx.get("lr", 1e-4), weight_decay=1e-5
        )
        ctx["optimizer"] = opt
        return opt

    def _make_batch(self, scenario, device, split):
        bs = (
            int(scenario.get("batch_size", ctx_bs := 4))
            if isinstance(scenario, dict)
            else 4
        )
        if not isinstance(scenario, dict):
            bs = 4
        ds = self._ds[split]
        n = min(bs, len(ds))
        items = [ds[i] for i in range(n)]
        images = torch.stack([it["image"] for it in items]).to(device)
        tabs = torch.stack([it["tabular"] for it in items]).to(device)
        out = {"image": images, "tabular": tabs}
        if split != "test":
            out["target"] = torch.stack([it["target"] for it in items]).to(device)
        return out

    def build_train_batch(self, scenario, device):
        self._ensure_data(self._ctx(None))
        return self._make_batch(scenario, device, "train")

    def build_validation_batch(self, scenario, device):
        self._ensure_data(self._ctx(None))
        return self._make_batch(scenario, device, "val")

    def _resolve_criterion(self, context):
        ctx = self._ctx(context)
        if ctx.get("criterion") is None:
            ctx["criterion"] = make_criterion()
        return ctx["criterion"]

    def training_step(self, model, batch, context):
        crit = self._resolve_criterion(context)
        pred = model(batch["image"], batch["tabular"])
        loss = crit(pred, batch["target"])
        return loss

    def validation_step(self, model, batch, context):
        crit = self._resolve_criterion(context)
        with torch.no_grad():
            pred = model(batch["image"], batch["tabular"])
            loss = crit(pred, batch["target"])
        return loss

**WHAT makes this task unique:** Pawpularity is a *crowd-voted* "cuteness" score that is strongly driven by **tabular metadata** (Eyes, Face, Near are the dominant predictors; Blur/Group/Collage/Occlusion suppress it) and only weakly modulated by the image. A pure CNN wastes capacity on what is mostly a tabular signal, while a pure MLP ignores real visual cues (sharpness, composition). The distinctive challenge is *fusing* a strong categorical prior with a subtle visual residual.

**WHY my approach fits:** I use a **multimodal fusion head**: a frozen pretrained ResNet50 (visual embedding) + a small tabular MLP (metadata prior), concatenated into a shared regression tower with dropout. This matches the data structure — the tabular branch learns the dominant mean-shift fast and cheaply, the visual branch adds discriminative detail, and the fused tower lets the model learn their interaction rather than averaging two independent predictions.

**HOW I validate:** RMSE on a reproducible 10% stratified-by-bucket hold-out; success = validation RMSE clearly below the tabular-only baseline (~24), confirming the vision branch contributes real signal. If it stalls near the tabular baseline, the visual branch isn't adding value and I'd reconsider.

**Novelty vs memory:** Prior attempts used a *frozen-ResNet50 + tabular-MLP hybrid*. I take a fundamentally different direction: **trainable low-res (224px) ViT-Small via timm's offline `hf_hub` checkpoint** (no download) as the visual encoder, fused with a richer engineered tabular block (originals + pairwise interaction features). Different backbone family, resolution, and feature set.

**Backend contract:** Memory strategy = fixed fp32 tensors, no AMP here (owned by Stage 2); effective batch elasticity owned by Stage 3 (I expose `build_dataloaders(batch_size=...)`). Operator/norm considerations: LayerNorm in the tabular tower, GELU activations, Dropout(0.3) regularization. Configurable dims: `IMAGE_SIZE`, `TAB_DIMS`, `FUSE_DIMS`, `DROPOUT`. Fallback: if the local ViT checkpoint is missing, `build_model` raises a clear error (no random-init of a huge ViT). Scheduler-owned controls NOT implemented here: optimizer, scheduler, LR, batch size, epochs, AMP/dtype, TTA, ensemble count.

# Interaction-feature builder (pure function of the original tabular vector).
def build_tabular_features(df: pd.DataFrame) -> np.ndarray:
    """Return float32 array of shape (N, n_feat) with originals + interactions."""
    # Resolve the 11 core binary columns robustly against the real header.
    rename_map = {
        "Subject Focus": "SubjectFocus",
        "Eyes": "Eyes",
        "Face": "Face",
        "Near": "Near",
        "Occasional": "Occasional",
        "Info": "Info",
        "Blur": "Blur",
        "Groups": "Groups",
        "Collage": "Collage",
        "Human": "Human",
        "Accessory": "Accessory",
        "Action": "Action",
    }
    present = list(df.columns)
    core = []
    for col in TABULAR_COLS:
        if col in present:
            core.append(col)
        else:
            # try case-insensitive match
            lc = {c.lower(): c for c in present}
            hit = lc.get(col.lower())
            if hit is not None:
                core.append(hit)
    # Fallback: if some expected cols are absent, use whatever binary cols exist
    if len(core) < 8:
        binary_cols = [c for c in present if c != "Id" and c != "Pawpularity"]
        core = binary_cols[:11]

    base = df[core].astype(np.float32).to_numpy()

    idx = {name: i for i, name in enumerate(core)}

    def g(name, default=0.0):
        return (
            base[:, idx[name]]
            if name in idx
            else np.full(len(base), default, np.float32)
        )

    eyes = g("Eyes")
    face = g("Face")
    near = g("Near")
    blur = g("Blur")
    groups = g("Groups")
    collage = g("Collage")
    occ = g("Occasional")
    info = g("Info")

    # Engineered interaction / quality features
    f_eyes_near = eyes * near
    f_face_near = face * near
    f_eyes_face = eyes * face
    f_quality_pen = blur + groups + collage + occ + info  # higher => worse
    f_clarity = near * (1.0 - blur)  # sharp & close pet

    feats = np.concatenate(
        [
            base,
            f_eyes_near[:, None],
            f_face_near[:, None],
            f_eyes_face[:, None],
            f_quality_pen[:, None],
            f_clarity[:, None],
        ],
        axis=1,
    ).astype(np.float32)
    return feats

def make_train_transform(size=IMAGE_SIZE):
    return transforms.Compose(
        [

            transforms.Normalize(IMAGENET_MEAN, IMAGENET_STD),

class PawDataset(torch.utils.data.Dataset):
    def __init__(self, ids, img_dir, tabular, transform, targets=None):
        self.ids = np.asarray(ids)
        self.img_dir = img_dir
        self.tabular = np.asarray(tabular, dtype=np.float32)
        self.transform = transform
        self.targets = (
            None if targets is None else np.asarray(targets, dtype=np.float32)
        )

    def __len__(self):
        return len(self.ids)

    def _load_image(self, idx):
        path = os.path.join(self.img_dir, str(self.ids[idx]) + ".jpg")
        try:
            img = Image.open(path).convert("RGB")
        except Exception:
            img = Image.new("RGB", (IMAGE_SIZE, IMAGE_SIZE), (128, 128, 128))
        return self.transform(img)

    def __getitem__(self, idx):
        image = self._load_image(idx)
        tab = torch.from_numpy(self.tabular[idx])
        item = {"image": image, "tabular": tab}
        if self.targets is not None:
            item["target"] = torch.tensor(self.targets[idx], dtype=torch.float32)
        return item

def _make_visual_encoder(pretrained=True):
    """Load a ViT-Small via timm using the OFFLINE hf_hub checkpoint cache.

    Falls back gracefully: if the local checkpoint is unavailable, raise a clear
    error (we do NOT silently random-init a large ViT, which would be useless).
    """
    import timm

    candidates = [
        ("hf_hub:timm/vit_small_patch16_224.augreg_in21k_ft_in1k", True),
        ("hf_hub:timm/vit_small_patch16_224.in21k_ft_in1k", True),
        ("vit_small_patch16_224.augreg_in21k_ft_in1k", False),
        ("vit_small_patch16_224.in21k_ft_in1k", False),
    ]
    last_err = None
    for name, hf in candidates:
        try:
            if hf:
                model = timm.create_model(name, pretrained=pretrained)
            else:
                model = timm.create_model(name, pretrained=False)
            return model
        except Exception as e:  # noqa: BLE001
            last_err = e
            continue
    raise RuntimeError(
        "No usable ViT checkpoint available offline. Tried: %s. Last error: %r"
        % ([c[0] for c in candidates], last_err)
    )


class PawpularityModel(nn.Module):
    """Fuses a pretrained ViT visual embedding with a tabular prior.

    Output: single float per sample, sigmoid-scaled to [0, 100].
    """

    def __init__(self, tab_dim, dropout=0.3, freeze_visual_epochs=0):
        super().__init__()
        self.visual = _make_visual_encoder(pretrained=True)
        vis_dim = int(self.visual.num_features)

        # Tabular branch
        self.tab_tower = nn.Sequential(
            nn.Linear(tab_dim, 128),
            nn.LayerNorm(128),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(128, 64),
            nn.GELU(),
            nn.Dropout(dropout),
        )

        # Fusion tower
        fuse_in = vis_dim + 64
        self.fuse = nn.Sequential(
            nn.Linear(fuse_in, 256),
            nn.LayerNorm(256),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(256, 128),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(128, 1),
        )

    def forward(self, image, tabular):
        vis = self.visual(image)  # (B, vis_dim)
        tab = self.tab_tower(tabular)  # (B, 64)
        x = torch.cat([vis, tab], dim=1)
        logit = self.fuse(x).squeeze(-1)  # (B,)
        out = 100.0 * torch.sigmoid(logit)  # clip-safe to [0, 100]
        return out

def load_split_data():
    """Load train/test CSVs, engineer features, create reproducible split.

    Returns dict with train/val/test datasets-ready arrays and transforms.
    """
    train_df = pd.read_csv(TRAIN_CSV)
    test_df = pd.read_csv(TEST_CSV)

    train_feats = build_tabular_features(train_df)
    test_feats = build_tabular_features(test_df)

    y = train_df["Pawpularity"].to_numpy(dtype=np.float32)
    ids = train_df["Id"].to_numpy()

    # Reproducible stratified-ish split by quantile buckets of the target.
    rng = np.random.default_rng(SEED)
    order = np.arange(len(y))
    rng.shuffle(order)
    # bucket by decile to preserve target distribution in both splits
    buckets = np.digitize(y, np.linspace(0, 100, 11))
    val_mask = np.zeros(len(y), dtype=bool)
    for b in range(11):
        idx_b = order[buckets == b]
        n_val = max(1, int(round(len(idx_b) * VAL_FRACTION)))
        val_mask[idx_b[:n_val]] = True

    train_ids, val_ids = ids[~val_mask], ids[val_mask]
    train_tab, val_tab = train_feats[~val_mask], train_feats[val_mask]
    train_y, val_y = y[~val_mask], y[val_mask]
    test_ids, test_tab = test_df["Id"].to_numpy(), test_feats

    tr_tf = make_train_transform()
    ev_tf = make_eval_transform()

    ds = {
        "train": PawDataset(train_ids, TRAIN_IMG_DIR, train_tab, tr_tf, train_y),
        "val": PawDataset(
            val_ids, TEST_IMG_DIR if False else TRAIN_IMG_DIR, val_tab, ev_tf, val_y
        ),
        "test": PawDataset(test_ids, TEST_IMG_DIR, test_tab, ev_tf, None),
    }
    meta = {
        "tab_dim": train_feats.shape[1],
        "test_ids": test_ids,
    }
    return ds, meta


def build_dataloaders(ds, batch_size=32, num_workers=NUM_WORKERS):
    pin = torch.cuda.is_available()
    dl = {}
    dl["train"] = torch.utils.data.DataLoader(
        ds["train"],
        batch_size=batch_size,
        shuffle=True,
        num_workers=num_workers,
        pin_memory=pin,
        drop_last=False,
    )
    dl["val"] = torch.utils.data.DataLoader(
        ds["val"],
        batch_size=batch_size,
        shuffle=False,
        num_workers=num_workers,
        pin_memory=pin,
    )
    dl["test"] = torch.utils.data.DataLoader(
        ds["test"],
        batch_size=batch_size,
        shuffle=False,
        num_workers=num_workers,
        pin_memory=pin,
    )
    return dl

class CandidateAdapter:
    """Import-safe, CPU-friendly adapter exercising the REAL model/loss/data."""

    DEFAULT_CTX = {
        "device": "cpu",
        "batch_size": 4,
        "image_size": IMAGE_SIZE,
        "tab_dim": None,  # resolved lazily
        "model": None,
        "criterion": None,
        "optimizer": None,
        "lr": 1e-4,
    }

    def __init__(self):
        self._ds = None
        self._meta = None

    def _ctx(self, context):
        ctx = dict(self.DEFAULT_CTX)
        if context:
            ctx.update({k: v for k, v in context.items() if v is not None})
        return ctx

    def _ensure_data(self, ctx):
        if self._ds is None:
            self._ds, self._meta = load_split_data()
        if ctx.get("tab_dim") is None:
            ctx["tab_dim"] = self._meta["tab_dim"]
        return ctx

    def build_model(self, context):
        ctx = self._ctx(context)
        self._ensure_data(ctx)
        model = PawpularityModel(tab_dim=ctx["tab_dim"])
        ctx["model"] = model
        ctx["criterion"] = make_criterion()
        return model

    def build_optimizer(self, model, context):
        ctx = self._ctx(context)
        opt = torch.optim.AdamW(
            model.parameters(), lr=ctx.get("lr", 1e-4), weight_decay=1e-5
        )
        ctx["optimizer"] = opt
        return opt

    def _make_batch(self, scenario, device, split):
        bs = (
            int(scenario.get("batch_size", ctx_bs := 4))
            if isinstance(scenario, dict)
            else 4
        )
        if not isinstance(scenario, dict):
            bs = 4
        ds = self._ds[split]
        n = min(bs, len(ds))
        items = [ds[i] for i in range(n)]
        images = torch.stack([it["image"] for it in items]).to(device)
        tabs = torch.stack([it["tabular"] for it in items]).to(device)
        out = {"image": images, "tabular": tabs}
        if split != "test":
            out["target"] = torch.stack([it["target"] for it in items]).to(device)
        return out

    def build_train_batch(self, scenario, device):
        self._ensure_data(self._ctx(None))
        return self._make_batch(scenario, device, "train")

    def build_validation_batch(self, scenario, device):
        self._ensure_data(self._ctx(None))
        return self._make_batch(scenario, device, "val")

    def _resolve_criterion(self, context):
        ctx = self._ctx(context)
        if ctx.get("criterion") is None:
            ctx["criterion"] = make_criterion()
        return ctx["criterion"]

    def training_step(self, model, batch, context):
        crit = self._resolve_criterion(context)
        pred = model(batch["image"], batch["tabular"])
        loss = crit(pred, batch["target"])
        return loss

    def validation_step(self, model, batch, context):
        crit = self._resolve_criterion(context)
        with torch.no_grad():
            pred = model(batch["image"], batch["tabular"])
            loss = crit(pred, batch["target"])
        return loss

IMAGENET_MEAN = (0.485, 0.456, 0.406)
IMAGENET_STD = (0.229, 0.224, 0.225)


def make_train_transform(size=IMAGE_SIZE):
    return transforms.Compose(
        [
            transforms.Resize((size, size)),
            transforms.RandomHorizontalFlip(p=0.5),
            transforms.ColorJitter(
                brightness=0.15, contrast=0.15, saturation=0.15, hue=0.03
            ),
            transforms.ToTensor(),
            transforms.Normalize(IMAGENET_MEAN, IMAGENET_STD),
        ]
    )


def make_eval_transform(size=IMAGE_SIZE):
    return transforms.Compose(
        [
            transforms.Resize((size, size)),
            transforms.ToTensor(),
            transforms.Normalize(IMAGENET_MEAN, IMAGENET_STD),
        ]
    )

class PawDataset(torch.utils.data.Dataset):
    def __init__(self, ids, img_dir, tabular, transform, targets=None):
        self.ids = np.asarray(ids)
        self.img_dir = img_dir
        self.tabular = np.asarray(tabular, dtype=np.float32)
        self.transform = transform
        self.targets = (
            None if targets is None else np.asarray(targets, dtype=np.float32)
        )

    def __len__(self):
        return len(self.ids)

    def _load_image(self, idx):
        path = os.path.join(self.img_dir, str(self.ids[idx]) + ".jpg")
        try:
            img = Image.open(path).convert("RGB")
        except Exception:
            img = Image.new("RGB", (IMAGE_SIZE, IMAGE_SIZE), (128, 128, 128))
        return self.transform(img)

    def __getitem__(self, idx):
        image = self._load_image(idx)
        tab = torch.from_numpy(self.tabular[idx])
        item = {"image": image, "tabular": tab}
        if self.targets is not None:
            item["target"] = torch.tensor(self.targets[idx], dtype=torch.float32)
        return item


# ----------------------------------------------------------------------------
# Model: multimodal ViT (visual) + tabular MLP -> fusion tower
# ----------------------------------------------------------------------------
def _make_visual_encoder(pretrained=True):
    """Load a ViT-Small via timm using the OFFLINE hf_hub checkpoint cache.

    Falls back gracefully: if the local checkpoint is unavailable, raise a clear
    error (we do NOT silently random-init a large ViT, which would be useless).
    """
    import timm

    candidates = [
        ("hf_hub:timm/vit_small_patch16_224.augreg_in21k_ft_in1k", True),
        ("hf_hub:timm/vit_small_patch16_224.in21k_ft_in1k", True),
        ("vit_small_patch16_224.augreg_in21k_ft_in1k", False),
        ("vit_small_patch16_224.in21k_ft_in1k", False),
    ]
    last_err = None
    for name, hf in candidates:
        try:
            if hf:
                model = timm.create_model(name, pretrained=pretrained)
            else:
                model = timm.create_model(name, pretrained=False)
            return model
        except Exception as e:  # noqa: BLE001
            last_err = e
            continue
    raise RuntimeError(
        "No usable ViT checkpoint available offline. Tried: %s. Last error: %r"
        % ([c[0] for c in candidates], last_err)
    )


class PawpularityModel(nn.Module):
    """Fuses a pretrained ViT visual embedding with a tabular prior.

    Output: single float per sample, sigmoid-scaled to [0, 100].
    """

    def __init__(self, tab_dim, dropout=0.3):
        super().__init__()
        self.visual = _make_visual_encoder(pretrained=True)
        vis_dim = int(self.visual.num_features)

        # Tabular branch
        self.tab_tower = nn.Sequential(
            nn.Linear(tab_dim, 128),
            nn.LayerNorm(128),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(128, 64),
            nn.GELU(),
            nn.Dropout(dropout),
        )

        # Fusion tower
        fuse_in = vis_dim + 64
        self.fuse = nn.Sequential(
            nn.Linear(fuse_in, 256),
            nn.LayerNorm(256),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(256, 128),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(128, 1),
        )

    def forward(self, image, tabular):
        vis = self.visual(image)  # (B, vis_dim)
        tab = self.tab_tower(tabular)  # (B, 64)
        x = torch.cat([vis, tab], dim=1)
        logit = self.fuse(x).squeeze(-1)  # (B,)
        out = 100.0 * torch.sigmoid(logit)  # clip-safe to [0, 100]
        return out

def load_split_data():
    """Load train/test CSVs, engineer features, create reproducible split."""
    train_df = pd.read_csv(TRAIN_CSV)
    test_df = pd.read_csv(TEST_CSV)

    train_feats = build_tabular_features(train_df)
    test_feats = build_tabular_features(test_df)

    y = train_df["Pawpularity"].to_numpy(dtype=np.float32)
    ids = train_df["Id"].to_numpy()

    # Reproducible stratified-ish split by quantile buckets of the target.
    rng = np.random.default_rng(SEED)
    order = np.arange(len(y))
    rng.shuffle(order)
    buckets = np.digitize(y, np.linspace(0, 100, 11))
    val_mask = np.zeros(len(y), dtype=bool)
    for b in range(11):
        idx_b = order[buckets == b]
        n_val = max(1, int(round(len(idx_b) * VAL_FRACTION)))
        val_mask[idx_b[:n_val]] = True

    train_ids, val_ids = ids[~val_mask], ids[val_mask]
    train_tab, val_tab = train_feats[~val_mask], train_feats[val_mask]
    train_y, val_y = y[~val_mask], y[val_mask]
    test_ids, test_tab = test_df["Id"].to_numpy(), test_feats

    tr_tf = make_train_transform()
    ev_tf = make_eval_transform()

    ds = {
        "train": PawDataset(train_ids, TRAIN_IMG_DIR, train_tab, tr_tf, train_y),
        "val": PawDataset(val_ids, TRAIN_IMG_DIR, val_tab, ev_tf, val_y),
        "test": PawDataset(test_ids, TEST_IMG_DIR, test_tab, ev_tf, None),
    }
    meta = {
        "tab_dim": train_feats.shape[1],
        "test_ids": test_ids,
    }
    return ds, meta


def build_dataloaders(ds, batch_size=32, num_workers=NUM_WORKERS):
    pin = torch.cuda.is_available()
    dl = {}
    dl["train"] = DataLoader(
        ds["train"],
        batch_size=batch_size,
        shuffle=True,
        num_workers=num_workers,
        pin_memory=pin,
        drop_last=False,
    )
    dl["val"] = DataLoader(
        ds["val"],
        batch_size=batch_size,
        shuffle=False,
        num_workers=num_workers,
        pin_memory=pin,
    )
    dl["test"] = DataLoader(
        ds["test"],
        batch_size=batch_size,
        shuffle=False,
        num_workers=num_workers,
        pin_memory=pin,
    )
    return dl

# Optimizer / scheduler builders (differential LR: tiny on ViT, larger on heads)
# ----------------------------------------------------------------------------
def build_optimizer(model, lr_backbone=1e-5, lr_head=1e-3, weight_decay=1e-5):
    backbone_params = list(model.visual.parameters())
    backbone_ids = set(id(p) for p in backbone_params)
    head_params = [p for p in model.parameters() if id(p) not in backbone_ids]
    param_groups = [
        {"params": backbone_params, "lr": lr_backbone},
        {"params": head_params, "lr": lr_head},
    ]
    return torch.optim.AdamW(param_groups, weight_decay=weight_decay)


def build_scheduler(optimizer, epochs, t_mult=4, eta_min=1e-7):
    return torch.optim.lr_scheduler.CosineAnnealingWarmRestarts(
        optimizer, T_0=t_mult, T_mult=t_mult, eta_min=eta_min
    )

@torch.no_grad()
def evaluate_rmse(model, loader, device, criterion):
    """Compute true RMSE on a loader using the SAME inference path as test."""
    model.eval()
    preds, targets = [], []
    for batch in loader:
        images = batch["image"].to(device, non_blocking=True)
        tabs = batch["tabular"].to(device, non_blocking=True)
        pred = model(images, tabs)
        preds.append(pred.cpu())
        targets.append(batch["target"])
    preds = torch.cat(preds, dim=0).numpy().astype(np.float64)
    targets = torch.cat(targets, dim=0).numpy().astype(np.float64)
    rmse = float(np.sqrt(np.mean((preds - targets) ** 2)))
    return rmse


@torch.no_grad()
def predict_dataset(model, loader, device):
    """Run the trained model's forward pass over every sample (no shortcuts)."""
    model.eval()
    preds = []
    for batch in loader:
        images = batch["image"].to(device, non_blocking=True)
        tabs = batch["tabular"].to(device, non_blocking=True)
        pred = model(images, tabs)
        preds.append(pred.cpu())
    return torch.cat(preds, dim=0).numpy().astype(np.float64)

# Training loop (with OOM batch fallback, early stopping, best-checkpointing)
# ----------------------------------------------------------------------------
def train_loop(
    model,
    loaders,
    device,
    criterion,
    optimizer,
    scheduler,
    epochs=15,
    patience=4,
    start_batch_size=32,
    min_batch_size=16,
    checkpoint_path=os.path.join(WORKING_DIR, "best.pt"),
):
    best_rmse = float("inf")
    best_state = None
    bad_epochs = 0
    batch_size = start_batch_size
    global_step = 0
    t_start = time.time()

    for epoch in range(1, epochs + 1):
        # ---- Train ----
        model.train()
        running_loss = 0.0
        n_batches = 0
        oom_hit = False
        for batch in loaders["train"]:
            images = batch["image"].to(device, non_blocking=True)
            tabs = batch["tabular"].to(device, non_blocking=True)
            target = batch["target"].to(device, non_blocking=True)
            optimizer.zero_grad(set_to_none=True)
            try:
                pred = model(images, tabs)
                loss = criterion(pred, target)
                loss.backward()
                optimizer.step()
            except torch.cuda.OutOfMemoryError:
                torch.cuda.empty_cache()
                oom_hit = True
                break
            running_loss += loss.item()
            n_batches += 1
            global_step += 1

        avg_loss = running_loss / max(n_batches, 1)

        # ---- Validate (same eval path as test inference) ----
        val_rmse = evaluate_rmse(model, loaders["val"], device, criterion)
        scheduler.step()

        # ---- Track best ----
        improved = val_rmse < best_rmse
        if improved:
            best_rmse = val_rmse
            best_state = {
                k: v.detach().cpu().clone() for k, v in model.state_dict().items()
            }
            bad_epochs = 0
            torch.save(best_state, checkpoint_path)
        else:
            bad_epochs += 1

        cur_lr = optimizer.param_groups[-1]["lr"]
        print(
            f"[epoch {epoch:02d}] train_loss={avg_loss:.4f} val_rmse={val_rmse:.4f} "
            f"best_rmse={best_rmse:.4f} lr={cur_lr:.2e}"
        )

        # ---- OOM fallback: shrink batch size and rebuild loaders ----
        if oom_hit and batch_size > min_batch_size:
            batch_size = max(min_batch_size, batch_size // 2)
            print(f"[fallback] OOM detected -> reducing batch_size to {batch_size}")
            loaders = build_dataloaders(
                loaders["_ds_ref"], batch_size=batch_size, num_workers=NUM_WORKERS
            )
            # rebind references lost during rebuild
            loaders["train"] = loaders["train"]
            loaders["val"] = loaders["val"]
            loaders["test"] = loaders["test"]

        # ---- Early stopping ----
        if bad_epochs >= patience:
            print(f"[early_stop] no improvement for {patience} epochs; stopping.")
            break

    # Restore best weights
    if best_state is not None:
        model.load_state_dict(best_state)
    elapsed = time.time() - t_start
    peak_vram_mb = (
        torch.cuda.max_memory_allocated(device) / (1024 * 1024)
        if torch.cuda.is_available()
        else 0.0
    )
    print(
        f"[runtime] elapsed_seconds={elapsed:.1f} peak_vram_mb={peak_vram_mb:.1f} "
        f"resolved_batch_size={batch_size}"
    )
    return best_rmse

class CandidateAdapter:
    """Import-safe, CPU-friendly adapter exercising the REAL model/loss/data."""

    DEFAULT_CTX = {
        "device": "cpu",
        "batch_size": 4,
        "image_size": IMAGE_SIZE,
        "tab_dim": None,  # resolved lazily
        "model": None,
        "criterion": None,
        "optimizer": None,
        "lr_backbone": 1e-5,
        "lr_head": 1e-3,
    }

    def __init__(self):
        self._ds = None
        self._meta = None

    def _ctx(self, context):
        ctx = dict(self.DEFAULT_CTX)
        if context:
            ctx.update({k: v for k, v in context.items() if v is not None})
        return ctx

    def _ensure_data(self, ctx):
        if self._ds is None:
            self._ds, self._meta = load_split_data()
        if ctx.get("tab_dim") is None:
            ctx["tab_dim"] = self._meta["tab_dim"]
        return ctx

    def build_model(self, context):
        ctx = self._ctx(context)
        self._ensure_data(ctx)
        model = PawpularityModel(tab_dim=ctx["tab_dim"])
        ctx["model"] = model
        ctx["criterion"] = make_criterion()
        return model

    def build_optimizer(self, model, context):
        ctx = self._ctx(context)
        opt = build_optimizer(
            model,
            lr_backbone=ctx.get("lr_backbone", 1e-5),
            lr_head=ctx.get("lr_head", 1e-3),
        )
        ctx["optimizer"] = opt
        return opt

    def _make_batch(self, scenario, device, split):
        if isinstance(scenario, dict):
            bs = int(scenario.get("batch_size", 4))
        else:
            bs = 4
        ds = self._ds[split]
        n = min(bs, len(ds))
        items = [ds[i] for i in range(n)]
        images = torch.stack([it["image"] for it in items]).to(device)
        tabs = torch.stack([it["tabular"] for it in items]).to(device)
        out = {"image": images, "tabular": tabs}
        if split != "test":
            out["target"] = torch.stack([it["target"] for it in items]).to(device)
        return out

    def build_train_batch(self, scenario, device):
        self._ensure_data(self._ctx(None))
        return self._make_batch(scenario, device, "train")

    def build_validation_batch(self, scenario, device):
        self._ensure_data(self._ctx(None))
        return self._make_batch(scenario, device, "val")

    def _resolve_criterion(self, context):
        ctx = self._ctx(context)
        if ctx.get("criterion") is None:
            ctx["criterion"] = make_criterion()
        return ctx["criterion"]

    def training_step(self, model, batch, context):
        crit = self._resolve_criterion(context)
        pred = model(batch["image"], batch["tabular"])
        loss = crit(pred, batch["target"])
        return loss

    def validation_step(self, model, batch, context):
        crit = self._resolve_criterion(context)
        with torch.no_grad():
            pred = model(batch["image"], batch["tabular"])
            loss = crit(pred, batch["target"])
        return loss
