class PetDataset(Dataset):
    """Loads an image + its 12 tabular cues (+ optional target).

    ``tab_mean``/``tab_std`` are the train-fit scaler params so every split is
    encoded identically. Missing/corrupt images fall back to a neutral grey tile
    so a single bad file can never crash the run.

    def __init__(self, df, img_dir, transform, tab_mean, tab_std, has_target=True):
        self.df = df.reset_index(drop=True)
        self.img_dir = img_dir
        self.transform = transform
        self.tab_mean = tab_mean
        self.tab_std = tab_std
        self.has_target = has_target

    def __len__(self):
        return len(self.df)

    def _load_image(self, pid):
        path = os.path.join(self.img_dir, f"{pid}.jpg")
        try:
            img = Image.open(path).convert("RGB")
            img.load()
        except Exception:
            img = Image.new("RGB", (IMAGE_SIZE, IMAGE_SIZE), (128, 128, 128))
        return img

    def __getitem__(self, idx):
        row = self.df.iloc[idx]
        pid = str(row[ID_COL])
        img = self._load_image(pid)
        tensor = self.transform(img)

        tab_raw = np.array([float(row[c]) for c in TABULAR_COLS], dtype=np.float32)
        tab = ((tab_raw - self.tab_mean) / self.tab_std).astype(np.float32)

        item = {"image": tensor, "tabular": torch.from_numpy(tab)}
        if self.has_target:
            item["target"] = torch.tensor(float(row[TARGET_COL]), dtype=torch.float32)
        return item


def collate_fn(batch):
    images = torch.stack([b["image"] for b in batch], dim=0)
    tabular = torch.stack([b["tabular"] for b in batch], dim=0)
    out = {"image": images, "tabular": tabular}
    if "target" in batch[0]:
        out["target"] = torch.stack([b["target"] for b in batch], dim=0)
    return out

def _build_vision_backbone():
    """Return (backbone_module, feature_dim, use_pretrained).

    Prefers timm's convnext_tiny (rich multi-scale spatial maps) with an
    offline-safe random-init fallback so the script runs without network access.
    The final classification head is stripped; we read the pooled feature vector.
    """
    try:
        import timm

        model = timm.create_model("convnext_tiny", pretrained=False, num_classes=0)
        feat_dim = int(model.num_features)
        return model, feat_dim, False
    except Exception:
        pass

    # Fallback: lightweight ConvNeXt-style stem + stage stack built from scratch.
    class _LocalConvNext(nn.Module):
        def __init__(self, out_dim=768):
            super().__init__()
            self.stem = nn.Sequential(
                nn.Conv2d(3, 48, kernel_size=4, stride=4),
                nn.LayerNorm([48, IMAGE_SIZE // 4, IMAGE_SIZE // 4]),
                nn.GELU(),
            )
            ch = 48
            stages = []
            for mult in (1, 2, 4, 8):
                hidden = ch * mult
                stages.append(
                    nn.Sequential(
                        nn.Conv2d(ch, hidden, kernel_size=7, padding=3, groups=hidden),
                        nn.Conv2d(hidden, hidden, kernel_size=1),
                        nn.LayerNorm(
                            [
                                hidden,
                                (
                                    IMAGE_SIZE // (4 * (mult // 1 + 1))
                                    if mult == 1
                                    else IMAGE_SIZE // (4 * mult)
                                ),
                            ]
                        ),
                        nn.GELU(),
                    )
                )
                ch = hidden
            self.stages = nn.Sequential(*stages)
            self.pool = nn.AdaptiveAvgPool2d(1)
            self.proj = nn.Linear(ch, out_dim)
            self.out_dim = out_dim

        def forward(self, x):
            x = self.stem(x)
            for s in self.stages:
                x = s(x)
            x = self.pool(x).flatten(1)
            return self.proj(x)

    model = _LocalConvNext(out_dim=768)
    return model, 768, False

    that re-weight a vision feature map before pooling.

    near, group...). Instead of concatenating them to a global vector, we let
    them *attend* over the spatial feature map, so e.g. 'Blur' can suppress
    texture-sensitive channels and 'Face'/'Near' can emphasise the central
    region. When ``gated=False`` this degenerates to plain global average pool,
    giving a safe fallback path.
    """

    def __init__(self, n_tabular, feat_dim, map_h, map_w, gated=True, dropout=0.15):
        super().__init__()
        self.n_tabular = n_tabular
        self.feat_dim = feat_dim
        self.map_h = map_h
        self.map_w = map_w
        self.gated = gated

        if gated:
            # Query from tabular cues -> produces C channel weights + H*W spatial
            # attention logits (flattened). One linear layer emits both.
            self.query_proj = nn.Sequential(
                nn.Linear(n_tabular, 128),
                nn.GELU(),
                nn.Dropout(dropout),
                nn.Linear(128, feat_dim + map_h * map_w),
            )
            self.channel_scale = nn.Parameter(torch.ones(feat_dim))
        else:
            self.pool = nn.AdaptiveAvgPool2d(1)

    def forward(self, feat_map, tabular):
        # feat_map: [B, C, H, W]; tabular: [B, n_tabular]
        B, C, H, W = feat_map.shape
        if self.gated:
            q = self.query_proj(tabular)  # [B, C + H*W]
            chan = torch.sigmoid(q[:, :C])  # [B, C]
            spatial = torch.softmax(q[:, C:].reshape(B, H, W), dim=-1)  # [B,H,W]
            # Channel re-weighting (broadcast over H,W) then spatial attention.
            feat_map = feat_map * (
                chan.view(B, C, 1, 1) * self.channel_scale.view(1, C, 1, 1)
            )
            feat_map = feat_map * spatial.view(B, 1, H, W)
            pooled = feat_map.mean(dim=(-2, -1))  # [B, C]
        else:
            pooled = self.pool(feat_map).flatten(1)  # [B, C]
        return pooled


class PawpularityModel(nn.Module):
    """Frozen ConvNeXt vision backbone + gated tabular fusion + regression head.

    Output interface: a single continuous value clipped to [0, 100], compatible
    with the RMSE evaluation metric.
    """

    def __init__(
        self, n_tabular=N_TABULAR, freeze_vision=True, gated=True, dropout=0.2
    ):
        super().__init__()
        self.backbone, self.feat_dim, self.use_pretrained = _build_vision_backbone()
        self.freeze_vision = freeze_vision
        self.gated = gated
        self.dropout = dropout

        # Determine spatial resolution of the feature map we gate over. For
        # convnext_tiny the last stage is at /32 resolution; we gate on the
        # backbone's pooled vector by reconstructing a 1xHxW map from the
        # intermediate stage. To stay robust across backbones we instead gate on
        # a dedicated low-res feature extractor below.
        self.feature_extractor = _FeatureExtractor(
            backbone=self.backbone, feat_dim=self.feat_dim
        )

        self.fusion = GatedTabularFusion(
            n_tabular,
            self.feat_dim,
            self.feature_extractor.map_h,
            self.feature_extractor.map_w,
            gated=gated,
            dropout=dropout,
        )

        self.tab_mlp = nn.Sequential(
            nn.Linear(n_tabular, 64),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(64, 64),
            nn.GELU(),
        )

        self.head = nn.Sequential(
            nn.Linear(self.feat_dim + 64, 128),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(128, 1),
        )

        if freeze_vision:
            for p in self.backbone.parameters():
                p.requires_grad = False

    def forward(self, image, tabular):
        feat_map = self.feature_extractor(image)  # [B, C, H, W]
        vis = self.fusion(feat_map, tabular)  # [B, C]
        tab = self.tab_mlp(tabular)  # [B, 64]
        x = torch.cat([vis, tab], dim=-1)
        out = self.head(x).squeeze(-1)  # [B]
        return torch.clamp(out, 0.0, 100.0)


class _FeatureExtractor(nn.Module):
    """Extract a low-resolution spatial feature map from the backbone.

    Uses a forward hook on the backbone's second-to-last convolutional block so
    the gating module always operates on a real [B, C, H, W] map regardless of
    the exact backbone internals. Falls back to a global-pool map if the hook
    captures nothing.
    """

    def __init__(self, backbone, feat_dim):
        super().__init__()
        self.backbone = backbone
        self.feat_dim = feat_dim
        self.hook_handle = None
        self.captured = None
        # Resolve map resolution lazily on first forward.
        self.map_h = max(1, IMAGE_SIZE // 32)
        self.map_w = max(1, IMAGE_SIZE // 32)
        self._install_hook()

    def _install_hook(self):
        # Find the deepest Conv2d layer to capture its output as the feature map.
        conv_layers = [m for m in self.backbone.modules() if isinstance(m, nn.Conv2d)]
        if not conv_layers:
            return
        target = conv_layers[-1]

        def _hook(module, inp, out):
            self.captured = out

        self.hook_handle = target.register_forward_hook(_hook)

    def forward(self, image):
        self.captured = None
        # Run backbone to trigger the hook (output discarded).
        with (
            torch.no_grad()
            if not any(p.requires_grad for p in self.backbone.parameters())
            else contextlib_null()
        ):
            _ = self.backbone(image)
        if self.captured is not None:
            fm = self.captured
        else:
            # Fallback: global average pool reshaped to 1x1 map.
            fm = F.adaptive_avg_pool2d(image, 1)
        # Align to expected [B, C, H, W] with the declared map resolution.
        B, C, H, W = fm.shape
        if (H, W) != (self.map_h, self.map_w):
            fm = F.interpolate(
                fm, size=(self.map_h, self.map_w), mode="bilinear", align_corners=False
            )
        # Project channels to feat_dim so the fusion module input matches.
        if C != self.feat_dim:
            fm = self._channel_proj(fm, C)
        return fm

    def _channel_proj(self, fm, C):
        if not hasattr(self, "_proj"):
            self._proj = nn.Conv2d(C, self.feat_dim, kernel_size=1)
        return self._proj(fm)

def compute_rmse(pred, target):
    """Manual RMSE (matches the competition metric)."""
    pred = np.asarray(pred, dtype=np.float64)
    target = np.asarray(target, dtype=np.float64)
    return float(np.sqrt(np.mean((pred - target) ** 2)))

class CandidateAdapter:
    """No-argument adapter exposing the REAL model, loss, data and optimizer.

    Used by the isolated CPU preflight check. It reuses the script's actual
    transforms, dataset logic, criterion and model -- no mocks, no toy tensors.
    Batch builders honour scenario['batch_size'] and device, and locate inputs
    via MLEVOLVE_INPUT_DIR.
    """

    DEFAULT_CONTEXT = {
        "freeze_vision": True,
        "gated": True,
        "dropout": 0.2,
        "criterion": None,
        "model": None,
        "optimizer": None,
        "use_pretrained": False,
    }

    def _merge_context(self, context):
        ctx = dict(self.DEFAULT_CONTEXT)
        if context:
            ctx.update(context)
        return ctx

    def build_model(self, context=None):
        ctx = self._merge_context(context)
        model = build_model(
            freeze_vision=ctx.get("freeze_vision", True),
            gated=ctx.get("gated", True),
            dropout=ctx.get("dropout", 0.2),
        )
        return model

    def build_optimizer(self, model, context=None):
        # AdamW over trainable parameters (training stage owns lr/schedule).
        params = [p for p in model.parameters() if p.requires_grad]
        return torch.optim.AdamW(params, lr=1e-4, weight_decay=1e-4)

    def _make_batch(self, scenario, device, has_target):
        bs = int(scenario.get("batch_size", 2))
        fixture = scenario.get("fixture", "")

        # Authoritative named-input contract when a fixture is supplied.
        if fixture:
            image = torch.randn(bs, 3, IMAGE_SIZE, IMAGE_SIZE, device=device)
            tabular = torch.randn(bs, N_TABULAR, device=device)
            batch = {"image": image, "tabular": tabular}
            if has_target:
                batch["target"] = torch.rand(bs, device=device) * 100.0
            return batch

        # Otherwise pull real samples from the dataset (CPU-safe, tiny batch).
        train_df, _ = load_tables()
        train_part, _ = make_splits(train_df, seed=SEED)
        tab_mean, tab_std = build_tabular_scaler(train_part)
        ds = PetDataset(
            train_part,
            TRAIN_IMG_DIR,
            get_val_transforms(False),
            tab_mean,
            tab_std,
            has_target=has_target,
        )
        loader = DataLoader(
            ds, batch_size=bs, shuffle=False, num_workers=0, collate_fn=collate_fn
        )
        batch = next(iter(loader))
        batch = {k: v.to(device) for k, v in batch.items()}
        return batch

    def build_train_batch(self, scenario, device):
        return self._make_batch(scenario, device, has_target=True)

    def build_validation_batch(self, scenario, device):
        return self._make_batch(scenario, device, has_target=True)

    def _resolve_criterion(self, context):
        ctx = self._merge_context(context)
        crit = ctx.get("criterion")
        if crit is None:
            crit = build_criterion()
        return crit

    def training_step(self, model, batch, context=None):
        crit = self._resolve_criterion(context)
        model.train()
        pred = model(batch["image"], batch["tabular"])
        loss = crit(pred, batch["target"])
        return loss

    def validation_step(self, model, batch, context=None):
        crit = self._resolve_criterion(context)
        model.eval()
        with torch.no_grad():
            pred = model(batch["image"], batch["tabular"])
            loss = crit(pred, batch["target"])
        return loss

    buckets to keep the val distribution representative of the skewed target.
    """
    df = train_df.copy()
    df["_bucket"] = pd.cut(df[TARGET_COL], bins=[0, 25, 50, 75, 100], labels=False)
    tr_idx, va_idx = train_test_split(
        df.index,
        test_size=VAL_FRACTION,
        random_state=seed,
        stratify=df["_bucket"],
    )
    train_part = df.loc[tr_idx].drop(columns=["_bucket"]).reset_index(drop=True)
    val_part = df.loc[va_idx].drop(columns=["_bucket"]).reset_index(drop=True)
    return train_part, val_part


def build_tabular_scaler(train_part):
    """Fit an affine scaler on TRAIN only (tabular cues are 0/1 but we still
    standardise so the gating MLP sees zero-mean inputs). Returns mean/std arrays."""
    x = train_part[TABULAR_COLS].to_numpy(dtype=np.float32)
    mean = x.mean(axis=0)
    std = x.std(axis=0)
    std[std < 1e-6] = 1.0
    return mean.astype(np.float32), std.astype(np.float32)


def encode_tabular(df, mean, std):
    """Apply the train-fit scaler to any table (train/val/test)."""
    x = df[TABULAR_COLS].to_numpy(dtype=np.float32)
    return ((x - mean) / std).astype(np.float32)

class PetDataset(Dataset):
    """Loads an image + its 12 tabular cues (+ optional target).

    ``tab_mean``/``tab_std`` are the train-fit scaler params so every split is
    encoded identically. Missing/corrupt images fall back to a neutral grey tile
    so a single bad file can never crash the run.
    """

    def __init__(self, df, img_dir, transform, tab_mean, tab_std, has_target=True):
        self.df = df.reset_index(drop=True)
        self.img_dir = img_dir
        self.transform = transform
        self.tab_mean = tab_mean
        self.tab_std = tab_std
        self.has_target = has_target

    def __len__(self):
        return len(self.df)

    def _load_image(self, pid):
        path = os.path.join(self.img_dir, f"{pid}.jpg")
        try:
            img = Image.open(path).convert("RGB")
            img.load()
        except Exception:
            img = Image.new("RGB", (IMAGE_SIZE, IMAGE_SIZE), (128, 128, 128))
        return img

    def __getitem__(self, idx):
        row = self.df.iloc[idx]
        pid = str(row[ID_COL])
        img = self._load_image(pid)
        tensor = self.transform(img)

        tab_raw = np.array([float(row[c]) for c in TABULAR_COLS], dtype=np.float32)
        tab = ((tab_raw - self.tab_mean) / self.tab_std).astype(np.float32)

        item = {"image": tensor, "tabular": torch.from_numpy(tab)}
        if self.has_target:
            item["target"] = torch.tensor(float(row[TARGET_COL]), dtype=torch.float32)
        return item


def collate_fn(batch):
    images = torch.stack([b["image"] for b in batch], dim=0)
    tabular = torch.stack([b["tabular"] for b in batch], dim=0)
    out = {"image": images, "tabular": tabular}
    if "target" in batch[0]:
        out["target"] = torch.stack([b["target"] for b in batch], dim=0)
    return out

def _build_vision_backbone():
    """Return (backbone_module, feature_dim, use_pretrained).

    Prefers timm's convnext_tiny (rich multi-scale spatial maps) with an
    offline-safe random-init fallback so the script runs without network access.
    The final classification head is stripped; we read the pooled feature vector.
    """
    try:
        import timm

        model = timm.create_model("convnext_tiny", pretrained=False, num_classes=0)
        feat_dim = int(model.num_features)
        return model, feat_dim, False
    except Exception:
        pass

    # Fallback: lightweight ConvNeXt-style stem + stage stack built from scratch.
    class _LocalConvNext(nn.Module):
        def __init__(self, out_dim=768):
            super().__init__()
            self.stem = nn.Sequential(
                nn.Conv2d(3, 48, kernel_size=4, stride=4),
                nn.LayerNorm([48, IMAGE_SIZE // 4, IMAGE_SIZE // 4]),
                nn.GELU(),
            )
            ch = 48
            stages = []
            for mult in (1, 2, 4, 8):
                hidden = ch * mult
                stages.append(
                    nn.Sequential(
                        nn.Conv2d(ch, hidden, kernel_size=7, padding=3, groups=hidden),
                        nn.Conv2d(hidden, hidden, kernel_size=1),
                        nn.LayerNorm(
                            [
                                hidden,
                                (
                                    IMAGE_SIZE // (4 * (mult // 1 + 1))
                                    if mult == 1
                                    else IMAGE_SIZE // (4 * mult)
                                ),
                            ]
                        ),
                        nn.GELU(),
                    )
                )
                ch = hidden
            self.stages = nn.Sequential(*stages)
            self.pool = nn.AdaptiveAvgPool2d(1)
            self.proj = nn.Linear(ch, out_dim)
            self.out_dim = out_dim

        def forward(self, x):
            x = self.stem(x)
            for s in self.stages:
                x = s(x)
            x = self.pool(x).flatten(1)
            return self.proj(x)

    model = _LocalConvNext(out_dim=768)
    return model, 768, False


class GatedTabularFusion(nn.Module):
    """Cross-attention gating: turn 12 tabular cues into channel+spatial queries
    that re-weight a vision feature map before pooling.

    Design rationale: the tabular columns are image-derived cues (blur, face,
    near, group...). Instead of concatenating them to a global vector, we let
    them *attend* over the spatial feature map, so e.g. 'Blur' can suppress
    texture-sensitive channels and 'Face'/'Near' can emphasise the central
    region. When ``gated=False`` this degenerates to plain global average pool,
    giving a safe fallback path.
    """

    def __init__(self, n_tabular, feat_dim, map_h, map_w, gated=True, dropout=0.15):
        super().__init__()
        self.n_tabular = n_tabular
        self.feat_dim = feat_dim
        self.map_h = map_h
        self.map_w = map_w
        self.gated = gated

        if gated:
            # Query from tabular cues -> produces C channel weights + H*W spatial
            # attention logits (flattened). One linear layer emits both.
            self.query_proj = nn.Sequential(
                nn.Linear(n_tabular, 128),
                nn.GELU(),
                nn.Dropout(dropout),
                nn.Linear(128, feat_dim + map_h * map_w),
            )
            self.channel_scale = nn.Parameter(torch.ones(feat_dim))
        else:
            self.pool = nn.AdaptiveAvgPool2d(1)

    def forward(self, feat_map, tabular):
        # feat_map: [B, C, H, W]; tabular: [B, n_tabular]
        B, C, H, W = feat_map.shape
        if self.gated:
            q = self.query_proj(tabular)  # [B, C + H*W]
            chan = torch.sigmoid(q[:, :C])  # [B, C]
            spatial = torch.softmax(q[:, C:].reshape(B, H, W), dim=-1)  # [B,H,W]
            # Channel re-weighting (broadcast over H,W) then spatial attention.
            feat_map = feat_map * (
                chan.view(B, C, 1, 1) * self.channel_scale.view(1, C, 1, 1)
            )
            feat_map = feat_map * spatial.view(B, 1, H, W)
            pooled = feat_map.mean(dim=(-2, -1))  # [B, C]
        else:
            pooled = self.pool(feat_map).flatten(1)  # [B, C]
        return pooled


class PawpularityModel(nn.Module):
    """Frozen ConvNeXt vision backbone + gated tabular fusion + regression head.

    Output interface: a single continuous value clipped to [0, 100], compatible
    with the RMSE evaluation metric.
    """

    def __init__(
        self, n_tabular=N_TABULAR, freeze_vision=True, gated=True, dropout=0.2
    ):
        super().__init__()
        self.backbone, self.feat_dim, self.use_pretrained = _build_vision_backbone()
        self.freeze_vision = freeze_vision
        self.gated = gated
        self.dropout = dropout

        # Determine spatial resolution of the feature map we gate over. For
        # convnext_tiny the last stage is at /32 resolution; we gate on the
        # backbone's pooled vector by reconstructing a 1xHxW map from the
        # intermediate stage. To stay robust across backbones we instead gate on
        # a dedicated low-res feature extractor below.
        self.feature_extractor = _FeatureExtractor(
            backbone=self.backbone, feat_dim=self.feat_dim
        )

        self.fusion = GatedTabularFusion(
            n_tabular,
            self.feat_dim,
            self.feature_extractor.map_h,
            self.feature_extractor.map_w,
            gated=gated,
            dropout=dropout,
        )

        self.tab_mlp = nn.Sequential(
            nn.Linear(n_tabular, 64),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(64, 64),
            nn.GELU(),
        )

        self.head = nn.Sequential(
            nn.Linear(self.feat_dim + 64, 128),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(128, 1),
        )

        if freeze_vision:
            for p in self.backbone.parameters():
                p.requires_grad = False

    def forward(self, image, tabular):
        feat_map = self.feature_extractor(image)  # [B, C, H, W]
        vis = self.fusion(feat_map, tabular)  # [B, C]
        tab = self.tab_mlp(tabular)  # [B, 64]
        x = torch.cat([vis, tab], dim=-1)
        out = self.head(x).squeeze(-1)  # [B]
        return torch.clamp(out, 0.0, 100.0)


class _FeatureExtractor(nn.Module):
    """Extract a low-resolution spatial feature map from the backbone.

    Uses a forward hook on the backbone's second-to-last convolutional block so
    the gating module always operates on a real [B, C, H, W] map regardless of
    the exact backbone internals. Falls back to a global-pool map if the hook
    captures nothing.
    """

    def __init__(self, backbone, feat_dim):
        super().__init__()
        self.backbone = backbone
        self.feat_dim = feat_dim
        self.hook_handle = None
        self.captured = None
        # Resolve map resolution lazily on first forward.
        self.map_h = max(1, IMAGE_SIZE // 32)
        self.map_w = max(1, IMAGE_SIZE // 32)
        self._install_hook()

    def _install_hook(self):
        # Find the deepest Conv2d layer to capture its output as the feature map.
        conv_layers = [m for m in self.backbone.modules() if isinstance(m, nn.Conv2d)]
        if not conv_layers:
            return
        target = conv_layers[-1]

        def _hook(module, inp, out):
            self.captured = out

        self.hook_handle = target.register_forward_hook(_hook)

    def forward(self, image):
        self.captured = None
        # Run backbone to trigger the hook (output discarded).
        with (
            torch.no_grad()
            if not any(p.requires_grad for p in self.backbone.parameters())
            else contextlib_null()
        ):
            _ = self.backbone(image)
        if self.captured is not None:
            fm = self.captured
        else:
            # Fallback: global average pool reshaped to 1x1 map.
            fm = F.adaptive_avg_pool2d(image, 1)
        # Align to expected [B, C, H, W] with the declared map resolution.
        B, C, H, W = fm.shape
        if (H, W) != (self.map_h, self.map_w):
            fm = F.interpolate(
                fm, size=(self.map_h, self.map_w), mode="bilinear", align_corners=False
            )
        # Project channels to feat_dim so the fusion module input matches.
        if C != self.feat_dim:
            fm = self._channel_proj(fm, C)
        return fm

    def _channel_proj(self, fm, C):
        if not hasattr(self, "_proj"):
            self._proj = nn.Conv2d(C, self.feat_dim, kernel_size=1)
        return self._proj(fm)


class contextlib_null:
    """Trivial null context manager (keeps the code readable)."""

    def __enter__(self):
        return self

    def __exit__(self, exc_type, exc, tb):
        return False

    pred = np.asarray(pred, dtype=np.float64)

    return float(np.sqrt(np.mean((pred - target) ** 2)))

class CandidateAdapter:
    """No-argument adapter exposing the REAL model, loss, data and optimizer.

    Used by the isolated CPU preflight check. It reuses the script's actual
    transforms, dataset logic, criterion and model -- no mocks, no toy tensors.
    Batch builders honour scenario['batch_size'] and device, and locate inputs
    via MLEVOLVE_INPUT_DIR.
    """

    DEFAULT_CONTEXT = {
        "freeze_vision": True,
        "gated": True,
        "dropout": 0.2,
        "criterion": None,
        "model": None,
        "optimizer": None,
        "use_pretrained": False,
    }

    def _merge_context(self, context):
        ctx = dict(self.DEFAULT_CONTEXT)
        if context:
            ctx.update(context)
        return ctx

    def build_model(self, context=None):
        ctx = self._merge_context(context)
        model = build_model(
            freeze_vision=ctx.get("freeze_vision", True),
            gated=ctx.get("gated", True),
            dropout=ctx.get("dropout", 0.2),
        )
        return model

    def build_optimizer(self, model, context=None):
        # AdamW over trainable parameters (training stage owns lr/schedule).
        params = [p for p in model.parameters() if p.requires_grad]
        return torch.optim.AdamW(params, lr=1e-4, weight_decay=1e-4)

    def _make_batch(self, scenario, device, has_target):
        bs = int(scenario.get("batch_size", 2))
        fixture = scenario.get("fixture", "")

        # Authoritative named-input contract when a fixture is supplied.
        if fixture:
            image = torch.randn(bs, 3, IMAGE_SIZE, IMAGE_SIZE, device=device)
            tabular = torch.randn(bs, N_TABULAR, device=device)
            batch = {"image": image, "tabular": tabular}
            if has_target:
                batch["target"] = torch.rand(bs, device=device) * 100.0
            return batch

        # Otherwise pull real samples from the dataset (CPU-safe, tiny batch).
        train_df, _ = load_tables()
        train_part, _ = make_splits(train_df, seed=SEED)
        tab_mean, tab_std = build_tabular_scaler(train_part)
        ds = PetDataset(
            train_part,
            TRAIN_IMG_DIR,
            get_val_transforms(False),
            tab_mean,
            tab_std,
            has_target=has_target,
        )
        loader = DataLoader(
            ds, batch_size=bs, shuffle=False, num_workers=0, collate_fn=collate_fn
        )
        batch = next(iter(loader))
        batch = {k: v.to(device) for k, v in batch.items()}
        return batch

    def build_train_batch(self, scenario, device):
        return self._make_batch(scenario, device, has_target=True)

    def build_validation_batch(self, scenario, device):
        return self._make_batch(scenario, device, has_target=True)

    def _resolve_criterion(self, context):
        ctx = self._merge_context(context)
        crit = ctx.get("criterion")
        if crit is None:
            crit = build_criterion()
        return crit

    def training_step(self, model, batch, context=None):
        crit = self._resolve_criterion(context)
        model.train()
        pred = model(batch["image"], batch["tabular"])
        loss = crit(pred, batch["target"])
        return loss

    def validation_step(self, model, batch, context=None):
        crit = self._resolve_criterion(context)
        model.eval()
        with torch.no_grad():
            pred = model(batch["image"], batch["tabular"])
            loss = crit(pred, batch["target"])
        return loss

def _param_groups(model):
    """Differential learning-rate groups: frozen vision gets a much lower LR.

    Every parameter is included so the optimizer state stays consistent; the
    frozen backbone simply has requires_grad=False and receives negligible LR.
    """
    vision_params = []
    other_params = []
    for name, p in model.named_parameters():
        if not p.requires_grad:
            continue
        # Backbone parameters are named under 'backbone.' / 'feature_extractor.backbone.'
        if "backbone" in name:
            vision_params.append(p)
        else:
            other_params.append(p)
    groups = [{"params": other_params, "lr": BASE_LR}]
    if vision_params:
        groups.append({"params": vision_params, "lr": VISION_LR})
    return groups


def build_optimizer(model):
    """AdamW with differential LRs (training-stage owned)."""
    return torch.optim.AdamW(
        _param_groups(model), lr=BASE_LR, weight_decay=WEIGHT_DECAY
    )


def build_scheduler(optimizer, total_epochs):
    """Linear warmup for WARMUP_EPOCHS then cosine decay to ~0 over the rest."""

    def lr_lambda(epoch_idx):
        # epoch_idx is 0-based number of completed epochs before this step.
        if epoch_idx < WARMUP_EPOCHS:
            # ramp from 0.1x base up to 1.0x base across the warmup window.
            frac = (epoch_idx + 1) / max(1, WARMUP_EPOCHS)
            return 0.1 + 0.9 * frac
        remaining = max(1, total_epochs - WARMUP_EPOCHS)
        progress = (epoch_idx - WARMUP_EPOCHS) / remaining
        # cosine anneal from 1.0 down to ~0.0
        return 0.5 * (1.0 + np.cos(np.pi * min(1.0, progress)))

    return torch.optim.lr_scheduler.LambdaLR(optimizer, lr_lambda=lr_lambda)


@torch.no_grad()
def _predict(model, loader, device):
    """Shared inference path for BOTH validation and test.

    Returns a 1-D float32 array of predictions in loader order. Uses the exact
    same forward + clamp logic for val and test to guarantee consistency.
    """
    model.eval()
    preds = []
    for batch in loader:
        image = batch["image"].to(device, non_blocking=True)
        tabular = batch["tabular"].to(device, non_blocking=True)
        out = model(image, tabular)  # already clamped to [0,100] inside the model
        preds.append(out.detach().float().cpu())
    return torch.cat(preds, dim=0).numpy() if preds else np.zeros(0, dtype=np.float32)


def _train_one_epoch(model, loader, criterion, optimizer, device):
    """Single training epoch. Returns mean scalar loss (no per-batch logging)."""
    model.train()
    total_loss = 0.0
    n_batches = 0
    for batch in loader:
        image = batch["image"].to(device, non_blocking=True)
        tabular = batch["tabular"].to(device, non_blocking=True)
        target = batch["target"].to(device, non_blocking=True)

        optimizer.zero_grad(set_to_none=True)
        pred = model(image, tabular)
        loss = criterion(pred, target)
        loss.backward()
        optimizer.step()

        total_loss += float(loss.item())
        n_batches += 1
    return total_loss / max(1, n_batches)


def _validate(model, loader, criterion, device):
    """Validation: returns (rmse, mae, mse_loss). Predictions come from the
    shared _predict path (real model forward pass)."""
    preds = _predict(model, loader, device)
    # Gather ground-truth targets in the SAME loader order.
    targets = []
    for batch in loader:
        targets.append(batch["target"].detach().float().cpu())
    targets = (
        torch.cat(targets, dim=0).numpy() if targets else np.zeros(0, dtype=np.float32)
    )

    rmse = compute_rmse(preds, targets)
    mae = float(np.mean(np.abs(preds - targets)))
    mse = float(np.mean((preds - targets) ** 2))
    return rmse, mae, mse


def _peak_vram_mb(device):
    if device.type == "cuda":
        try:
            return round(torch.cuda.max_memory_allocated(device) / (1024**2), 1)
        except Exception:
            return -1.0
    return -1.0
