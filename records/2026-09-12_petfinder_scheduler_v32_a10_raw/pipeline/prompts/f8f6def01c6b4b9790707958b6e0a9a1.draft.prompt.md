class _RandomErasing(nn.Module):
    """Randomly erase a rectangular patch (probabilistic occlusion robustness)."""

    def __init__(self, p: float = 0.4, scale: tuple = (0.02, 0.25)):
        super().__init__()
        self.p = p
        self.scale = scale

    def forward(self, img: torch.Tensor) -> torch.Tensor:
        if self.p <= 0 or not self.training:
            return img
        h, w = img.shape[-2:]
        area = h * w
        for _ in range(10):
            target_area = random.uniform(*self.scale) * area
            aspect = random.uniform(0.3, 3.3)
            eh = max(1, int(math.sqrt(target_area * aspect)))
            ew = max(1, int(math.sqrt(target_area / aspect)))
            if eh <= h and ew <= w:
                y = random.randint(0, h - eh)
                x = random.randint(0, w - ew)
                mask = torch.ones_like(img)
                mask[..., y : y + eh, x : x + ew] = 0.0
                return img * mask
        return img

def load_meta(split: str) -> pd.DataFrame:
    """Load the metadata CSV for a given split ('train' or 'test')."""

def make_splits(train_df: pd.DataFrame, val_frac: float = VAL_FRAC, seed: int = SEED):
    """Stratified-by-bucket reproducible train/validation split.

    Returns (train_ids, val_ids) preserving original row order within each set.
    The split is performed BEFORE any stateful transformation so no leakage.
    """
    rng = np.random.default_rng(seed)
    buckets = np.digitize(
        train_df["Pawpularity"].to_numpy(), bins=np.linspace(0, 100, 11)
    )
    ids = train_df["Id"].to_numpy()
    train_idx, val_idx = [], []
    for b in np.unique(buckets):
        mask = np.where(buckets == b)[0]
        perm = rng.permutation(mask)
        n_val = max(1, int(round(len(mask) * val_frac)))
        val_idx.extend(perm[:n_val])
        train_idx.extend(perm[n_val:])
    train_ids = list(ids[train_idx])
    val_ids = list(ids[val_idx])
    return train_ids, val_ids


class PetDataset(Dataset):
    """Loads one JPEG + its 12 tabular flags; returns tensors ready for the model."""

    def __init__(self, meta: pd.DataFrame, img_dir: str, transform, has_target: bool):
        self.meta = meta.reset_index(drop=True)
        self.img_dir = img_dir
        self.transform = transform
        self.has_target = has_target

    def __len__(self) -> int:
        return len(self.meta)

    def _load_image(self, path: str) -> torch.Tensor:
        try:
            from PIL import Image, ImageOps

            img = Image.open(path).convert("RGB")
            img = ImageOps.exif_transpose(img)
        except Exception:
            img = Image.new("RGB", (IMG_SIZE, IMG_SIZE), (128, 128, 128))
        return self.transform(img)

    def __getitem__(self, idx: int):
        row = self.meta.iloc[idx]
        path = os.path.join(self.img_dir, f"{row['Id']}.jpg")
        image = self._load_image(path)
        tab = torch.tensor(
            row[TABULAR_COLS].astype(np.float32).to_numpy(), dtype=torch.float32
        )
        item = {"image": image, "tabular": tab}
        if self.has_target:
            item["target"] = torch.tensor(
                float(row["Pawpularity"]), dtype=torch.float32
            )
        return item


def collate_fn(batch):
    images = torch.stack([b["image"] for b in batch], dim=0)
    tabular = torch.stack([b["tabular"] for b in batch], dim=0)
    out = {"image": images, "tabular": tabular}
    if "target" in batch[0]:
        out["target"] = torch.stack([b["target"] for b in batch], dim=0)
    return out


def build_datasets_and_loaders():
    """Construct datasets + dataloaders for train/val/test.

    Dataloader hyper-parameters (batch size, workers, pin_memory) are intentionally
    generic defaults owned by the training stage; this function only wires the
    Stage-1 data objects together.
    """
    train_df = load_meta("train")
    test_df = load_meta("test")
    train_ids, val_ids = make_splits(train_df)

    train_meta = train_df[train_df["Id"].isin(train_ids)].reset_index(drop=True)
    val_meta = train_df[train_df["Id"].isin(val_ids)].reset_index(drop=True)

    train_ds = PetDataset(
        train_meta, TRAIN_IMG_DIR, build_train_transform(), has_target=True
    )
    val_ds = PetDataset(
        val_meta, VAL_IMG_DIR_OR_TRAIN(), build_eval_transform(), has_target=True
    )
    test_ds = PetDataset(
        test_df, TEST_IMG_DIR, build_eval_transform(), has_target=False
    )

    common = dict(num_workers=2, pin_memory=torch.cuda.is_available())
    train_loader = DataLoader(
        train_ds,
        batch_size=32,
        shuffle=True,
        collate_fn=collate_fn,
        drop_last=False,
        **common,
    )
    val_loader = DataLoader(
        val_ds, batch_size=32, shuffle=False, collate_fn=collate_fn, **common
    )
    test_loader = DataLoader(
        test_ds, batch_size=32, shuffle=False, collate_fn=collate_fn, **common
    )
    return train_loader, val_loader, test_loader, train_df, test_df

def _load_pretrained_resnet50() -> nn.Module:
    """Offline-safe pretrained ResNet50 loader with random-init fallback."""
    try:
        net = tv_models.resnet50(weights=tv_models.ResNet50_Weights.DEFAULT)

        net = tv_models.resnet50(weights=None)

class GatedCrossAttention(nn.Module):
    """Small gated cross-attention: tabular query attends to visual keys/values."""

    def __init__(self, dim: int, heads: int = 4, dropout: float = DROPOUT):
        super().__init__()
        assert dim % heads == 0, "dim must be divisible by heads"
        self.heads = heads
        self.head_dim = dim // heads
        self.q_proj = nn.Linear(dim, dim)
        self.k_proj = nn.Linear(dim, dim)
        self.v_proj = nn.Linear(dim, dim)
        self.out_proj = nn.Linear(dim, dim)
        self.gate = nn.Sigmoid()
        self.dropout = nn.Dropout(dropout)
        self.norm_q = nn.LayerNorm(dim)
        self.norm_kv = nn.LayerNorm(dim)

    def forward(self, q: torch.Tensor, kv: torch.Tensor) -> torch.Tensor:
        B, Lq, C = q.shape
        Lk = kv.shape[1]
        q = self.norm_q(q)
        kv = self.norm_kv(kv)
        q = self.q_proj(q).view(B, Lq, self.heads, self.head_dim).transpose(1, 2)
        k = self.k_proj(kv).view(B, Lk, self.heads, self.head_dim).transpose(1, 2)
        v = self.v_proj(kv).view(B, Lk, self.heads, self.head_dim).transpose(1, 2)
        scale = math.sqrt(self.head_dim)
        attn = torch.matmul(q, k.transpose(-2, -1)) / scale
        attn = F.softmax(attn, dim=-1)
        attn = self.dropout(attn)
        out = torch.matmul(attn, v)
        out = out.transpose(1, 2).reshape(B, Lq, C)
        out = self.out_proj(out)
        return out


class PawpularityModel(nn.Module):
    """Dual-pathway fusion: frozen ResNet50 (vision) + MLP (tabular) -> scalar."""

    def __init__(self, freeze_backbone: bool = True, dropout: float = DROPOUT):
        super().__init__()
        self.backbone = _load_pretrained_resnet50()
        # Replace final FC with an identity; we read the pooled feature instead.
        feat_dim = self.backbone.fc.in_features  # 2048
        self.backbone.fc = nn.Identity()
        if freeze_backbone:
            for p in self.backbone.parameters():
                p.requires_grad = False

        # Vision projection down to fusion width.
        self.vis_proj = nn.Sequential(
            nn.Linear(feat_dim, FUSE_DIM),
            nn.GELU(),
            nn.Dropout(dropout),
        )
        # Tabular MLP -> fusion width.
        self.tab_mlp = nn.Sequential(
            nn.Linear(TAB_DIM, 64),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(64, FUSE_DIM),
            nn.GELU(),
        )
        # Cross-attention fusion (tabular queries attend to visual tokens).
        self.attn = GatedCrossAttention(FUSE_DIM, heads=4, dropout=dropout)
        self.fuse_norm = nn.LayerNorm(FUSE_DIM)
        # Residual regression head.
        self.head = nn.Sequential(
            nn.Linear(FUSE_DIM * 2, FUSE_DIM),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(FUSE_DIM, 1),
        )

    def forward(self, image: torch.Tensor, tabular: torch.Tensor) -> torch.Tensor:
        vis_feat = self.backbone(image)  # [B, 2048]
        vis = self.vis_proj(vis_feat)  # [B, FUSE_DIM]
        tab = self.tab_mlp(tabular)  # [B, FUSE_DIM]
        # Treat each pathway as a single token for cross-attention.
        q = tab.unsqueeze(1)  # [B, 1, FUSE_DIM]
        kv = vis.unsqueeze(1)  # [B, 1, FUSE_DIM]
        attended = self.attn(q, kv)  # [B, 1, FUSE_DIM]
        attended = attended.squeeze(1)  # [B, FUSE_DIM]
        fused = self.fuse_norm(attended + tab)  # residual gate
        combined = torch.cat([fused, vis], dim=1)  # [B, 2*FUSE_DIM]
        logit = self.head(combined).squeeze(-1)  # [B]
        return logit

# Loss / criterion                                                             #
# --------------------------------------------------------------------------- #
class SmoothedMSELoss(nn.Module):
    """MSE against a label smoothed toward the training-set mean.

    y_smooth = (1 - eps) * y_true + eps * mu_train
    This damps overfitting to individual noisy subjective ratings while keeping
    the gradient direction aligned with the true target.
    """

    def __init__(self, mu: float, eps: float = SMOOTH_EPS):
        super().__init__()
        self.register_buffer("mu", torch.tensor(float(mu)))
        self.eps = float(eps)

    def forward(self, pred: torch.Tensor, target: torch.Tensor) -> torch.Tensor:
        smooth = (1.0 - self.eps) * target + self.eps * self.mu.to(target.dtype)
        return F.mse_loss(pred, smooth)


def compute_rmse(pred: torch.Tensor, target: torch.Tensor) -> float:
    """Root mean squared error (the competition selection metric)."""
    pred = pred.detach().float()
    target = target.detach().float()
    return float(torch.sqrt(F.mse_loss(pred, target)).item())

    model: nn.Module, image: torch.Tensor, tabular: torch.Tensor

    model.eval()

        ctx = self._ctx(context)

        trainable = [p for p in model.parameters() if p.requires_grad]
        return torch.optim.AdamW(trainable, lr=lr, weight_decay=wd)

class _RandomErasing(nn.Module):
    """Randomly erase a rectangular patch (probabilistic occlusion robustness)."""

    def __init__(self, p: float = 0.4, scale: tuple = (0.02, 0.25)):
        super().__init__()
        self.p = p
        self.scale = scale

    def forward(self, img: torch.Tensor) -> torch.Tensor:
        if self.p <= 0 or not self.training:
            return img
        h, w = img.shape[-2:]
        area = h * w
        for _ in range(10):
            target_area = random.uniform(*self.scale) * area
            aspect = random.uniform(0.3, 3.3)
            eh = max(1, int(math.sqrt(target_area * aspect)))
            ew = max(1, int(math.sqrt(target_area / aspect)))
            if eh <= h and ew <= w:
                y = random.randint(0, h - eh)
                x = random.randint(0, w - ew)
                mask = torch.ones_like(img)
                mask[..., y : y + eh, x : x + ew] = 0.0
                return img * mask
        return img

def make_splits(train_df: pd.DataFrame, val_frac: float = VAL_FRAC, seed: int = SEED):
    """Stratified-by-bucket reproducible train/validation split.

    Returns (train_ids, val_ids) preserving original row order within each set.
    The split is performed BEFORE any stateful transformation so no leakage.
    """
    rng = np.random.default_rng(seed)
    buckets = np.digitize(
        train_df["Pawpularity"].to_numpy(), bins=np.linspace(0, 100, 11)
    )
    ids = train_df["Id"].to_numpy()
    train_idx, val_idx = [], []
    for b in np.unique(buckets):
        mask = np.where(buckets == b)[0]
        perm = rng.permutation(mask)
        n_val = max(1, int(round(len(mask) * val_frac)))
        val_idx.extend(perm[:n_val])
        train_idx.extend(perm[n_val:])
    train_ids = list(ids[train_idx])
    val_ids = list(ids[val_idx])
    return train_ids, val_ids


class PetDataset(Dataset):
    """Loads one JPEG + its 12 tabular flags; returns tensors ready for the model."""

    def __init__(self, meta: pd.DataFrame, img_dir: str, transform, has_target: bool):
        self.meta = meta.reset_index(drop=True)
        self.img_dir = img_dir
        self.transform = transform
        self.has_target = has_target

    def __len__(self) -> int:
        return len(self.meta)

    def _load_image(self, path: str) -> torch.Tensor:
        try:
            from PIL import Image, ImageOps

            img = Image.open(path).convert("RGB")
            img = ImageOps.exif_transpose(img)
        except Exception:
            img = Image.new("RGB", (IMG_SIZE, IMG_SIZE), (128, 128, 128))
        return self.transform(img)

    def __getitem__(self, idx: int):
        row = self.meta.iloc[idx]
        path = os.path.join(self.img_dir, f"{row['Id']}.jpg")
        image = self._load_image(path)
        tab = torch.tensor(
            row[TABULAR_COLS].astype(np.float32).to_numpy(), dtype=torch.float32
        )
        item = {"image": image, "tabular": tab}
        if self.has_target:
            item["target"] = torch.tensor(
                float(row["Pawpularity"]), dtype=torch.float32
            )
        return item


def collate_fn(batch):
    images = torch.stack([b["image"] for b in batch], dim=0)
    tabular = torch.stack([b["tabular"] for b in batch], dim=0)
    out = {"image": images, "tabular": tabular}
    if "target" in batch[0]:
        out["target"] = torch.stack([b["target"] for b in batch], dim=0)
    return out

class GatedCrossAttention(nn.Module):
    """Small gated cross-attention: tabular query attends to visual keys/values."""

    def __init__(self, dim: int, heads: int = 4, dropout: float = DROPOUT):
        super().__init__()
        assert dim % heads == 0, "dim must be divisible by heads"
        self.heads = heads
        self.head_dim = dim // heads
        self.q_proj = nn.Linear(dim, dim)
        self.k_proj = nn.Linear(dim, dim)
        self.v_proj = nn.Linear(dim, dim)
        self.out_proj = nn.Linear(dim, dim)
        self.gate = nn.Sigmoid()
        self.dropout = nn.Dropout(dropout)
        self.norm_q = nn.LayerNorm(dim)
        self.norm_kv = nn.LayerNorm(dim)

    def forward(self, q: torch.Tensor, kv: torch.Tensor) -> torch.Tensor:
        B, Lq, C = q.shape
        Lk = kv.shape[1]
        q = self.norm_q(q)
        kv = self.norm_kv(kv)
        q = self.q_proj(q).view(B, Lq, self.heads, self.head_dim).transpose(1, 2)
        k = self.k_proj(kv).view(B, Lk, self.heads, self.head_dim).transpose(1, 2)
        v = self.v_proj(kv).view(B, Lk, self.heads, self.head_dim).transpose(1, 2)
        scale = math.sqrt(self.head_dim)
        attn = torch.matmul(q, k.transpose(-2, -1)) / scale
        attn = F.softmax(attn, dim=-1)
        attn = self.dropout(attn)
        out = torch.matmul(attn, v)
        out = out.transpose(1, 2).reshape(B, Lq, C)
        out = self.out_proj(out)
        return out


class PawpularityModel(nn.Module):
    """Dual-pathway fusion: frozen ResNet50 (vision) + MLP (tabular) -> scalar."""

    def __init__(self, freeze_backbone: bool = True, dropout: float = DROPOUT):
        super().__init__()
        self.backbone = _load_pretrained_resnet50()
        # Replace final FC with an identity; we read the pooled feature instead.
        feat_dim = self.backbone.fc.in_features  # 2048
        self.backbone.fc = nn.Identity()
        if freeze_backbone:
            for p in self.backbone.parameters():
                p.requires_grad = False

        # Vision projection down to fusion width.
        self.vis_proj = nn.Sequential(
            nn.Linear(feat_dim, FUSE_DIM),
            nn.GELU(),
            nn.Dropout(dropout),
        )
        # Tabular MLP -> fusion width.
        self.tab_mlp = nn.Sequential(
            nn.Linear(TAB_DIM, 64),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(64, FUSE_DIM),
            nn.GELU(),
        )
        # Cross-attention fusion (tabular queries attend to visual tokens).
        self.attn = GatedCrossAttention(FUSE_DIM, heads=4, dropout=dropout)
        self.fuse_norm = nn.LayerNorm(FUSE_DIM)
        # Residual regression head.
        self.head = nn.Sequential(
            nn.Linear(FUSE_DIM * 2, FUSE_DIM),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(FUSE_DIM, 1),
        )

    def forward(self, image: torch.Tensor, tabular: torch.Tensor) -> torch.Tensor:
        vis_feat = self.backbone(image)  # [B, 2048]
        vis = self.vis_proj(vis_feat)  # [B, FUSE_DIM]
        tab = self.tab_mlp(tabular)  # [B, FUSE_DIM]
        # Treat each pathway as a single token for cross-attention.
        q = tab.unsqueeze(1)  # [B, 1, FUSE_DIM]
        kv = vis.unsqueeze(1)  # [B, 1, FUSE_DIM]
        attended = self.attn(q, kv)  # [B, 1, FUSE_DIM]
        attended = attended.squeeze(1)  # [B, FUSE_DIM]
        fused = self.fuse_norm(attended + tab)  # residual gate
        combined = torch.cat([fused, vis], dim=1)  # [B, 2*FUSE_DIM]
        logit = self.head(combined).squeeze(-1)  # [B]
        return logit

# Loss / criterion                                                             #
# --------------------------------------------------------------------------- #
class SmoothedMSELoss(nn.Module):
    """MSE against a label smoothed toward the training-set mean.

    y_smooth = (1 - eps) * y_true + eps * mu_train
    This damps overfitting to individual noisy subjective ratings while keeping
    the gradient direction aligned with the true target.
    """

    def __init__(self, mu: float, eps: float = SMOOTH_EPS):
        super().__init__()
        self.register_buffer("mu", torch.tensor(float(mu)))
        self.eps = float(eps)

    def forward(self, pred: torch.Tensor, target: torch.Tensor) -> torch.Tensor:
        smooth = (1.0 - self.eps) * target + self.eps * self.mu.to(target.dtype)
        return F.mse_loss(pred, smooth)


def compute_rmse(pred: torch.Tensor, target: torch.Tensor) -> float:
    """Root mean squared error (the competition selection metric)."""
    pred = pred.detach().float()
    target = target.detach().float()
    return float(torch.sqrt(F.mse_loss(pred, target)).item())

def predict_scalar(
    model: nn.Module, image: torch.Tensor, tabular: torch.Tensor
) -> torch.Tensor:
    """Single scalar per sample in [0, 100]; clipped at inference time."""
    model.eval()
    with torch.no_grad():
        out = model(image, tabular)
    return torch.clamp(out, 0.0, 100.0)

class CandidateAdapter:
    """Import-safe, CPU-runnable adapter exercising the REAL model + loss.

    All methods treat the caller-supplied ``context`` as a partial mapping merged
    over internal defaults, preserving caller-provided values.
    """

    DEFAULTS = {
        "img_size": IMG_SIZE,
        "tab_dim": TAB_DIM,
        "criterion": None,  # resolved lazily to a real SmoothedMSELoss
        "mu": 50.0,
        "eps": SMOOTH_EPS,
        "freeze_backbone": True,
    }

    def __init__(self):
        self._model = None
        self._criterion = None

    # -- context helpers ---------------------------------------------------- #
    def _ctx(self, context) -> dict:
        merged = dict(self.DEFAULTS)
        if isinstance(context, dict):
            merged.update({k: v for k, v in context.items() if v is not None})
        return merged

    def _resolve_criterion(self, ctx: dict) -> nn.Module:
        crit = ctx.get("criterion")
        if crit is None:
            crit = SmoothedMSELoss(
                mu=float(ctx.get("mu", 50.0)), eps=float(ctx.get("eps", SMOOTH_EPS))
            )
            ctx["criterion"] = crit
        return crit

    # -- builders ----------------------------------------------------------- #
    def build_model(self, context=None) -> nn.Module:
        ctx = self._ctx(context)
        model = PawpularityModel(freeze_backbone=bool(ctx.get("freeze_backbone", True)))
        self._model = model
        return model

    def build_optimizer(self, model: nn.Module, context=None):
        # Optimiser choice belongs to the training stage; provide a safe default
        # (AdamW on trainable params) so the preflight can still run a step.
        ctx = self._ctx(context)
        lr = float(ctx.get("lr", 1e-4))
        wd = float(ctx.get("weight_decay", 1e-4))
        trainable = [p for p in model.parameters() if p.requires_grad]
        return torch.optim.AdamW(trainable, lr=lr, weight_decay=wd)

    def _synthetic_batch(self, scenario: dict, device: torch.device, with_target: bool):
        bs = int(scenario.get("batch_size", 2))
        img = torch.randn(bs, 3, IMG_SIZE, IMG_SIZE, device=device)
        tab = torch.randint(0, 2, (bs, TAB_DIM), device=device).float()
        out = {"image": img, "tabular": tab}
        if with_target:
            out["target"] = torch.rand(bs, device=device) * 100.0
        return out

    def build_train_batch(self, scenario: dict, device: torch.device) -> dict:
        return self._synthetic_batch(scenario, device, with_target=True)

    def build_validation_batch(self, scenario: dict, device: torch.device) -> dict:
        return self._synthetic_batch(scenario, device, with_target=True)

    # -- steps -------------------------------------------------------------- #
    def training_step(self, model: nn.Module, batch: dict, context=None):
        ctx = self._ctx(context)
        crit = self._resolve_criterion(ctx)
        model.train()
        pred = model(batch["image"], batch["tabular"])
        loss = crit(pred, batch["target"])
        return loss

    def validation_step(self, model: nn.Module, batch: dict, context=None):
        ctx = self._ctx(context)
        crit = self._resolve_criterion(ctx)
        model.eval()
        with torch.no_grad():
            pred = model(batch["image"], batch["tabular"])
            loss = crit(pred, batch["target"])
        return loss

# --------------------------------------------------------------------------- #
# Parameter groups: differential LR with selective backbone-layer4 unfreeze.     #

def _unfreeze_layer4(model: PawpularityModel) -> None:
    """Re-enable gradients on the backbone's final residual stage only."""
    for p in model.backbone.layer4.parameters():
        p.requires_grad = True


def build_param_groups(model: PawpularityModel) -> list:
    """Two param groups: (a) backbone layer4 @ LAYER4_LR, (b) everything else @ HEAD_LR."""
    layer4_ids = {id(p) for p in model.backbone.layer4.parameters()}
    layer4_params = [p for p in model.parameters() if id(p) in layer4_ids]
    other_params = [p for p in model.parameters() if id(p) not in layer4_ids]
    return [
        {"params": layer4_params, "lr": LAYER4_LR},
        {"params": other_params, "lr": HEAD_LR},
    ]


def make_optimizer(model: PawpularityModel):
    """AdamW with differential learning rates (weight decay applied to all groups)."""
    groups = build_param_groups(model)
    for g in groups:
        g["weight_decay"] = WEIGHT_DECAY
    return torch.optim.AdamW(groups)


class WarmupCosineScheduler:
    """Linear warmup for WARMUP_EPOCHS then cosine decay down to ETA_MIN.

    Step once per epoch (after the optimizer step). Keeps the base LRs intact so
    differential rates are preserved throughout training.
    """

    def __init__(
        self,
        optimizer,
        total_epochs: int,
        warmup_epochs: int = WARMUP_EPOCHS,
        eta_min: float = ETA_MIN,
    ):
        self.optimizer = optimizer
        self.total = max(1, int(total_epochs))
        self.warmup = max(0, min(int(warmup_epochs), self.total - 1))
        self.eta_min = float(eta_min)
        self.base_lrs = [g["lr"] for g in optimizer.param_groups]
        self.epoch = 0

    def step(self):
        self.epoch += 1
        e = self.epoch
        if e <= self.warmup:
            frac = (e + 1) / (self.warmup + 1)  # ramp 1/(w+1) -> 1
        else:
            prog = (e - self.warmup) / max(1, (self.total - self.warmup))
            cos = 0.5 * (1.0 + math.cos(math.pi * prog))
            frac = self.eta_min / self.base_lrs[0] + cos * (
                1.0 - self.eta_min / self.base_lrs[0]
            )
        for g, base in zip(self.optimizer.param_groups, self.base_lrs):
            g["lr"] = base * frac

@torch.no_grad()
def _predict(
    model: PawpularityModel, loader: DataLoader, device: torch.device
) -> np.ndarray:
    """Run the trained model over a loader; returns a 1-D float array (clipped [0,100]).

    Uses IDENTICAL processing for validation and test (same transform via the
    loader, same forward pass, same clipping) to avoid any val/test mismatch.
    """
    model.eval()
    outs = []
    for batch in loader:
        img = batch["image"].to(device, non_blocking=True)
        tab = batch["tabular"].to(device, non_blocking=True)
        pred = model(img, tab)
        pred = torch.clamp(pred, 0.0, 100.0)
        outs.append(pred.detach().cpu())
    return torch.cat(outs, dim=0).numpy() if outs else np.zeros(0, dtype=np.float32)


def compute_rmse(pred: np.ndarray, target: np.ndarray) -> float:
    """Root mean squared error -- the competition selection metric (lower is better)."""
    pred = np.asarray(pred, dtype=np.float64)
    target = np.asarray(target, dtype=np.float64)
    return float(np.sqrt(np.mean((pred - target) ** 2)))

def run_training():
    t_start = time.time()
    set_global_seed(SEED)

    # ---- data (consumed from Stage 1) ------------------------------------- #
    train_loader, val_loader, test_loader, train_df, test_df = (
        build_datasets_and_loaders()

    n_train = len(train_loader.dataset)
    n_val = len(val_loader.dataset)
    n_test = len(test_loader.dataset)

    model = PawpularityModel(freeze_backbone=True).to(DEVICE)
    _unfreeze_layer4(model)  # selective adaptation of vision tail
    mu = float(train_df["Pawpularity"].mean())
    criterion = SmoothedMSELoss(mu=mu, eps=SMOOTH_EPS).to(DEVICE)

    # ---- optimizer + scheduler (Stage 3) ---------------------------------- #
    optimizer = make_optimizer(model)
    scheduler = WarmupCosineScheduler(optimizer, total_epochs=EPOCHS)

    # ---- training state --------------------------------------------------- #
    best_rmse = float("inf")
    best_state = None
    patience_left = PATIENCE
    batch_size = INITIAL_BATCH_SIZE
    resolved_batch_size = batch_size

    def _rebuild_train_loader(bs: int) -> DataLoader:
        common = dict(num_workers=2, pin_memory=torch.cuda.is_available())
        return DataLoader(
            train_loader.dataset,
            batch_size=bs,
            shuffle=True,
            collate_fn=train_loader.collate_fn,
            drop_last=False,
            **common,
        )

    for epoch in range(1, EPOCHS + 1):
        # Elastic OOM fallback: rebuild the train loader if batch size changed.
        if batch_size != getattr(run_training, "_last_bs", INITIAL_BATCH_SIZE):
            train_loader = _rebuild_train_loader(batch_size)
            run_training._last_bs = batch_size

        model.train()
        running_loss = 0.0
        n_seen = 0
        try:
            for it, batch in enumerate(train_loader):
                img = batch["image"].to(DEVICE, non_blocking=True)
                tab = batch["tabular"].to(DEVICE, non_blocking=True)
                tgt = batch["target"].to(DEVICE, non_blocking=True)

                optimizer.zero_grad(set_to_none=True)
                pred = model(img, tab)
                loss = criterion(pred, tgt)
                (loss / ACCUM_STEPS).backward()
                if (it + 1) % ACCUM_STEPS == 0 or (it + 1) == len(train_loader):
                    torch.nn.utils.clip_grad_norm_(model.parameters(), max_norm=1.0)
                    optimizer.step()
                running_loss += loss.item() * img.size(0)
                n_seen += img.size(0)
        except RuntimeError as exc:
            if "out of memory" in str(exc).lower() and batch_size > MIN_BATCH_SIZE:
                batch_size = max(MIN_BATCH_SIZE, batch_size // 2)
                resolved_batch_size = batch_size
                if DEVICE.type == "cuda":
                    torch.cuda.empty_cache()
                print(
                    f"[epoch {epoch}] OOM -> halved batch to {batch_size}; restarting epoch"
                )
                continue  # restart this epoch with the smaller batch
            raise

        scheduler.step()

        # ---- validation (shared inference path) --------------------------- #
        val_pred = _predict(model, val_loader, DEVICE)
        val_target = val_loader.dataset.meta["Pawpularity"].to_numpy(dtype=np.float64)
        val_rmse = compute_rmse(val_pred, val_target)
        avg_loss = running_loss / max(1, n_seen)

        # One line per epoch (minimal logging).
        print(
            f"[epoch {epoch:02d}/{EPOCHS}] loss={avg_loss:.4f} val_rmse={val_rmse:.4f} "
            f"bs={resolved_batch_size} lr={optimizer.param_groups[-1]['lr']:.2e}"
        )

        # ---- best-model tracking + early stopping ------------------------- #
        if val_rmse < best_rmse:
            best_rmse = val_rmse
            best_state = {
                k: v.detach().cpu().clone() for k, v in model.state_dict().items()
            }
            patience_left = PATIENCE
        else:
            patience_left -= 1
            if patience_left <= 0:
                print(
                    f"[early-stop] no improvement for {PATIENCE} epochs; stopping at {epoch}"
                )
                break

    # ---- restore best weights --------------------------------------------- #
    if best_state is not None:
        model.load_state_dict(best_state)

    # ---- test inference (identical path to validation) -------------------- #
    test_pred = _predict(model, test_loader, DEVICE)
    test_pred = np.clip(test_pred, 0.0, 100.0)

    # ---- submission (original test-row order) ----------------------------- #
    sub_dir = os.path.join(os.getcwd(), "submission")
    os.makedirs(sub_dir, exist_ok=True)
    submission = pd.DataFrame(
        {
            "Id": test_df["Id"].values,
            "Pawpularity": test_pred.astype(np.float64),
        }
    )
    submission.to_csv(os.path.join(sub_dir, "submission.csv"), index=False)

    # ---- runtime metrics (Stage-3 contract) ------------------------------- #
    elapsed = time.time() - t_start
    peak_vram_mb = (
        torch.cuda.max_memory_allocated(DEVICE) / (1024 * 1024)
        if DEVICE.type == "cuda"
        else 0.0
    )
    print(
        f"[metrics] elapsed_seconds={elapsed:.1f} peak_vram_mb={peak_vram_mb:.1f} "
        f"resolved_batch_size={resolved_batch_size}"
    )
    print(f"[done] best_val_rmse={best_rmse:.4f} submission_rows={len(submission)}")

    score = best_rmse
    print(f"Final Validation Score: {score}")
    return score
