class TabularCrossAttention(nn.Module):
    def __init__(
        self, tab_dim, vis_dim, hidden=FUSION_HIDDEN, n_heads=4, dropout=DROPOUT_RATE
    ):
        super().__init__()
        self.n_heads = n_heads
        self.head_dim = hidden // n_heads
        assert hidden % n_heads == 0
        self.q_proj = nn.Linear(tab_dim, hidden)
        self.k_proj = nn.Linear(vis_dim, hidden)
        self.v_proj = nn.Linear(vis_dim, hidden)
        self.out_proj = nn.Linear(hidden, hidden)
        self.norm_q = nn.LayerNorm(hidden)
        self.norm_kv = nn.LayerNorm(hidden)
        self.dropout = nn.Dropout(dropout)

    def forward(self, tab, vis):
        # tab: [B, T], vis: [B, V]
        B = tab.size(0)
        q = (
            self.norm_q(self.q_proj(tab))
            .view(B, 1, self.n_heads, self.head_dim)
            .transpose(1, 2)
        )
        kv = self.norm_kv(self.k_proj(vis))
        k = kv.view(B, 1, self.n_heads, self.head_dim).transpose(1, 2)
        v = self.v_proj(vis).view(B, 1, self.n_heads, self.head_dim).transpose(1, 2)
        attn = torch.matmul(q, k.transpose(-2, -1)) / math.sqrt(self.head_dim)
        attn = F.softmax(attn, dim=-1)
        attn = self.dropout(attn)
        ctx = torch.matmul(attn, v)  # [B, H, 1, D]
        ctx = ctx.transpose(1, 2).reshape(B, self.n_heads * self.head_dim)
        return self.out_proj(ctx)

    """Dual-branch multi-scale CNN + tabular cross-attention fusion.

    Returns a dict with 'fused' (main head) and 'tab_only' (baseline head),
    each a scalar logit in [0, 100]. The main output interface is 'fused'.
    """

    def __init__(
        self,
        tab_dim=NUM_TABULAR,
        vis_feat_a=1280,
        vis_feat_b=512,
        hidden=FUSION_HIDDEN,
        dropout=DROPOUT_RATE,
    ):
        super().__init__()
        self.backbone_a = _load_efficientnet_b0()  # multi-scale view A
        self.backbone_b = _load_resnet34()  # multi-scale view B
        self.vis_a_proj = nn.Linear(vis_feat_a, hidden)
        self.vis_b_proj = nn.Linear(vis_feat_b, hidden)
        self.vis_fuse = nn.Sequential(
            nn.Linear(hidden * 2, hidden),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.LayerNorm(hidden),
        )
        self.cross_attn = TabularCrossAttention(
            tab_dim, hidden, hidden=hidden, dropout=dropout
        )
        self.fusion_mlp = nn.Sequential(
            nn.Linear(hidden * 3, hidden),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(hidden, hidden // 2),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(hidden // 2, 1),
        )
        # Tabular-only baseline head (reference for signal confirmation).
        self.tab_mlp = nn.Sequential(
            nn.Linear(tab_dim, 64),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(64, 32),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(32, 1),
        )

    def _vision(self, image):
        fa = self.backbone_a(image)
        fb = self.backbone_b(image)
        fa = self.vis_a_proj(fa)
        fb = self.vis_b_proj(fb)
        return self.vis_fuse(torch.cat([fa, fb], dim=1))

    def forward(self, image, tabular):
        vis = self._vision(image)  # [B, H]
        tab = tabular.float()  # [B, T]
        attn_out = self.cross_attn(tab, vis)  # [B, H]
        fused_in = torch.cat([vis, tab, attn_out], dim=1)
        fused = torch.sigmoid(self.fusion_mlp(fused_in)) * 100.0  # [B, 1]
        tab_only = torch.sigmoid(self.tab_mlp(tab)) * 100.0  # [B, 1]
        return {"fused": fused.squeeze(-1), "tab_only": tab_only.squeeze(-1)}

# Loss / criterion (MSE on the fused head; auxiliary small MSE on tab head)
# ----------------------------------------------------------------------------
class PawpularityLoss(nn.Module):
    def __init__(self, aux_weight=0.25):
        super().__init__()
        self.aux_weight = aux_weight

    def forward(self, outputs, targets):
        targets = targets.float().view(-1)
        fused = outputs["fused"].view(-1)
        tab_only = outputs["tab_only"].view(-1)
        main = F.mse_loss(fused, targets)
        aux = F.mse_loss(tab_only, targets)
        return main + self.aux_weight * aux

class TabularCrossAttention(nn.Module):
    def __init__(
        self, tab_dim, vis_dim, hidden=FUSION_HIDDEN, n_heads=4, dropout=DROPOUT_RATE
    ):
        super().__init__()
        self.n_heads = n_heads
        self.head_dim = hidden // n_heads
        assert hidden % n_heads == 0
        self.q_proj = nn.Linear(tab_dim, hidden)
        self.k_proj = nn.Linear(vis_dim, hidden)
        self.v_proj = nn.Linear(vis_dim, hidden)
        self.out_proj = nn.Linear(hidden, hidden)
        self.norm_q = nn.LayerNorm(hidden)
        self.norm_kv = nn.LayerNorm(hidden)
        self.dropout = nn.Dropout(dropout)

    def forward(self, tab, vis):
        # tab: [B, T], vis: [B, V]
        B = tab.size(0)
        q = (
            self.norm_q(self.q_proj(tab))
            .view(B, 1, self.n_heads, self.head_dim)
            .transpose(1, 2)
        )
        kv = self.norm_kv(self.k_proj(vis))
        k = kv.view(B, 1, self.n_heads, self.head_dim).transpose(1, 2)
        v = self.v_proj(vis).view(B, 1, self.n_heads, self.head_dim).transpose(1, 2)
        attn = torch.matmul(q, k.transpose(-2, -1)) / math.sqrt(self.head_dim)
        attn = F.softmax(attn, dim=-1)
        attn = self.dropout(attn)
        ctx = torch.matmul(attn, v)  # [B, H, 1, D]
        ctx = ctx.transpose(1, 2).reshape(B, self.n_heads * self.head_dim)
        return self.out_proj(ctx)

        self.backbone_b = _load_resnet34()  # multi-scale view B
        self.vis_a_proj = nn.Linear(vis_feat_a, hidden)
        self.vis_b_proj = nn.Linear(vis_feat_b, hidden)
        self.vis_fuse = nn.Sequential(
            nn.Linear(hidden * 2, hidden),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.LayerNorm(hidden),
        )
        self.cross_attn = TabularCrossAttention(
            tab_dim, hidden, hidden=hidden, dropout=dropout
        )
        self.fusion_mlp = nn.Sequential(
            nn.Linear(hidden * 3, hidden),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(hidden, hidden // 2),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(hidden // 2, 1),
        )
        # Tabular-only baseline head (reference for signal confirmation).
        self.tab_mlp = nn.Sequential(
            nn.Linear(tab_dim, 64),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(64, 32),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(32, 1),
        )

    def _vision(self, image):
        fa = self.backbone_a(image)
        fb = self.backbone_b(image)
        fa = self.vis_a_proj(fa)
        fb = self.vis_b_proj(fb)
        return self.vis_fuse(torch.cat([fa, fb], dim=1))

    def forward(self, image, tabular):
        vis = self._vision(image)  # [B, H]
        tab = tabular.float()  # [B, T]
        attn_out = self.cross_attn(tab, vis)  # [B, H]
        fused_in = torch.cat([vis, tab, attn_out], dim=1)
        fused = torch.sigmoid(self.fusion_mlp(fused_in)) * 100.0  # [B, 1]
        tab_only = torch.sigmoid(self.tab_mlp(tab)) * 100.0  # [B, 1]
        return {"fused": fused.squeeze(-1), "tab_only": tab_only.squeeze(-1)}


# ----------------------------------------------------------------------------
# Loss / criterion (MSE on the fused head; auxiliary small MSE on tab head)
# ----------------------------------------------------------------------------
class PawpularityLoss(nn.Module):
    def __init__(self, aux_weight=0.25):
        super().__init__()
        self.aux_weight = aux_weight

    def forward(self, outputs, targets):
        targets = targets.float().view(-1)
        fused = outputs["fused"].view(-1)
        tab_only = outputs["tab_only"].view(-1)
        main = F.mse_loss(fused, targets)
        aux = F.mse_loss(tab_only, targets)
        return main + self.aux_weight * aux

    return pred.cpu()

class CandidateAdapter:
    def __init__(self):
        self._defaults = {
            "criterion": None,
            "optimizer": None,
            "device": "cpu",
            "batch_size": 4,
        }

    def _ctx(self, context):
        merged = dict(self._defaults)
        if isinstance(context, dict):
            merged.update(context)
        return merged

    def build_model(self, context):
        ctx = self._ctx(context)
        model = PetModel(tab_dim=NUM_TABULAR, vis_feat_a=1280, vis_feat_b=512)
        model = model.to(ctx.get("device", "cpu"))
        return model

    def build_optimizer(self, model, context):
        ctx = self._ctx(context)
        lr = float(ctx.get("lr", 1e-4))
        wd = float(ctx.get("weight_decay", 1e-4))
        return torch.optim.AdamW(model.parameters(), lr=lr, weight_decay=wd)

    def _make_batch(self, scenario, device, with_target=True):
        bs = int(scenario.get("batch_size", 4))
        image = torch.randn(bs, 3, IMAGE_SIZE, IMAGE_SIZE, dtype=torch.float32)
        tabular = torch.randint(0, 2, (bs, NUM_TABULAR)).to(torch.float32)
        batch = {"image": image.to(device), "tabular": tabular.to(device)}
        if with_target:
            batch["target"] = torch.rand(bs, dtype=torch.float32) * 100.0
            batch["target"] = batch["target"].to(device)
        return batch

    def build_train_batch(self, scenario, device):
        return self._make_batch(scenario, device, with_target=True)

    def build_validation_batch(self, scenario, device):
        return self._make_batch(scenario, device, with_target=True)

    def _resolve_criterion(self, context):
        ctx = self._ctx(context)
        crit = ctx.get("criterion")
        if crit is None:
            crit = PawpularityLoss()
        return crit

    def training_step(self, model, batch, context):
        ctx = self._ctx(context)
        model.train()
        crit = self._resolve_criterion(context)
        out = model(batch["image"], batch["tabular"])
        loss = crit(out, batch["target"])
        return loss

    def validation_step(self, model, batch, context):
        ctx = self._ctx(context)
        model.eval()
        crit = self._resolve_criterion(context)
        with torch.no_grad():
            out = model(batch["image"], batch["tabular"])
            loss = crit(out, batch["target"])
        return loss

    y = np.asarray(y, dtype=np.float64)
    try:
        edges = np.unique(np.quantile(y, np.linspace(0, 1, STRATIFY_BINS + 1)))
        if len(edges) < 2:
            return np.zeros(len(y), dtype=int)
        bins = np.clip(
            np.searchsorted(edges[1:-1], y, side="right"), 0, STRATIFY_BINS - 1
        )
        return bins.astype(int)
    except Exception:
        return np.zeros(len(y), dtype=int)

class TabularCrossAttention(nn.Module):
    def __init__(
        self, tab_dim, vis_dim, hidden=FUSION_HIDDEN, n_heads=4, dropout=DROPOUT_RATE
    ):
        super().__init__()
        self.n_heads = n_heads
        self.head_dim = hidden // n_heads
        assert hidden % n_heads == 0
        self.q_proj = nn.Linear(tab_dim, hidden)
        self.k_proj = nn.Linear(vis_dim, hidden)
        self.v_proj = nn.Linear(vis_dim, hidden)
        self.out_proj = nn.Linear(hidden, hidden)
        self.norm_q = nn.LayerNorm(hidden)
        self.norm_kv = nn.LayerNorm(hidden)
        self.dropout = nn.Dropout(dropout)

    def forward(self, tab, vis):
        B = tab.size(0)
        q = (
            self.norm_q(self.q_proj(tab))
            .view(B, 1, self.n_heads, self.head_dim)
            .transpose(1, 2)
        )
        kv = self.norm_kv(self.k_proj(vis))
        k = kv.view(B, 1, self.n_heads, self.head_dim).transpose(1, 2)
        v = self.v_proj(vis).view(B, 1, self.n_heads, self.head_dim).transpose(1, 2)
        attn = torch.matmul(q, k.transpose(-2, -1)) / math.sqrt(self.head_dim)
        attn = F.softmax(attn, dim=-1)
        attn = self.dropout(attn)
        ctx = torch.matmul(attn, v)
        ctx = ctx.transpose(1, 2).reshape(B, self.n_heads * self.head_dim)
        return self.out_proj(ctx)


class PetModel(nn.Module):
    """Dual-branch multi-scale CNN + tabular cross-attention fusion."""

    def __init__(
        self,
        tab_dim=NUM_TABULAR,
        vis_feat_a=1280,
        vis_feat_b=512,
        hidden=FUSION_HIDDEN,
        dropout=DROPOUT_RATE,
    ):
        super().__init__()
        self.backbone_a = _load_efficientnet_b0()
        self.backbone_b = _load_resnet34()
        self.vis_a_proj = nn.Linear(vis_feat_a, hidden)
        self.vis_b_proj = nn.Linear(vis_feat_b, hidden)
        self.vis_fuse = nn.Sequential(
            nn.Linear(hidden * 2, hidden),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.LayerNorm(hidden),
        )
        self.cross_attn = TabularCrossAttention(
            tab_dim, hidden, hidden=hidden, dropout=dropout
        )
        self.fusion_mlp = nn.Sequential(
            nn.Linear(hidden * 3, hidden),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(hidden, hidden // 2),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(hidden // 2, 1),
        )
        self.tab_mlp = nn.Sequential(
            nn.Linear(tab_dim, 64),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(64, 32),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(32, 1),
        )

    def _vision(self, image):
        fa = self.vis_a_proj(self.backbone_a(image))
        fb = self.vis_b_proj(self.backbone_b(image))
        return self.vis_fuse(torch.cat([fa, fb], dim=1))

    def forward(self, image, tabular):
        vis = self._vision(image)
        tab = tabular.float()
        attn_out = self.cross_attn(tab, vis)
        fused_in = torch.cat([vis, tab, attn_out], dim=1)
        fused = torch.sigmoid(self.fusion_mlp(fused_in)) * 100.0
        tab_only = torch.sigmoid(self.tab_mlp(tab)) * 100.0
        return {"fused": fused.squeeze(-1), "tab_only": tab_only.squeeze(-1)}


# ----------------------------------------------------------------------------
# Loss (MSE on fused head + small auxiliary MSE on tab head)
# ----------------------------------------------------------------------------
class PawpularityLoss(nn.Module):
    def __init__(self, aux_weight=0.25):
        super().__init__()
        self.aux_weight = aux_weight

    def forward(self, outputs, targets):
        targets = targets.float().view(-1)
        fused = outputs["fused"].view(-1)
        tab_only = outputs["tab_only"].view(-1)
        return F.mse_loss(fused, targets) + self.aux_weight * F.mse_loss(
            tab_only, targets
        )


# ----------------------------------------------------------------------------
# Differential-LR parameter groups (backbone vs fusion/head)
# ----------------------------------------------------------------------------
def get_param_groups(
    model, backbone_lr=BACKBONE_LR, fusion_lr=FUSION_LR, wd=WEIGHT_DECAY
):
    backbone_ids = set(id(p) for p in model.backbone_a.parameters())
    backbone_ids.update(id(p) for p in model.backbone_b.parameters())
    backbone_params, other_params = [], []
    for name, p in model.named_parameters():
        if not p.requires_grad:
            continue
        if id(p) in backbone_ids:
            backbone_params.append(p)
        else:
            other_params.append(p)
    return [
        {"params": backbone_params, "lr": backbone_lr, "weight_decay": wd},
        {"params": other_params, "lr": fusion_lr, "weight_decay": wd},
    ]


# ----------------------------------------------------------------------------
# Mixup (applied to image + tabular + target)
# ----------------------------------------------------------------------------
def mixup_batch(images, tabular, targets, alpha=MIXUP_ALPHA):
    lam = float(np.random.beta(alpha, alpha))
    bs = images.size(0)
    idx = torch.randperm(bs, device=images.device)
    mixed_img = lam * images + (1 - lam) * images[idx]
    mixed_tab = lam * tabular + (1 - lam) * tabular[idx]
    mixed_tgt = lam * targets + (1 - lam) * targets[idx]
    return mixed_img, mixed_tab, mixed_tgt

def train_one_epoch(model, loader, criterion, optimizer, device, use_mixup=True):
    model.train()
    total_loss, n_batches = 0.0, 0
    for batch in loader:
        image = batch["image"].to(device, non_blocking=True)
        tabular = batch["tabular"].to(device, non_blocking=True)
        targets = batch["target"].to(device, non_blocking=True)
        if use_mixup and image.size(0) > 1:
            image, tabular, targets = mixup_batch(image, tabular, targets, MIXUP_ALPHA)
        optimizer.zero_grad(set_to_none=True)
        out = model(image, tabular)
        loss = criterion(out, targets)
        loss.backward()
        optimizer.step()
        total_loss += float(loss.item())
        n_batches += 1
    return total_loss / max(n_batches, 1)
