    """Stratification buckets so the validation split mirrors the target spread."""

def make_split(train_df, test_size=0.1, seed=42):

def build_tabular_features(df):
    """Raw flags + one interaction -> float array [N, N_TABULAR]."""
    base = df[TABULAR_COLS].astype(float).values.copy()
    eyes_face = (base[:, 0] * base[:, 1]).reshape(-1, 1)  # Eyes AND Face interaction
    feats = np.concatenate([base, eyes_face], axis=1)
    return feats.astype(np.float32)


class TabularNormalizer:
    """Standardizes tabular features using statistics fitted ONLY on training data."""

    def __init__(self):
        self.mean = None
        self.std = None

    def fit(self, X):
        X = np.asarray(X, dtype=np.float64)
        self.mean = X.mean(axis=0)
        self.std = X.std(axis=0)
        self.std[self.std < 1e-6] = 1.0
        return self

    def transform(self, X):
        X = np.asarray(X, dtype=np.float64)
        return ((X - self.mean) / self.std).astype(np.float32)

        return len(self.ids)

class PetFinderModel(nn.Module):
    """ViT image encoder + tabular MLP fused through a learned attention gate."""

    def __init__(
        self,
        n_tabular=N_TABULAR,
        img_dim=768,
        tab_hidden=128,
        dropout=0.3,
        pretrained=True,
    ):
        super().__init__()
        self.backbone = _load_backbone(pretrained=pretrained)
        self.img_dim = getattr(self.backbone, "num_features", img_dim)

        self.tab_mlp = nn.Sequential(
            nn.Linear(n_tabular, tab_hidden),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
            nn.Linear(tab_hidden, tab_hidden),
            nn.ReLU(inplace=True),
        )

        # Learned gate: how much to trust the tabular branch vs the image branch.
        self.gate = nn.Sequential(
            nn.Linear(self.img_dim + tab_hidden, 128),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
            nn.Linear(128, 1),
            nn.Sigmoid(),
        )

        self.head = nn.Sequential(
            nn.Linear(self.img_dim + tab_hidden, 256),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
            nn.Linear(256, 1),
        )

    def forward(self, image, tabular):
        img_feat = self.backbone(image)  # [B, img_dim]
        tab_feat = self.tab_mlp(tabular)  # [B, tab_hidden]
        g = self.gate(torch.cat([img_feat, tab_feat], dim=1))  # [B, 1]
        fused = g * tab_feat + (1.0 - g) * img_feat  # element-wise blend
        out = self.head(fused)  # [B, 1]
        # Sigmoid keeps predictions inside the valid 0..100 Pawpularity range.
        return 100.0 * torch.sigmoid(out)

    """MSE on the 0..100 scale (matches the RMSE evaluation metric)."""

    def __init__(self):
        super().__init__()
        self.mse = nn.MSELoss()

    def forward(self, pred, target):
        return self.mse(pred, target)


def rmse_metric(pred, target):
    pred = np.asarray(pred, dtype=np.float64)
    target = np.asarray(target, dtype=np.float64)
    return float(np.sqrt(np.mean((pred - target) ** 2)))

class CandidateAdapter:
    def __init__(self):
        self._ctx = _default_context()

    def _merge(self, context):
        ctx = dict(self._ctx)
        if context:
            for k, v in context.items():
                if v is not None:
                    ctx[k] = v
        return ctx

    def build_model(self, context=None):
        ctx = self._merge(context)
        if ctx.get("model") is None:
            ctx["model"] = PetFinderModel(pretrained=bool(ctx.get("pretrained", False)))
        self._ctx = ctx
        return ctx["model"]

    def build_optimizer(self, model, context=None):
        ctx = self._merge(context)
        lr = float(ctx.get("lr", 1e-4))
        opt = torch.optim.AdamW(model.parameters(), lr=lr, weight_decay=1e-5)
        ctx["optimizer"] = opt
        self._ctx = ctx
        return opt

    def _get_dataset(self, split, ctx):
        input_dir = get_input_dir()
        train_df, test_df = load_tables(input_dir)
        if split == "train":
            tr, _ = make_split(train_df)
            ids = tr["Id"].values
            label = tr["Pawpularity"].values
            root = os.path.join(input_dir, "train")
            tfm = ctx.get("transform") or build_train_transform(
                ctx.get("img_size", IMG_SIZE)
            )
            norm = ctx.get("normalizer")
            if norm is None:
                norm = TabularNormalizer().fit(build_tabular_features(tr))
                ctx["normalizer"] = norm
            tab = norm.transform(build_tabular_features(tr))
            ds = PetFinderDataset(ids, tab, label, root, tfm, has_label=True)
        else:
            _, va = make_split(train_df)
            ids = va["Id"].values
            label = va["Pawpularity"].values
            root = os.path.join(input_dir, "train")
            tfm = ctx.get("eval_transform") or build_eval_transform(
                ctx.get("img_size", IMG_SIZE)
            )
            norm = ctx.get("normalizer")
            if norm is None:
                tr, _ = make_split(train_df)
                norm = TabularNormalizer().fit(build_tabular_features(tr))
                ctx["normalizer"] = norm
            tab = norm.transform(build_tabular_features(va))
            ds = PetFinderDataset(ids, tab, label, root, tfm, has_label=True)
        return ds

    def _build_batch(self, split, scenario, device, ctx):
        bs = int(scenario.get("batch_size", ctx.get("batch_size", 8)))
        ds = self._get_dataset(split, ctx)
        n = min(bs, len(ds))
        loader = torch.utils.data.DataLoader(
            ds,
            batch_size=n,
            shuffle=(split == "train"),
            num_workers=0,
            collate_fn=default_collate,
            drop_last=False,
        )
        batch = next(iter(loader))
        batch = {
            k: (v.to(device) if torch.is_tensor(v) else v) for k, v in batch.items()
        }
        return batch

    def build_train_batch(self, scenario, device):
        ctx = self._merge({})
        return self._build_batch("train", scenario, device, ctx)

    def build_validation_batch(self, scenario, device):
        ctx = self._merge({})
        return self._build_batch("val", scenario, device, ctx)

    def _resolve_criterion(self, ctx):
        crit = ctx.get("criterion")
        if crit is None:
            crit = PawpularityLoss()
            ctx["criterion"] = crit
        return crit

    def training_step(self, model, batch, context=None):
        ctx = self._merge(context)
        model.train()
        image = batch["image"].to(ctx.get("device", "cpu"))
        tabular = batch["tabular"].to(ctx.get("device", "cpu"))
        target = batch["target"].to(ctx.get("device", "cpu"))
        pred = model(image, tabular)
        loss = self._resolve_criterion(ctx)(pred, target)
        self._ctx = ctx
        return loss

    def validation_step(self, model, batch, context=None):
        ctx = self._merge(context)
        model.eval()
        image = batch["image"].to(ctx.get("device", "cpu"))
        tabular = batch["tabular"].to(ctx.get("device", "cpu"))
        target = batch["target"].to(ctx.get("device", "cpu"))
        with torch.no_grad():
            pred = model(image, tabular)
        loss = self._resolve_criterion(ctx)(pred, target)
        self._ctx = ctx
        return loss

        num_workers=2,

            if "target" in batch:

class TabularNormalizer:
    """Standardizes tabular features using statistics fitted ONLY on training data."""

    def __init__(self):
        self.mean = None
        self.std = None

    def fit(self, X):
        X = np.asarray(X, dtype=np.float64)
        self.mean = X.mean(axis=0)
        self.std = X.std(axis=0)
        self.std[self.std < 1e-6] = 1.0
        return self

    def transform(self, X):
        X = np.asarray(X, dtype=np.float64)
        return ((X - self.mean) / self.std).astype(np.float32)

class PetFinderModel(nn.Module):
    """ViT image encoder + tabular MLP fused through a learned attention gate."""

    def __init__(
        self,
        n_tabular=N_TABULAR,
        img_dim=768,
        tab_hidden=128,
        dropout=0.3,
        pretrained=True,
    ):
        super().__init__()
        self.backbone = _load_backbone(pretrained=pretrained)
        self.img_dim = getattr(self.backbone, "num_features", img_dim)

        self.tab_mlp = nn.Sequential(
            nn.Linear(n_tabular, tab_hidden),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
            nn.Linear(tab_hidden, tab_hidden),
            nn.ReLU(inplace=True),
        )

        # Learned gate: how much to trust the tabular branch vs the image branch.
        self.gate = nn.Sequential(
            nn.Linear(self.img_dim + tab_hidden, 128),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
            nn.Linear(128, 1),
            nn.Sigmoid(),
        )

        self.head = nn.Sequential(
            nn.Linear(self.img_dim + tab_hidden, 256),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
            nn.Linear(256, 1),
        )

    def forward(self, image, tabular):
        img_feat = self.backbone(image)  # [B, img_dim]
        tab_feat = self.tab_mlp(tabular)  # [B, tab_hidden]
        g = self.gate(torch.cat([img_feat, tab_feat], dim=1))  # [B, 1]
        fused = g * tab_feat + (1.0 - g) * img_feat  # element-wise blend
        out = self.head(fused)  # [B, 1]
        # Sigmoid keeps predictions inside the valid 0..100 Pawpularity range.
        return 100.0 * torch.sigmoid(out)

def rmse_metric(pred, target):
    pred = np.asarray(pred, dtype=np.float64)
    target = np.asarray(target, dtype=np.float64)
    return float(np.sqrt(np.mean((pred - target) ** 2)))

class CandidateAdapter:
    def __init__(self):
        self._ctx = _default_context()

    def _merge(self, context):
        ctx = dict(self._ctx)
        if context:
            for k, v in context.items():
                if v is not None:
                    ctx[k] = v
        return ctx

    def build_model(self, context=None):
        ctx = self._merge(context)
        if ctx.get("model") is None:
            ctx["model"] = PetFinderModel(pretrained=bool(ctx.get("pretrained", False)))
        self._ctx = ctx
        return ctx["model"]

    def build_optimizer(self, model, context=None):
        ctx = self._merge(context)
        lr = float(ctx.get("lr", 1e-4))
        opt = torch.optim.AdamW(model.parameters(), lr=lr, weight_decay=1e-5)
        ctx["optimizer"] = opt
        self._ctx = ctx
        return opt

    def _get_dataset(self, split, ctx):
        input_dir = get_input_dir()
        train_df, test_df = load_tables(input_dir)
        if split == "train":
            tr, _ = make_split(train_df)
            ids = tr["Id"].values
            label = tr["Pawpularity"].values
            root = os.path.join(input_dir, "train")
            tfm = ctx.get("transform") or build_train_transform(
                ctx.get("img_size", IMG_SIZE)
            )
            norm = ctx.get("normalizer")
            if norm is None:
                norm = TabularNormalizer().fit(build_tabular_features(tr))
                ctx["normalizer"] = norm
            tab = norm.transform(build_tabular_features(tr))
            ds = PetFinderDataset(ids, tab, label, root, tfm, has_label=True)
        else:
            _, va = make_split(train_df)
            ids = va["Id"].values
            label = va["Pawpularity"].values
            root = os.path.join(input_dir, "train")
            tfm = ctx.get("eval_transform") or build_eval_transform(
                ctx.get("img_size", IMG_SIZE)
            )
            norm = ctx.get("normalizer")
            if norm is None:
                tr, _ = make_split(train_df)
                norm = TabularNormalizer().fit(build_tabular_features(tr))
                ctx["normalizer"] = norm
            tab = norm.transform(build_tabular_features(va))
            ds = PetFinderDataset(ids, tab, label, root, tfm, has_label=True)
        return ds

    def _build_batch(self, split, scenario, device, ctx):
        bs = int(scenario.get("batch_size", ctx.get("batch_size", 8)))
        ds = self._get_dataset(split, ctx)
        n = min(bs, len(ds))
        loader = torch.utils.data.DataLoader(
            ds,
            batch_size=n,
            shuffle=(split == "train"),
            num_workers=0,
            collate_fn=default_collate,
            drop_last=False,
        )
        batch = next(iter(loader))
        batch = {
            k: (v.to(device) if torch.is_tensor(v) else v) for k, v in batch.items()
        }
        return batch

    def build_train_batch(self, scenario, device):
        ctx = self._merge({})
        return self._build_batch("train", scenario, device, ctx)

    def build_validation_batch(self, scenario, device):
        ctx = self._merge({})
        return self._build_batch("val", scenario, device, ctx)

    def _resolve_criterion(self, ctx):
        crit = ctx.get("criterion")
        if crit is None:
            crit = PawpularityLoss()
            ctx["criterion"] = crit
        return crit

    def training_step(self, model, batch, context=None):
        ctx = self._merge(context)
        model.train()
        image = batch["image"].to(ctx.get("device", "cpu"))
        tabular = batch["tabular"].to(ctx.get("device", "cpu"))
        target = batch["target"].to(ctx.get("device", "cpu"))
        pred = model(image, tabular)
        loss = self._resolve_criterion(ctx)(pred, target)
        self._ctx = ctx
        return loss

    def validation_step(self, model, batch, context=None):
        ctx = self._merge(context)
        model.eval()
        image = batch["image"].to(ctx.get("device", "cpu"))
        tabular = batch["tabular"].to(ctx.get("device", "cpu"))
        target = batch["target"].to(ctx.get("device", "cpu"))
        with torch.no_grad():
            pred = model(image, tabular)
        loss = self._resolve_criterion(ctx)(pred, target)
        self._ctx = ctx
        return loss

def predict_dataset(model, dataset, device, batch_size=16):
    """Run real model inference over a dataset; returns (ids, preds)."""
    model.eval()
    loader = torch.utils.data.DataLoader(
        dataset,
        batch_size=batch_size,
        shuffle=False,
        num_workers=2,
        collate_fn=default_collate,
    )
    ids, preds = [], []
    with torch.no_grad():
        for batch in loader:
            image = batch["image"].to(device)
            tabular = batch["tabular"].to(device)
            out = model(image, tabular)
            preds.extend(out.squeeze(-1).cpu().numpy())
            if "target" in batch:
                pass
    # ids come from dataset order
    ids = list(dataset.ids)
    return ids, np.clip(np.asarray(preds, dtype=np.float64), 0.0, 100.0)

        n_splits=1, test_size=test_size, random_state=seed
    )
    tr_idx, va_idx = next(splitter.split(np.zeros(len(y)), labels))
    return train_df.iloc[tr_idx].reset_index(drop=True), train_df.iloc[
        va_idx
    ].reset_index(drop=True)


def build_tabular_features(df):
    """Raw flags + one interaction -> float array [N, N_TABULAR+1]."""
    base = df[TABULAR_COLS].astype(float).values.copy()
    eyes_face = (base[:, 0] * base[:, 1]).reshape(-1, 1)  # Eyes AND Face interaction
    feats = np.concatenate([base, eyes_face], axis=1)
    return feats.astype(np.float32)


class TabularNormalizer:
    """Standardizes tabular features using statistics fitted ONLY on training data."""

    def __init__(self):
        self.mean = None
        self.std = None

    def fit(self, X):
        X = np.asarray(X, dtype=np.float64)
        self.mean = X.mean(axis=0)
        self.std = X.std(axis=0)
        self.std[self.std < 1e-6] = 1.0
        return self

    def transform(self, X):
        X = np.asarray(X, dtype=np.float64)
        return ((X - self.mean) / self.std).astype(np.float32)

def default_collate(batch):
    images = torch.stack([b["image"] for b in batch], dim=0)
    tabular = torch.stack([b["tabular"] for b in batch], dim=0)
    out = {"image": images, "tabular": tabular}
    if "target" in batch[0]:
        out["target"] = torch.stack([b["target"] for b in batch], dim=0)
    return out

def _load_backbone(pretrained=True):

class PetFinderModel(nn.Module):
    """ViT image encoder + tabular MLP fused through a learned attention gate."""

    def __init__(
        self,
        n_tabular=N_TABULAR + 1,
        img_dim=768,
        tab_hidden=128,
        dropout=0.3,
        pretrained=True,
    ):
        super().__init__()
        self.backbone = _load_backbone(pretrained=pretrained)
        self.img_dim = getattr(self.backbone, "num_features", img_dim)

        self.tab_mlp = nn.Sequential(
            nn.Linear(n_tabular, tab_hidden),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
            nn.Linear(tab_hidden, tab_hidden),
            nn.ReLU(inplace=True),
        )

        # Learned gate: how much to trust the tabular branch vs the image branch.
        self.gate = nn.Sequential(
            nn.Linear(self.img_dim + tab_hidden, 128),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
            nn.Linear(128, 1),
            nn.Sigmoid(),
        )

        self.head = nn.Sequential(
            nn.Linear(self.img_dim + tab_hidden, 256),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout),
            nn.Linear(256, 1),
        )

    def forward(self, image, tabular):
        img_feat = self.backbone(image)  # [B, img_dim]
        tab_feat = self.tab_mlp(tabular)  # [B, tab_hidden]
        g = self.gate(torch.cat([img_feat, tab_feat], dim=1))  # [B, 1]
        fused = g * tab_feat + (1.0 - g) * img_feat  # element-wise blend
        out = self.head(fused)  # [B, 1]
        # Sigmoid keeps predictions inside the valid 0..100 Pawpularity range.
        return 100.0 * torch.sigmoid(out)

class PawpularityLoss(nn.Module):
    """MSE on the 0..100 scale (matches the RMSE evaluation metric)."""

    def __init__(self):
        super().__init__()
        self.mse = nn.MSELoss()

    def forward(self, pred, target):
        return self.mse(pred, target)


def rmse_metric(pred, target):
    pred = np.asarray(pred, dtype=np.float64)
    target = np.asarray(target, dtype=np.float64)
    return float(np.sqrt(np.mean((pred - target) ** 2)))

class CandidateAdapter:

    def _get_dataset(self, split, ctx):

            tab = norm.transform(build_tabular_features(va))

    def _build_batch(self, split, scenario, device, ctx):
        bs = int(scenario.get("batch_size", ctx.get("batch_size", 8)))
        fixture = scenario.get("fixture")
        if fixture:
            # Authoritative named-input contract: honor declared non-batch shapes.
            image_shape = [int(x) for x in scenario.get("image_shape", [3, 256, 256])]
            tab_shape = [int(x) for x in scenario.get("tabular_shape", [13])]
            image = torch.randn(bs, *image_shape, dtype=torch.float32)
            tabular = torch.randn(bs, *tab_shape, dtype=torch.float32)
            target = torch.rand(bs, dtype=torch.float32) * 100.0
            batch = {"image": image.to(device), "tabular": tabular.to(device)}
            if "target" in fixture:
                batch["target"] = target.to(device)
            return batch
        ds = self._get_dataset(split, ctx)
        n = min(bs, len(ds))
        loader = torch.utils.data.DataLoader(
            ds,
            batch_size=n,
            shuffle=(split == "train"),
            num_workers=0,
            collate_fn=default_collate,
            drop_last=False,
        )
        batch = next(iter(loader))
        batch = {
            k: (v.to(device) if torch.is_tensor(v) else v) for k, v in batch.items()
        }
        return batch

    def build_train_batch(self, scenario, device):
        ctx = self._merge({})
        return self._build_batch("train", scenario, device, ctx)

    def build_validation_batch(self, scenario, device):
        ctx = self._merge({})
        return self._build_batch("val", scenario, device, ctx)

    def _resolve_criterion(self, ctx):
        crit = ctx.get("criterion")
        if crit is None:

            ctx["criterion"] = crit
        return crit

    def training_step(self, model, batch, context=None):

        image = batch["image"].to(dev)
        tabular = batch["tabular"].to(dev)
        target = batch["target"].to(dev)
        pred = model(image, tabular)
        loss = self._resolve_criterion(ctx)(pred, target)
        self._ctx = ctx
        return loss

    def validation_step(self, model, batch, context=None):
        ctx = self._merge(context)
        model.eval()
        dev = ctx.get("device", "cpu")
        image = batch["image"].to(dev)
        tabular = batch["tabular"].to(dev)
        target = batch["target"].to(dev)
        with torch.no_grad():
            pred = model(image, tabular)
        loss = self._resolve_criterion(ctx)(pred, target)
        self._ctx = ctx
        return loss

def resolve_device():
    if torch.cuda.is_available():
        return torch.device("cuda")
    return torch.device("cpu")


def pick_batch_size(device):
    """Physical batch policy: fixed 32 on GPU, 8 on CPU (safe defaults)."""
    if device.type == "cuda":
        return 32
    return 8


def build_param_groups(model, backbone_lr=1e-5, head_lr=1e-4, weight_decay=1e-5):
    """Differential LR: gentle fine-tuning of the pretrained ViT, faster head."""
    backbone_params = list(model.backbone.parameters())
    backbone_ids = {id(p) for p in backbone_params}
    head_params = []
    for name, p in model.named_parameters():
        if not p.requires_grad:
            continue
        if id(p) in backbone_ids:
            continue
        head_params.append(p)
    groups = [
        {"params": backbone_params, "lr": backbone_lr},
        {"params": head_params, "lr": head_lr},
    ]
    return groups


def make_scheduler(optimizer, warmup_epochs, total_epochs):
    """Linear warmup then cosine decay to eta_min."""

    def lr_lambda(epoch):
        if epoch < warmup_epochs:
            return (epoch + 1) / max(1, warmup_epochs)
        progress = (epoch - warmup_epochs) / max(1, (total_epochs - warmup_epochs))
        return 1e-6 + 0.5 * (1 + math.cos(math.pi * progress))

    return torch.optim.lr_scheduler.LambdaLR(optimizer, lr_lambda)


def peak_vram_mb(device):
    if device.type == "cuda":
        return torch.cuda.max_memory_allocated(device) / (1024**2)
    return 0.0


def evaluate_rmse(model, dataset, device, batch_size=16):
    """Run real model inference over a dataset; returns RMSE (or preds if no labels)."""
    model.eval()
    loader = torch.utils.data.DataLoader(
        dataset,
        batch_size=batch_size,
        shuffle=False,
        num_workers=2,
        pin_memory=(device.type == "cuda"),
        collate_fn=default_collate,
    )
    preds, targets = [], []
    with torch.no_grad():
        for batch in loader:
            image = batch["image"].to(device)
            tabular = batch["tabular"].to(device)
            out = model(image, tabular)
            preds.extend(out.squeeze(-1).cpu().numpy())
            if "target" in batch:
                targets.extend(batch["target"].cpu().numpy())
    preds = np.clip(np.asarray(preds, dtype=np.float64), 0.0, 100.0)
    if len(targets) > 0:
        targets = np.asarray(targets, dtype=np.float64)
        return rmse_metric(preds, targets), preds
    return None, preds


def predict_dataset(model, dataset, device, batch_size=16):
    """Run real model inference over a dataset; returns (ids, preds)."""
    model.eval()
    loader = torch.utils.data.DataLoader(
        dataset,
        batch_size=batch_size,
        shuffle=False,
        num_workers=2,
        pin_memory=(device.type == "cuda"),
        collate_fn=default_collate,
    )
    preds = []
    with torch.no_grad():
        for batch in loader:
            image = batch["image"].to(device)
            tabular = batch["tabular"].to(device)
            out = model(image, tabular)
            preds.extend(out.squeeze(-1).cpu().numpy())
    ids = list(dataset.ids)
    return ids, np.clip(np.asarray(preds, dtype=np.float64), 0.0, 100.0)

def run_training(
    epochs=15,
    backbone_lr=1e-5,
    head_lr=1e-4,
    weight_decay=1e-5,
    warmup_epochs=2,
    patience=5,
    val_frac=0.1,
    seed=42,
    img_size=IMG_SIZE,
    pretrained=True,
):
    t0 = time.time()
    device = resolve_device()
    print(f"[setup] device={device} | model_family={MODEL_FAMILY}")

    # ---- Data preparation (split FIRST, fit normalizer on train only) ----
    input_dir = get_input_dir()
    train_df, test_df = load_tables(input_dir)
    tr, va = make_split(train_df, test_size=val_frac, seed=seed)

    tr_tab_raw = build_tabular_features(tr)
    va_tab_raw = build_tabular_features(va)
    te_tab_raw = build_tabular_features(test_df)

    normalizer = TabularNormalizer().fit(tr_tab_raw)
    tr_tab = normalizer.transform(tr_tab_raw)
    va_tab = normalizer.transform(va_tab_raw)
    te_tab = normalizer.transform(te_tab_raw)

    tr_tfm = build_train_transform(img_size)
    ev_tfm = build_eval_transform(img_size)
    train_root = os.path.join(input_dir, "train")
    test_root = os.path.join(input_dir, "test")

    train_ds = PetFinderDataset(
        tr["Id"].values, tr_tab, tr["Pawpularity"].values, train_root, tr_tfm, True
    )
    val_ds = PetFinderDataset(
        va["Id"].values, va_tab, va["Pawpularity"].values, train_root, ev_tfm, True
    )
    test_ds = PetFinderDataset(
        test_df["Id"].values, te_tab, None, test_root, ev_tfm, False
    )

    # ---- Model / criterion ----
    model = PetFinderModel(pretrained=pretrained).to(device)
    criterion = PawpularityLoss()

    # ---- Optimizer / scheduler (differential LR + warmup/cosine) ----
    batch_size = pick_batch_size(device)
    param_groups = build_param_groups(model, backbone_lr, head_lr, weight_decay)
    optimizer = torch.optim.AdamW(param_groups, weight_decay=weight_decay)
    scheduler = make_scheduler(optimizer, warmup_epochs, epochs)

    train_loader = torch.utils.data.DataLoader(
        train_ds,
        batch_size=batch_size,
        shuffle=True,
        num_workers=4,
        pin_memory=(device.type == "cuda"),
        collate_fn=default_collate,
        drop_last=False,
    )

    # ---- Training loop with early stopping + best checkpoint ----
    best_rmse = float("inf")
    best_state = None
    bad_epochs = 0
    grad_accum = 1
    resolved_bs = batch_size

    for epoch in range(1, epochs + 1):
        model.train()
        running_loss = 0.0
        n_batches = 0
        optimizer.zero_grad(set_to_none=True)
        for step, batch in enumerate(train_loader):
            image = batch["image"].to(device)
            tabular = batch["tabular"].to(device)
            target = batch["target"].to(device)
            try:
                pred = model(image, tabular)
                loss = criterion(pred, target)
                loss = loss / grad_accum
                loss.backward()
            except RuntimeError as e:
                if (
                    "out of memory" in str(e).lower()
                    and device.type == "cuda"
                    and grad_accum == 1
                ):
                    # Fallback: reduce batch to 16 and enable gradient accumulation of 2.
                    torch.cuda.empty_cache()
                    grad_accum = 2
                    resolved_bs = 16
                    print(f"[fallback] OOM -> batch=16, grad_accum=2")
                    # Re-run this step with smaller effective handling via re-split
                    chunk = max(1, image.shape[0] // 2)
                    for s in range(0, image.shape[0], chunk):
                        sl = slice(s, s + chunk)
                        p = model(image[sl], tabular[sl])
                        l = criterion(p, target[sl])
                        (l / grad_accum).backward()
                    running_loss += float(loss.item()) * grad_accum
                    n_batches += 1
                    if (step + 1) % grad_accum == 0:
                        torch.nn.utils.clip_grad_norm_(model.parameters(), 1.0)
                        optimizer.step()
                        optimizer.zero_grad(set_to_none=True)
                    continue
                raise
            running_loss += float(loss.item()) * grad_accum
            n_batches += 1
            if (step + 1) % grad_accum == 0:
                torch.nn.utils.clip_grad_norm_(model.parameters(), 1.0)
                optimizer.step()
                optimizer.zero_grad(set_to_none=True)

        scheduler.step()
        avg_loss = running_loss / max(1, n_batches)

        # Validation RMSE (same inference path as test)
        val_rmse, _ = evaluate_rmse(model, val_ds, device, batch_size=resolved_bs)

        print(
            f"Epoch {epoch:02d}/{epochs} | train_loss={avg_loss:.4f} | "
            f"val_rmse={val_rmse:.4f} | lr={optimizer.param_groups[0]['lr']:.2e}"
        )

        if val_rmse < best_rmse - 1e-4:
            best_rmse = val_rmse
            best_state = {
                k: v.detach().cpu().clone() for k, v in model.state_dict().items()
            }
            bad_epochs = 0
        else:
            bad_epochs += 1
            if bad_epochs >= patience:
                print(f"[early-stop] no improvement for {patience} epochs")
                break

    elapsed = time.time() - t0
    peak = peak_vram_mb(device)
    print(
        f"[metrics] elapsed_seconds={elapsed:.1f} | peak_vram_mb={peak:.1f} | "
        f"resolved_batch_size={resolved_bs}"
    )

    # Restore best weights
    if best_state is not None:
        model.load_state_dict(best_state)

    # ---- Test inference (identical processing to validation) ----
    ids, test_preds = predict_dataset(model, test_ds, device, batch_size=resolved_bs)

    # Re-align predictions to original test-row order (predict_dataset preserves
    # dataset order which equals test_df order, but be explicit/safe).
    order_map = {iid: i for i, iid in enumerate(test_df["Id"].values)}
    aligned = np.full(len(test_df), np.nan, dtype=np.float64)
    for iid, p in zip(ids, test_preds):
        aligned[order_map[iid]] = p
    if np.isnan(aligned).any():
        aligned = test_preds  # fallback to dataset-order alignment

    sub_path = os.path.join(SUBMISSION_DIR, "submission.csv")
    write_submission(test_df, aligned, sub_path)
    print(f"[submission] wrote {sub_path} ({len(aligned)} rows)")

    score = best_rmse
    print(f"Final Validation Score: {score}")
    return score


if __name__ == "__main__":
    run_training()
