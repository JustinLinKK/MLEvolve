    """Return a float array [N, N_TABULAR_FEAT] built defensively from df."""
    present = {c: df[c].astype(np.float32).to_numpy() for c in df.columns}
    # base flags (fill missing with 0)

        "Eyes",
        "Near",

        "Occupy",
        "Group",
        "Collage",
        "Info",
        "Blur",
        "Accessory",
        "Action",

        if c in present:
            base[:, j] = present[c]
    face, eyes, near, subj, occupy, group, collage, info, blur, acc, act, human = (

        base[:, 7],
        base[:, 8],
        base[:, 9],

        base[:, 11],

    # engineered interaction / synergy features (rubric knowledge)
    inter = np.stack(
        [
            near * face,  # close-up AND face visible -> big positive
            blur * eyes,  # blur hurts when eyes expected
            face * eyes,  # full-face + eyes open
            near * subj,  # near subject focus
            group,  # group shots tend lower

            eyes,  # eye contact
            subj,  # subject focus
            1.0 - group,  # solo bonus
        ],
        axis=1,
    ).astype(np.float32)

    feat = np.concatenate([base, inter], axis=1)

def rank_smooth_targets(y_raw, alpha=RANK_SMOOTH_ALPHA):
    """Blend raw label with its rank-percentile (0-100) for noise robustness."""
    y_raw = np.asarray(y_raw, dtype=np.float32)
    order = np.argsort(np.argsort(y_raw))
    n = max(len(y_raw) - 1, 1)
    ranks = (order.astype(np.float32) / n) * 100.0
    return alpha * y_raw + (1.0 - alpha) * ranks

def collate_fn(batch):

class TabularTower(nn.Module):
    def __init__(self, in_dim=N_TABULAR_FEAT, hidden=(128, 64)):
        super().__init__()
        layers, d = [], in_dim
        for h in hidden:
            layers += [
                nn.Linear(d, h),
                nn.BatchNorm1d(h),
                nn.ReLU(inplace=True),
                nn.Dropout(0.2),
            ]
            d = h
        self.net = nn.Sequential(*layers)
        self.out = nn.Linear(d, 1)

    def forward(self, x):
        return self.net(x).squeeze(-1)


class DualPathModel(nn.Module):
    """Residual fusion: pred = tab_pred + gate * vision_pred.

    The tabular tower is initialized to carry the main signal; the vision path
    starts small (gate ~0) so the network can add image evidence gradually.
    """

    def __init__(self, tab_in=N_TABULAR_FEAT, freeze_backbone=True):
        super().__init__()
        self.backbone = resnet18(
            weights=None
        )  # offline: random init, then fine-tune lightly
        in_f = self.backbone.fc.in_features
        self.backbone.fc = nn.Identity()
        self.vision_head = nn.Sequential(
            nn.Linear(in_f, 64),
            nn.ReLU(inplace=True),
            nn.Dropout(0.3),
            nn.Linear(64, 1),
        )
        self.tab_tower = TabularTower(tab_in)
        # learned scalar gate, start near 0 (vision contributes little at first)
        self.gate = nn.Parameter(torch.tensor(-2.0))  # sigmoid(-2)=0.12
        if freeze_backbone:
            for p in self.backbone.parameters():
                p.requires_grad_(False)

    def forward(self, image, tabular):
        vfeat = self.backbone(image)
        vpred = self.vision_head(vfeat).squeeze(-1)
        tpred = self.tab_tower(tabular)
        g = torch.sigmoid(self.gate)
        return tpred + g * vpred


def build_model(context=None):
    context = dict(context or {})
    freeze = context.get("freeze_backbone", True)
    return DualPathModel(tab_in=N_TABULAR_FEAT, freeze_backbone=freeze)

# Step results

        ### Step 1: model_design
        **Plan:** **Plan**

1. **WHAT makes this task unique?** Pawpularity is a *subjective human* rating of pet photos, not an objective property — the 12 tabular flags (Face, Eyes, Near, Blur, Group…) are essentially the crowd's own rubric for what makes a photo "paw-pular," and they dominate the signal over fine visual detail. This means a model that *under-weights* raw pixels and *over-weights* the structured flags will generalize better than a pure vision net; the best prior attempt (20.75) was convnext+gated cross-attention, but every heavier/fusion variant since has regressed or broken, so the winning direction is to make the **tabular channel first-class and dominant**, with a lightweight CNN only as a complementary feature source.
2. **WHY my approach fits:** I use a *dual-path residual architecture*: a strong MLP tower on the engineered tabular features (including interaction terms like `Near×Face`, `Blur×Eyes`) plus a frozen ResNet18 image encoder, fused by a learned **scalar gate** that lets the network decide how much to trust each modality. A rank-smoothed target (robust to the noisy integer labels) + Huber loss keeps training stable against label noise. This directly targets the "flags carry most of the truth" hypothesis while still learning from images.
3. **HOW I validate:** If the tabular-dominant design beats the prior 20.75 baseline, it confirms the rubric-flags hypothesis; if it underperforms, the gate weights tell us whether to shift toward the vision path. Memory strategy: fp32, single-GPU, fixed batch (elastic OOM-halving owned by Stage 3); no scheduler/optimizer/batch logic here.

def _build_tabular_vector(df):
    """Return a float array [N, N_TABULAR_FEAT] built defensively from df."""
    present = {c: df[c].astype(np.float32).to_numpy() for c in df.columns}
    # base flags (fill missing with 0)
    base_cols = [
        "Face",
        "Eyes",
        "Near",
        "Subject Focus",
        "Occupy",
        "Group",
        "Collage",
        "Info",
        "Blur",
        "Accessory",
        "Action",
        "Human",
    ]
    base = np.zeros((len(df), len(base_cols)), dtype=np.float32)
    for j, c in enumerate(base_cols):
        if c in present:
            base[:, j] = present[c]
    face, eyes, near, subj, occupy, group, collage, info, blur, acc, act, human = (
        base[:, 0],
        base[:, 1],
        base[:, 2],
        base[:, 3],
        base[:, 4],
        base[:, 5],
        base[:, 6],
        base[:, 7],
        base[:, 8],
        base[:, 9],
        base[:, 10],
        base[:, 11],
    )

    # engineered interaction / synergy features (rubric knowledge)
    inter = np.stack(
        [
            near * face,  # close-up AND face visible -> big positive
            blur * eyes,  # blur hurts when eyes expected
            face * eyes,  # full-face + eyes open
            near * subj,  # near subject focus
            group,  # group shots tend lower
            collage,  # collages tend lower
            info,  # text/info overlays lower
            (1.0 - blur),  # sharpness bonus
            (1.0 - occu) if False else (1.0 - occupy),  # clarity bonus
            human,  # human presence
            act,  # action shot
            acc,  # accessory
            near,  # proximity
            face,  # face visibility
            eyes,  # eye contact
            subj,  # subject focus
            1.0 - group,  # solo bonus
        ],
        axis=1,
    ).astype(np.float32)

    feat = np.concatenate([base, inter], axis=1)
    # pad/truncate to exactly N_TABULAR_FEAT for a stable contract
    if feat.shape[1] < N_TABULAR_FEAT:
        feat = np.pad(feat, ((0, 0), (0, N_TABULAR_FEAT - feat.shape[1])))
    elif feat.shape[1] > N_TABULAR_FEAT:
        feat = feat[:, :N_TABULAR_FEAT]
    return feat

def rank_smooth_targets(y_raw, alpha=RANK_SMOOTH_ALPHA):
    """Blend raw label with its rank-percentile (0-100) for noise robustness."""
    y_raw = np.asarray(y_raw, dtype=np.float32)
    order = np.argsort(np.argsort(y_raw))
    n = max(len(y_raw) - 1, 1)
    ranks = (order.astype(np.float32) / n) * 100.0
    return alpha * y_raw + (1.0 - alpha) * ranks

def collate_fn(batch):
    images = torch.stack([b["image"] for b in batch], dim=0)
    tabular = torch.stack([b["tabular"] for b in batch], dim=0)
    out = {"image": images, "tabular": tabular}
    if "target" in batch[0]:

    return out

# Model: dual-path residual (dominant tabular MLP + frozen ResNet18 gate fusion)
# --------------------------------------------------------------------------- #
class TabularTower(nn.Module):
    def __init__(self, in_dim=N_TABULAR_FEAT, hidden=(128, 64)):
        super().__init__()
        layers, d = [], in_dim
        for h in hidden:
            layers += [
                nn.Linear(d, h),
                nn.BatchNorm1d(h),
                nn.ReLU(inplace=True),
                nn.Dropout(0.2),
            ]
            d = h
        self.net = nn.Sequential(*layers)
        self.out = nn.Linear(d, 1)

    def forward(self, x):
        return self.net(x).squeeze(-1)


class DualPathModel(nn.Module):
    """Residual fusion: pred = tab_pred + gate * vision_pred.

    The tabular tower is initialized to carry the main signal; the vision path
    starts small (gate ~0) so the network can add image evidence gradually.
    """

    def __init__(self, tab_in=N_TABULAR_FEAT, freeze_backbone=True):
        super().__init__()
        self.backbone = resnet18(
            weights=None
        )  # offline: random init, then fine-tune lightly
        in_f = self.backbone.fc.in_features
        self.backbone.fc = nn.Identity()
        self.vision_head = nn.Sequential(
            nn.Linear(in_f, 64),
            nn.ReLU(inplace=True),
            nn.Dropout(0.3),
            nn.Linear(64, 1),
        )
        self.tab_tower = TabularTower(tab_in)
        # learned scalar gate, start near 0 (vision contributes little at first)
        self.gate = nn.Parameter(torch.tensor(-2.0))  # sigmoid(-2)=0.12
        if freeze_backbone:
            for p in self.backbone.parameters():
                p.requires_grad_(False)

    def forward(self, image, tabular):
        vfeat = self.backbone(image)
        vpred = self.vision_head(vfeat).squeeze(-1)
        tpred = self.tab_tower(tabular)
        g = torch.sigmoid(self.gate)
        return tpred + g * vpred


def build_model(context=None):
    context = dict(context or {})
    freeze = context.get("freeze_backbone", True)
    return DualPathModel(tab_in=N_TABULAR_FEAT, freeze_backbone=freeze)

class CandidateAdapter:
    def __init__(self):
        self._defaults = {
            "huber_delta": HUBER_DELTA,
            "freeze_backbone": True,
            "lr": 1e-3,
            "weight_decay": 1e-4,
        }
        self._model = None
        self._criterion = None
        self._optimizer = None

    def _ctx(self, context):
        merged = dict(self._defaults)
        if context:
            merged.update({k: v for k, v in context.items() if v is not None})
        return merged

    def build_model(self, context=None):
        ctx = self._ctx(context)
        model = build_model(ctx)
        self._model = model
        self._criterion = build_criterion(ctx)
        return model

    def build_optimizer(self, model, context=None):
        ctx = self._ctx(context)
        lr = ctx.get("lr", 1e-3)
        wd = ctx.get("weight_decay", 1e-4)
        opt = torch.optim.AdamW(model.parameters(), lr=lr, weight_decay=wd)
        self._optimizer = opt
        return opt

def _build_tabular_vector(df):
    """Return float array [N, N_TABULAR_FEAT]; defensive against missing cols."""
    present = {c: df[c].astype(np.float32).to_numpy() for c in df.columns}
    base_cols = [
        "Face",
        "Eyes",
        "Near",
        "Subject Focus",
        "Occupy",
        "Group",
        "Collage",
        "Info",
        "Blur",
        "Accessory",
        "Action",
        "Human",
    ]
    base = np.zeros((len(df), len(base_cols)), dtype=np.float32)
    for j, c in enumerate(base_cols):
        if c in present:
            base[:, j] = present[c]

    face, eyes, near, subj, occupy, group = (
        base[:, 0],
        base[:, 1],
        base[:, 2],
        base[:, 3],
        base[:, 4],
        base[:, 5],
    )
    collage, info, blur, acc, act, human = (
        base[:, 6],
        base[:, 7],
        base[:, 8],
        base[:, 9],
        base[:, 10],
        base[:, 11],
    )

    inter = np.stack(
        [
            near * face,  # close-up AND face visible
            blur * eyes,  # blur hurts when eyes expected
            face * eyes,  # full-face + eyes open
            near * subj,  # near subject focus
            group,  # group shots tend lower
            collage,  # collages tend lower
            info,  # text/info overlays lower
            (1.0 - blur),  # sharpness bonus
            (1.0 - occupy),  # clarity bonus
            human,  # human presence
            act,  # action shot
            acc,  # accessory
            near,  # proximity
            face,  # face visibility
            eyes,  # eye contact
            subj,  # subject focus
            1.0 - group,  # solo bonus
        ],
        axis=1,
    ).astype(np.float32)

    feat = np.concatenate([base, inter], axis=1)
    if feat.shape[1] < N_TABULAR_FEAT:
        feat = np.pad(feat, ((0, 0), (0, N_TABULAR_FEAT - feat.shape[1])))
    elif feat.shape[1] > N_TABULAR_FEAT:
        feat = feat[:, :N_TABULAR_FEAT]
    return feat

def rank_smooth_targets(y_raw, alpha=RANK_SMOOTH_ALPHA):
    """Blend raw label with its rank-percentile (0-100) for noise robustness."""
    y_raw = np.asarray(y_raw, dtype=np.float32)
    order = np.argsort(np.argsort(y_raw))
    n = max(len(y_raw) - 1, 1)
    ranks = (order.astype(np.float32) / n) * 100.0
    return alpha * y_raw + (1.0 - alpha) * ranks

# Model: dual-path residual (dominant tabular MLP + ResNet18 gated fusion)

class TabularTower(nn.Module):
    def __init__(self, in_dim=N_TABULAR_FEAT, hidden=(128, 64)):
        super().__init__()
        layers, d = [], in_dim
        for h in hidden:
            layers += [
                nn.Linear(d, h),
                nn.BatchNorm1d(h),
                nn.ReLU(inplace=True),
                nn.Dropout(0.2),
            ]
            d = h
        self.net = nn.Sequential(*layers)
        self.out = nn.Linear(d, 1)

    def forward(self, x):
        return self.net(x).squeeze(-1)


class DualPathModel(nn.Module):
    """pred = tab_pred + sigmoid(gate) * vision_pred. Gate starts ~0.12."""

    def __init__(self, tab_in=N_TABULAR_FEAT, freeze_backbone=False):
        super().__init__()
        self.backbone = resnet18(weights=None)  # offline random init, light fine-tune
        in_f = self.backbone.fc.in_features
        self.backbone.fc = nn.Identity()
        self.vision_head = nn.Sequential(
            nn.Linear(in_f, 64),
            nn.ReLU(inplace=True),
            nn.Dropout(0.3),
            nn.Linear(64, 1),
        )
        self.tab_tower = TabularTower(tab_in)
        self.gate = nn.Parameter(torch.tensor(-2.0))  # sigmoid(-2)=0.12
        if freeze_backbone:
            for p in self.backbone.parameters():
                p.requires_grad_(False)

    def forward(self, image, tabular):
        vfeat = self.backbone(image)
        vpred = self.vision_head(vfeat).squeeze(-1)
        tpred = self.tab_tower(tabular)
        g = torch.sigmoid(self.gate)
        return tpred + g * vpred

    freeze = context.get("freeze_backbone", False)

# Scheduler: linear warmup -> cosine decay (LambdaLR over epochs)
# --------------------------------------------------------------------------- #
def build_scheduler(optimizer, total_epochs, warmup_epochs=WARMUP_EPOCHS):
    def lr_lambda(epoch):
        if epoch < warmup_epochs:
            return (epoch + 1) / max(warmup_epochs, 1)
        progress = (epoch - warmup_epochs) / max(total_epochs - warmup_epochs, 1)
        return 0.5 * (1.0 + math.cos(math.pi * min(progress, 1.0)))

    return torch.optim.lr_scheduler.LambdaLR(optimizer, lr_lambda)

def _mixup(images, tabs, targets, prob=MIXUP_PROB, alpha=MIXUP_ALPHA):
    if torch.rand(1).item() > prob:
        return images, tabs, targets
    lam = np.random.beta(alpha, alpha)
    bs = images.size(0)
    idx = torch.randperm(bs, device=images.device)
    mixed_img = lam * images + (1 - lam) * images[idx]
    mixed_tab = lam * tabs + (1 - lam) * tabs[idx]
    mixed_tgt = lam * targets + (1 - lam) * targets[idx]
    return mixed_img, mixed_tab, mixed_tgt
