# Target transformation (rank-based, fit ONLY on train)
# --------------------------------------------------------------------------- #
class RankTargetTransform:
    """Maps raw Pawpularity scores to their empirical CDF rank on the TRAIN set.

    Using ranks (in [0, 1]) stabilises regression against heavy-tailed /
    outlier-heavy subjective ratings. Inverse mapping uses the sorted train
    values, so no validation/test target information leaks in.
    """

    def __init__(self):
        self.sorted_values = None

    def fit(self, y):
        y = np.asarray(y, dtype=np.float64)
        order = np.argsort(y, kind="stable")
        self.sorted_values = y[order]
        self.n = len(y)
        return self

    def transform(self, y):
        y = np.asarray(y, dtype=np.float64)
        ranks = np.searchsorted(self.sorted_values, y, side="right").astype(np.float64)
        return ranks / max(self.n, 1)

    def inverse_transform(self, r):
        r = np.clip(np.asarray(r, dtype=np.float64), 0.0, 1.0)
        pos = r * (self.n - 1)
        lo = np.floor(pos).astype(int)
        hi = np.minimum(lo + 1, self.n - 1)
        frac = pos - lo
        return self.sorted_values[lo] * (1.0 - frac) + self.sorted_values[hi] * frac

# --------------------------------------------------------------------------- #
class CrossAttentionFusion(nn.Module):
    """Cross-attention fusion of tabular queries over spatial vision tokens."""

    def __init__(self, feat_dim, tab_dim, hidden, num_heads=4, dropout=0.3):
        super().__init__()
        self.num_heads = num_heads
        self.head_dim = hidden // num_heads
        assert hidden % num_heads == 0, "hidden must be divisible by num_heads"

        self.tab_proj = nn.Sequential(
            nn.Linear(tab_dim, hidden), nn.GELU(), nn.LayerNorm(hidden)
        )
        self.vision_proj = nn.Linear(feat_dim, hidden)
        self.q = nn.Linear(hidden, hidden)
        self.k = nn.Linear(hidden, hidden)
        self.v = nn.Linear(hidden, hidden)
        self.out = nn.Linear(hidden, hidden)
        self.norm = nn.LayerNorm(hidden)
        self.dropout = nn.Dropout(dropout)

    def _reshape(self, x, bsz):
        return x.view(bsz, self.num_heads, -1, self.head_dim).transpose(1, 2)

    def forward(self, tab, vision):
        # tab: [B, T, H]; vision: [B, N, D]
        B = tab.shape[0]
        q = self._reshape(self.q(self.tab_proj(tab)), B)  # [B, h, T, hd]
        k = self._reshape(self.k(self.vision_proj(vision)), B)  # [B, h, N, hd]
        v = self._reshape(self.v(self.vision_proj(vision)), B)  # [B, h, N, hd]
        scale = math.sqrt(self.head_dim)
        attn = torch.matmul(q, k.transpose(-2, -1)) / scale  # [B, h, T, N]
        attn = torch.softmax(attn, dim=-1)
        attn = self.dropout(attn)
        ctx = torch.matmul(attn, v)  # [B, h, T, hd]
        ctx = ctx.transpose(1, 2).reshape(B, -1, self.head_dim * self.num_heads)
        ctx = self.out(ctx)
        return self.norm(tab + ctx)


class PawpularityModel(nn.Module):
    """Frozen ResNet50 feature-map backbone + cross-attention tabular fusion head.

    Forward inputs:
        image:   [B, 3, IMG_SIZE, IMG_SIZE]
        tabular: [B, TABULAR_DIM]
    Output:
        pred:    [B]  (raw score in the target-transform space, i.e. [0, 1] when
                      rank targets are used)
    """

    def __init__(
        self,
        tab_dim=TABULAR_DIM,
        feat_dim=BACKBONE_FEAT,
        hidden=ATTN_HIDDEN,
        dropout=DROPOUT,
    ):
        super().__init__()
        self.backbone = resnet50(weights=None)  # offline-safe: no remote download
        # Keep the convolutional stem + stages; replace the classifier head.
        self.feat = nn.Sequential(
            *list(self.backbone.children())[:-1]
        )  # -> [B,1024,H,W]
        self.pool = nn.AdaptiveAvgPool2d((4, 4))  # -> [B,1024,16]

        self.fusion = CrossAttentionFusion(
            feat_dim, tab_dim, hidden, num_heads=4, dropout=dropout
        )
        self.head = nn.Sequential(
            nn.Linear(hidden, hidden),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(hidden, hidden // 2),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(hidden // 2, 1),
        )

        # Freeze the backbone (feature extractor only).
        for p in self.feat.parameters():
            p.requires_grad = False
        self.feat.eval()

    def train(self, mode=True):
        super().train(mode)
        # Backbone stays in eval mode regardless (BatchNorm running stats fixed).
        self.feat.eval()
        return self

    def extract_vision(self, image):
        with torch.no_grad():
            x = self.feat(image)  # [B, 1024, H', W']
            x = self.pool(x)  # [B, 1024, 4, 4]
        B = x.shape[0]
        x = x.flatten(2).transpose(1, 2)  # [B, 16, 1024]
        return x

    def forward(self, image, tabular):
        vision = self.extract_vision(image)  # [B, N, D]
        tab = tabular.unsqueeze(1)  # [B, 1, D]
        fused = self.fusion(tab, vision)  # [B, 1, H]
        fused = fused.squeeze(1)  # [B, H]
        pred = self.head(fused).squeeze(-1)  # [B]
        return pred

    # -- context handling -------------------------------------------------- #
    def _merge_context(self, context):
        merged = dict(self.DEFAULT_CONTEXT)
        merged.update(self._ctx)
        if context:
            merged.update(context)
        self._ctx = merged
        return merged

    def _resolve_model(self, ctx):
        model = ctx.get("model")
        if model is None:
            model = PawpularityModel()
            ctx["model"] = model
        return model

    def _resolve_criterion(self, ctx):
        crit = ctx.get("criterion")
        if crit is None:
            crit = get_criterion()
            ctx["criterion"] = crit
        return crit

    def _resolve_dataset(self, ctx):
        ds = ctx.get("dataset")
        if ds is None:
            transform = ctx.get("transform") or build_eval_transform()
            df = load_csv("train")
            df = engineer_tabular(df)
            ds = PetDataset(df, "train", transform, FEATURE_COLUMNS)
            ctx["dataset"] = ds
        return ds

    # -- required API ------------------------------------------------------ #
    def build_model(self, context=None):
        ctx = self._merge_context(context)
        device = ctx.get("device", "cpu")
        model = self._resolve_model(ctx)
        return model.to(device)

    def build_optimizer(self, model, context=None):
        ctx = self._merge_context(context)
        lr = ctx.get("lr", 1e-3)
        wd = ctx.get("weight_decay", 1e-4)
        # Only optimise parameters that require grad (the fusion head).
        params = [p for p in model.parameters() if p.requires_grad]
        opt = torch.optim.AdamW(params, lr=lr, weight_decay=wd)
        ctx["optimizer"] = opt
        return opt

    def _build_batch_from_scenario(self, scenario, device):
        bs = int(scenario.get("batch_size", 2))
        fixture = scenario.get("fixture", "")
        if fixture:
            # Authoritative named input contract: build declared tensors.

                "image": image.to(device),
                "tabular": tabular.to(device),
                "target": target.to(device),

        ds = self._resolve_dataset(self._ctx)
        idxs = list(range(min(bs, len(ds))))

        batch = default_collate(items)
        return {
            k: (v.to(device) if torch.is_tensor(v) else v) for k, v in batch.items()
        }

    def build_train_batch(self, scenario, device):
        return self._build_batch_from_scenario(scenario, device)

    def build_validation_batch(self, scenario, device):
        return self._build_batch_from_scenario(scenario, device)

    def training_step(self, model, batch, context=None):
        ctx = self._merge_context(context)
        criterion = self._resolve_criterion(ctx)
        model.train()
        image = batch["image"]
        tabular = batch["tabular"]
        target = batch["target"]
        pred = model(image, tabular)
        loss = criterion(pred, target)
        return loss

    def validation_step(self, model, batch, context=None):
        ctx = self._merge_context(context)
        criterion = self._resolve_criterion(ctx)
        model.eval()
        with torch.no_grad():
            image = batch["image"]
            tabular = batch["tabular"]
            target = batch["target"]
            pred = model(image, tabular)
            loss = criterion(pred, target)
        return loss

# Target transformation (rank-based, fit ONLY on train)

class RankTargetTransform:
    """Maps raw Pawpularity scores to their empirical CDF rank on the TRAIN set.

    Using ranks (in [0, 1]) stabilises regression against heavy-tailed /
    outlier-heavy subjective ratings. Inverse mapping uses the sorted train
    values, so no validation/test target information leaks in.
    """

    def __init__(self):
        self.sorted_values = None

    def fit(self, y):
        y = np.asarray(y, dtype=np.float64)
        order = np.argsort(y, kind="stable")
        self.sorted_values = y[order]
        self.n = len(y)
        return self

    def transform(self, y):
        y = np.asarray(y, dtype=np.float64)
        ranks = np.searchsorted(self.sorted_values, y, side="right").astype(np.float64)
        return ranks / max(self.n, 1)

    def inverse_transform(self, r):
        r = np.clip(np.asarray(r, dtype=np.float64), 0.0, 1.0)
        pos = r * (self.n - 1)
        lo = np.floor(pos).astype(int)
        hi = np.minimum(lo + 1, self.n - 1)
        frac = pos - lo
        return self.sorted_values[lo] * (1.0 - frac) + self.sorted_values[hi] * frac

class CrossAttentionFusion(nn.Module):
    """Cross-attention fusion of tabular queries over spatial vision tokens."""

    def __init__(self, feat_dim, tab_dim, hidden, num_heads=4, dropout=0.3):
        super().__init__()
        self.num_heads = num_heads
        self.head_dim = hidden // num_heads
        assert hidden % num_heads == 0, "hidden must be divisible by num_heads"

        self.tab_proj = nn.Sequential(
            nn.Linear(tab_dim, hidden), nn.GELU(), nn.LayerNorm(hidden)
        )
        self.vision_proj = nn.Linear(feat_dim, hidden)
        self.q = nn.Linear(hidden, hidden)
        self.k = nn.Linear(hidden, hidden)
        self.v = nn.Linear(hidden, hidden)
        self.out = nn.Linear(hidden, hidden)
        self.norm = nn.LayerNorm(hidden)
        self.dropout = nn.Dropout(dropout)

    def _reshape(self, x, bsz):
        return x.view(bsz, self.num_heads, -1, self.head_dim).transpose(1, 2)

    def forward(self, tab, vision):
        # tab: [B, T, H]; vision: [B, N, D]
        B = tab.shape[0]
        q = self._reshape(self.q(self.tab_proj(tab)), B)  # [B, h, T, hd]
        k = self._reshape(self.k(self.vision_proj(vision)), B)  # [B, h, N, hd]
        v = self._reshape(self.v(self.vision_proj(vision)), B)  # [B, h, N, hd]
        scale = math.sqrt(self.head_dim)
        attn = torch.matmul(q, k.transpose(-2, -1)) / scale  # [B, h, T, N]
        attn = torch.softmax(attn, dim=-1)
        attn = self.dropout(attn)
        ctx = torch.matmul(attn, v)  # [B, h, T, hd]
        ctx = ctx.transpose(1, 2).reshape(B, -1, self.head_dim * self.num_heads)
        ctx = self.out(ctx)
        return self.norm(tab + ctx)

    """Frozen ResNet50 feature-map backbone + cross-attention tabular fusion head.

    Forward inputs:
        image:   [B, 3, IMG_SIZE, IMG_SIZE]
        tabular: [B, TABULAR_DIM]
    Output:
        pred:    [B]  (raw score in the target-transform space, i.e. [0, 1] when
                      rank targets are used)
    """

    def __init__(
        self,
        tab_dim=TABULAR_DIM,
        feat_dim=BACKBONE_FEAT,
        hidden=ATTN_HIDDEN,
        dropout=DROPOUT,
    ):
        super().__init__()
        self.backbone = resnet50(weights=None)  # offline-safe: no remote download
        # Keep the convolutional stem + stages; replace the classifier head.
        self.feat = nn.Sequential(
            *list(self.backbone.children())[:-1]
        )  # -> [B,1024,H,W]
        self.pool = nn.AdaptiveAvgPool2d((4, 4))  # -> [B,1024,16]

        self.fusion = CrossAttentionFusion(
            feat_dim, tab_dim, hidden, num_heads=4, dropout=dropout
        )
        self.head = nn.Sequential(
            nn.Linear(hidden, hidden),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(hidden, hidden // 2),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(hidden // 2, 1),
        )

        # Freeze the backbone (feature extractor only).
        for p in self.feat.parameters():
            p.requires_grad = False
        self.feat.eval()

    def train(self, mode=True):
        super().train(mode)
        # Backbone stays in eval mode regardless (BatchNorm running stats fixed).
        self.feat.eval()
        return self

    def extract_vision(self, image):
        with torch.no_grad():
            x = self.feat(image)  # [B, 1024, H', W']
            x = self.pool(x)  # [B, 1024, 4, 4]
        B = x.shape[0]
        x = x.flatten(2).transpose(1, 2)  # [B, 16, 1024]
        return x

    def forward(self, image, tabular):
        vision = self.extract_vision(image)  # [B, N, D]
        tab = tabular.unsqueeze(1)  # [B, 1, D]
        fused = self.fusion(tab, vision)  # [B, 1, H]
        fused = fused.squeeze(1)  # [B, H]
        pred = self.head(fused).squeeze(-1)  # [B]
        return pred

# --------------------------------------------------------------------------- #
# Scheduler: linear warmup -> cosine decay (step-based, LambdaLR).
# --------------------------------------------------------------------------- #
def build_scheduler(optimizer, total_steps, warmup_fraction=WARMUP_FRACTION):
    warmup_steps = max(1, int(total_steps * warmup_fraction))

    def lr_lambda(step):
        if step < warmup_steps:
            return (step + 1) / float(warmup_steps)
        progress = (step - warmup_steps) / max(1, (total_steps - warmup_steps))
        progress = min(max(progress, 0.0), 1.0)
        return 0.5 * (1.0 + math.cos(math.pi * progress))

    return torch.optim.lr_scheduler.LambdaLR(optimizer, lr_lambda)


# --------------------------------------------------------------------------- #
# Differential parameter groups: high LR on trainable head, low LR elsewhere.
# --------------------------------------------------------------------------- #
def build_optimizer(
    model, head_lr=HEAD_LR, backbone_lr=BACKBONE_LR, weight_decay=WEIGHT_DECAY
):
    head_params, backbone_params = [], []
    for name, p in model.named_parameters():
        if not p.requires_grad:
            continue
        # Backbone feature extractor lives under `feat.`; everything else is the
        # trainable fusion/head (and pool adapter).
        if name.startswith("feat."):
            backbone_params.append(p)
        else:
            head_params.append(p)

    groups = [{"params": head_params, "lr": head_lr}]
    if backbone_params:
        groups.append({"params": backbone_params, "lr": backbone_lr})
    return torch.optim.AdamW(groups, lr=head_lr, weight_decay=weight_decay)

def _predict(model, loader, device, rank_tf=None):
    """Run the trained model over a loader.

    Returns raw model outputs (rank space) and, when ``rank_tf`` is provided,
    the inverse-transformed scores clamped to [0, 100].
    """
    model.eval()
    raw_out = []
    for batch in loader:
        img = batch["image"].to(device, non_blocking=True)
        tab = batch["tabular"].to(device, non_blocking=True)
        pred = model(img, tab)  # [B] in target-transform space
        raw_out.append(pred.detach().cpu())
    raw = torch.cat(raw_out, dim=0).numpy()
    if rank_tf is None:
        return raw, None
    scores = rank_tf.inverse_transform(raw)
    scores = np.clip(scores, 0.0, 100.0)
    return raw, scores
